def exp_interval(x, terms=140):
    x = F(x)
    if x < 0:
        l, h = exp_interval(-x, terms)
        return 1 / h, 1 / l
    total, term = F(1), F(1)
    for n in range(1, terms):
        term *= x / n
        total += term
    first = term * x / terms
    ratio = x / (terms + 1)
    assert ratio < 1
    return total, total + first / (1 - ratio)


def log_interval(x, terms=240):
    x = F(x)
    assert x > 0
    if x < 1:
        l, h = log_interval(1 / x, terms)
        return -h, -l
    # Bound the enormous Taylor fraction by 45-digit rationals before the
    # logarithm series; monotonicity preserves a rigorous enclosure and avoids
    # needless hundreds-of-thousands-digit intermediate denominators.
    scale = 10**45
    low = F((x.numerator*scale)//x.denominator, scale)
    high = F(-((-x.numerator*scale)//x.denominator), scale)
    def series(y):
        z = (y - 1) / (y + 1)
        s = sum((2 * z ** (2 * j + 1) / (2 * j + 1)
                 for j in range(terms)), F(0))
        tail = 2 * z ** (2 * terms + 1) / ((2 * terms + 1) * (1 - z * z))
        return s, s + tail
    def reduced(y):
        power = 0
        while y >= 2:
            y /= 2
            power += 1
        a,b = series(y)
        c,d = series(F(2))
        return a+power*c,b+power*d
    a,_ = reduced(low)
    _,b = reduced(high)
    return a,b


def gibbs(H, beta):
    E, U = np.linalg.eigh(H)
    logz = float(logsumexp(-beta * E))
    rho = (U * np.exp(-beta * E - logz)) @ U.conj().T
    return -logz / beta, rho, logz


def entropy(rho):
    ev = np.linalg.eigvalsh(rho)
    ev = ev[ev > 1e-16]
    return float(-np.dot(ev, np.log(ev)))


def binary_f(x):
    x = min(1.0, max(0.0, float(x)))
    z = 0.5 * (1 + x) * np.log1p(x)
    if x < 1:
        z += 0.5 * (1 - x) * np.log1p(-x)
    return float(z)


def histogram_bounds(p, C, v, beta):
    offset = min(v)
    u = np.exp(-beta * (v - offset))
    mass, lower = 1 / C, 0.0
    for i in np.argsort(u):
        take = min(mass, p[i])
        mass -= take
        lower += take * u[i]
    assert mass < 1e-12
    return (max(offset, offset - np.log(p @ u) / beta),
            min(max(v), offset - np.log(lower) / beta))


def fractional_extrema(p, C, u0, u1, error=0.0):
    k = len(p)
    if error == 0:
        # q formulation: variables y[0:k], t.
        A = np.zeros((k, k + 1))
        A[:, :k] = np.eye(k)
        A[:, k] = -C * p
        eq = np.zeros((2, k + 1))
        eq[0, :k], eq[1, :k], eq[1, k] = u0, 1, -1
        rhs, cost = [1, 0], np.r_[u1, 0]
    else:
        # w,p formulation: variables y[0:k], x[0:k], t.
        A = np.zeros((3 * k, 2 * k + 1))
        A[:k, :k], A[:k, k:2*k] = np.eye(k), -np.eye(k)
        A[k:2*k, k:2*k], A[k:2*k, -1] = np.eye(k), -np.minimum(1, p + error)
        A[2*k:, k:2*k], A[2*k:, -1] = -np.eye(k), np.maximum(0, p - error)
        eq = np.zeros((3, 2 * k + 1))
        eq[0, :k] = u0
        eq[1, k:2*k], eq[1, -1] = 1, -1
        eq[2, :k], eq[2, -1] = 1, -1 / C
        rhs, cost = [1, 0, 0], np.r_[u1, np.zeros(k + 1)]
    ans = []
    for sign in [1, -1]:
        result = linprog(sign * cost, A_ub=A, b_ub=np.zeros(len(A)),
                         A_eq=eq, b_eq=rhs, bounds=[(0, None)] * len(cost),
                         method="highs")
        assert result.success, result.message
        ans.append(float(sign * result.fun))
    return ans
