def tau_witness(gamma: float) -> float:
    x = max(gamma - 0.2, 0.0) / 0.05
    return 0.020 * x * x / (1 + x * x)


def beta(omega: float, tau: float) -> float:
    return 1 / (1 + (omega * tau) ** 2)
