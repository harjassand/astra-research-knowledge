def transitions(x, k):
    a, b, c = x
    return [(k*a*c, (a, b+1, c-1)), (a*b, (a, b-1, c+1)),
            (a*b, (a-1, b, c)), (b, (a+1, b, c)),
            (c, (a, b, c-1)), (1, (a, b, c+1))]


def gen(fn, x, k):
    old = fn(x)
    return sum((rate*(fn(y)-old) for rate, y in transitions(x, k) if rate), F(0))
