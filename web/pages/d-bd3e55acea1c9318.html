"""Independent original-factor signs and entire real factor2 M-curve domain.

Checks every algebraic identity used in the written inequality proof, without
importing a producer or checker. Raw M-curve identities remain separate.
"""
import hashlib
import json
import time
from pathlib import Path
import sympy as s

BASE=Path(__file__).resolve().parents[3]

def zero(v):assert s.cancel(v)==0,str(v)

def main():
    start=time.monotonic()
    a,b,c,A,E,F,d,S,x,y,z,w=s.symbols('a b c A E F d S x y z w')
    path=BASE/'sources/metric_use10/cycle16/generic_d8_divisor_factors.json'
    if not path.is_file():path=BASE/'results/metric_prior/cycle13/generic_d8_divisor_factors.json'
    fs=[s.sympify(v,locals={'a':a,'b':b,'c':c}) for v in json.loads(path.read_text())['factors']]
    rel=s.Poly(b*c-a*d,a,b,c)
    congruence=lambda v:s.rem(s.Poly(v,a,b,c),rel).is_zero
    V=lambda e,g:(7*d*d-24*d+63)*e+(d*d*(7-16*A)+57*d)*g+(14*d-15)*g*g-A*d*d*(12*d+18)-63*d
    M=lambda e:2*e*(1+d)-(d-3)**2
    largest=[v.subs({a:c,b:b,c:a},simultaneous=True) for v in fs]
    middle=[v.subs({a:b,b:c,c:a},simultaneous=True) for v in fs]
    assert congruence(largest[8]-a*a*V(c*c,b*b).subs(A,a*a))
    W=((14*d*d-16*A*d*d+33*d+63)*S+(14*d-15)*S*S)/2-A*d*d*(26*d+3)-63*d
    assert s.rem(s.Poly((V(E,F)+V(F,E))/2-W.subs(S,E+F),E,F),s.Poly(E*F-A*d*d,E,F)).is_zero
    zero(V(E,F)-V(F,E)-(E-F)*(16*A*d*d-81*d+63-(14*d-15)*(E+F)))
    da=7/(2*A+5);k=1-A+2*A*da
    zero(2-k-(A-1)*(2*A-5)/(2*A+5))
    zero(W.subs({S:k,d:da},simultaneous=True)+(A-1)**2*(120*A**3-668*A*A+1690*A+4395)/(2*(2*A+5)**3))
    zero(16*A*da*da-81*da+65-10*(A-1)*(26*A+121)/(2*A+5)**2)
    assert 4395-668>0
    Tlower=96-s.Rational(98,25)
    assert Tlower/2-2==s.Rational(1101,25)>0
    assert congruence(largest[5]-a*a*M(c*c))
    assert congruence(largest[18]+a*a*M(b*b))
    assert congruence(middle[5]-a*a*M(b*b))
    assert congruence(middle[18]+a*a*M(c*c))
    zero(largest[6]-middle[6]-2*a*(c*c-b*b)*(c*c+b*b+4))
    # In any strong-first/weak-last chart, weak*first/second=x²/d.
    ff=[v.subs({a:x,b:y,c:z},simultaneous=True) for v in fs]
    xyrel=s.Poly(y*z-x*y*y/d,x,y,z)
    # Substitute z=x*y/d (bc=ad under the chart); this parameter-only
    # substitution is for signs of factor ratios, not auxiliary-variable inversion.
    actual={j:s.cancel(ff[j].subs(z,x*y/d)) for j in (0,1,2,3,19,22,23)}
    expected={0:y*(14*x**4/d+7*x*x+6*x*x/d-27),
              1:(x*y*y/d)*(d-6+x*x*(2+3/d)),
              2:x*(y*y*(2*x*x+5)/d-7),
              3:x*(2*x*x-7+5*y*y/d),
              19:y*(x+2*x*x/d-3),
              22:y*(x+3-2*x*x/d),23:y*(x*x/d-1)}
    for j in expected:zero(actual[j]-expected[j])
    Tk=(2*x*x-9)*w*w+2*x*(x*x+3)*w-x*x
    zero(ff[4]-z*z*Tk.subs(w,y/z))
    zero(Tk.subs(w,x)+4*x*x*(1-x*x))
    zero(s.diff(Tk,w).subs(w,x)-6*x*(x*x-2))
    # Sign inequalities are written in the gate; these identities make each
    # original-index assignment checkable and retain all remaining factors.
    U=(3*A+4)**2/((A+6)*(2*A+5));C=A*da*da/U
    h=9*A**3+87*A*A+114*A-112
    det=1-A-U-C+2*A*da
    zero(U-A+2*(A-1)*(A*A+5*A+8)/((A+6)*(2*A+5)))
    zero(C-A+A*(A-1)*(18*A*A+111*A+214)/((2*A+5)*(3*A+4)**2))
    zero(U-1-7*(A-1)*(A+2)/((A+6)*(2*A+5)))
    zero(C-1+2*(A-1)**2*(9*A+40)/((2*A+5)*(3*A+4)**2))
    zero(det+2*(A-1)**2*h/((A+6)*(2*A+5)*(3*A+4)**2))
    assert h.subs(A,0)<0<h.subs(A,s.Rational(2,3))
    assert s.Poly(h,A).count_roots(0,1)==1
    assert s.Rational(2,3)<s.Rational(25,36)
    # Both2x2 minors and determinant give exactly the claimed real PD domain.
    P3=162*A**3+599*A*A+619*A-8
    Cstar=49*A/(6*A+1)**2
    zero(Cstar-U+2*(A-1)*P3/((A+6)*(2*A+5)*(6*A+1)**2))
    assert s.Poly(P3,A).is_irreducible
    assert P3.subs(A,s.Rational(1,80))<0<P3.subs(A,s.Rational(13,1000))
    # Cubic point lies strictly inside the full curve PD domain.
    assert h.subs(A,s.Rational(13,1000))<0
    result={'status':'PASS independent original-factor sign/domain reductions',
        'safe_largest_indices':sorted(set(range(28))-{5,6,18}),
        'safe_middle_indices':sorted(set(range(28))-{5,6,8,18}),
        'exact_ratio_factor_identities':{str(j):str(expected[j]) for j in expected},
        'ratio_quadratic':str(Tk),'giant_Wk':str(s.factor(W.subs({S:k,d:da},simultaneous=True))),
        'W_derivative_lower_bound':'1101/25','difference_base_plus2':str(s.factor(16*A*da*da-81*da+65)),
        'U':str(U),'C':str(C),'determinant':str(s.factor(det)),
        'full_real_PD_domain':'0<A<rho,unique positive root9rho3+87rho2+114rho-112=0;rho<2/3;0<r<5/6',
        'old_irreducible_cubic_point_in_M_curve':True,
        'raw_D8_membership_proved_here':False,'source_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
        'seconds':time.monotonic()-start}
    out=BASE/'results/metric_use10/cycle16/p2_domain_independent_checks.json'
    out.write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps({'status':result['status'],'seconds':result['seconds'],'output':str(out)}))
