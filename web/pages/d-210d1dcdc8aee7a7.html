def ratio(c,i,k):

def threshold(c,k,eta):
    target=max(F(1),*(ratio(c,k,j) for j in range(3) if j!=k))/eta
    lo=-1;hi=sum(c)+s.ceil_log9_integer(eta.denominator)+1

        else:lo=mid
    return hi

def relative_T(c,k,p):

def accept(c,k,p):

def propose(n,k,rng):

    return tuple(c)

def repair(p,observe,rng,ledger):

def convert(p,observe,rng,ledger):
