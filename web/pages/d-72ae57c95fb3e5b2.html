#!/usr/bin/env python3
"""Checks exact formulas numerically; never a general proof certificate."""
from pathlib import Path
import json
import numpy as np
from explore_quadratic import evaluate, hermitian_basis

HERE=Path(__file__).resolve().parent
sx=np.array([[0,1],[1,0]],complex)
sy=np.array([[0,-1j],[1j,0]],complex)
sz=np.diag([1.,-1.]).astype(complex)


def k(x):
    return x*np.sinh(x)-np.cosh(x)+1


def actual(alpha,beta,c,B):
    sigma=np.array([(1+np.tanh(alpha))/2,(1-np.tanh(alpha))/2])
    r=np.tanh(beta)
    rho=(np.eye(2)+r*(np.sqrt(1-c*c)*sx+c*sz))/2
    s=np.sqrt(sigma)
    d=np.power(sigma,.25)
    V=2*(B@np.diag(s)@B)/(s[:,None]+s[None,:])
    def H(X):
        return (V@X+X@V)/2-B@X@B
    vals,U=np.linalg.eigh(rho)
    root=(U*np.sqrt(vals))@U.conj().T
    logs=(U*np.log(vals))@U.conj().T-np.diag(np.log(sigma))
    Lrho=H(rho/np.outer(d,d))*np.outer(d,d)
    assert abs(np.trace(Lrho))<1e-8
    assert np.linalg.norm(H(np.diag(s)))<1e-8
    J=np.trace(Lrho@logs).real
    E=np.trace(root@H(root)).real
    return J,E


def formulas(alpha,beta,c,a):
    C,S=np.cosh(alpha),np.sinh(alpha)
    A=np.cosh(alpha/2)
    r=np.tanh(beta)
    h=1-1/np.cosh(beta)
    G=beta*r
    K=G-h
    E0=C-1+h*c*c-S*r*c
    E1=2*np.sqrt(1-c*c)*(np.tanh(alpha/2)*r-h*c)
    E2=h*(1-c*c)
    J0=G*((C-1)+(C+1)*c*c)-2*c*(alpha*r*C+beta*S)+2*alpha*S
    J1=2*np.sqrt(1-c*c)*((beta*S+alpha*r)/A-2*A*G*c)
    J2=2*G*(1-c*c)
    U=(beta*S+alpha*r)/A-2*np.tanh(alpha/2)*r
    W=2*(h-A*G)
    F0,Jminus=J0-2*E0, J1-2*E1
    assert abs(Jminus-2*np.sqrt(1-c*c)*(U+W*c))<1e-8
    concave=F0-(U+W*c)**2/(2*K)
    concave_coefficient=((C+1)*G-2*h)-W*W/(2*K)
    predicted=-2*G*h*(A-1)**2/K
    assert abs(concave_coefficient-predicted)<1e-8
    return J0+a*J1+a*a*J2,E0+a*E1+a*a*E2,concave


def endpoint(alpha,beta,sign):
    b=sign*beta
    delta=b-alpha
    T=b*np.sinh(delta)+delta*np.sinh(b)-4*np.sinh(b/2)*np.sinh(delta/2)
    numerator=4*np.cosh(alpha/2)**2*k(beta)*k(delta)-T*T
    denominator=2*np.cosh(alpha/2)**2*np.cosh(beta)*k(beta)
    return numerator/denominator


def main():
    rng=np.random.default_rng(6006312)
    count=0
    worst=0
    for i in range(1000):
        alpha=rng.uniform(0,5)
        beta=rng.uniform(.05,5)

        err=max(abs(J-J0),abs(E-E0))
        assert err<1e-7*max(1,abs(J),abs(E))

        B=1.3*sx+.7*sy-1.1*sz+2.3*np.eye(2)

    print(json.dumps(report,indent=2))
