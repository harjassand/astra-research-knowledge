def sample(seed):
    s0,s1,s2,s3,s4,s5=seed
    def local(v,mask,own,other):
        if mask==N:
            first=own
            second=tuple(other[j] if j==v else 0 for j in range(2))

            first=second=Z

    nt=(s1==s4==N and s3==s5==N)
