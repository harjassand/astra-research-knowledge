import itertools

def expected_length(q,mass):
    indices = [i for i in range(3) if mass[i]]
    if not indices:return 8
    if len(indices)==1:
        i=indices[0];j,k=[a for a in range(3) if a!=i]
        if q[i,j]==0 and q[i,k]==0:return 8
        if q[i,j]==0 or q[i,k]==0:return 0
        return 4 if q[k,k]*q[i,j]**2==q[j,j]*q[i,k]**2 else 0
    if len(indices)==2:
        i,j=indices;k=next(a for a in range(3) if a not in indices)
        off=[q[i,j],q[i,k],q[j,k]]
        if all(a==0 for a in off):return 8
        if any(a==0 for a in off):return 0
        r=q[i,k]/q[j,k];t=q[i,j]/q[j,k]
        return 2 if q[i,i]==q[j,j]*r*r and q[i,i]+q[i,j]*r==2*q[k,k]*t*t else 0
    off=[q[0,1],q[0,2],q[1,2]]
    if all(a==0 for a in off):return 8
    if any(a==0 for a in off):return 0
    r1=q[1,2]/q[0,2];r3=q[0,1]/q[0,2]
    return 2 if q[0,0]*r1*r1==q[1,1]==q[2,2]*r3*r3 else 0


def standard_exponents(gb,u):
    lead=[g.LM(order=gb.order).exponents for g in gb.polys]
    if (0,0,0) in lead:return []
    def allowed(a):return not any(all(a[i]>=lm[i] for i in range(3)) for lm in lead)
    basis={(0,0,0)};pending=[(0,0,0)]
    while pending:
        a=pending.pop()
        for i in range(3):
            nxt=tuple(a[k]+(i==k) for k in range(3))
            if nxt not in basis and allowed(nxt):
                basis.add(nxt);pending.append(nxt)
                assert len(basis)<=64,"unexpected unbounded branch quotient"
    return sorted(basis,key=lambda a:(sum(a),a))


def branch(q,mass):
    u=s.symbols('u0:3')
    eq=[]
    for i in range(3):
        if mass[i]:
            eq.extend(q[i,i]*u[i]**2+q[i,j]*u[i]*u[j]-2 for j in range(3) if j!=i)
        else:eq.append(q[i,i]*u[i]**2-1)
    gb=s.groebner(eq,*u,order='grlex',domain=s.QQ)
    exponents=standard_exponents(gb,u)
    assert len(exponents)==expected_length(q,mass),(q,mass,exponents)
    typ=''.join('A' if a else 'B' for a in mass)
    if not exponents:return {'branch':typ,'length':0,'status':'PASS'}
    basis=[s.prod(u[i]**a[i] for i in range(3)) for a in exponents]
    def matrix(expr):
        columns=[]
        for mon in basis:
            _,rem=gb.reduce(s.expand(mon*expr))
            poly=s.Poly(rem,*u)
            columns.append(s.Matrix([poly.coeff_monomial(a) for a in basis]))
        return s.Matrix.hstack(*columns)
    ms=[matrix(a) for a in basis]
    gram=s.Matrix(len(basis),len(basis),lambda i,j:s.trace(ms[i]*ms[j]))
    gram_minors=[gram[:i,:i].det() for i in range(1,len(basis)+1)]
    assert all(a>0 for a in gram_minors)
    # Positive-definite trace pairing verifies reducedness and total reality
    # for this finite quotient. This replay is not the universal proof.
    ss=[]
    for i in range(3):
        if mass[i]:
            j=(i+1)%3
            ss.append((q[i,i]*u[i]+q[i,j]*u[j])/2)
        else:ss.append(q[i,i]*u[i])
    b=3*ss[0]*ss[1]*ss[2]-q[1,2]*ss[0]-q[0,2]*ss[1]-q[0,1]*ss[2]
    mb=matrix(b)
    assert mb.det()!=0
    md=matrix(b*s.prod(u))
    weighted=s.Matrix(len(basis),len(basis),lambda i,j:s.trace(ms[i]*ms[j]*md))
    weighted_minors=[weighted[:i,:i].det() for i in range(1,len(basis)+1)]
    assert all(a>0 for a in weighted_minors)
    return {'branch':typ,'length':len(basis),'basis':[str(a) for a in basis],
            'trace_Gram_principal_minors':[str(a) for a in gram_minors],
            'B_multiplication_determinant':str(mb.det()),
            'normalized_B_weighted_Gram_principal_minors':[str(a) for a in weighted_minors],
            'status':'PASS'}
