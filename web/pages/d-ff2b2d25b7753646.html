"""Exact polynomial expansion of the CPV natural-gauge potential.

This is exact algebraic verification of the natural-gauge identity, not
an independent proof of the source-to-physical-operator map or the spectral
bound. Variables are a,b,R,delta,alpha,j. The main polynomial first writes
C3=alpha*C2; the exact central-relation remainder is checked separately.
"""
from fractions import Fraction as F

NVAR = 6


class P:
    def __init__(self, v=0):
        self.c = dict(v) if isinstance(v, dict) else {(0,) * NVAR: F(v)}
        self.c = {k: v for k, v in self.c.items() if v}
    def __add__(self, y):
        y = y if isinstance(y, P) else P(y)
        c = dict(self.c)
        for k, v in y.c.items():
            c[k] = c.get(k, F(0)) + v
        return P(c)
    __radd__ = __add__
    def __neg__(self):
        return P({k: -v for k, v in self.c.items()})
    def __sub__(self, y):
        return self + -(y if isinstance(y, P) else P(y))
    def __rsub__(self, y):
        return P(y) - self
    def __mul__(self, y):
        y = y if isinstance(y, P) else P(y)
        c = {}
        for k, v in self.c.items():
            for l, w in y.c.items():
                key = tuple(a + b for a, b in zip(k, l))
                c[key] = c.get(key, F(0)) + v * w
        return P(c)
    __rmul__ = __mul__
    def __truediv__(self, y):
        return self * (1 / F(y))
    def __pow__(self, n):
        result = P(1)
        for _ in range(n):
            result = result * self
        return result
    def __eq__(self, y):
        return self.c == (y if isinstance(y, P) else P(y)).c
    def coefficient(self, variable, exponent):
        c = {}
        for k, v in self.c.items():
            if k[variable] == exponent:
                key = list(k)
                key[variable] = 0
                c[tuple(key)] = v
        return P(c)
    def evaluate(self, values):
        return sum(v * prod(F(x) ** n for x, n in zip(values, k))
                   for k, v in self.c.items())
    def description(self):
        names = ['a', 'b', 'R', 'delta', 'alpha', 'j']
        return ' + '.join(str(v) + '*' + '*'.join(name + (f'^{n}' if n != 1 else '')
                     for name, n in zip(names, k) if n)
                     for k, v in sorted(self.c.items(), reverse=True))


def prod(xs):
    v = 1
    for x in xs:
        v *= x
    return v


def var(i):
    k = [0] * NVAR
    k[i] = 1
    return P({tuple(k): F(1)})


def potential_polynomial():
    a, b, R, d, z, j = [var(i) for i in range(NVAR)]
    n, Q = a + b, R + d
    c2 = (a * a + a * b + b * b + 3 * n) / 3
    c = R * R + R * d + d * d + 2 * R + d
    xs = [j + a + F(1, 2) - Q, j + F(1, 2), j + F(1, 2) - d,
          j - Q - F(1, 2), j - R - F(1, 2), j - b - F(1, 2)]
    E = {k: sum(x ** k for x in xs) for k in range(1, 5)}
    L = n + Q + 3
    source_x = -(F(7, 2) * E[1] ** 3 - 18 * E[1] * E[2] + 18 * E[3]) / 108
    source_x -= E[1] * (L * L + 2) / 24
    source_y = (F(5, 2) * E[1] ** 4 + 32 * E[1] * E[3]
                + 6 * (E[2] ** 2 - 3 * E[1] ** 2 * E[2] - 4 * E[4])
                + 6 * E[2] * (L * L + 2) - 3 * E[1] ** 2 * (L * L - 2)
                - F(3, 2) * L ** 4 + 6 * L * L - 36) / 288
    scalar = -c * c / 12 + (c2 / 3 + F(1, 4) + z * z) * c - F(4, 3) * z * z * c2
    downward = (a - Q + j) * j * (j - d) * ((a + 2 * b) / 3 + 2 * z + Q + 2 - j)
    upward = (Q - j) * (R - j) * (b - j) * ((2 * a + b) / 3 - 2 * z + j + 2)
    return -source_y - 2 * z * source_x + scalar - downward - upward


def main():
    polynomial = potential_polynomial()
    a, b, R, d, alpha, j = [var(i) for i in range(NVAR)]
    N = a + b
    C2 = (a * a + a * b + b * b + 3 * N) / 3
    C3 = (a - b) * (2 * a + b + 3) * (a + 2 * b + 3) / 18
    v = (2 * a + b + 3) / 3 - 2 * alpha
    z, w = (v - b) / 2, a + 1 - v
    L1 = (d + 1) * ((N + 2) * w - d * (d + 2) - R * (2 * z + d + 1))
    L0 = (R * R * z * (z + d) + R * d * ((z + b) ** 2 + v)
          + R * d * d * z + R * (2 * z * z + b * (v + 1))
          + d * (d + 1) * (z + b) * (z + b + 1))
    remainder = F(4, 3) * alpha * (C3 - alpha * C2)
    assert polynomial.coefficient(5, 2) == (d + 1) * (d + 2)
    assert polynomial.coefficient(5, 3) == 0
    assert polynomial.coefficient(5, 4) == 0
    assert polynomial.coefficient(5, 1) == L1
    assert polynomial.coefficient(5, 0) - L0 == remainder
    assert polynomial - remainder == (d + 1) * (d + 2) * j * j + L1 * j + L0
    # The rational identity v=(2a+b+3)b(b+2)/(3 C2) uses alpha=C3/C2.
    assert (2 * a + b + 3) * C2 - 6 * C3 == (2 * a + b + 3) * b * (b + 2)
    print('PASS: exact j^3/j^4 cancellation, j^2 coefficient, factored L1/L0,')
    print('central-relation remainder, full potential identity, and rational v identity.')


if __name__ == '__main__':
    main()
