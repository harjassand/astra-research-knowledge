def equal(a, b, label):

def positive(value, label):

S = s.diag(1, s.sqrt(2)) / 5 ** Q(1, 4)


def LH(a):
    return (Qop * a + a * Qop) / 2 - V.T * a * V


def LS(a):
    return (Qop * a + a * Qop) / 2 - V * a * V.T


def L(a):
    return s.simplify(S.inv() * LS(S * a * S) * S.inv())
