def star(q,bits,p):
    h=2**bits;columns=[(h+2*i+1,h//2+i+1) for i in range(q)]
    return {'dimensions':[1]+[2]*q,'p':p,'arrows':[{'tail':0,'head':i+1,'matrix':[[a],[b]]}
             for i,(a,b) in enumerate(columns)],'modular_bases':[[[1]]]+[[[a%p,b%p]] for a,b in columns],
            'weight':[q+1]+[-1]*q,'support_upper':1,'height_bound':2**(bits+q+2),
            'max_precision_exponent':512,'elementary_support_proof':'nonzero source forces every image line; score<=1, attained uniquely by source plus all image lines; each 2b-a=1 ensures every prime nonzero'}


def main():
    start=time.perf_counter();checks=[];reports=[]
    def check(label,value):
        checks.append({'label':label,'pass':bool(value)})
        if not value:raise AssertionError(label)
    inputs={
      'modular_chart_avoids_Q_RREF_denominator':{'dimensions':[1,2],'p':5,
          'arrows':[{'tail':0,'head':1,'matrix':[[5],[1]]}],
          'modular_bases':[[[1]],[[0,1]]],'weight':[2,-1],'support_upper':1,
          'height_bound':5,'max_precision_exponent':8},
      'nonlinear_closure_Newton':{'dimensions':[1,2,2],'p':5,
          'arrows':[{'tail':0,'head':1,'matrix':[[1],[2]]},
                    {'tail':1,'head':2,'matrix':[[1,1],[0,1]]}],
          'modular_bases':[[[1]],[[1,2]],[[1,4]]],'weight':[3,-1,-1],
          'support_upper':1,'height_bound':3,'max_precision_exponent':8},
      'bad_prime_omitted_row_stat':{'dimensions':[2,1],'p':5,
          'arrows':[{'tail':0,'head':1,'matrix':[[0,1]]},
                    {'tail':0,'head':1,'matrix':[[5,-1]]}],
          'modular_bases':[[[1,0]],[]],'weight':[1,-2],'support_upper':1,
          'max_precision_exponent':8},
      'bad_prime_zero_variable':{'dimensions':[1,1],'p':5,
          'arrows':[{'tail':0,'head':1,'matrix':[[5]]}],
          'modular_bases':[[[1]],[]],'weight':[1,-1],'support_upper':1,
          'max_precision_exponent':8},
      'noncanonical_flat_point':{'dimensions':[2],'p':5,'arrows':[],
          'modular_bases':[[[1,0]]],'weight':[0],'support_upper':0,
          'max_precision_exponent':8},
      'high_height_star_16':star(16,256,101),
      'declared_height_budget_UNKNOWN':{**star(4,128,5),'max_precision_exponent':8},
    }
    INPUTS.write_text(json.dumps(inputs,indent=2)+'\n')
    for name,data in inputs.items():
        c=chart(data);r=lift(data);r['name']=name;reports.append(r)
        check(name+'_quadratic_chart',all(sum(m)<=2 for eq in c['equations'] for m in eq.terms))
        if name in ('bad_prime_omitted_row_stat','bad_prime_zero_variable'):
            check(name+'_REJECTED',r['status']=='REJECTED')
            check(name+'_surviving_full_residual',r['precision_trace'][-1]['all_residuals_zero'] is False)
        elif name in ('noncanonical_flat_point','declared_height_budget_UNKNOWN'):
            check(name+'_UNKNOWN',r['status']=='UNKNOWN')
        else:
            check(name+'_certified',r['status']=='CERTIFIED_CONDITIONAL')
            bases=[[[Q(x) for x in row] for row in b] for b in r['rational_bases']]
            check(name+'_original_arrows',original_closure(data,bases))
            check(name+'_supplied_support',Q(r['score'])==Q(data['support_upper']))
    bad=chart(inputs['bad_prime_omitted_row_stat'])
    check('omitted_row_literal_equations',bad['equations'][0].terms=={(1,):1}
          and bad['equations'][1].terms=={(0,):5,(1,):-1})
    nonlinear=chart(inputs['nonlinear_closure_Newton'])
    check('literal_nonlinear_cross_term',any(m==(1,1) for eq in nonlinear['equations'] for m in eq.terms))
    chart_control=next(r for r in reports if r['name']=='modular_chart_avoids_Q_RREF_denominator')
    check('modular_second_pivot',chart_control['chart_pivots']==[(0,),(1,)])
    check('target_coordinate_5',chart_control['rational_coordinates']==['5'])
    high=next(r for r in reports if r['name']=='high_height_star_16')
    check('arbitrary_dimension_33',sum(inputs['high_height_star_16']['dimensions'])==33)
    check('high_height_precision_charged',high['max_modulus_bits']>500)
    check('high_height_all_residuals',all(t['all_residuals_zero'] for t in high['precision_trace']))
