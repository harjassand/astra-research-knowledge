"""Exact affine stationary-cofactor construction and finite EP checks."""
from pathlib import Path
import json,math
import sympy as s
import numpy as np

root=Path(__file__).resolve().parents[2]
records=[]
for n in [3,4,5,6]:
    q=s.zeros(n)
    for i in range(n-1):
        q[i,i+1]=s.Rational(3+i,3)
        q[i+1,i]=s.Rational(8+(i%2),4)
    u,v=0,n-1;a=s.Rational(3,2);q[v,u]=a
    for i in range(n):q[i,i]=-sum(q[i,j] for j in range(n) if j!=i)
    perturb=s.zeros(n);perturb[u,v]=1;perturb[u,u]=-1
    b=s.symbols('b',real=True)
    taus=[(-q-b*perturb).minor_submatrix(i,i).det() for i in range(n)]
    Ai=[z.subs(b,0) for z in taus];Bi=[s.expand(z).coeff(b) for z in taus]
    assert all(s.cancel(taus[i]-Ai[i]-b*Bi[i])==0 for i in range(n))
    assert all(x>0 for x in Ai) and all(x>=0 for x in Bi)
    A=sum(Ai);Bb=sum(Bi)
    pi=s.Matrix(1,n,[(Ai[i]+b*Bi[i])/(A+b*Bb) for i in range(n)])
    assert all(s.cancel(z)==0 for z in pi*(q+b*perturb))
    Ki=[abs(A*Bi[i]-Ai[i]*Bb)/A**2 for i in range(n)]
    Ci=[];ci=[]
    for i in range(n):
        cb=s.Rational(0);ce=s.Rational(0)
        for j in range(n):
            if i==j or {i,j}=={u,v} or q[i,j]==0:continue
            assert q[j,i]>0
            ratio=q[i,j]/q[j,i]
            cb+=q[i,j]*(max(ratio,1/ratio)-1)
            ce+=q[i,j]*s.log(ratio)
        Ci.append(cb);ci.append(ce)
    H=a*Ki[v]+1+Ki[u];La=max(a,1/a)-1
    Cep=s.factor(sum(Ki[i]*Ci[i] for i in range(n))+H*(La+1)+(a+1))
    J=a*pi[v]-b*pi[u]
    sigma=sum(pi[i]*ci[i] for i in range(n))+J*s.log(a/b)
    ds=s.diff(sigma,b)
    f=s.lambdify(b,b*ds,'numpy')
    grid=np.logspace(-10,0,500)
    maximum=float(np.max(np.abs(f(grid))))
    assert maximum<=float(Cep)*(1+1e-10)
    records.append({'n':n,'A_i':[str(x) for x in Ai],'B_i':[str(x) for x in Bi],
