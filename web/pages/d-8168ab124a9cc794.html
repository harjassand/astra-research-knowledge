def entropy(r):
    p=np.linalg.eigvalsh((r+r.T)/2)
    p=p[p>1e-14]
    return float(-np.dot(p,np.log(p)))

def gibbs(h,beta):
    e,u=eigh(h)
    w=np.exp(-beta*(e-e[0])); z=w.sum()
    return (u*(w/z))@u.T,float(np.log(z)-beta*e[0])
