def plus(P, R):
    if P is None:
        return R
    if R is None:
        return P
    x, y = P
    u, v = R
    if x == u and y == -v:
        return None
    slope = (v-y)/(u-x) if x != u else (3*x*x-25)/(2*y)
    xx = slope*slope-x-u
    return (xx, slope*(x-xx)-y)


def mul(n, P):
    answer = None
    for bit in bin(n)[2:]:
        answer = plus(answer, answer)
        if bit == "1":
            answer = plus(answer, P)
    return answer


def hx(P):
    return 1 if P is None else max(abs(P[0].numerator), P[0].denominator)


def unpack(d):
    if d["identity"]:
        return None
    return (Q(d["x_num"], d["x_den"]), Q(d["y_num"], d["y_den"]))


def pack(P):
    if P is None:
        return {"identity": True}
    return {"identity": False, "x_num": P[0].numerator,
            "x_den": P[0].denominator, "y_num": P[1].numerator,
            "y_den": P[1].denominator}


def padd(A, B):
    C = dict(A)
    for i, v in B.items():
        C[i] = C.get(i, 0)+v
    return {i: v for i, v in C.items() if v}


def pmul(A, B):
    C = {}
    for i, v in A.items():
        for j, w in B.items():
            C[i+j] = C.get(i+j, 0)+v*w
    return {i: v for i, v in C.items() if v}
