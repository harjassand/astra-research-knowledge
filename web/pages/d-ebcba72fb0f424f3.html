"""Independent finite symbolic check of the reconstructed response ledger.

The mathematical proof is in the owned gate. This checks ODE solutions,
moving-endpoint expressions and mean boundary fluxes using only explicit
Fourier moments. It does not run the author's script or solve the actual PDE.
"""
import json
from pathlib import Path
import sympy as sp

u, d, h, z, v = sp.symbols("u d h z v", positive=True)
s, c = sp.sin(d), sp.cos(d)
k = (c - d/s)/2
H = s*(c-k)/(u**2*c**3)
L = lambda f: sp.diff(f,h,2) + sp.diff(f,z,2) + f
simp = lambda f: sp.simplify(sp.trigsimp(sp.expand_trig(f)))
ru = (h*sp.cos(h)-d*sp.cot(d)*sp.sin(h))/2
rv = (h**2-d**2)*sp.cos(z)/2
du = -k*sp.cos(z)/(u*c)
dv = -d*sp.sin(h)/(2*u*c*s) - d*sp.sinh(sp.sqrt(3)*h)*sp.cos(2*z)/(2*u*c*sp.sinh(sp.sqrt(3)*d))
ddu = H*sp.sin(h)/(2*s) + H*sp.sinh(sp.sqrt(3)*h)*sp.cos(2*z)/(2*sp.sinh(sp.sqrt(3)*d))
checks = []
def check(name, expression):
    assert simp(expression) == 0, name
    checks.append(name)

check("unperturbed_u_ODE", L(ru)+sp.sin(h))
check("unperturbed_v_ODE", L(rv)-sp.cos(z))
for name, f in [("u_shape1",du),("v_shape1",dv),("u_shape2",ddu)]:
    check(name+"_ODE", L(f))
for sign in (-1,1):
    a = sign*d
    hp = sp.cos(z)/(u*c)
    hpp = sign*s*sp.cos(z)**2/(u**2*c**3)
    check(f"u_trace0_{sign}",ru.subs(h,a))
    check(f"v_trace0_{sign}",rv.subs(h,a))
    check(f"u_trace1_{sign}",(du+sp.diff(ru,h)*hp).subs(h,a))
    check(f"v_trace1_{sign}",(dv+sp.diff(rv,h)*hp).subs(h,a))
    check(f"u_trace2_{sign}",(ddu+2*sp.diff(du,h)*hp+sp.diff(ru,h,2)*hp**2+sp.diff(ru,h)*hpp).subs(h,a))

# First two even outside antiderivative integrals over the full two-strip
# complement. Endpoint movement is kept separately, never folded into bulk.
I2 = 4*c/s
I4 = 4*(c/s+c**3/(3*s**3))
B = 2*(u**2-2*u*v*sp.sin(h)*sp.cos(z)+v**2)/(-u*sp.sin(h)+v*sp.cos(z))**3
B1 = simp(sp.diff(B,v).subs(v,0))
check("B_first_variation",B1-sp.cos(z)/u**2*(4/sp.sin(h)**2-6/sp.sin(h)**4))
B2 = sp.diff(B,v,2).subs(v,0)
mean_cos_moments = lambda f: simp(sp.integrate(sp.expand_trig(f),(z,0,2*sp.pi))/(2*sp.pi))
check("u_bulk_second_variation",mean_cos_moments(-sp.sin(h)*B2)-(12/sp.sin(h)**4-8/sp.sin(h)**2)/u**3)

v_bulk = -(4*I2-6*I4)/(4*sp.pi*u**2)
v_endpoints = -2/(sp.pi*u**2*c*s**3)
v_regular = simp(v_bulk+v_endpoints)
check("v_regular_row",v_regular+2/(sp.pi*u**2*c*s))

u_bulk = -(12*I4-8*I2)/(2*sp.pi*u**3)
# Three moving-endpoint terms: 2F_v h', F_h(h')², and F h''.
u_endpoints = ((-8/s+12/s**3)/c - 4/(c*s**3) + 2/(s*c**3))/(sp.pi*u**3)
u_regular = simp(u_bulk+u_endpoints)
check("u_regular_row",u_regular-2/(sp.pi*u**3*c**3*s))

# Boundary fluxes include r_h(1+h_z²). Their second-variation mean
# has the four terms below; the last is the surface factor.
mean_u_flux2 = H*c/(2*s) + (-c-k)/(2*u**2*c**2) - s**2/(2*u**2*c**3) + k/(u**2*c**2)
check("mean_u_boundary_flux2",mean_u_flux2+s**2/(2*u**2*c**3))
u_defect = (4*mean_u_flux2/(u*s**3)+8/(u**3*c*s))/sp.pi
check("u_defect_row",u_defect-(8/c-2/c**3)/(sp.pi*u**3*s))

mean_v_flux1 = -d/(2*u*s)+1/(2*u*c)
v_defect = (4*mean_v_flux1/(u*s**3)-4*d/(u**2*s**2))/sp.pi
check("v_defect_row",v_defect-(-2*d/s**4+2/(c*s**3)-4*d/s**2)/(sp.pi*u**2))

Cu2 = simp(u_regular+u_defect)
Cv1 = simp(v_regular+v_defect)
check("total_Cu_vv",Cu2-8/(sp.pi*u**3*c*s))
check("total_Cv_v",Cv1-2/(sp.pi*u**2)*(c/s**3-d/s**4-2*d/s**2))
# tau is fixed: d=arcsin(tau/u), d_u=−tan(d)/u.
curl = simp(Cu2-sp.diff(Cv1,u)+sp.tan(d)/u*sp.diff(Cv1,d))
target = 2*(2*d-sp.sin(2*d))/(sp.pi*u**3*s**4)
check("fixed_tau_curl",curl-target)
out = {"status":"PASS", "scope":"Independent symbolic transcription check of explicit shape/endpoint/full-defect algebra; not a global PDE or spectral computation", "checks":checks, "Cu_vv":str(Cu2),"Cv_v":str(Cv1),"curl_coefficient_per_v":str(target),"positive_coefficient_reason":"2d−sin(2d)>0 for d>0; derivative2−2cos(2d)>0 on0<d<1/10"}
root = Path(__file__).resolve().parents[3]
(root/'results/pde_gate10/cycle11/response_curl_reconstruction.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps({"status":out['status'],"checks":len(checks),"scope":out['scope']},indent=2))
