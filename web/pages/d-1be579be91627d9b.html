from fractions import Fraction as F
from math import isqrt

def ceil_sqrt_fraction(x: F) -> int:
    assert x >= 0
    n, d = x.numerator, x.denominator
    j = isqrt(n // d)
    while j * j * d < n:
        j += 1
    while j > 0 and (j - 1) * (j - 1) * d >= n:
        j -= 1
    return j
