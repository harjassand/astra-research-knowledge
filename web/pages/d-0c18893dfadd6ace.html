def ceil_log2_rational(v):
    n,d=int(v.numerator),int(v.denominator)
    if n<=d:return 0
    k=n.bit_length()-d.bit_length()
    if (1<<k)*d<n:k+=1
    return k


def cayley(x,u):
    assert not sum(x[a]*u[a] for a in range(3))
    s=sum(v*v for v in u)/4
    y=tuple(((1-s)*x[a]+u[a])/(1+s) for a in range(3))
    assert sum(v*v for v in y)==1
    return y


def means(polys,points):
    return [sum((p_evaluate(p,x) for x in points),Q(0))/len(points) for p in polys]


def chart_reset(point,bits):
    x,y,z=point; side=1 if z>=0 else -1
    a=x/(1+side*z); b=y/(1+side*z)
    aa,bb=round_dyadic(a,bits),round_dyadic(b,bits)
    s=aa*aa+bb*bb
    result=(2*aa/(1+s),2*bb/(1+s),side*(1-s)/(1+s))
    assert sum(v*v for v in result)==1
    return result,dict(side=side,input_parameters=[str(a),str(b)],
                       output_parameters=[str(aa),str(bb)],mesh_bits=bits)
