CHARGES = {"candidate_tests": 0, "trial_divisions": 0, "order_power_tests": 0}


def check(name, condition, evidence=None):
    if not condition:
        raise AssertionError((name, evidence))
    CHECKS.append({"name": name, "pass": True, "evidence": evidence})


def floor_log2(n):
    assert n >= 1
    return n.bit_length() - 1


def ceil_log2(n):
    assert n >= 1
    return (n - 1).bit_length()


def ceil_div(a, b):
    return (a + b - 1) // b


def charge_delta(before):
    return {key: CHARGES[key] - before[key] for key in CHARGES}


def is_prime(n, charged=False):
    if charged:
        CHARGES["candidate_tests"] += 1
    if n < 2:
        return False
    for t in range(2, math.isqrt(n) + 1):
        if charged:
            CHARGES["trial_divisions"] += 1
        if n % t == 0:
            return False
    return True


def primes_upto(bound):
    return [p for p in range(2, bound + 1) if is_prime(p)]


def rank_mod(matrix, p):
    rows = [[int(v) % p for v in row] for row in matrix.tolist()]
    m, n = len(rows), len(rows[0])
    rank = 0
    for c in range(n):
        pivot = next((r for r in range(rank, m) if rows[r][c]), None)
        if pivot is None:
            continue
        rows[rank], rows[pivot] = rows[pivot], rows[rank]
        inv = pow(rows[rank][c], -1, p)
        rows[rank] = [(v * inv) % p for v in rows[rank]]
        for r in range(m):
            if r != rank and rows[r][c]:
                a = rows[r][c]
                rows[r] = [(v - a * u) % p for v, u in zip(rows[r], rows[rank])]
        rank += 1
        if rank == m:
            break
    return rank


def primitive_roots(q, p):
    return [a for a in range(2, p) if pow(a, q, p) == 1]


def evaluate_poly(poly, a, p):
    return sum(int(c) * pow(a, int(e[0]), p) for e, c in sp.Poly(poly, Z).terms()) % p


def quotient_matrix(q, poly):
    phi = sum(Z**j for j in range(q))
    cols = []
    for j in range(q - 1):
        rem = sp.Poly(sp.rem(poly * Z**j, phi, Z), Z)
        cols.append([rem.nth(i) for i in range(q - 1)])
    return sp.Matrix(q - 1, q - 1, lambda i, j: cols[j][i]), phi


def root_budget_case(q, poly):
    matrix, phi = quotient_matrix(q, poly)
    resultant = int(sp.resultant(phi, poly, Z))
    norm = sum(abs(int(c)) for c in sp.Poly(poly, Z).all_coeffs())
    check(f"q{q}:{poly}:integer determinant", int(matrix.det()) == resultant)
    check(f"q{q}:{poly}:nonzero norm and triangle bound",
          0 < abs(resultant) <= norm ** (q - 1), {"R": resultant, "W": norm})
    pool = [p for p in primes_upto(1000) if p % q == 1]
    root_rows = []
    capacity_numerator = 0
    total_weight = 0
    product = 1
    for p in pool:
        roots = primitive_roots(q, p)
        r = sum(evaluate_poly(poly, a, p) == 0 for a in roots)
        nullity = q - 1 - rank_mod(matrix, p)
        assert len(roots) == q - 1 and r == nullity
        assert resultant % (p ** r) == 0
        product *= p ** r
        ell = floor_log2(p)
        capacity_numerator += ell * r
        total_weight += ell
        if r:
            root_rows.append({"p": p, "roots": r, "nullity": nullity})
    check(f"q{q}:{poly}:all modular roots equal nullity and divide determinant", True,
          {"p_count": len(pool), "nonzero_root_rows": root_rows})
    check(f"q{q}:{poly}:distinct-prime product", abs(resultant) % product == 0)
    probability = Fraction(capacity_numerator, (q - 1) * total_weight)
    upper = Fraction(ceil_log2(norm), total_weight)
    check(f"q{q}:{poly}:weighted exact probability", probability <= upper,
          {"probability": str(probability), "upper": str(upper)})
    return {"q": q, "C": str(poly), "R": resultant, "W": norm,
            "pool_size": len(pool), "nonzero_root_rows": root_rows,
            "weighted_probability": str(probability), "capacity_bound": str(upper)}


def acquire_q(U, target):
    pool, capacity, q = [], 0, U + 1
    while capacity < target:
        if is_prime(q, charged=True):
            pool.append(q)
            capacity += floor_log2(q)
        q += 1
    return pool, capacity


def order_element(q, p):
    for h in range(2, p):
        CHARGES["order_power_tests"] += 1
        gamma = pow(h, (p - 1) // q, p)
        if gamma != 1:
            assert 1 < gamma < p and pow(gamma, q, p) == 1
            return gamma
    raise AssertionError("Cyclic field group supplies an order-q element")


def acquire_p(q, target):
    pool, capacity, j = [], 0, 1
    while capacity < target:
        p = j * q + 1
        if is_prime(p, charged=True):
            pool.append({"p": p, "gamma": order_element(q, p)})
            capacity += floor_log2(p)
        j += 1
    return pool, capacity


def exact_composed_tester():

    k, d, w = 2, 1, 2
    L = k + k * (k - 1) // 2
    U = max(k, 6 * d * (k * k + 1))

    outer_target = 6 * ceil_log2(k * w)

    q_pool, Tq = acquire_q(U, inner_target)

        pool, Tp = acquire_p(q, outer_target)
        root_pool_charges = charge_delta(before_root_pool)
        capacities.append(Tp)
        q_probability = Fraction(0)
        p_rows = []
        for item in pool:
            p, gamma = item["p"], item["gamma"]
            p_min = p if p_min is None else min(p_min, p)
            roots = [pow(gamma, r, p) for r in range(1, q)]
            check(f"prototype q{q},p{p}: order certificate and distinct primitive roots",
                  len(set(roots)) == q - 1 and set(roots) == set(primitive_roots(q, p)))
            failures = sum((1 - 2 * pow(a, v, p)) % p == 0

                           "failures": failures, "sample_pairs": q * (q - 1),
                           "exact_false_accept": str(probability_p)})
        probability += Fraction(floor_log2(q), Tq) * q_probability
        rows.append({"q": q, "ell": floor_log2(q), "Tp": Tp, "pool": p_rows,
                     "acquisition_charges": root_pool_charges,
                     "exact_false_accept_given_q": str(q_probability)})

             + Fraction(d * k * k, min(q_pool))

def ordinary_supply_checks():
    for n in range(3, 65):
        bc = math.comb(2 * n, n)
        pi = len(primes_upto(2 * n))
        assert bc * (2 * n + 1) >= 4 ** n
        assert bc <= (2 * n) ** pi
        assert 2 * n + 1 <= 2 ** n
    check("ordinary supply: central binomial instances n=3..64", True)
    rows = []
    for U, M in [(1, 1), (10, 4), (30, 5), (73, 20)]:
        N = 16 * (U + M) ** 2
        # Check only the first M certified primes in the legal interval.
        got = []
        for p in range(U + 1, N + 1):
            if is_prime(p):
                got.append(p)
                if len(got) == M:
                    break
        assert len(got) == M
        rows.append({"U": U, "M": M, "bound": N, "primes": got})
    check("ordinary supply: finite U,M fixtures", True, rows)
    return rows


def crt_checks():
    rows = []
    for q, M in [(2, 3), (3, 4), (7, 5), (31, 6)]:
        V = max(q, M)
        ell = next(p for p in range(V + 1, 16 * (V + 1) ** 2 + 1) if is_prime(p))
        modulus = q * ell
        classes, witnesses = [], []
        for t in range(1, M + 1):
            a = next(a for a in range(modulus) if a % q == 1 and a % ell == t)
            assert math.gcd(a, modulus) == 1
            classes.append(a)
            p = a
            while not is_prime(p):
                p += modulus
            assert p % q == 1 and p % ell == t
            witnesses.append(p)
        check(f"CRT q{q},M{M}: distinct units and distinct prime witnesses",
              len(set(classes)) == M and len(set(witnesses)) == M,
              {"ell": ell, "classes": classes, "prime_witnesses": witnesses})
        rows.append({"q": q, "M": M, "ell": ell, "classes": classes,
                     "prime_witnesses": witnesses})
    return rows


def excluded_interface_counterexamples():
    # Composite q: sparse C can equal Phi_q, so its cyclotomic norm is zero.
    C = 1 + Z**2
    composite = int(sp.resultant(sp.cyclotomic_poly(4, Z), C, Z))
    check("exclusion: composite q=4 invalidates sparse nonzero norm", composite == 0)
    # Repetition buys no additional divisibility.  Both primitive cube roots vanish.
    q, p, C = 3, 7, sp.Integer(7)
    r = sum(evaluate_poly(C, a, p) == 0 for a in primitive_roots(q, p))
    repeated_capacity = 6 * floor_log2(p)
    fake_bound = Fraction(ceil_log2(7), repeated_capacity)
    check("exclusion: repeated prime artificial capacity contradicts actual root probability",
          r == q - 1 and fake_bound == Fraction(1, 4))
    # Each per-query input respects the same w, but is chosen after the characteristic.
    p_pool = [p for p in primes_upto(100) if p > 2][:20]
    w_p, Tp = max(p_pool), sum(floor_log2(p) for p in p_pool)
    p_fake_bound = Fraction(ceil_log2(w_p), Tp)
    check("exclusion: p-adaptive constant C_p=p vanishes on every sampled field",
          all(p % p == 0 for p in p_pool) and p_fake_bound < 1,
          {"pool": p_pool, "w": w_p, "capacity": Tp, "illegal_bound": str(p_fake_bound)})
    q_pool = primes_upto(100)[:20]
    w_q, Tq = max(q_pool), sum(floor_log2(q) for q in q_pool)
    q_fake_bound = Fraction(ceil_log2(w_q), Tq)
    check("exclusion: q-adaptive denominator h_q=q makes every query undefined",
          all(q % q == 0 for q in q_pool) and q_fake_bound < 1,
          {"pool": q_pool, "w": w_q, "capacity": Tq, "illegal_bound": str(q_fake_bound)})
    return {"composite_q_norm": composite,
            "repeated_p": {"q": 3, "p": 7, "copies": 6,
                           "actual_probability": "1", "fake_bound": str(fake_bound)},
            "p_adaptive": {"actual_probability": "1", "fake_bound": str(p_fake_bound),
