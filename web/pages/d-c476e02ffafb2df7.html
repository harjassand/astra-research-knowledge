from fractions import Fraction
from math import ceil, isqrt, lcm
import time


class ExactField:
    def __init__(self, modulus=None):
        self.modulus = modulus
        self.operations = {"add": 0, "sub": 0, "mul": 0, "inv": 0}
        self.max_operand_bits = 0
        self.max_result_bits = 0
        self.conversion_calls = 0
        self.modular_conversion_inverses = 0
        self.modular_conversion_multiplications = 0
        self.max_conversion_num_den_bits = 0

    def value(self, x):
        self.conversion_calls += 1
        x = Fraction(x)
        self.max_conversion_num_den_bits = max(self.max_conversion_num_den_bits, abs(x.numerator).bit_length(), x.denominator.bit_length())
        if self.modulus is None:
            return x
        self.modular_conversion_inverses += 1
        self.modular_conversion_multiplications += 1
        return x.numerator * pow(x.denominator, -1, self.modulus) % self.modulus

    def bits(self, x):
        if self.modulus is None:
            return max(abs(x.numerator).bit_length(), x.denominator.bit_length())
        return abs(x).bit_length()

    def operation(self, name, *xs):
        self.operations[name] += 1
        for x in xs:
            self.max_operand_bits = max(self.max_operand_bits, self.bits(x))
        p = self.modulus
        if name == "add":
            y = xs[0] + xs[1]
        elif name == "sub":
            y = xs[0] - xs[1]
        elif name == "mul":
            y = xs[0] * xs[1]
        elif name == "inv":
            if xs[0] == 0:
                raise ZeroDivisionError("zero field inverse")
            y = 1 / xs[0] if p is None else pow(xs[0], -1, p)
        else:
            raise ValueError(name)
        if p is not None:
            y %= p
        self.max_result_bits = max(self.max_result_bits, self.bits(y))
        return y

    def add(self, x, y): return self.operation("add", x, y)
    def sub(self, x, y): return self.operation("sub", x, y)
    def mul(self, x, y): return self.operation("mul", x, y)
    def inv(self, x): return self.operation("inv", x)


def identity(n, field):
    return [[field.value(int(i == j)) for j in range(n)] for i in range(n)]


def matrix_product(a, b, field):
    n, inner, width = len(a), len(b), len(b[0])
    answer = [[field.value(0) for _ in range(width)] for _ in range(n)]
    for i in range(n):
        for k in range(inner):
            if a[i][k] != 0:
                for j in range(width):
                    if b[k][j] != 0:
                        answer[i][j] = field.add(answer[i][j], field.mul(a[i][k], b[k][j]))
    return answer


def inverse(a, field):
    n = len(a)
    one = identity(n, field)
    rows = [list(a[i]) + one[i] for i in range(n)]
    for j in range(n):
        pivot = next((i for i in range(j, n) if rows[i][j] != 0), None)
        if pivot is None:
            raise ValueError("singular basis or invalid modulus")
        rows[j], rows[pivot] = rows[pivot], rows[j]
        scale = field.inv(rows[j][j])
        rows[j] = [field.mul(x, scale) for x in rows[j]]
        for i in range(n):
            if i != j and rows[i][j] != 0:
                scale = rows[i][j]
                rows[i] = [field.sub(x, field.mul(scale, y)) for x, y in zip(rows[i], rows[j])]
    return [row[n:] for row in rows]


def flag_gauge(v, field):
    """Canonical column representatives, preserving each ordered prefix span."""
    n = len(v)
    columns, pivots = [], []
    for j in range(n):
        column = [v[i][j] for i in range(n)]
        for old, pivot in zip(columns, pivots):
            coefficient = column[pivot]
            if coefficient != 0:
                column = [field.sub(x, field.mul(coefficient, y)) for x, y in zip(column, old)]
        pivot = next((i for i in reversed(range(n)) if column[i] != 0), None)
        if pivot is None:
            raise ValueError("dependent flag basis")
        scale = field.inv(column[pivot])
        column = [field.mul(x, scale) for x in column]
        columns.append(column)
        pivots.append(pivot)
    return [[columns[j][i] for j in range(n)] for i in range(n)]


def align_flag(v, s, field):
    """Lemma 2.3: reverse-row reduced column echelon of V^{-1} S."""
    n, d = len(v), len(s[0])
    coordinates = matrix_product(inverse(v, field), s, field)
    columns = [[coordinates[i][j] for i in range(n)] for j in range(d)]
    finished, pivots = 0, []
    for row in reversed(range(n)):
        pivot = next((j for j in range(finished, d) if columns[j][row] != 0), None)
        if pivot is None:
            continue
        columns[finished], columns[pivot] = columns[pivot], columns[finished]
        scale = field.inv(columns[finished][row])
        columns[finished] = [field.mul(x, scale) for x in columns[finished]]
        for j in range(d):
            if j != finished and columns[j][row] != 0:
                scale = columns[j][row]
                columns[j] = [field.sub(x, field.mul(scale, y)) for x, y in zip(columns[j], columns[finished])]
        pivots.append(row)
        finished += 1
        if finished == d:
            break
    if finished != d:
        raise ValueError("input subspace lost rank or invalid modulus")
    e = identity(n, field)
    for column, pivot in zip(columns, pivots):
        for i in range(n):
            e[i][pivot] = column[i]
    apartment = matrix_product(v, e, field)
    return apartment, sorted(pivots)


def common_coordinate_denominator(caps, m, sweeps):
    """All source weight coordinates are multiples of 1/D at this horizon."""
    cap_denominator = lcm(*(u.denominator for u in caps))
    kmax = isqrt(sweeps)
    if kmax * kmax < sweeps:
        kmax += 1
    step_denominator = lcm(*range(1, kmax + 1))
    return (m + 1) * cap_denominator * step_denominator


def snapshot(v, ticks, denominator):
    return {"basis": [[str(x) for x in row] for row in v],
            "weights": [str(Fraction(x, denominator)) for x in ticks]}


def full_objective(v, ticks, denominator, subspaces, costs, caps, field):
    value = Fraction(sum(ticks), denominator)
    hinges, jumps = [], []
    for s, c, u in zip(subspaces, costs, caps):
        _, indices = align_flag(v, s, field)
        hinge = max(0, c * denominator - sum(ticks[j] for j in indices))
        value += u * Fraction(hinge, denominator)
        hinges.append(str(Fraction(hinge, denominator)))
        jumps.append(indices)
    return value, hinges, jumps
