def monomials(n,d):
    if n==1:
        yield (d,);return
    for k in range(d+1):
        for rest in monomials(n-1,d-k):yield (k,)+rest


PERMS=list(itertools.permutations(range(3)))


def permute(mon,p):
    out=[0]*6
    for i in range(3):
        out[p[i]]=mon[i];out[p[i]+3]=mon[i+3]
    return tuple(out)


@functools.lru_cache(None)
def canonical(mon):return min(permute(mon,p) for p in PERMS)


def shifted(poly,mon):return {tuple(a+b for a,b in zip(k,mon)):v for k,v in poly.items()}


def combine(polys):
    result={}
    for poly in polys:
        for mon,c in poly.items():result[mon]=result.get(mon,0)+c
    return {k:s.expand(v) for k,v in result.items() if v!=0}


def matrix(r,degree=8):
    m=s.symbols('m0:3');ss=s.symbols('s0:3');var=m+ss
    expr={f'A{i}':(1-2*m[i]*ss[i])*(1-ss[i]**2) for i in range(3)}
    for i in range(3):
        for j in range(3):
            if i!=j:expr[f'H{i}{j}']=r*(1-4*m[i]*ss[i])+ss[j]*(-ss[i]-2*m[i]+6*m[i]*ss[i]**2)
    expr['B']=3*s.prod(ss)-r*sum(ss)
    polydict={key:dict(s.Poly(s.expand(value),*var).terms()) for key,value in expr.items()}
    even=list(itertools.chain.from_iterable(monomials(6,d) for d in range(0,degree-3,2)))
    odd=list(itertools.chain.from_iterable(monomials(6,d) for d in range(1,degree-2,2)))
    sources=[]
    swap=(0,2,1)
    sources.extend(('A',mon) for mon in even if mon<=permute(mon,swap))
    sources.extend(('H',mon) for mon in even)
    sources.extend(('B',mon) for mon in odd if mon==canonical(mon))
    rows=[]
    for typ,mon in sources:
        summands=[]
        for p in PERMS:
            key=f'A{p[0]}' if typ=='A' else f'H{p[0]}{p[1]}' if typ=='H' else 'B'
            summands.append(shifted(polydict[key],permute(mon,p)))
        full=combine(summands)
        # S3 invariance makes each orbit's coefficients equal. Check rather
        # than averaging away a potentially nonsymmetric equation.
        orbitrow={}
        for a,c in full.items():
            can=canonical(a)
            if can in orbitrow:assert orbitrow[can]==c
            else:orbitrow[can]=c
        rows.append(orbitrow)
    basis=sorted(set().union(*(set(a) for a in rows)),key=lambda a:(sum(a),a))
    index={a:i for i,a in enumerate(basis)}
    target=[s.Integer(a==(0,)*6) for a in basis]
    sparse={(index[mon],j):c for j,row in enumerate(rows) for mon,c in row.items() if c!=0}
    M=s.SparseMatrix(len(basis),len(rows),sparse)
    return M,s.Matrix(target),sources,basis,rows,expr,var
