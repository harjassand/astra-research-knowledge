from __future__ import annotations

OUT=Path(__file__).resolve().parent
I=np.eye(2,dtype=complex)
X=np.array([[0,1],[1,0]],complex)
Y=np.array([[0,-1j],[1j,0]],complex)
Z=np.diag([1.,-1.]).astype(complex)
PS=[I,X,Y,Z]
def kron_all(xs):
    a=np.array([[1.]],complex)
    for x in xs:a=np.kron(a,x)
    return a
def pauli(n, entries):
    return kron_all([PS[entries.get(i,0)] for i in range(n)])
def herm(rng,d):
    a=rng.normal(size=(d,d))+1j*rng.normal(size=(d,d))
    return (a+a.conj().T)/(2*np.sqrt(d))
def norm(a): return float(np.linalg.norm(a,2))
def free(a,beta):return float(-logsumexp(-beta*eigvalsh(a))/beta)
def comm(a,b):return a@b-b@a

def pairing_test():
    rng=np.random.default_rng(892271)
    worst=0.; count=0; min_s=1.
    for j in range(200):
        d=int(rng.integers(2,10))
        h0=herm(rng,d); h1=herm(rng,d); w=float(rng.normal())
        a=rng.normal(size=(d,d))+1j*rng.normal(size=(d,d))
        # Ensure well-defined invertibility but permit varying conditioning.
        u,_,vh=svd(a)
        aa=(u*np.geomspace(0.12,1.9,d))@vh
        s=float(svd(aa,compute_uv=False)[-1]); min_s=min(min_s,s)
        q=h1@aa-aa@h0-w*aa
        rad=norm(q)/s
        uu,pp=polar(aa)
        dres=uu.conj().T@h1@uu-h0-w*np.eye(d)
        ratio=norm(dres)/rad
        worst=max(worst,ratio)
        assert ratio<=1+1e-10
        assert np.max(np.abs(eigvalsh(h1)-eigvalsh(h0)-w))<=rad+1e-10
        for beta in [0.1,0.5,2.,8.]:
            assert abs(free(h1,beta)-free(h0,beta)-w)<=rad+1e-9
            count+=1
    return dict(matrix_pairs=200,free_energy_checks=count,max_ratio=worst,min_s=min_s)

def local_search():
    # A 3-spin collar calculation. Any additional interacting host attached to
    # spin 2 cancels, because the proposed intertwiner acts only on spins 0,1.
    terms=[(.70,{0:3}),(.43,{1:1}),(.31,{1:3}),
           (.83,{0:1,1:1}),(.61,{0:2,1:2}),(.97,{0:3,1:3}),
           (.79,{1:1,2:1}),(.57,{1:2,2:2}),(.88,{1:3,2:3})]
    k0=sum(c*pauli(3,p) for c,p in terms)
    # Exact Clifford unitary exp(i pi Z0 X1/4) with local support {0,1}.
    p=pauli(2,{0:3,1:1}); u=(np.eye(4)+1j*p)/np.sqrt(2)
    ue=np.kron(u,I)
    w=.20; delta=.03
    r=delta*pauli(3,{0:3,2:3})
    k1=ue@k0@ue.conj().T+w*np.eye(8)+r
    basis=[np.kron(a,b) for a in PS for b in PS]
    # Normalized Hilbert-Schmidt coefficient convention: A=sum a_j P_j.
    cols=[]
    for b in basis:
        be=np.kron(b,I)
        cols.append((k1@be-be@k0-w*be).reshape(-1)/np.sqrt(8))
    mat=np.stack(cols,axis=1)
    _,sing,vh=svd(mat,full_matrices=False)
    coeff=vh[-1].conj(); a=sum(c*b for c,b in zip(coeff,basis))
    ae=np.kron(a,I)
    q=k1@ae-ae@k0-w*ae
    s=float(svd(a,compute_uv=False)[-1]); rad=norm(q)/s
    naive=eigvalsh(k1-k0)
    out=dict(smallest_HS_residual=float(sing[-1]),sigma_min=s,
             sigma_max=float(svd(a,compute_uv=False)[0]),
             op_residual=norm(q),radius=rad,interval=[w-rad,w+rad],
             unoptimized_witness_radius=delta,
             direct_difference_interval=[float(naive[0]),float(naive[-1])],
             reset_collar_spins_for_algebra=3,
             environment_size_scope="arbitrary; an 80-spin extension has Hilbert dimension 2^80, but was not diagonalized",
             provenance="constructed positive family, not an independently specified physical benchmark")
    # Small full-Hamiltonian validation, not part of large-environment certificate.
    betas=[.2,1.,3.,10.]
    out['finite_3_spin_deltaF']=[dict(beta=b,delta=free(k1,b)-free(k0,b)) for b in betas]
    for row in out['finite_3_spin_deltaF']:
        assert abs(row['delta']-w)<=rad+1e-10
    np.savez(OUT/'local_search_matrices.npz',k0=k0,k1=k1,a=a,q=q,u=u,
             coefficients=coeff)
    return out

def negative_search_control():
    # Same general interaction pattern; independently chosen modification,
    # not a conjugate Hamiltonian. No selective reporting of successful fits.
    rng=np.random.default_rng(6518)
    basis=[np.kron(a,b) for a in PS for b in PS]
    successes=0; singular_rejections=0; widths=[]; cases=[]
    for case in range(40):
        k0=sum(float(rng.uniform(-1,1))*pauli(3,{site:p})
               for site in [0,1] for p in [1,2,3])
        k0+=sum(float(rng.uniform(-1,1))*pauli(3,{i:p,j:q})
                for i,j in [(0,1),(1,2)] for p in [1,2,3] for q in [1,2,3])
        # A conventional field/exchange modification with a declared cost.
        w=.2
        k1=k0+w*np.eye(8)+.8*pauli(3,{0:3})+.4*pauli(3,{0:1,1:1})
        cols=[]
        for b in basis:
            be=np.kron(b,I)
            cols.append((k1@be-be@k0-w*be).reshape(-1)/np.sqrt(8))
        _,_,vh=svd(np.stack(cols,axis=1),full_matrices=False)
        a=sum(c*b for c,b in zip(vh[-1].conj(),basis))
        s=float(svd(a,compute_uv=False)[-1])
        if s<1e-10:
            singular_rejections+=1;continue
        ae=np.kron(a,I);rad=norm(k1@ae-ae@k0-w*ae)/s
        widths.append(2*rad)
        success=rad<w
        successes+=int(success)
        cases.append(dict(case=case,s=s,radius=rad,sign_certified=success,
                         beta_2_deltaF=free(k1,2)-free(k0,2)))
    return dict(cases=40,certified_signs=successes,singular_rejections=singular_rejections,
                median_interval_width=float(np.median(widths)),details=cases,
                note="diagnostic family, not a materials benchmark; search is sufficient and not complete")

def hidden_sector():
    beta=2.; cost=.5
    out=[]
    for b in [-4.,4.]:
        # index (n,z); both may be dual-rail single-particle encodings.
        ev=np.array([n*(1-2*z)+b*z for n in [0,1] for z in [0,1]],float)
        vv=np.array([cost-n for n in [0,1] for z in [0,1]],float)
        df=(-logsumexp(-beta*(ev+vv))+logsumexp(-beta*ev))/beta
        out.append(dict(hidden_offset=b,deltaF=float(df),energies=ev.tolist()))
    assert out[0]['deltaF']<0<out[1]['deltaF']
    return dict(beta=beta,cost=cost,models=out,
                statement="From hidden z=0, all interventions commuting with z give identical records at zero hidden tunnelling.")

def random_time_test():
    # Two reset spins, one inaccessible evolving/retained memory spin.
    h=.7*pauli(3,{0:3})+.5*pauli(3,{0:3,1:1})+.3*pauli(3,{0:3,1:2})
    h+=1.1*pauli(3,{1:3,2:3})+.9*pauli(3,{1:1})+.4*pauli(3,{2:1})
    o=pauli(3,{0:1}); pp=pauli(2,{0:2,1:2})
    d1=1j*comm(h,o);d2=1j*comm(h,d1);d3=1j*comm(h,d2)
    dcoef=float(np.trace(np.kron(pp,I)@d1).real/8)
    B3=norm(d3)
    # Signal uses sigma_s=(I+sP)/4 and arbitrary memory z eigenstates.
    times=np.array([.08,.16]);hstep=float(times[1])
    mu=np.zeros((2,2,2))
    for ti,t in enumerate(times):
        ut=expm(-1j*h*t);ot=ut.conj().T@o@ut
        for zi,z in enumerate([1,-1]):
            tau=(I+z*Z)/2
            for si,s in enumerate([1,-1]):
                rho=np.kron((np.eye(4)+s*pp)/4,tau)
                mu[ti,zi,si]=np.trace(rho@ot).real
    fp= (mu[:,:,0]-mu[:,:,1])/2
    exact_conditional=(4*fp[0,:]-fp[1,:])/hstep
    bias=B3*hstep*hstep/12
    assert np.max(np.abs(exact_conditional-dcoef))<=bias+1e-12
    # Batch at different times with different memory preparations: invalid
    # cancellation can create first-order-in-h errors.
    batched=(4*fp[0,0]-fp[1,1])/hstep
    # Seeded memory changes on every shot. Past-output dependence is allowed
    # by the theorem; here use a history-feedback sign with a periodic drive.
    rng=np.random.default_rng(746384)
    N=1_000_000
    running=0.;sumz=0.;sum_cond=0.;memory_counts=[0,0]
    for k in range(N):
        zi=int(((k//317)%2) ^ (running>0))
        memory_counts[zi]+=1
        ti=0 if rng.random()<.8 else 1
        si=int(rng.integers(2));s=1 if si==0 else -1
        y=1 if rng.random()<(1+mu[ti,zi,si])/2 else -1
        z=5*s*y*(1 if ti==0 else -1)
        sumz+=z;running=.999*running+z
        sum_cond+=exact_conditional[zi]
    est=sumz/(N*hstep)
    alpha=.01
    stat=5/hstep*math.sqrt(2*math.log(2/alpha)/N)
    interval=[est-stat-bias,est+stat+bias]
    return dict(true_derivative=dcoef,third_derivative_norm=B3,h=hstep,
                N=N,alpha=alpha,estimate=est,statistical_radius=stat,
                finite_time_radius=bias,confidence_interval=interval,
                exact_conditional_derivatives=exact_conditional.tolist(),
                mean_conditional_derivative=sum_cond/N,memory_state_counts=memory_counts,
                wrongly_batched_derivative=float(batched),
                provenance="synthetic outcomes with adaptive memory, not a laboratory run")

def rank_failure():
    h0=np.diag([0.,.4]);h1=np.diag([.3,-2.])
    a=np.diag([1.,0.]);w=.3
    q=h1@a-a@h0-w*a
    return dict(residual=norm(q),sigma_min=0.,observed_shift=w,
                actual_beta2_deltaF=free(h1,2)-free(h0,2),
                verdict="rejected: singular transition operator does not cover unseen states")

def resources():
    # Certified Richardson bound per scalar derivative, noise-free controls.
    rows=[]
    for e in [.1,.03,.01]:
        B3=100.;M=32;alpha=.01
        h=math.sqrt(6*e/B3)
        N=math.ceil((100/3)*B3/e**3*math.log(2*M/alpha))
        shots=M*N
        rows.append(dict(per_coefficient_error=e,dimensionless_B3=B3,coefficients=M,
                    h=h,shots_per_coefficient=N,total_shots=shots,
                    ideal_mean_evolution_time_units=shots*.6*h,
                    wall_seconds_at_assumed_10_microsecond_reset=shots*1e-5,
                    caveat="a hypothetical accounting scenario, not device performance"))
    return rows
