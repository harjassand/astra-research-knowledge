"""Exact acquisition for a bare four-layer catalytic chain.

The universal proof is in FOUR_LAYER_PHASE_PROOF.txt. This file checks the
actual falling-factorial generator, selecting each independent rate-box
endpoint by the sign of the actual potential increment. No peer imports.
"""
from fractions import Fraction as Q
from itertools import product
from pathlib import Path
import json
import time


def interval(v):
    if (not isinstance(v, (list, tuple)) or len(v) != 2
            or any(isinstance(x, bool) or not isinstance(x, (str, int)) for x in v)):
        raise ValueError("Two exact rational strings/integers per interval required")
    lo, hi = (Q(str(x)) for x in v)
    if not 0 < lo <= hi:
        raise ValueError("Strictly positive ordered rate intervals required")
    return lo, hi


def ceilq(x):
    return -((-x.numerator) // x.denominator)


def cubic_envelope(birth_hi, death_lo):
    u = 3 * (birth_hi + death_lo) + max(3 * birth_hi - death_lo, Q(0)) + birth_hi
    threshold = 1 + 2 * u / (5 * death_lo)
    return u * threshold**2


def acquire(spec):
    if not isinstance(spec, dict) or set(spec) != {"names", "birth", "death"}:
        raise ValueError("Complete four-layer chain syntax required")
    names = spec["names"]
    if (not isinstance(names, list) or len(names) != 4
            or any(not isinstance(x, str) or not x for x in names)
            or len(set(names)) != 4
            or not isinstance(spec["birth"], list) or len(spec["birth"]) != 4
            or not isinstance(spec["death"], list) or len(spec["death"]) != 4):
        raise ValueError("Four distinct species and four paired rate intervals required")
    births = [interval(x) for x in spec["birth"]]
    deaths = [interval(x) for x in spec["death"]]
    b, alpha, a, e = births
    delta, beta, t, u = deaths

    # Cubic B and C correctors use half of their physical death lower bound.
    cb = min(Q(1), beta[0] / (2 * delta[1]))
    gb = min(cb * b[0] / 2, beta[0] / 4)
    sbc = 1 + 4 * delta[1] / alpha[0]
    cc = min(Q(1), (t[0] / 2) / (delta[1] / sbc + beta[1]))
    gc = min(cc * b[0] / (sbc * (sbc + 1)),
             cc * alpha[0] / (4 * (sbc + 1)), t[0] / 4)
    Gb, Gc = 1 + cb, 1 + cc / sbc

    # New, unbounded quadratic phase representation for D's killed particles.
    s = 1 + delta[1] / beta[0] + 2 * delta[1] / alpha[0]
    k = 2 * alpha[1] + 3 * beta[1]
    kap = min(beta[0] / (s + 1), b[0] / (2 * s * (s + 1)))
    J = b[0] / (s * (s + 1))
    gh = min(J, alpha[0] / (2 * (s + 1)))
    eps = a[0] / (4 * k)
    F = eps * J + (eps * k + t[1] / 2)**2 / (4 * eps * kap)
    M = 1 + (F + 1) / (u[0] / 2)
    gd = min(eps * gh, a[0] / 4, Q(1))

    Ks = [cubic_envelope(x[1], y[0]) for x, y in zip(births, deaths)]
    tB = Ks[3] * 4 * eps / (3 * s)
    tC = Ks[3] * (M + 1 + 8 * eps / (3 * s))
    wC = (1 + tC) / gc
    wB = (1 + tB + wC * Gc * Ks[2]) / gb
    wA = 2 * (1 + wB * Gb * Ks[1]) / delta[0]
    C0 = wA * Ks[0]
    kappa = min(Q(1), gd)
    radius = max(1, ceilq((2 * C0 + 2) / kappa))
    turnover = b[1] + delta[1] + alpha[1] + a[1] + e[1] + beta[1] + t[1] + u[1]
    firing_factor = 8 * turnover / kappa
    return {"spec": spec, "births": births, "deaths": deaths,
            "cb": cb, "gammaB": gb, "sC": sbc, "cC": cc, "gammaC": gc,
            "GB": Gb, "GC": Gc, "s": s, "K": k, "kappa_h": kap,
            "J": J, "gamma_h": gh, "epsilon": eps, "F": F, "M": M,
            "gammaD": gd, "cubic_envelopes": Ks,
            "weights": [wA, wB, wC], "constant": C0,
            "kappa": kappa, "core_radius": radius,
            "turnover": turnover, "firing_bound_factor": firing_factor}


def acquire_from_reactions(species, reactions):
    """Consume the full reaction table and acquire its root/chain ordering.

    Parallel identical channels are aggregated exactly. Unknown reactions,
    extra roots, branches, cycles, self-catalysis, bursts and unpaired channels
    reject rather than silently being discarded.
    """
    if (not isinstance(species, list) or len(species) != 4
            or any(not isinstance(x, str) or not x for x in species)
            or len(set(species)) != 4 or not isinstance(reactions, list)):
        raise ValueError("Exactly four explicit distinct species required")
    buckets = {}
    for r in reactions:
        if not isinstance(r, dict) or set(r) != {"source", "target", "rate"}:
            raise ValueError("Exactly source/target/rate required")
        src, dst = r["source"], r["target"]
        if (not isinstance(src, (list, tuple)) or not isinstance(dst, (list, tuple))
                or len(src) != 4 or len(dst) != 4
                or any(isinstance(x, bool) or not isinstance(x, int) or x < 0
                       for x in (*src, *dst))):
            raise ValueError("Nonnegative integer source/target vectors required")
        changes = [i for i in range(4) if src[i] != dst[i]]
        if len(changes) != 1 or abs(dst[changes[0]] - src[changes[0]]) != 1:
            raise ValueError("Only one unit count change admitted")
        child = changes[0]
        sign = dst[child] - src[child]
        residual = src if sign > 0 else dst
        if sum(residual) not in (0, 1):
            raise ValueError("Only zero or one unchanged catalyst admitted")
        parent = None if sum(residual) == 0 else residual.index(1)
        if parent == child:
            raise ValueError("Self-catalysis outside the chain grammar")
        bounds = interval(r["rate"])
        key = (child, parent, sign)
        old = buckets.get(key, (Q(0), Q(0)))
        buckets[key] = tuple(x + y for x, y in zip(old, bounds))
    pairs = {(j, i) for j, i, _ in buckets}
    if any((j, i, 1) not in buckets or (j, i, -1) not in buckets for j, i in pairs):
        raise ValueError("Every channel needs both positive directions")
    roots = [j for j, i in pairs if i is None]
    if len(roots) != 1 or len(pairs) != 4:
        raise ValueError("One root and three chain edges required")
    order = roots[:]
    while len(order) < 4:
        nxt = [j for j, i in pairs if i == order[-1]]
        if len(nxt) != 1 or nxt[0] in order:
            raise ValueError("Branches, cycles or disconnected species rejected")
        order.append(nxt[0])
    expected = {(order[0], None), *((order[j], order[j-1]) for j in range(1, 4))}
    if pairs != expected:
        raise ValueError("Unexpected paired edge rejected")
    spec = {"names": [species[i] for i in order],
            "birth": [[str(x) for x in buckets[(j, i, 1)]] for j, i in
                      [(order[0], None), *[(order[k], order[k-1]) for k in range(1, 4)]]],
            "death": [[str(x) for x in buckets[(j, i, -1)]] for j, i in
                      [(order[0], None), *[(order[k], order[k-1]) for k in range(1, 4)]]]}
    return acquire(spec)


def gateB(c, A):
    return 1 + c["cb"] / (A + 1)


def gateC(c, A, B):
    return 1 + c["cC"] / ((A + c["sC"]) * (B + 1))


def gateD(c, A, B, C):
    return c["M"] + Q(1, C + 1) + c["epsilon"] * (B - 1)**2 / (A + c["s"])


def V(c, x):
    A, B, C, D = x
    wA, wB, wC = c["weights"]
    return (wA * A**3 + wB * gateB(c, A) * B**3
            + wC * gateC(c, A, B) * C**3 + gateD(c, A, B, C) * D**3)


def raw(c):
    rs = []
    for j in range(4):
        src = [0] * 4
        if j:
            src[j-1] = 1
        dst = src.copy()
        dst[j] = 1
        rs.append((tuple(src), tuple(dst), c["births"][j]))
        rs.append((tuple(dst), tuple(src), c["deaths"][j]))
    return rs


def generator(c, x, f):
    value = f(x)
    total = Q(0)
    for src, dst, bounds in raw(c):
        rate = 1
        for n, k in zip(x, src):
            for j in range(k):
                rate *= max(0, n - j)
        if rate:
            y = tuple(n + z - a for n, a, z in zip(x, src, dst))
            jump = f(y) - value
            total += rate * (bounds[1] if jump > 0 else bounds[0]) * jump
    return total


def killedD(c, A, B, C):
    x = (A, B, C, 0)
    drift = generator(c, x, lambda z: gateD(c, *z[:3]))
    return drift - c["deaths"][3][0] / 2 * C * gateD(c, A, B, C)


def dumpable(x):
    if isinstance(x, Q):
        return str(x)
    if isinstance(x, dict):
        return {k: dumpable(v) for k, v in x.items()}
    if isinstance(x, (tuple, list)):
        return [dumpable(v) for v in x]
    return x


def verify():
    start = time.perf_counter()
    specs = [
        {"names": ["A", "B", "C", "D"], "birth": [[1, 1]] * 4, "death": [[1, 1]] * 4},
        {"names": ["A", "B", "C", "D"], "birth": [[1, 2], [1, 3], [2, 4], [1, 2]],
         "death": [[2, 5], [1, 4], [1, 3], [2, 6]]},
        {"names": ["A", "B", "C", "D"], "birth": [["1/3", "2/3"], ["1/7", 2], ["1/5", 4], ["1/11", 3]],
         "death": [[1, 3], ["1/2", 5], ["1/9", 2], ["1/13", 3]]},
    ]
    checks = killed = raw_admission = reject = 0
    records = []
    for spec in specs:
        c = acquire(spec)
        margin = None
        for x in product(range(5), repeat=4):
            lv = generator(c, x, lambda z: V(c, z))
            A, B, C, D = x
            upper = c["constant"] - A**3 - B**3 - C**3 - c["gammaD"] * D**3
            assert lv <= upper, (spec, x, lv, upper)
            checks += 1
            margin = upper - lv if margin is None else min(margin, upper - lv)
        for A, B, C in product(range(9), repeat=3):
            assert killedD(c, A, B, C) <= -c["gammaD"]
            killed += 1
        for A, B, C in product([0, 1, 10, 1000, 1000000], [0, 1, 2, 10, 1000], [0, 1, 2]):
            assert killedD(c, A, B, C) <= -c["gammaD"]
            killed += 1
        reactions = [{"source": list(src), "target": list(dst), "rate": list(map(str, bounds))}
                     for src, dst, bounds in raw(c)]
        c2 = acquire_from_reactions(spec["names"], reactions)
        assert {k: v for k, v in c2.items() if k != "spec"} == {k: v for k, v in c.items() if k != "spec"}
        raw_admission += 1
        extra = {"source": [0]*4, "target": [0, 0, 0, 2], "rate": [1, 1]}
        try:
            acquire_from_reactions(spec["names"], reactions + [extra])
        except ValueError:
            reject += 1
        else:
            raise AssertionError("Unrecognized extra reaction silently admitted")
        records.append({"certificate": dumpable(c), "minimum_full_drift_margin": str(margin)})
    unit = acquire(specs[0])
    n = 100
    q_new = gateD(unit, n, n, 0)
    kd_new = killedD(unit, n, n, 0)
    assert kd_new < 0
    witness = {"upstream_state": [n, n, 0], "new_gate_value": str(q_new),
               "new_killed_drift": str(kd_new), "old_product_route":"proved positive on this scaling in independent fourth-layer counterexample"}
    return {"status": "EXACT_FINITE_DIAGNOSTICS_PASSED", "full_generator_states": checks,
            "killed_gate_states": killed, "raw_table_acquisitions": raw_admission,
            "extra_reaction_rejections": reject, "fixtures": records,
            "old_obstruction_witness": witness, "elapsed_seconds": time.perf_counter()-start,
            "scope": "Finite diagnostics only; infinite-lattice proof in FOUR_LAYER_PHASE_PROOF.txt"}
