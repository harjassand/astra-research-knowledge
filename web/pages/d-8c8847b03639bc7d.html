from fractions import Fraction as F
import json

# Periodic, nonuniform partition used only as an exact finite algebra fixture.
gaps = [F(1, 8), F(1, 4), F(1, 8), F(1, 2)]
D = len(gaps)
assert sum(gaps) == 1
knots = [F(0)]
for h in gaps[:-1]:
    knots.append(knots[-1] + h)

# A periodic linear hat has mass (h_{j-1}+h_j)/2 and circular
# centroid offset (h_j-h_{j-1})/3 from knot j.
masses = [(gaps[(j - 1) % D] + gaps[j]) / 2 for j in range(D)]
offsets = [(gaps[j] - gaps[(j - 1) % D]) / 3 for j in range(D)]
assert sum(masses) == 1

# Check the exact local first-moment drift formula at rational points in every cell.
bias_checks = 0
for i, h in enumerate(gaps):
    left_change = gaps[i] - gaps[(i - 1) % D]
    right_change = gaps[(i + 1) % D] - gaps[i]
    max_change = max(abs(left_change), abs(right_change))
    for lam in (F(0), F(1, 4), F(1, 2), F(3, 4), F(1)):
        drift = (1 - lam) * offsets[i] + lam * offsets[(i + 1) % D]
        assert abs(drift) <= max_change / 3
        bias_checks += 1

# Exact invariance of Lebesgue measure for a nonconstant periodic piecewise-linear f.
values = [F(1, 3), F(7, 5), F(-2, 7), F(4, 3)]
int_f = F(0)
coefficients = []
for j in range(D):
    h_left = gaps[(j - 1) % D]
    h_right = gaps[j]
    v_left = values[(j - 1) % D]
    v = values[j]
    v_right = values[(j + 1) % D]
    # Integral phi_j f on the two intervals supporting hat j.
    integral_phi_f = h_left * (v_left / 6 + v / 3) + h_right * (v / 3 + v_right / 6)
    coefficients.append(integral_phi_f / masses[j])
    int_f += h_left * (v_left + v) / 2
# Compute integral Kf = sum_j m_j * coefficient_j.
int_Kf = sum(masses[j] * coefficients[j] for j in range(D))
assert int_Kf == int_f

# K is positive and fixes 1 because hats are nonnegative and sum to one.
assert all(m > 0 for m in masses)
assert sum(masses) == 1
