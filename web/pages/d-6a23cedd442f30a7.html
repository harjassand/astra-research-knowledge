def image(a,u,m,field):
    return span([tuple(sum_field((field.mul(a[r][j],v[j]) for j in range(len(v))),field)
                       for r in range(m)) for v in u],m,field)


def preimage(a,w,n,m,field):
    rows=[[*(a[r][j] for j in range(n)),*(field.sub(field.elt(0),v[r]) for v in w)] for r in range(m)]
    return span([v[:n] for v in nullspace(rows,n+len(w),field)],n,field)


def contains(u,v,n,field):return len(span(u+v,n,field))==len(u)


def block_image(generators,u,n,m,d,field):
    # (Mat_d tensor B)U is direct sum of d copies of B times all block projections.
    projected=span([v[j*n:(j+1)*n] for v in u for j in range(d)],n,field)
    out=[]
    for gen in generators:
        for v in projected:
            row=[field.elt(0) for _ in range(m)]
            for r,c,a in gen:row[r]=field.add(row[r],field.mul(a,v[c]))
            out.append(tuple(row))
    base=span(out,m,field);zero=field.elt(0)
    return tuple(tuple((*([zero]*(i*m)),*v,*([zero]*((d-i-1)*m)))) for i in range(d) for v in base)


def weighted_copy_space(data):
    """Direct bipartite input when legal; otherwise identity-penalty doubling."""
    dims=data['dimensions'];sigma=data['weight'];vcount=len(dims)
    if len(sigma)!=vcount or any(type(z) is not int for z in sigma):raise ValueError('integer signed weights required')
    if any(type(z) is not int or z<1 for z in dims):raise ValueError('active positive integer dimensions required')
    # Binary-encoded isolated dimensions must be bounded before constructing
    # the dense identity arrows used by penalty doubling.
    original_total=sum(dims);original_square=sum(dim*dim for dim in dims)
    if (original_total>data.get('max_expanded_dimension',256)
        or original_square>data.get('max_dense_entries',1000000)):
        return {'status':'BUDGET_UNKNOWN','mode':'original_dimension_preflight',
                'Nplus':None,'Nminus':None,
                'original_expanded_dimension':original_total,
                'estimated_original_identity_entries':original_square}
    for a in data['arrows']:
        t,h=a['tail'],a['head'];mat=a['matrix']
        if len(mat)!=dims[h] or any(len(row)!=dims[t] for row in mat):raise ValueError('arrow shape')
        if any(type(x) is not int for row in mat for x in row):raise ValueError('integer arrows required')
    direct=all(sigma) and all(sigma[a['tail']]>0>sigma[a['head']] for a in data['arrows'])
    if direct:
        bdims=dims;bweights=sigma;arrows=data['arrows'];mode='direct_bipartite';penalty=None
    else:
        t=sum(max(0,w)*dim for w,dim in zip(sigma,dims));penalty=t+max([0]+[-w for w in sigma])+1
        bdims=dims+dims;bweights=[w+penalty for w in sigma]+[-penalty]*vcount
        arrows=[{'tail':v,'head':v+vcount,'matrix':[[int(r==c) for c in range(dim)] for r in range(dim)]}
                for v,dim in enumerate(dims)]
        arrows += [{'tail':a['tail'],'head':a['head']+vcount,'matrix':a['matrix']} for a in data['arrows']]
        mode='identity_penalty_doubling'
    plus=[v for v,w in enumerate(bweights) if w>0];minus=[v for v,w in enumerate(bweights) if w<0]
    noff={};moff={};n=m=0
    for v in plus:noff[v]=n;n+=bweights[v]*bdims[v]
    for v in minus:moff[v]=m;m-=bweights[v]*bdims[v]
    copy_slots=sum(bweights[a['tail']]*(-bweights[a['head']]) for a in arrows)
    sparse_entries=sum(bweights[a['tail']]*(-bweights[a['head']])*
                       sum(x!=0 for row in a['matrix'] for x in row) for a in arrows)
    if (sparse_entries>data.get('max_generator_entries',200000)
        or copy_slots>data.get('max_copy_slots',200000)
        or n+m>data.get('max_expanded_dimension',256)
        or (n+m)**2>data.get('max_dense_entries',1000000)):
        return {'status':'BUDGET_UNKNOWN','mode':mode,'Nplus':n,'Nminus':m,
                'estimated_copy_slots_including_zero_arrows':copy_slots,
                'estimated_sparse_entries':sparse_entries,'estimated_base_dense_entries':(n+m)**2}
    gens=[]
    for a in arrows:
        t,h=a['tail'],a['head'];mat=a['matrix']
        for j in range(-bweights[h]):
            for i in range(bweights[t]):
                g=tuple((moff[h]+j*bdims[h]+r,noff[t]+i*bdims[t]+c,x)
                        for r,row in enumerate(mat) for c,x in enumerate(row) if x)
                if g:gens.append(g)
    return {'status':'BUILT','mode':mode,'penalty':penalty,'Nplus':n,'Nminus':m,
            'generators':gens,'source_offsets':noff,'target_offsets':moff,'dimensions':bdims,
            'weights':bweights,'arrows':arrows,'sparse_entries':sparse_entries,
            'copy_slots_including_zero_arrows':copy_slots}


def acquire(data):

    result={k:v for k,v in spec.items() if k not in ('generators','arrows')}
