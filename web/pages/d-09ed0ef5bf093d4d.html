def cross_bkm(c,b,tau,p):
    vals,u=np.linalg.eigh(c)
    mask=vals>1e-12
    vals=vals[mask];u=u[:,mask]

    diff=np.log(x)-np.log(y)
    divided=diff/(x-y)
    return float(2*p*q*np.sum(np.abs(entries)**2*divided))
