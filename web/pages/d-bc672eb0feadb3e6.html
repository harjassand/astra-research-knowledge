def binrisk(p,m):
    n=len(p)
    reciprocals=[]
    for py in p:
        direct=sum(F(comb(m,c))*py**c*(1-py)**(m-c)/F(c+1)
                   for c in range(m+1))
        closed=(1-(1-py)**(m+1))/((m+1)*py)
        assert direct==closed
        reciprocals.append(direct)
    risk=(m+n)*sum(py*py*rec for py,rec in zip(p,reciprocals))-1
    assert risk<=F(n-1,m+1)
    return float(risk/F(n-1,m+1))

def exact_adaptive(Q,horizon):
    """Rational toy endpoint kernels, adaptive two-lag selection and masking."""
    # Branch = full state history, its true probability, ratio wealth,
    #           predictor counts keyed by(start,lag), threshold crossed.
    branches=[((0,),F(1),F(1),{},False)]
    expectations=[]
    threshold=F(10,3)
    for i in range(horizon):
        new=[]
        for history,prob,z,counts,crossed in branches:
            x=history[-1]
            lag=(sum(history)+i)%2  # chosen before next endpoint
            mask=(x==0 or i%3==0)   # includes omitted reads in filtration
            cell=(x,lag)
            old=counts.get(cell,(0,0))
            q=[F(old[y]+1,sum(old)+2) for y in range(2)]
            for y in range(2):
                py=Q[lag][x][y]
                zy=z*q[y]/py if mask else z
                nc=counts.copy()
                if mask:
                    row=list(old);row[y]+=1;nc[cell]=tuple(row)
                new.append((history+(y,),prob*py,zy,nc,crossed or zy>=threshold))
        branches=new
        ez=sum(prob*z for _,prob,z,_,_ in branches)
        assert ez==1
        expectations.append(str(ez))
    crossing=sum(prob for _,prob,_,_,crossed in branches if crossed)
    assert crossing<=F(3,10)
    return {'expected_ratio_each_step':expectations,
            'exact_crossing_probability':str(crossing),
            'crossing_probability_float':float(crossing),
            'ville_bound':.3,'scope':'rational toy conditional kernels, not a CTMC fixture'}

def saturated_regret(counts):
    n=counts.shape[1]
    ans=0.
    for row in counts:
        N=int(sum(row))
        if not N:continue
        log_sat=sum(c*log(c/N) for c in row if c)
        # lgamma is sufficient for a diagnostic; no certification claimed.
        from math import lgamma
        log_m=lgamma(n)+sum(lgamma(int(c)+1) for c in row)-lgamma(N+n)
        ans+=log_sat-log_m
    return ans
