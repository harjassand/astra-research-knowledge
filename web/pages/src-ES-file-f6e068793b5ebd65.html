#!/usr/bin/env python3
"""Numerical cross-checks of independent analytic identities.

The proofs are in FINITE_TIME_INDEPENDENT_CHECK.md. Quadrature agreement is
not claimed as a proof. All routines use Python's standard library.
"""
import json
import math
from pathlib import Path

PI = math.pi
GAMMA = 0.5772156649015328606
MU2 = PI*PI/4


def quad(fun, a, b, tol=2e-11):
    if a == b:
        return 0.
    fa, fb = fun(a), fun(b)
    m = (a+b)/2
    fm = fun(m)
    total = (b-a)*(fa+4*fm+fb)/6
    def recurse(a, b, fa, fm, fb, old, tol, depth):
        m = (a+b)/2
        left, right = (a+m)/2, (m+b)/2
        fl, fr = fun(left), fun(right)
        sl = (m-a)*(fa+4*fl+fm)/6
        sr = (b-m)*(fm+4*fr+fb)/6
        if depth <= 0 or abs(sl+sr-old) <= 15*tol:
            return sl+sr+(sl+sr-old)/15
        return (recurse(a, m, fa, fl, fm, sl, tol/2, depth-1)
                + recurse(m, b, fm, fr, fb, sr, tol/2, depth-1))
    return recurse(a, b, fa, fm, fb, total, tol, 24)


def sinc(x):
    return math.sin(x)/x if abs(x)>1e-7 else 1-x*x/6+x**4/120


def phi(x):
    what = sinc(x+PI/2)+sinc(x-PI/2)
    return what*what/(2*PI)


def rphi(a):
    if a == 0:
        return 0.
    return 2*a*quad(lambda y: (1-y*y)*phi(a*y), 0., 1.)


def m2(a):
    return 2*a**3*quad(lambda y: y*y*phi(a*y), 0., 1.)


def j_tonelli(a):
    # x=a exp(-t) removes the logarithmic endpoint.
    return 2*a*quad(lambda t: math.exp(-t)*phi(a*math.exp(-t)) *
                   (t-.5+.5*math.exp(-2*t)), 0., 40.)


def test_constants():
    si = quad(sinc, 0., PI, 2e-14)
    ci = GAMMA+math.log(PI)+quad(
        lambda t: -2*math.sin(t/2)**2/t if t else 0., 0., PI, 2e-14)
    c = .5+math.log(PI/2)-ci-si/PI
    assert abs(phi(0)-8/PI**3) < 2e-15
    assert abs(phi(PI/2)-1/(2*PI)) < 2e-15
    assert abs(c-0.2884249210069454) < 2e-14
    for t in (0., .2, 1., 1.9, 2.):
        overlap = quad(lambda s: math.cos(PI*s/2)*math.cos(PI*(s+t)/2),
                       -1., 1.-t)
        closed = (1-t/2)*math.cos(PI*t/2)+math.sin(PI*t/2)/PI
        assert abs(overlap-closed) < 2e-11
    # Independent direct log-moment integration; the omitted tail is positive.
    negative = -2*quad(lambda t: t*math.exp(-t)*phi(math.exp(-t)), 0., 40.)
    cutoff = 1024.
    positive = 2*sum(quad(lambda x: phi(x)*math.log(x), float(j), float(j+1),
                          1e-12) for j in range(1, int(cutoff)))
    direct = .5+negative+positive
    tail_bound = 16*PI/(9*cutoff**3)*(math.log(cutoff)/3+1/9)
    assert -1e-10 <= c-direct <= tail_bound+1e-10
    return c, {"Si_pi": si, "Ci_pi": ci, "cosine_c": c,
               "direct_partial_c": direct,
               "omitted_positive_tail_upper_bound": tail_bound}


def test_angular_and_unscreened(c):
    records = []
    for a in (.05, .2, 1., 3., 10., 32., 64.):
        r = rphi(a)
        assert -1e-11 <= r <= 1+1e-11
        assert 1-r <= MU2/(a*a)+1e-10
        j = j_tonelli(a)
        eps = j-math.log(a)+c
        assert eps >= -2e-9
        assert eps <= MU2/(2*a*a)+2e-9
        if a <= 10:
            direct = quad(lambda t: 2*quad(
                lambda y: (1-y*y)*phi(t*y), 0., 1.), 0., a)
            assert abs(direct-j) < 2e-9
            raw_longitudinal = 4*quad(lambda t: quad(
                lambda y: y*y*phi(t*y), 0., 1.), 0., a)
            assert abs(raw_longitudinal-r) < 2e-9
        records.append({"A": a, "R": r, "J": j, "epsilon": eps,
                        "A2_epsilon": a*a*eps})
    # Direct angular integration includes the Fourier radial measure.
    time, radius, q = 1.2, .7, 1.3
    a = time*radius*q
    direct = PI*time*q**4*quad(
        lambda mu: (1-mu*mu)*phi(a*mu), -1., 1.)
    reduced = PI/radius*q**3*rphi(a)
    assert abs(direct-reduced) < 2e-11
    return records


def test_yukawa_difference():
    time, radius, kappa = 1.2, .7, .9
    records = []
    for cutoff in (.4, 1., 5.):
        a = cutoff*time*radius
        direct = PI/radius*quad(
            lambda q: q**3/(q*q+kappa*kappa)**2 *
                      (1-rphi(time*radius*q)), 0., cutoff)
        inside = 2*quad(
            lambda x: phi(x)*(math.log1p((x/(time*radius*kappa))**2)
                    -(x/(time*radius))**2/(kappa*kappa+cutoff*cutoff)),
            0., a)
        mass_out = 1-2*quad(phi, 0., a)
        outside_value = math.log1p((cutoff/kappa)**2) - (
            cutoff*cutoff/(kappa*kappa+cutoff*cutoff))
        tonelli = PI/(2*radius)*(inside+outside_value*mass_out)
        jensen = PI/(2*radius)*math.log1p(MU2/(kappa*kappa*time*time*radius*radius))
        algebraic = PI*MU2/(2*kappa*kappa*time*time*radius**3)
        assert abs(direct-tonelli) < 2e-9
        assert 0 <= direct <= jensen <= algebraic
        records.append({"K": cutoff, "direct_difference": direct,
                        "Tonelli_difference": tonelli,
                        "Jensen_upper_bound": jensen,
                        "algebraic_upper_bound": algebraic})
    # Independent exact expression for the leading-log radial discrepancy.
    for z in (1., 2., 10., 1e3, 1e12):
        difference = math.log1p(1/z)+1/(1+z)-1
        assert abs(difference) <= 1
    return records


def main():
    c, constants = test_constants()
    data = {
        "status": "NUMERICAL_CROSS_CHECKS_PASS_ANALYTIC_PROOFS_IN_COMPANION_FILE",
        "constants": constants,
        "unscreened_angular": test_angular_and_unscreened(c),
        "yukawa": test_yukawa_difference(),
        "qualification": "Unscreened raw longitudinal coefficient tends to pi/r; "
                         "only its normalized leading-log contribution vanishes.",
    }
    Path(__file__).with_name("FINITE_TIME_TEST_RESULTS.json").write_text(
        json.dumps(data, indent=2)+"\n")
    print(json.dumps(data, indent=2))


if __name__ == "__main__":
    main()
