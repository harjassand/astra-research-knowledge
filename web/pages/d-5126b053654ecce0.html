def cycle_fixture(n=5):

def split_axes(projectors,n):
    assert all(P*R==R*P for P,R in itertools.combinations(projectors,2))
    spaces=[s.eye(n)]
    for P in projectors:
        nxt=[]
        for W in spaces:
            for U in [P*W,(s.eye(n)-P)*W]:
                cols=U.columnspace()
                if cols:nxt.append(s.Matrix.hstack(*cols))
        spaces=nxt
    if len(spaces)!=n or any(U.cols!=1 for U in spaces):raise ValueError('unresolved axes')
    return s.Matrix.hstack(*spaces)


def pencil_edge_acquisition(B,Minv,rng,max_trials=30):

            trials.append('non split or degenerate Hessian');continue
