def matrix(rows):
    return Mat([[Q(v) for v in row] for row in rows])


def eye(n):
    return matrix([[int(i == j) for j in range(n)] for i in range(n)])


def rows(m):
    return [[str(m[i,j]) for j in range(m.ncols())] for i in range(m.nrows())]


def number(q):
    return int(q.numerator) / int(q.denominator)


def p_add(a,b):
    c=dict(a)
    for e,x in b.items():
        c[e]=c.get(e,Q(0))+x
        if not c[e]: del c[e]
    return c


def p_scale(a,x):
    return {e:v*x for e,v in a.items() if v*x}


def p_mul(a,b):
    c={}
    for e,v in a.items():
        for f,w in b.items():
            g=tuple(e[k]+f[k] for k in range(3))
            c[g]=c.get(g,Q(0))+v*w
    return {e:v for e,v in c.items() if v}


def p_pow(a,n):
    c={(0,0,0):Q(1)}
    for _ in range(n): c=p_mul(c,a)
    return c


def p_derivative(a,k):
    c={}
    for e,v in a.items():
        if e[k]:
            f=list(e); f[k]-=1
            c[tuple(f)]=v*e[k]
    return c


def laplacian(a):
    c={}
    for k in range(3): c=p_add(c,p_derivative(p_derivative(a,k),k))
    return c


def harmonic_projection(p,degree):
    rr={(2,0,0):Q(1),(0,2,0):Q(1),(0,0,2):Q(1)}
    delta=dict(p); power={(0,0,0):Q(1)}; a=Q(1); result={}
    for j in range(degree//2+1):
        if j: a=-a/Q(2*j*(2*degree-2*j+1))
        result=p_add(result,p_scale(p_mul(power,delta),a))
        delta=laplacian(delta); power=p_mul(power,rr)
    assert not laplacian(result)
    return result


def p_evaluate(p,x):
    degree=max((sum(e) for e in p),default=0)
    powers=[[Q(1)] for _ in range(3)]
    for k in range(3):
        for j in range(degree): powers[k].append(powers[k][-1]*x[k])
    return sum((v*powers[0][e[0]]*powers[1][e[1]]*powers[2][e[2]]
                for e,v in p.items()),Q(0))


@lru_cache(None)
def sphere_monomial(e):
    if any(x%2 for x in e): return Q(0)
    numerator=1
    for x in e:
        for j in range(1,x,2): numerator*=j
    denominator=1
    for j in range(1,sum(e)+2,2): denominator*=j
    return Q(numerator,denominator)


def inner(a,b):
    return sum((v*w*sphere_monomial(tuple(e[k]+f[k] for k in range(3)))
               for e,v in a.items() for f,w in b.items()),Q(0))


def basis_decode(data):

def basis_encode(polys):

                 terms=[dict(powers=list(e),coefficient=str(v))

def verify_basis(polys,norms):

def tangent_projection(anchors):
    S=len(anchors); pp=[[Q(0) for _ in range(3*S)] for _ in range(3*S)]
    for i,x in enumerate(anchors):
        assert sum(a*a for a in x)==1
        for a in range(3):
            for b in range(3): pp[3*i+a][3*i+b]=Q(int(a==b))-x[a]*x[b]
    return matrix(pp)


def operator(polys,anchors):
    S=len(anchors); mean=[]; jac=[]
    for p in polys:
        mean.append(sum((p_evaluate(p,x) for x in anchors),Q(0))/S)
        row=[]
        derivatives=[p_derivative(p,a) for a in range(3)]
        for x in anchors:
            grad=[p_evaluate(d,x) for d in derivatives]
            radial=sum(x[a]*grad[a] for a in range(3))
            row.extend((grad[a]-x[a]*radial)/S for a in range(3))
        jac.append(row)
    return matrix(jac),mean


def norm2(z,nu):
    return sum((v*v/n for v,n in zip(z,nu)),Q(0))


def round_dyadic(q,bits):
    scale=1<<bits
    n=int(q.numerator)*scale; d=int(q.denominator)
    return Q((2*n+d)//(2*d),scale)


def floor_sqrt_dyadic(q,bits):
    assert q>=0
    n=int(q.numerator)*(1<<(2*bits))//int(q.denominator)
    return Q(math.isqrt(n),1<<bits)


def null_vector(a):
    rr=[[a[i,j] for j in range(a.ncols())] for i in range(a.nrows())]
    piv=[]; row=0
    for col in range(a.ncols()):
        at=next((i for i in range(row,len(rr)) if rr[i][col]),None)
        if at is None: continue
        rr[row],rr[at]=rr[at],rr[row]
        z=rr[row][col]; rr[row]=[v/z for v in rr[row]]
        for i in range(len(rr)):
            if i!=row and rr[i][col]:
                z=rr[i][col]; rr[i]=[x-z*y for x,y in zip(rr[i],rr[row])]
        piv.append(col); row+=1
        if row==len(rr): break
    free=[c for c in range(a.ncols()) if c not in piv]
    if not free: return row,None
    v=[Q(0)]*a.ncols(); v[free[0]]=Q(1)
    for i,c in enumerate(piv): v[c]=-rr[i][free[0]]
    assert all(not sum(a[i,j]*v[j] for j in range(a.ncols()))
               for i in range(a.nrows()))
    return row,v


def orbit_points(anchors):
    out=[]
    for x in anchors:

                out.append(tuple(signs[a]*x[perm[a]] for a in range(3)))
    assert len(set(out))==len(out)
    return out
