def det(a):
    if not a:return Fraction(1)
    t=Fraction(0)
    for p in permutations(range(len(a))):
        inv=sum(p[i]>p[j] for i in range(len(p)) for j in range(i+1,len(p)))
        x=Fraction(-1 if inv%2 else 1)
        for i,j in enumerate(p):x*=a[i][j]
        t+=x
    return t
