def spectral_power(a, exponent):
    vals, vecs = np.linalg.eigh((a+a.conj().T)/2)
    assert vals.min() > 0, vals
    return (vecs * vals**exponent) @ vecs.conj().T

def kernel(x, y, t):

def choi(fn, n):
    out = np.zeros((n*n,n*n), complex)
    for a in range(n):
        for b in range(n):
            e = np.zeros((n,n), complex)
            e[a,b] = 1
            out += np.kron(fn(e),e)
    return out
