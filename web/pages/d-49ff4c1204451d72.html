"""Small independent matrix checks for TRANSVERSE_GATE.txt, not proof.

Uses NumPy only. Includes k=0, m=0, physical null channels and t=m edge.
Does not rerun any qubit precision check or test an asymptotic theorem.
"""
import json
import math
from pathlib import Path

import numpy as np


def falling(a, ell):
    ans = 1
    for j in range(ell):
        ans *= a - j
    return ans


def matrices(k, m, nmax):
    states = [(u, n-u) for n in range(nmax+1) for u in range(n+1)]
    index = {uv:i for i, uv in enumerate(states)}
    f = len(states)
    eye_f = np.eye(f)
    eye_s = np.eye(k+1)
    bos = []
    for which in (0, 1):
        mat = np.zeros((f, f))
        for col, uv in enumerate(states):
            if uv[which]:
                target = list(uv)
                target[which] -= 1
                mat[index[tuple(target)], col] = math.sqrt(uv[which])
        bos.append(np.kron(eye_s, mat))
    number = np.kron(eye_s, np.diag([sum(uv) for uv in states]))
    sp = np.zeros((k+1, k+1))
    for r in range(1, k+1):
        sp[r-1, r] = math.sqrt(r*(k-r+1))
    spin = [[np.kron(np.diag([m+k-r for r in range(k+1)]), eye_f),
             np.kron(sp, eye_f)],
            [np.kron(sp.T, eye_f),
             np.kron(np.diag([m+r for r in range(k+1)]), eye_f)]]
    e12 = spin[0][1] - bos[1].T @ bos[0]
    jz = (spin[0][0]-spin[1][1]-bos[0].T@bos[0]+bos[1].T@bos[1])/2
    casimir = jz@jz+(e12@e12.T+e12.T@e12)/2
    projections = {}
    for n in range(nmax+1):
        ix = np.flatnonzero(np.diag(number) == n)
        eig, vec = np.linalg.eigh(casimir[np.ix_(ix, ix)])
        for s in range(min(k, n)+1):
            j = (k+n-2*s)/2
            keep = np.abs(eig-j*(j+1)) < 1e-8
            assert int(keep.sum()) == int(2*j+1)
            p = np.zeros(number.shape)
            p[np.ix_(ix, ix)] = vec[:, keep] @ vec[:, keep].T
            projections[s, n] = p
    gram = np.zeros(number.shape)
    gsqrt = np.zeros(number.shape)
    ginvsqrt = np.zeros(number.shape)
    physical = np.zeros(number.shape)
    for (s, n), p in projections.items():
        gamma = falling(m+k+1, s)*falling(m, n-s)
        assert gamma >= 0
        gram += gamma*p
        if gamma:
            gsqrt += math.sqrt(gamma)*p
            ginvsqrt += p/math.sqrt(gamma)
            physical += p
    dyson = [sum(spin[i][j]@bos[j] for j in (0, 1))-number@bos[i]
             for i in (0, 1)]
    roots_dag = [gsqrt@a.T@ginvsqrt for a in bos]
    return bos, number, spin, projections, gram, dyson, roots_dag, physical


def norm(mat):
    return float(np.linalg.norm(mat, 2))


def main():
    max_gram = max_lie = max_formula = 0.0
    fixtures = 0
    for k in range(4):
        for m in range(4):
            a, num, spin, ps, gram, dyson, bd, phys = matrices(k, m, k+m+1)
            for i in (0, 1):
                max_gram = max(max_gram, norm(gram@a[i].T-dyson[i].T@gram))
                formula = np.zeros(gram.shape)
                for (s, n), p in ps.items():
                    t = n-s
                    if t>m:
                        continue
                    out0 = ps.get((s, n+1))
                    out1 = ps.get((s+1, n+1))
                    if out0 is not None:
                        formula += math.sqrt(m-t)*out0@a[i].T@p
                    if out1 is not None:
                        formula += math.sqrt(m+k+1-s)*out1@a[i].T@p
                max_formula = max(max_formula, norm(formula-bd[i]))
                for j in (0, 1):
                    rhs = spin[i][j]-a[j].T@a[i]-(num if i==j else 0)
                    err = bd[i].T@bd[j]-bd[j]@bd[i].T-rhs
                    max_lie = max(max_lie, norm(phys@err@phys))
                    max_lie = max(max_lie, norm(bd[i]@bd[j]-bd[j]@bd[i]))
            fixtures += 1
    assert max_gram < 1e-9
    assert max_formula < 1e-10
    assert max_lie < 1e-10

    remainder_cases = []
    z = (0.3+0.2j, -0.1+0.4j)
    z0 = math.sqrt(sum(abs(zi)**2 for zi in z))
    for k in (0, 1, 3):
        for m in (16, 30):
            cutoff = 3
            assert k+cutoff+1 <= m/2
            a, num, spin, ps, gram, dyson, bd, phys = matrices(k, m, cutoff+1)
            q = np.diag((np.diag(num)<=cutoff).astype(float))
            ncopy = k+2*m
            h = 1j*sum(z[i]*bd[i].T-z[i].conjugate()*bd[i]
                        for i in (0, 1))/math.sqrt(ncopy)
            for order in (1, 2):
                cd = []
                for i in (0, 1):
                    mat = np.zeros(gram.shape)
                    for (s, n), p in ps.items():
                        if n>=cutoff:
                            continue
                        for sout, x in ((s, -(n-s)/m), (s+1, (k+1-s)/m)):
                            out = ps.get((sout, n+1))
                            if out is None:
                                continue
                            coeff = 1+x/2-(x*x/8 if order==2 else 0)
                            mat += math.sqrt(m)*coeff*out@a[i].T@p
                    cd.append(mat)
                hc = 1j*sum(z[i]*cd[i].T-z[i].conjugate()*cd[i]
                             for i in (0, 1))/math.sqrt(ncopy)
                actual = norm(q@h@q-hc)
                ratio = (k+cutoff+1)/m
                bound = 8*z0*math.sqrt(m/ncopy)*math.sqrt(cutoff+1)*ratio**(order+1)
                assert actual <= bound+1e-12
                remainder_cases.append({'k':k,'m':m,'K':cutoff,'order':order,
                                        'actual':actual,'bound':bound})
    record = {
        'status':'FINITE-EVIDENCE',
        'scope':'Gram adjointness, polar root formula, Lie/boundary conventions and cutoff Taylor norm only',
        'gram_physical_fixtures':fixtures,
        'max_gram_adjoint_error':max_gram,
        'max_root_formula_error':max_formula,
        'max_physical_lie_error':max_lie,
        'cutoff_remainder_cases':remainder_cases,
        'asymptotic_or_external_validation':False,
        'qubit_tests_rerun':False,
    }
    path = Path(__file__).with_suffix('.json')
    path.write_text(json.dumps(record, indent=2)+'\n')
    print(json.dumps({k:v for k,v in record.items() if k!='cutoff_remainder_cases'},indent=2))


if __name__ == '__main__':
    main()
