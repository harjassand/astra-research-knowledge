def parity_coefficient(types):
    return sum(all(t=='a' or t==b for t,b in zip(types,bs)) for bs in itertools.product('cd',repeat=len(types)) if bs.count('d')%2==0)

def gf2_rank(cols,n):
    piv={}
    for col in cols:
        while col:
            p=col.bit_length()-1
            if p in piv:col^=piv[p]
            else:piv[p]=col;break
    return len(piv)
