def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()
def mat(rows):
    return s.Matrix([[s.Rational(v) for v in row] for row in rows])
def ldlt(A):
    assert A==A.T
    L=s.eye(A.rows)
    pivots=[]
    for j in range(A.rows):
        d=A[j,j]-sum(L[j,k]**2*pivots[k] for k in range(j))
        assert d>0
        pivots.append(d)
        for i in range(j+1,A.rows):
            L[i,j]=(A[i,j]-sum(L[i,k]*L[j,k]*pivots[k] for k in range(j)))/d
    assert L*s.diag(*pivots)*L.T==A
    return L,pivots
def primitive(v):
    scale=s.ilcm(*[a.q for a in v])
    entries=[int(a*scale) for a in v]
    divisor=reduce(gcd,[abs(a) for a in entries])
    entries=[a//divisor for a in entries]
    if next(a for a in entries if a)!=abs(next(a for a in entries if a)):
        entries=[-a for a in entries]
    return s.Matrix(entries)
def serialized(A):
    return [[str(v) for v in row] for row in A.tolist()]

input_path=PEER/'SOS_OBSERVED_INPUT.json'
output_path=PEER/'SOS_LEARNER_OUTPUT.json'
space_path=PEER/'SOS_QUADRATIC_SPACE_CANDIDATE.json'
freeze=json.loads((PEER/'SOS_LEARNER_FREEZE.json').read_text())

proposal=json.loads(space_path.read_text())
assert proposal['input_hash']==input_hash and proposal['learner_output_hash']==output_hash
n=data['n']
pairs=[(i,i) for i in range(n)]+[(i,j) for i in range(n) for j in range(i+1,n)]
N=len(pairs)
features=[]
basis=[]
for i,j in pairs:
    e=[0]*n
    e[i]+=1
    e[j]+=1
    features.append(tuple(e))
    B=s.zeros(n)
    if i==j:
        B[i,i]=1
    else:
        B[i,j]=B[j,i]=s.Rational(1,2)
    basis.append(B)
space=mat(proposal['candidate_space'])
assert space.shape==(N,n) and space.rank()==n
quadratics=[sum((space[i,j]*basis[i] for i in range(N)),s.zeros(n)) for j in range(n)]
Mc=mat(certificate['M'])
mc=s.Matrix([Mc[i,j] if i==j else 2*Mc[i,j] for i,j in pairs])
# Acquire an SPD member by exact projection in the rational Mc inner product;
# no reported H or latent metric is supplied here. This projection has the
# separately proved guarantee after exact D_L recovery and small confidence.
Minv=Mc.inv()
W=s.Matrix(N,N,lambda i,j:s.trace(Minv*basis[i]*Minv*basis[j]))
ldlt(W)
a=s.Rational(certificate['a_outward'])
zeta=a/(1-a)
assert n*zeta*zeta<(1-a)**2
projected=space*(space.T*W*space).inv()*space.T*W*mc
H=sum((projected[i]*basis[i] for i in range(N)),s.zeros(n))
_,H_pivots=ldlt(H)

trials=[]
chosen=None
bad_parameter_bound=(n-1)*n*(n-1)//2
for t in range(bad_parameter_bound+1):
    A=sum((s.Integer(t)**j*quadratics[j] for j in range(n)),s.zeros(n))
    operator=H.inv()*A
    characteristic=operator.charpoly().as_poly()
    unit,factors=s.factor_list(characteristic)
    success=len(factors)==n and all(p.degree()==1 and multiplicity==1 for p,multiplicity in factors)
    trials.append({'t':t,'distinct_rational_eigenvalues':success})
    if success:
        eigenvalues=sorted([-p.all_coeffs()[1]/p.all_coeffs()[0] for p,multiplicity in factors])
        columns=[]
        for value in eigenvalues:
            kernel=(operator-value*s.eye(n)).nullspace()
            assert len(kernel)==1
            columns.append(primitive(kernel[0]))
        V=s.Matrix.hstack(*columns)
        assert V.det()!=0
        chosen=t
        break
assert chosen is not None
for B in quadratics:
    transformed=V.T*B*V
    assert transformed==s.diag(*[transformed[i,i] for i in range(n)])
assert (V.T*H*V).is_diagonal()
frame_inverse=V.inv()

x=s.symbols('x:'+str(n))
z=s.symbols('z:'+str(n))
f=sum(s.Rational(term['coefficient'])*s.prod(x[i]**term['exponent'][i] for i in range(n)) for term in data['terms'])
substitution={x[i]:sum(V[i,j]*z[j] for j in range(n)) for i in range(n)}
aligned=s.Poly(s.expand(f.subs(substitution,simultaneous=True)),*z)
Q=s.zeros(n)
for exponent,coefficient in aligned.terms():
    support=[i for i,v in enumerate(exponent) if v]
    if len(support)==1 and exponent[support[0]]==4:
        Q[support[0],support[0]]=coefficient
    elif len(support)==2 and all(exponent[i]==2 for i in support):
        i,j=support
        Q[i,j]=Q[j,i]=coefficient/2
    else:
        raise AssertionError('terminal quartic contains a non-even mass term')

mass=s.Matrix([v*v for v in z])
assert s.expand(aligned.as_expr()-(mass.T*Q*mass)[0])==0

# Independently export a rational positive-square identity in observed x.
ell=[sum(frame_inverse[i,j]*x[j] for j in range(n)) for i in range(n)]
square_coefficients=s.zeros(N,n)
for j in range(n):
    quadratic=s.Poly(s.expand(sum(Lq[i,j]*ell[i]**2 for i in range(n))),*x)
    for i,exponent in enumerate(features):
        square_coefficients[i,j]=quadratic.coeff_monomial(exponent)
terminal_Gram=square_coefficients*s.diag(*weights)*square_coefficients.T
observed={tuple(t['exponent']):s.Rational(t['coefficient']) for t in data['terms']}
recovered={}
for i,e in enumerate(features):
    for j,d in enumerate(features):
        exponent=tuple(a+b for a,b in zip(e,d))
        recovered[exponent]=recovered.get(exponent,0)+terminal_Gram[i,j]
recovered={e:v for e,v in recovered.items() if v!=0}
assert recovered==observed
