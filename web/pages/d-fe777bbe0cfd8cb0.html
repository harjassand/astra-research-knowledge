def cadd(a,b): return (a[0]+b[0],a[1]+b[1])
def csub(a,b): return (a[0]-b[0],a[1]-b[1])
def cmul(a,b): return (a[0]*b[0]-a[1]*b[1],a[0]*b[1]+a[1]*b[0])
def cconj(a): return (a[0],-a[1])
def cabs2(a): return a[0]*a[0]+a[1]*a[1]
def cd(v): return (Fraction(v[0]),Fraction(v[1])) if isinstance(v,tuple) else (Fraction(v),Fraction(0))
def det(a):
    n=len(a)
    if n==0:return (Fraction(1),Fraction(0))
    if n==1:return a[0][0]
    out=(Fraction(0),Fraction(0))
    for j,x in enumerate(a[0]):
        minor=[row[:j]+row[j+1:] for row in a[1:]]
        term=cmul(x,det(minor))
        out=cadd(out,term if j%2==0 else (-term[0],-term[1]))
    return out

def direct_prefix(F,k,U=(),D=(),H=()):
    n=len(F); f=[[cd(x) for x in row] for row in F]
    U,D,H=set(U),set(D),set(H)
    total=Fraction(0)
    for I in combinations(range(n),k):
        I=set(I)
        if not U<=I or I&D or I&H: continue
        for J0 in combinations(range(n),k):
            J=set(J0)
            if I&J or not D<=J or J&H or (I|J)&H: continue
            total += cabs2(det([[f[i][j] for j in sorted(J)] for i in sorted(I)]))
    return total

def prefix_assignments(n):

        D={i for i,x in enumerate(labels) if x==2}
        H={i for i,x in enumerate(labels) if x==3}
        yield U,D,H
