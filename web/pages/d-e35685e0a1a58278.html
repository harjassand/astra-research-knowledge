This deliberately reuses c06_s03's generic local exchange implementation

def eye(n):
    return [[F(int(i == j)) for j in range(n)] for i in range(n)]


def mm(A, B):
    return [[sum((A[i][k] * B[k][j] for k in range(len(B))), F(0))
             for j in range(len(B[0]))] for i in range(len(A))]


def add(A, B):
    return [[A[i][j] + B[i][j] for j in range(len(A[0]))]
            for i in range(len(A))]


def scale(c, A):
    return [[c * x for x in row] for row in A]


def p2(A, h):
    return add(add(eye(len(A)), scale(h, A)), scale(h * h / 2, mm(A, A)))


def p2_edge_table(h, alpha, gamma):
    h, alpha, gamma = map(F, (h, alpha, gamma))
    A = [[3 * alpha + gamma, 0, 0, 0],
         [0, 3 * alpha - gamma, 2 * alpha, 0],
         [0, 2 * alpha, 3 * alpha - gamma, 0],
         [0, 0, 0, 3 * alpha + gamma]]
    G = p2(A, h)
    return {mask: G[i][j] for mask, (i, j) in {
        3: (0, 0), 12: (3, 3), 6: (1, 1), 9: (2, 2),
        5: (1, 2), 10: (2, 1)}.items() if G[i][j] > 0}


def p2_field_table(h, b, c):
    h, b, c = map(F, (h, b, c))
    r = b + abs(c)
    G = p2([[r + c, b], [b, r - c]], h)
    return {3: G[0][1], 12: G[1][0],
            5: G[0][0] / 2, 9: G[0][0] / 2,
            6: G[1][1] / 2, 10: G[1][1] / 2}


def compile_p2_trace(qubits, gates):
    factors, timeline, dummy = [], [[] for _ in range(qubits)], []
    fields = 0
    for index, (kind, vertices, h, a, g) in enumerate(gates):
        start = 4 * index
        coords = tuple(range(start, start + 4))
        if kind == "edge":
            u, v = vertices
            assert u != v and 0 <= u < qubits and 0 <= v < qubits
            table = p2_edge_table(h, a, g)
            for j, vertex in enumerate((u, v)):
                timeline[vertex].append((start + j, start + 2 + j))
        elif kind == "field":
            (u,) = vertices
            assert 0 <= u < qubits
            table = p2_field_table(h, a, g)
            timeline[u].append((start, start + 1))
            dummy.extend((start + 2, start + 3))
            fields += 1
        else:
            raise ValueError(kind)
        factors.append(chain.LocalFactor(coords, table))
    groups, inactive = [], 0
    for visits in timeline:
        if not visits:
            inactive += 1
            continue
        for j, (row, _) in enumerate(visits):
            groups.append(((visits[j - 1][1], row), 1))
    if dummy:
        groups.append((tuple(dummy), fields))
    return chain.TraceInstance(4 * len(gates), factors, groups, inactive)


def gate_matrix(qubits, gate):
    kind, vertices, h, a, g = gate
    h, a, g = map(F, (h, a, g))
    dim = 1 << qubits
    A = [[F(0) for _ in range(dim)] for _ in range(dim)]
    if kind == "edge":
        u, v = vertices
        for bits in range(dim):
            zu = 1 - 2 * ((bits >> u) & 1)
            zv = 1 - 2 * ((bits >> v) & 1)
            A[bits][bits] = 3 * a + g * zu * zv
            if zu != zv:
                A[bits ^ (1 << u) ^ (1 << v)][bits] = 2 * a
    else:
        (u,) = vertices
        r = a + abs(g)
        for bits in range(dim):
            z = 1 - 2 * ((bits >> u) & 1)
            A[bits][bits] = r + g * z
            A[bits ^ (1 << u)][bits] = a
    return p2(A, h)


def mat_product(A, B):
    return mm(A, B)


def exact_trace(qubits, gates):
    dim = 1 << qubits
    W = eye(dim)
    for gate in gates:
        W = mat_product(W, gate_matrix(qubits, gate))
    return sum((W[i][i] for i in range(dim)), F(0))


def hard_coefficient(instance):
    total = F(0)
    local_choices = []
    for factor in instance.factors:
        local_choices.append([(frozenset(u for j, u in enumerate(factor.coordinates)
                                         if mask >> j & 1), weight)
                              for mask, weight in factor.table.items()])
    ground = frozenset(range(instance.n))
    for parts in product(*local_choices):
        S = frozenset().union(*(chosen for chosen, _ in parts))
        if instance.nu(ground - S):
            total += chain.product_fraction(weight for _, weight in parts)
    return total * (1 << instance.inactive_qubits)


def global_row(instance, state, z, t, pins):
    """Literal all-coordinate implementation of source Eq. (19)."""
    S, T = map(set, state)
    row = {state: F(1)}
    for side, selected, other in ((0, S, T), (1, T, S)):
        included, excluded = pins[2 * side:2 * side + 2]
        oracle = instance.mu if side == 0 else instance.nu
        for u in selected:
            rest = selected - {u}
            weights = []
            for v in range(instance.n):
                candidate = rest | {v}
                if (v not in rest and included <= candidate and
                        not excluded.intersection(candidate)):
                    w = z[v] * oracle(candidate)
                    if w > 0:
                        weights.append((v, w))
            denominator = sum((w for _, w in weights), F(0))
            assert denominator > 0
            factor = (F(1, 2) if u in other else t / 2) / instance.n
            for v, w in weights:
                if v == u:
                    continue
                moved = frozenset(rest | {v})
                target = (moved, frozenset(T)) if side == 0 else (frozenset(S), moved)
                prob = factor * w / denominator
                row[target] = row.get(target, F(0)) + prob
                row[state] -= prob
    return row


def run_kernel_case(gates):
    instance = compile_p2_trace(2, gates)
    assert instance.complement_symmetric()
    assert hard_coefficient(instance) == exact_trace(2, gates)
    states = chain.enumerate_supports(instance, pair_cap=1000)
    z = [F(1)] * instance.n
    t = F(1, 256 * instance.n * instance.n)
    weights, rows = {}, {}
    for state in states:
        sampler = chain.LocalExchange(instance, z, t, state=state)
        weights[state] = sampler.supported_weight()
        rows[state] = sampler.kernel_row()
        assert rows[state] == global_row(instance, state, z, t, (frozenset(),) * 4)
        assert sum(rows[state].values(), F(0)) == 1
        assert min(rows[state].values()) >= 0
    for state, row in rows.items():
        for target, prob in row.items():
            assert target in weights
            assert weights[state] * prob == weights[target] * rows[target].get(state, F(0))
    Z = sum(weights.values(), F(0))
    phard = sum((w for (S, T), w in weights.items() if S.isdisjoint(T)), F(0)) / Z
    assert phard >= F(3, 4)
    return {"ground_size": instance.n, "supported_pairs": len(states),
            "hard_probability": str(phard), "all_rows_match_literal": True,
            "all_directed_detailed_balance_checks_pass": True,
            "coefficient_equals_dense_trace": True}


def run():
    edge = ("edge", (0, 1), F(1, 7), F(2, 3), F(1, 4))
    field = ("field", (0,), F(1, 9), F(3, 5), F(0))
    # A single W W* layer, preserving exact rational p2 entries.
    palindrome = [edge, field, field, edge]
    projected_case = compile_p2_trace(2, palindrome)
    assert projected_case.complement_symmetric()
    assert hard_coefficient(projected_case) == exact_trace(2, palindrome)

    # Smaller cases permit all supported pair states and every transition row.
    edge_only = [("edge", (0, 1), F(1, 10), F(1), F(1, 3))]
    field_only = [("field", (0,), F(1, 11), F(2), F(0))]
    mixed = [edge, field]
    longitudinal = [("field", (0,), F(1, 11), F(2), F(-1, 3))]
    nonsymmetric = compile_p2_trace(2, longitudinal)
    assert not nonsymmetric.complement_symmetric()
    results = {"status": "PASS_EXACT_FINITE_P2_COMPLEMENT_CHAIN",
               "kernel_cases": {"edge": run_kernel_case(edge_only),
                                "field_c_zero": run_kernel_case(field_only),
                                "edge_plus_field_c_zero": run_kernel_case(mixed)},
               "palindromic_p2_layer": {
                   "gate_count": len(palindrome),
                   "ground_size": projected_case.n,
                   "complement_symmetric": projected_case.complement_symmetric(),
                   "coefficient_equals_dense_trace": True},
               "nonzero_longitudinal_field_boundary": {
                   "complement_symmetric": nonsymmetric.complement_symmetric(),
                   "scope": "confirms only that this shortcut's symmetry premise fails; does not refute field-general p2 counting"},
               "scope": "finite rational lift/kernel identities only; no all-size gap, FPRAS wrapper, or practical runtime claim"}
    return results
