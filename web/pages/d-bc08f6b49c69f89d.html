def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
def mat(rows):
    return s.Matrix([[s.Rational(value) for value in row] for row in rows])
def ldlt(A, positive=False):
    assert A == A.T
    n=A.rows
    L=s.eye(n)
    D=[]
    for j in range(n):
        pivot=A[j,j]-sum(L[j,k]*L[j,k]*D[k] for k in range(j))
        assert pivot != 0
        if positive:
            assert pivot > 0
        D.append(pivot)
        for i in range(j+1,n):
            L[i,j]=(A[i,j]-sum(L[i,k]*L[j,k]*D[k] for k in range(j)))/pivot
    assert L*s.diag(*D)*L.T == A
    return D
def sqrt_upper(q, denominator=10**12):
    q=s.Rational(q)
    assert q >= 0
    numerator=int(q.p)*denominator*denominator
    div=int(q.q)
    k=isqrt(numerator//div)
    if k*k*div < numerator:
        k+=1
    upper=s.Rational(k,denominator)
    assert upper*upper >= q
    return upper

N=n*(n+1)//2
assert n==5 and N==15
pairs=[(i,i) for i in range(n)]+[(i,j) for i in range(n) for j in range(i+1,n)]
features=[]
basis=[]
for i,j in pairs:
    e=[0]*n
    e[i]+=1
    e[j]+=1
    features.append(tuple(e))
    A=s.zeros(n)
    if i==j:
        A[i,i]=1
    else:
        A[i,j]=A[j,i]=s.Rational(1,2)
    basis.append(A)

M=mat(certificate['M'])
G=mat(certificate['G'])
assert M.shape==(n,n) and G.shape==(N,N)
assert M==M.T and G==G.T
metric_coeff=s.Matrix([M[i,i] if i==j else 2*M[i,j] for i,j in pairs])
S=G-metric_coeff*metric_coeff.T
assert S==mat(certificate['residual_Gram'])
mp=ldlt(M,True)
sp=ldlt(S,True)

observed={tuple(t['exponent']):s.Rational(t['coefficient']) for t in data['terms']}
gram_polynomial={}
for i,e in enumerate(features):
    for j,d in enumerate(features):
        exp=tuple(a+b for a,b in zip(e,d))
        gram_polynomial[exp]=gram_polynomial.get(exp,0)+G[i,j]
gram_polynomial={e:v for e,v in gram_polynomial.items() if v!=0}
assert gram_polynomial==observed

dual={tuple(t['exponent']):s.Rational(t['value']) for t in certificate['lambda']}

Minv=M.inv()

assert zeta==s.Rational(certificate['zeta'])

# Observable quadratic coefficient inner product in Mc whitening. All
# entries are rational; no matrix square root or hidden metric is evaluated.
W=s.Matrix(N,N,lambda i,j:s.trace(Minv*basis[i]*Minv*basis[j]))

assert Tc>0

inertia_checks=[]

for gamma in [s.Rational(1,2),s.Rational(1,4),s.Rational(1,8),
              s.Rational(1,16),s.Rational(1,32),s.Rational(1,64)]:
    try:
        pivots=ldlt(S-gamma*Winv)
    except AssertionError:
        inertia_checks.append({'gamma':str(gamma),'status':'zero-pivot UNKNOWN'})
        continue

    negative=sum(int(bool(p<0)) for p in pivots)

           'gamma_above_Gram_error':bool(gamma>error_candidate)}
    inertia_checks.append(entry)
    if positive==n-1 and gamma>error_candidate:
        chosen=gamma
        break

# Canonical contact moment is an algebraic identity, not fixture contacts
# supplied to this checker or an algorithm for discovering them.
moment=s.zeros(N)
for signs in product([-1,1],repeat=n):
    v=s.Matrix([s.Rational(1,n) if i==j else s.sqrt(2)*signs[i]*signs[j]/n for i,j in pairs])
    moment+=v*v.T/(2**n)

    expected[i,i]=s.Rational(2,n*n)
assert moment==expected and moment.rank()==N-(n-1)

# Derive diagonal Gram/tensor norm by polynomial multinomial coefficients.
