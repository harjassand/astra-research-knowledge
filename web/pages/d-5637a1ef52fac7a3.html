F=[[Q(0) for _ in range(6)] for _ in range(6)]
for i in range(3):
 for j in range(3):
  F[i][j]=c*P[i][j]; F[i][j+3]=-s*P[i][j]

V=[]

def det_cols(sel):
 total=0
 for p in perms:
  inv=sum(p[i]>p[j] for i in range(6) for j in range(i+1,6))
  val=(-1 if inv&1 else 1)
  for i,j in enumerate(p): val*=V[i][sel[j]]
  total+=val
 return total
