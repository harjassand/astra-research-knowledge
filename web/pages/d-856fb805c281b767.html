def ceilf(x):return (x.numerator+x.denominator-1)//x.denominator
def ceil_sqrt(x):

def exp_upper(x,J=40):
    s=sum(x**j/F(factorial(j)) for j in range(J+1))
    nxt=x**(J+1)/F(factorial(J+1))
    return s+nxt/(1-x/F(J+2))
