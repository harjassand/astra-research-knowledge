def enc(x):
    if isinstance(x, F):
        return str(x)
    if isinstance(x, (fmpq,)):
        return str(x)
    if isinstance(x, np.generic):
        return x.item()
    if isinstance(x, np.ndarray):
        return enc(x.tolist())
    if isinstance(x, Path):
        return str(x.relative_to(BASE))
    if isinstance(x, dict):
        return {str(k): enc(v) for k, v in x.items()}
    if isinstance(x, (list, tuple)):
        return [enc(v) for v in x]
    return x

def filehash(p):

    return h.hexdigest()

def posdef(a):
    """Exact Sylvester/LDL certificate; no numerical tolerance."""
    a = [row[:] for row in a]
    pivots = []
    for k in range(len(a)):
        p = a[k][k]
        pivots.append(p)
        if p <= 0:
            return False, pivots
        for i in range(k + 1, len(a)):
            for j in range(i, len(a)):
                a[j][i] = a[i][j] = a[i][j] - a[i][k] * a[k][j] / p
    return True, pivots


def metric_shift(a, nu, lam, sign=1):
    return [[sign * a[i][j] + (lam * nu[i] if i == j else F(0))
             for j in range(len(nu))] for i in range(len(nu))]


def eigen_bound(a, nu, kind='upper'):
    """Numeric proposal + exact weighted LDL; at most 8 deterministic margins."""
    v = np.array([[float(x) for x in row] for row in a])
    weights = np.sqrt(np.array([float(x) for x in nu]))
    v = v / weights[:, None] / weights[None, :]
    eig = np.linalg.eigvalsh(v)
    guess = float(eig[-1] if kind == 'upper' else eig[0])
    den = 1 << 40
    for j in range(8):
        margin = max(1., abs(guess)) * 2. ** (-34 + 3 * j)
        if kind == 'upper':
            lam = F(math.ceil((guess + margin) * den), den)
            ok, piv = posdef(metric_shift(a, nu, lam, -1))
        else:
            lam = F(math.floor((guess - margin) * den), den)
            ok, piv = posdef(metric_shift(a, nu, -lam, +1))
        if ok:
            return lam, dict(kind=kind, proposed_spectrum=eig.tolist(), value=lam,
                             exact_LDL_pivots=piv, margin_index=j)
    raise ArithmeticError('failed exact spectral proposal verification')


def opnorm(a, nu):
    lp, cp = eigen_bound(a, nu)
    lm, cm = eigen_bound(scale(a, -1), nu)
    return max(lp, lm, F(0)), dict(positive=cp, negative=cm)


def sqrtup(x, bits=88):
    if x < 0:
        raise ArithmeticError('negative square root')
    den = 1 << bits
    k = math.isqrt(x.numerator * den * den // x.denominator)
    if k * k * x.denominator < x.numerator * den * den:
        k += 1
    return F(k, den)


def logup(x, terms=84):
    """atanh series, range reduction to [1,2), positive exact tail."""
    if x < 1:
        raise ArithmeticError('log interface requires x>=1')
    s = 0
    while x >= 2:
        x /= 2
        s += 1

    def partial(y):
        t = (y - 1) / (y + 1)
        p = t
        val = F(0)
        for k in range(terms):
            val += 2 * p / (2 * k + 1)
            p *= t * t
        tail = 2 * p / ((2 * terms + 1) * (1 - t * t))
        return val + tail

    upper = s * partial(F(2)) + partial(x)
    grid = 1 << 100
    # Outward compression avoids carrying the series' enormous denominators
    # into every downstream certificate; the exact tail proof is unchanged.
    return F((upper.numerator * grid + upper.denominator - 1)//upper.denominator, grid)


def empirical_bernstein(count, dim, v, alpha):
    # Two-sided at failure alpha; one-sided theorem invoked at alpha/2.
    L = logup(F(count * dim, count - 1) / (alpha / 2))
    J = logup(F(2 * count * dim) / (alpha / 2))
    terms = [L / (3 * count), sqrtup(2 * v * L / count),
             (sqrtup(F(5, 3)) + 1) * sqrtup(L * J) / count]
    return sum(terms), dict(count=count, dimension=dim, two_sided_failure=alpha,
                            variance_upper=v, log_L_upper=L, log_J_upper=J,
                            radius_terms=terms, series_terms=84, log_outward_bits=100,
                            sqrt_bits=88)


def quaternion_rotation(q):
    """Homogeneous rational quaternion; q order scalar,x,y,z."""
    w, x, y, z = map(F, q)
    norm = w*w + x*x + y*y + z*z
    if norm == 0:
        raise ArithmeticError('zero quaternion')
    return scale([[w*w+x*x-y*y-z*z, 2*(x*y-w*z), 2*(x*z+w*y)],
                  [2*(x*y+w*z), w*w-x*x+y*y-z*z, 2*(y*z-w*x)],
                  [2*(x*z-w*y), 2*(y*z+w*x), w*w-x*x-y*y+z*z]], 1/norm)


def rational_frame(v):
    v = v.copy()
    if np.linalg.det(v) < 0:
        v[:, 0] *= -1
    xyzw = Rotation.from_matrix(v).as_quat()
    nums = [round(float(x) * (1 << 30)) for x in [xyzw[3], *xyzw[:3]]]
    R = quaternion_rotation(nums)
    assert mm(tr(R), R) == eye(3)
    return R, nums

def observed_cache(raw, offset, count, name):

    k = len(atoms)
    pairs = np.bincount(labels[0::2]*k + labels[1::2], minlength=k*k).reshape(k, k)
    assert int(counts.sum()) == count and int(pairs.sum()) == count//2
    assert np.array_equal(pairs.sum(0) + pairs.sum(1), counts)
    for suffix, arr in [('atoms', atoms), ('counts', counts), ('ordered_pairs', pairs)]:
        np.save(OUT / f'{name}_{suffix}.npy', arr)
    dhash = hashlib.sha256(d.tobytes(order='C')).hexdigest()
    lhash = hashlib.sha256(labels.tobytes(order='C')).hexdigest()
    cache = dict(atom_vectors=atoms, counts=counts, pairs=pairs, difference_count=count,
                 distinct_difference_atoms=k, difference_order_sha256=dhash,

def exact_rankone_statistics(cache, features, norms, denom, label):
    """Exact mean and genuine paired matrix variance in orthogonal basis.

    features[i,p] = <B_p,phi_i>*denom; basis norms are norms[p].
    Stored matrices are bilinear: S_pq=<B_p,K B_q>. For pilot phi=Y,
    denom=sqrt(2)*rawden; its powers below are specified by denom_squared.
    """
    t = time.monotonic()
    f = [[int(x) for x in row] for row in features]
    counts = [int(x) for x in cache['counts']]
    dim = len(norms)
    sample_count = cache['difference_count']
    k = len(f)
    den2 = denom  # square of feature common denominator, always rational integer.
    S = [[F(sum(counts[i]*f[i][p]*f[i][q] for i in range(k)), sample_count*den2)
          for q in range(dim)] for p in range(dim)]
    # First summand: each feature observation appears once in a genuine pair.
    V = [[sum((F(counts[i]*f[i][p]*f[i][q]) *
                    sum((F(f[i][s]*f[i][s])/norms[s] for s in range(dim)), F(0))
                    for i in range(k)), F(0))
          for q in range(dim)] for p in range(dim)]
    C = fmpz_mat(cache['pairs'].tolist())
    for s in range(dim):
        Fs = fmpz_mat([[f[i][p]*f[i][s] for p in range(dim)] for i in range(k)])
        cross = Fs.transpose() * (C * Fs)
        for p in range(dim):
            for q in range(dim):
                V[p][q] -= F(int(cross[p,q]) + int(cross[q,p])) / norms[s]
    V = scale(V, F(1, sample_count*den2*den2))
    # Nonnegative covariance and paired-variance are also exactly certified below.
    mark(label + '_exact_mean_and_paired_variance', t)
    save(label + '_EXACT_STATISTICS.json', dict(basis_norm_squared=norms,
          rankone_mean_bilinear=S, paired_variance_bilinear=V,
          denominator_squared=den2, samples=sample_count,
          operation='observed rankone sums and original paired differences squared'))
    return S, V


def make_whitener(Sigma):
    t = time.monotonic()
    a = np.array([[float(x) for x in row] for row in Sigma])
    ev, eig = np.linalg.eigh(a)
    if min(ev) <= 0:
        raise ArithmeticError('pilot covariance not SPD')
    w = (eig / np.sqrt(ev)) @ eig.T
    wden = 1 << 20
    wnum = [[round(float(x)*wden) for x in row] for row in w]
    W = [[F(x,wden) for x in row] for row in wnum]
    Wi = inv(W)
    wtw = mm(tr(W), W)
    kw2, kwcert = eigen_bound(wtw, NU3)
    lw2, lwcert = eigen_bound(wtw, NU3, 'lower')
    if lw2 <= 0:
        raise ArithmeticError('rounded whitener not certified invertible')
    delta, deltacert = opnorm(add(mm(mm(tr(W), Sigma), W), eye(3), -1), NU3)
    mark('pilot_whitener_and_exact_metric_checks', t)
    return W, Wi, wnum, wden, dict(W=W, W_inverse=Wi, round_bits=20,
               pilot_numerical_eigenvalues=ev.tolist(), K_w_squared_upper=kw2,
               K_w_squared_lower=lw2, K_w_cert=kwcert, lower_cert=lwcert,
               delta_w=delta, delta_cert=deltacert, condition_squared_upper=kw2/lw2)

def propose(S, whitenum, counts, zd):
    proposals = {}
    t = time.monotonic()
    norm = np.sqrt(np.array([float(x) for x in NU]))
    A = np.array([[float(x) for x in row] for row in S])/norm[:,None]/norm[None,:]
    ev, eig = np.linalg.eigh(A)
    top = eig[:,-2:]
    basis = np.array(BASIS, dtype=float)/norm[:,None,None]
    probe = np.outer(np.arange(1,4),np.arange(1,4))
    pc = np.einsum('pij,ij->p',basis,probe)
    form = np.einsum('p,pij->ij',top@(top.T@pc),basis)
    probe_eigs, r = np.linalg.eigh(form)
    R, q = rational_frame(r)
    proposals['eigenmatrix'] = dict(R=R, quaternion_integer=q,
            seconds=time.monotonic()-t, observed_operator_spectrum=ev.tolist(),
            probe_spectrum=probe_eigs.tolist(),
            method='Classical top-2 trace-free eigenmatrix subspace, fixed rank-one probe, diagonalization')
    t = time.monotonic()
    z = np.array(whitenum, dtype=float)/float(zd)/math.sqrt(2)
    weight = np.array(counts, dtype=float)/N
    tensor = np.einsum('i,ia,ib,ic,id->abcd',weight,z,z,z,z,optimize=True)
    tw=tensor.copy(); rot=np.eye(3); initial=sum(tw[i,i,i,i] for i in range(3))
    history=[]
    for sweep in range(30):
        largest=0.
        for i in range(3):
            for j in range(i+1,3):
                aa=(tw[i,i,i,i]+tw[j,j,j,j]-6*tw[i,i,j,j])/4
                bb=tw[i,i,i,j]-tw[i,j,j,j]
                theta=np.arctan2(bb,aa)/4
                largest=max(largest,abs(theta))
                c,s=np.cos(theta),np.sin(theta)
                G=np.eye(3);G[i,i]=G[j,j]=c;G[i,j]=-s;G[j,i]=s
                tw=np.einsum('ia,jb,kc,ld,ijkl->abcd',G,G,G,G,tw,optimize=True)
                rot=rot@G
        history.append(dict(sweep=sweep+1,max_angle=largest,
                            contrast=float(sum(tw[i,i,i,i] for i in range(3)))))
        if largest<1e-11:
            break
    R,q=rational_frame(rot)
    proposals['maxkurt']=dict(R=R,quaternion_integer=q,seconds=time.monotonic()-t,
            sweeps=len(history),sweep_history=history,initial_contrast=float(initial),
            final_contrast=float(sum(tw[i,i,i,i] for i in range(3))),
            method='Established ordinary MaxKurt Jacobi, fixed 30-sweep cap, no convergence theorem')
    save('FROZEN_PROPOSALS.json',proposals)
    mark('both_fixed_classical_proposals_and_freeze', t)
    return proposals


def posterior(S, common, proposal, W, Wi):
    t=time.monotonic()
    R=proposal['R']
    Rbasis=[mm(mm(R,fm(B)),tr(R)) for B in BASIS]
    T=[[dotmat(fm(BASIS[i]),Rbasis[j])/NU[i] for j in range(5)] for i in range(5)]
    Sprime=mm(mm(tr(T),S),T)
    assert mm(mm(tr(T),[[NU[i] if i==j else F(0) for j in range(5)] for i in range(5)]),T)==[[NU[i] if i==j else F(0) for j in range(5)] for i in range(5)]
    top=[row[:2] for row in Sprime[:2]]
    bottom=[row[2:] for row in Sprime[2:]]
    a,ac=eigen_bound(top,NU[:2],'lower')
    c,cc=eigen_bound(bottom,NU[2:])
    b2=sum((Sprime[i][j]**2/(NU[i]*NU[j]) for i in range(2,5) for j in range(2)),F(0))
    b=sqrtup(b2)
    e=common['total_operator_error']
    g=a-c-2*e
    result=dict(a=a,c_obs=c,b=b,cross_Frobenius_squared=b2,g=g,
                top_lower_certificate=ac,bottom_upper_certificate=cc,
                basis_rotation=T,candidate_operator_bilinear=Sprime,
                proposed_physical_mixing_columns=mm(tr(Wi),R),
                proposed_source_linear_forms_columns=mm(W,R))
    if g<=0:
        result.update(status='UNKNOWN_SPECTRAL_SEPARATION',seconds=time.monotonic()-t)
        return result
    eta=(b+e)/g
    r=F(2,3)*eta**2/(1+eta**2)
    zeta=common['zeta']
    s=sqrtup(r)
    sqrtminuslo=1/sqrtup(1/(1-zeta))
    cond=sqrtup(common['condition_squared_upper'])
    plus=1-sqrtminuslo
    minus=sqrtup(1/(1-zeta))-1
    physical=min(F(1),cond*(s+plus)/sqrtminuslo)
    demixing=min(F(1),cond*sqrtup(1+zeta)*(s+minus))
    result.update(eta=eta,r_axis=r,ideal_axis_sine_upper=s,
                  physical_column_sine_upper=physical,source_row_sine_upper=demixing,
                  unique_projective_matching=(r<F(1,2)),
                  status='INFORMATIVE_ALL_AXIS_CERTIFICATE' if r<F(1,2) and max(physical,demixing)<F(1,2)
                         else 'VALID_BUT_UNINFORMATIVE_AMBIGUITY',seconds=time.monotonic()-t)
    return result


def learner(path,den,raw_meta):
    """Only actual raw rows, lattice scale, D/model promise and fixed protocol."""
    raw=np.load(path,mmap_mode='r')
    pilot=observed_cache(raw,0,M,'pilot')
    Sigma,V2=exact_rankone_statistics(pilot,pilot['atom_vectors'],NU3,2*den*den,'pilot')
    v2,cv2=eigen_bound(V2,NU3)
    assert v2>=0
    rad2,eb2=empirical_bernstein(M,3,v2/(U*U),ALPHA2)
    eps2=U*rad2
    W,Wi,wnum,wden,white=make_whitener(Sigma)
    zeta=white['delta_w']+white['K_w_squared_upper']*eps2
    if zeta>=1:
        raise ArithmeticError('UNKNOWN pilot whitening radius>=1')
    save('PILOT_CONFIDENCE.json',dict(Sigmahat=Sigma,raw_paired_variance=V2,
          variance_LDL=cv2,EB=eb2,epsilon2=eps2,zeta=zeta,whitener=white))
    confirm=observed_cache(raw,2*M,N,'confirmation')
    features,whitenum=confirmation_features(confirm['atom_vectors'],wnum)
    zd=den*wden
    S,V4=exact_rankone_statistics(confirm,features,NU,4*zd**4,'confirmation')
    # Feature denominator is 2*(den*wden)^2; square is 4*(den*wden)^4.
    H=F(2,3)*U*U*white['K_w_squared_upper']**2
    v4,cv4=eigen_bound(V4,NU)
    rad4,eb4=empirical_bernstein(N,5,v4/(H*H),ALPHA4)
    eps4=H*rad4
    mhat,cm=eigen_bound(S,NU)
    k=(2*zeta+zeta*zeta)*F(5,2)+F(3,2)*zeta*zeta
    if k>=1:
        raise ArithmeticError('UNKNOWN k>=1')
    e=eps4+k*(mhat+eps4)/(1-k)
    common=dict(epsilon2=eps2,epsilon4=eps4,zeta=zeta,k_transport=k,
                observed_operator_upper=mhat,total_operator_error=e,
                condition_squared_upper=white['condition_squared_upper'],

def exact_population_operator(W,private):
    C=mm(mm(tr(W),private['covariance']),W)
    A=mm(tr(W),private['A'])
    atoms=[[[A[i][k]*A[j][k] for j in range(3)] for i in range(3)] for k in range(3)]
    S=[]
    for p in range(5):
        row=[];Bp=fm(BASIS[p])
        for q in range(5):
            Bq=fm(BASIS[q])
            isotropic=dotmat(Bp,C)*dotmat(Bq,C)+2*dotmat(Bp,mm(mm(C,Bq),C))
            diagonal=sum((private['difference_beta'][k]*dotmat(Bp,atoms[k])*dotmat(Bq,atoms[k]) for k in range(3)),F(0))
            row.append(private['difference_cd']*isotropic+diagonal)
        S.append(row)
    return S


def projective_match(true, proposed):
    # Columns; all projective sine squares exact, one permutation covers all axes.
    cost=[]
    for perm in itertools.permutations(range(3)):
        sins=[]
        for j in range(3):
            a=[true[i][perm[j]] for i in range(3)]
            b=[proposed[i][j] for i in range(3)]
            dot=sum((x*y for x,y in zip(a,b)),F(0))
            sins.append(1-dot*dot/(sum(x*x for x in a)*sum(y*y for y in b)))
        cost.append((max(sins),list(perm),sins))
    return min(cost,key=lambda x:x[0])


def private_evaluate(fit,private):
    t=time.monotonic()
    eps2=fit['common']['epsilon2'];eps4=fit['common']['epsilon4']
    cov_error,covcert=opnorm(add(fit['Sigma'],private['covariance'],-1),NU3)
    pop=exact_population_operator(fit['W'],private)
    op_error,opcert=opnorm(add(fit['S'],pop,-1),NU)
    evaluation=dict(learner_freeze_sha256=fit['freeze_sha256'],
                    true_covariance_error_upper=cov_error,covariance_error_LDL=covcert,
                    pilot_error_event_holds=cov_error<=eps2,
                    true_candidate_population_operator=pop,
                    true_confirmation_operator_error_upper=op_error,
                    operator_error_LDL=opcert,confirmation_error_event_holds=op_error<=eps4,
                    scope='single private finite truth check, not empirical coverage validation')
    rows={}
    true_demixing=tr(inv(private['A']))
    for name,p in fit['posteriors'].items():
        mix=projective_match(private['A'],p['proposed_physical_mixing_columns'])
        demix=projective_match(true_demixing,p['proposed_source_linear_forms_columns'])
        rows[name]=dict(physical_column_max_sine_squared=mix[0],physical_permutation=mix[1],
             physical_per_axis_sine_squared=mix[2],source_row_max_sine_squared=demix[0],
             source_permutation=demix[1],source_per_axis_sine_squared=demix[2],
             physical_advertised_bound_holds=mix[0]<=p.get('physical_column_sine_upper',F(1))**2,
             source_advertised_bound_holds=demix[0]<=p.get('source_row_sine_upper',F(1))**2)
    evaluation['all_controls']=rows
    save('PRIVATE_POST_FREEZE_EVALUATION.json',evaluation)
    mark('private_post_freeze_exact_evaluation',t)
    return evaluation
