def qfloat(x):
    # python-flint 0.8.0 has no fmpq.__float__; Python long true division
    # converts the ratio without converting either giant integer separately.
    return int(x.numerator)/int(x.denominator)

def dot(a,b):

def qfloor(a):

def bits(a):

def sqrt_bounds(a,precision):
    assert a>=0
    den=1<<precision
    integer=math.isqrt((int(a.numerator)<<(2*precision))//int(a.denominator))
    lo,hi=Q(integer,den),Q(integer+1,den)
    assert lo*lo<=a<=hi*hi
    return lo,hi


@dataclass
class IV:
    lo: Q
    hi: Q

    @staticmethod
    def point(a):
        return IV(Q(a),Q(a))

    def __add__(self,other):
        if not isinstance(other,IV): other=IV.point(other)
        return IV(self.lo+other.lo,self.hi+other.hi)

    __radd__=__add__

    def __neg__(self):
        return IV(-self.hi,-self.lo)

    def __sub__(self,other):
        return self+-other if isinstance(other,IV) else self+(-Q(other))

    def __mul__(self,other):
        if not isinstance(other,IV): other=IV.point(other)
        products=[x*y for x in (self.lo,self.hi) for y in (other.lo,other.hi)]
        return IV(min(products),max(products))

    __rmul__=__mul__

    def __truediv__(self,other):
        if not isinstance(other,IV): other=IV.point(other)
        assert not other.lo<=0<=other.hi
        inverse=IV(min(1/other.lo,1/other.hi),max(1/other.lo,1/other.hi))
        return self*inverse

    def square(self):
        upper=max(self.lo*self.lo,self.hi*self.hi)
        lower=Q(0) if self.lo<=0<=self.hi else min(self.lo*self.lo,self.hi*self.hi)
        return IV(lower,upper)

    def sqrt(self,precision):
        assert self.lo>=0
        return IV(sqrt_bounds(self.lo,precision)[0],sqrt_bounds(self.hi,precision)[1])

    def quantize(self,precision):
        den=1<<precision
        return IV(Q(qfloor(self.lo*den),den),Q(-qfloor(-self.hi*den),den))

    def json(self):
        return [qs(self.lo),qs(self.hi)]


def atan_interval(den,precision):
    x=Q(1,den)
    total=Q(0)
    j=0
    term=x
    while True:
        total+=term/(2*j+1)
        j+=1
        term*=-x*x
        tail=abs(term)/(2*j+1)
        if tail<Q(1,1<<(precision+8)):
            return IV(total-tail,total+tail),j


def pi_interval(precision):
    a,ka=atan_interval(5,precision)
    b,kb=atan_interval(239,precision)
    interval=(16*a-4*b).quantize(precision+4)
    assert Q(3)<interval.lo<interval.hi<Q(22,7)
    return interval,{'atan5_terms':ka,'atan239_terms':kb}


def trig_interval(arg,kind,precision):
    arg=arg.quantize(precision+4)
    midpoint=(arg.lo+arg.hi)/2
    uncertainty=(arg.hi-arg.lo)/2
    k=precision//2+12
    exponent=1 if kind=='sin' else 0
    term=midpoint if exponent else Q(1)
    total=term
    for j in range(1,k):
        term*=-midpoint*midpoint/Q((2*j+exponent-1)*(2*j+exponent))
        total+=term
    # Taylor's derivative bound is 1 for every real argument.
    remainder=abs(midpoint)**(2*k+exponent)/Q(math.factorial(2*k+exponent))
    radius=remainder+uncertainty
    return IV(total-radius,total+radius).quantize(precision+4)


def sphere_map(s,t,sign):
    den=1+s*s+t*t
    return [2*s/den,2*t/den,sign*(1-s*s-t*t)/den]


def reset(parameters,delta):
    count=len(parameters)
    lam=delta/(16*count)
    seen=set()
    points=[]
    choices=[]
    for s,t,sign in parameters:
        sr=lam*qfloor(s/lam+Q(1,2))
        tr=lam*qfloor(t/lam+Q(1,2))
        for shift in range(count+1):
            ss=sr+shift*lam
            candidate=sphere_map(ss,tr,sign)
            key=tuple(qs(x) for x in candidate)
            if key not in seen:
                seen.add(key)
                points.append(candidate)
                choices.append({'sign':sign,'input':[qs(s),qs(t)],'rounded':[qs(sr),qs(tr)],
                                'shift':shift,'selected':[qs(ss),qs(tr)]})
                break
        else:
            raise RuntimeError('distinct reset failed')
        assert dot(candidate,candidate)==1
        # d_sphere <= 2 |p-p'|. Avoid evaluating an angle or square root.
        assert 4*((ss-s)**2+(tr-t)**2)<=delta*delta
    return points,{'lambda':qs(lam),'choices':choices}


def reset_points(points,delta):
    parameters=[]
    for x,y,z in points:
        sign=1 if z>=0 else -1
        parameters.append((x/(1+sign*z),y/(1+sign*z),sign))
    return reset(parameters,delta)


def centroid_initializer(q,delta):
    tau=delta/(2**20*q**3)
    precision=(int(tau.denominator).bit_length()-int(tau.numerator).bit_length())+24
    series_length=precision+8
    pi,pi_guard=pi_interval(precision)
    sqrt2=IV(*sqrt_bounds(Q(2),precision+4))
    parameters=[]
    cells=[]
    for sign in (1,-1):
        for j in range(1,q+1):
            m=2*j-1
            r0,r1=Q(j-1,q),Q(j,q)
            mz=1-(r0*r0+r1*r1)/2
            coeff=Q(1)
            integral=Q(0)
            for k in range(series_length):
                if k:
                    coeff*=Q(2*k-3,4*k)
                integral+=coeff*(r1**(2*k+3)-r0**(2*k+3))/(2*k+3)
            # |binomial(1/2,k)|<=1 and r^2/2<=1/2.
            # Integral tail is <=2^(2-L)/q since r<=1 and sqrt2<2.
            tail=Q(1,q*(1<<(series_length-2)))
            area=(sqrt2*integral+IV(-tail,tail))*(Q(2*q*q,m))
            angle=pi/m
            sinc=IV.point(0) if m==1 else trig_interval(angle,'sin',precision)/angle
            for k in range(m):
                if m==1:
                    mx,my=IV.point(0),IV.point(0)
                else:
                    phi=pi*Q(2*k+1,m)
                    mx=area*sinc*trig_interval(phi,'cos',precision)
                    my=area*sinc*trig_interval(phi,'sin',precision)
                norm=(mx.square()+my.square()+IV.point(mz*mz)).sqrt(precision+4)
                schart=(mx/(norm+mz)).quantize(precision)
                tchart=(my/(norm+mz)).quantize(precision)
                s=(schart.lo+schart.hi)/2
                t=(tchart.lo+tchart.hi)/2
                assert (schart.hi-schart.lo)**2+(tchart.hi-tchart.lo)**2 <= (delta/32)**2
                parameters.append((s,t,sign))
                cells.append({'hemisphere':sign,'ring':j,'wedge':k,'wedges':m,
                              'r_bounds':[qs(r0),qs(r1)],'mz':qs(mz),
                              'area_interval':area.json(),'mx_interval':mx.json(),'my_interval':my.json(),
                              'norm_interval':norm.json(),'chart_intervals':[schart.json(),tchart.json()],
                              'chart_midpoints':[qs(s),qs(t)]})
    points,reset_record=reset(parameters,delta)
    return points,{'q':q,'N':2*q*q,'tau':qs(tau),'precision_bits':precision,
                   'binomial_terms':series_length,'pi_interval':pi.json(),'pi_guard':pi_guard,
                   'cells':cells,'reset':reset_record}


def harmonic_basis(T):
    x,y,z=sp.symbols('x y z',real=True)
    basis=[]
    for ell in range(1,T+1):
        for m in range(ell+1):
            H=(x+sp.I*y)**m*sum(
                sp.Rational((-1)**j*math.factorial(2*ell-2*j),
                            2**ell*math.factorial(j)*math.factorial(ell-j)*math.factorial(ell-m-2*j))
                *z**(ell-m-2*j)*(x*x+y*y+z*z)**j for j in range((ell-m)//2+1))
            nu=Q(1,2*ell+1) if m==0 else Q(math.factorial(ell+m),2*(2*ell+1)*math.factorial(ell-m))
            for part in (['real'] if m==0 else ['real','imag']):
                f=sp.expand(sp.re(H) if part=='real' else sp.im(H))
                terms=[(tuple(power),Q(str(coef))) for power,coef in sp.Poly(f,x,y,z).terms()]
                deriv=[]
                for a in range(3):
                    item=[]
                    for power,coef in terms:
                        if power[a]:
                            p=list(power);p[a]-=1
                            item.append((tuple(p),coef*power[a]))
                    deriv.append(item)
                basis.append({'ell':ell,'m':m,'part':part,'nu':nu,'terms':terms,'derivatives':deriv})
    assert len(basis)==T*(T+2)
    return basis


def polynomial_value(terms,powers):
    return sum((c*powers[0][p[0]]*powers[1][p[1]]*powers[2][p[2]] for p,c in terms),Q(0))


def evaluate(points,basis):
    T=max(item['ell'] for item in basis)
    values=[[] for _ in basis]
    gradients=[[] for _ in basis]
    for point in points:
        powers=[[Q(1)] for _ in range(3)]
        for a in range(3):
            for k in range(T): powers[a].append(powers[a][-1]*point[a])
        for row,item in enumerate(basis):
            values[row].append(polynomial_value(item['terms'],powers))
            ambient=[polynomial_value(term,powers) for term in item['derivatives']]
            normal=dot(ambient,point)
            gradients[row].extend([v-normal*x for v,x in zip(ambient,point)])
    return values,gradients


def mean_squared(values,basis):
    N=len(values[0])
    return sum(((sum(row,Q(0))/N)**2/item['nu'] for row,item in zip(values,basis)),Q(0))


def basis_json(basis):

            for item in basis]


def matvec(rows,vector):
    return [dot(row,vector) for row in rows]


def exact_active_candidate(J,b,free,signs):
    n=len(J[0]); M=len(J)
    active=[j for j in range(n) if j not in free]
    B=[[row[j] for j in free]+[sum((signs[j]*row[j] for j in active),Q(0))] for row in J]
    matrix=QM(B)
    solved=matrix.solve(QM([[v] for v in b]))
    d=solved[M-1,0]
    w=[signs[j]*d for j in range(n)]
    for k,j in enumerate(free): w[j]=solved[k,0]
    primal_residual=[v-t for v,t in zip(matvec(J,w),b)]
    margins=[d-abs(v) for v in w]
    primal=all(v==0 for v in primal_residual) and 0<=d<=1 and min(margins)>=0
    dual_rhs=[[Q(0)] for _ in range(M)]; dual_rhs[-1]=[Q(-1)]
    dual_solution=matrix.transpose().solve(QM(dual_rhs))
    multiplier=[dual_solution[j,0] for j in range(M)]
    mu=[-signs[j]*sum((J[k][j]*multiplier[k] for k in range(M)),Q(0)) if j in active else Q(0) for j in range(n)]
    stationarity=[sum((J[k][j]*multiplier[k] for k in range(M)),Q(0))+signs[j]*mu[j] for j in range(n)]
    dual=all(v==0 for v in stationarity) and min(mu)>=0 and sum(mu,Q(0))==1 and -dot(multiplier,b)==d
    return w,d,{'kind':'exact_active_LP_vertex','free_columns':free,'active_columns':active,
                'active_signs':signs,'residuals':vector_json(primal_residual),'box_margins':vector_json(margins),
                'primal_accepted':primal,'dual_accepted':dual,'lambda':vector_json(multiplier),'mu':vector_json(mu),
                'stationarity':vector_json(stationarity),'dual_objective':qs(-dot(multiplier,b))}


def independent_active_free(J_float,u,t):
    M,n=J_float.shape
    rowspace=np.column_stack((J_float,np.zeros(M)))
    orthogonal,_=np.linalg.qr(rowspace.T)
    vectors=[orthogonal[:,j] for j in range(M)]
    order=sorted(range(n),key=lambda j:abs(abs(u[j])-t))
    active=[]
    for j in order:
        row=np.zeros(n+1); row[j]=1 if u[j]>=0 else -1; row[n]=-1
        rem=row.copy()
        for v in vectors: rem-=np.dot(rem,v)*v
        # Reorthogonalize: selection is only a numerical proposal.
        for v in vectors: rem-=np.dot(rem,v)*v
        norm=np.linalg.norm(rem)
        if norm>1e-8:
            vectors.append(rem/norm); active.append(j)
            if len(active)==n+1-M: break
    if len(active)!=n+1-M:
        raise RuntimeError('numeric active rank selection incomplete')
    return [j for j in range(n) if j not in active]


def exact_feasible_repair(J,b,proposal,J_float):
    M,n=len(J),len(J[0])
    _,_,pivot=qr(J_float,mode='economic',pivoting=True)
    selected=[int(j) for j in pivot[:M]]
    den=1<<80
    w=[Q(round(float(a)*den),den) for a in proposal]
    residual=[target-value for target,value in zip(b,matvec(J,w))]
    block=QM([[row[j] for j in selected] for row in J])
    solved=block.solve(QM([[a] for a in residual]))

    d=max(abs(a) for a in w)
    errors=[v-t for v,t in zip(matvec(J,w),b)]
    assert all(a==0 for a in errors) and 0<=d<=1

                'residuals':vector_json(errors),'box_margins':vector_json([d-abs(v) for v in w]),

def ldl_certificate(matrix):

    lower=[[Q(0)]*n for _ in range(n)]

    for i in range(n):
        lower[i][i]=Q(1)
        d=matrix[i][i]-sum((lower[i][k]**2*diagonal[k] for k in range(i)),Q(0))
        diagonal.append(d)
        if d<=0:
            return {'positive_definite':False,'failure_index':i,'diagonal':vector_json(diagonal)}
        for j in range(i+1,n):
            lower[j][i]=(matrix[j][i]-sum((lower[j][k]*lower[i][k]*diagonal[k] for k in range(i)),Q(0)))/d
    return {'positive_definite':True,'diagonal':vector_json(diagonal),'lower':matrix_json(lower)}


def final_certificate(points,basis,T):
    start=time.monotonic()
    values,J=evaluate(points,basis)
    mean2=mean_squared(values,basis)
    beta=Q(1,2048*(T+1)**5)
    gram=gram_matrix(values)
    shifted=[[a-(basis[j]['nu']/2 if j==k else 0) for k,a in enumerate(row)] for j,row in enumerate(gram)]
    ldl=ldl_certificate(shifted)
    certificate={'beta':qs(beta),'mean_squared':qs(mean2),'mean_pass':mean2<=beta*beta,
                 'mean_margin_squared':qs(beta*beta-mean2),'gram':matrix_json(gram),
                 'gram_shifted':matrix_json(shifted),'gram_LDL':ldl,
                 'gram_pass':ldl['positive_definite'],'unit_exact':all(dot(p,p)==1 for p in points),
                 'distinct_exact':len({tuple(qs(a) for a in p) for p in points})==len(points),
                 'coordinate_bits':max(bits(a) for p in points for a in p),
                 'seconds':time.monotonic()-start}
    return certificate,(values,J,gram)


def right_inverse_certificate(points,basis,T,values,J,gram):
    start=time.monotonic();M=len(basis);N=len(points)
    normalized=np.asarray([[qfloat(a) for a in row] for row in J])
    normalized/=np.asarray([math.sqrt(qfloat(item['nu'])) for item in basis])[:,None]
    _,_,pivot=qr(normalized,mode='economic',pivoting=True)
    selected=[int(j) for j in pivot[:M]]
    block=QM([[row[j]/N for j in selected] for row in J]).inv()
    ambient=[[Q(0)]*M for _ in range(3*N)]
    for k,index in enumerate(selected):
        ambient[index]=[block[k,j] for j in range(M)]
    R=[]
    for i,point in enumerate(points):
        rows=ambient[3*i:3*i+3]
        tangent_rows=[[rows[a][j]-point[a]*sum((point[k]*rows[k][j] for k in range(3)),Q(0))
                       for j in range(M)] for a in range(3)]
        R.extend(tangent_rows)
    # Tangent projection preserves the derivative action.
    residual=[[sum((J[j][k]*R[k][ell] for k in range(3*N)),Q(0))/N-(1 if j==ell else 0)
               for ell in range(M)] for j in range(M)]
    assert all(a==0 for row in residual for a in row)
    C2=max(sum((R[3*i+a][j]**2*basis[j]['nu'] for a in range(3) for j in range(M)),Q(0)) for i in range(N))
    C=sqrt_bounds(C2,24)[1]
    mean2=mean_squared(values,basis)
    mean_upper=sqrt_bounds(mean2,80)[1]
    rho=4*mean_upper
    A2=Q(sum(ell*(ell+1)*(2*ell+1) for ell in range(1,T+1)),2)
    B2=Q(sum(ell*(ell+1)*(3*ell*(ell+1)-2)*(2*ell+1) for ell in range(1,T+1)),8)
    A=sqrt_bounds(A2,24)[1];B=sqrt_bounds(B2,24)[1]
    r=C*rho
    K=B+(r/2)*A
    lambda0=Q(513,1024)
    gram_shift=[[a-(lambda0*basis[j]['nu'] if j==k else 0) for k,a in enumerate(row)] for j,row in enumerate(gram)]
    gram_slack=ldl_certificate(gram_shift)
    sqrtM=sqrt_bounds(Q(M),24)[1]
    checks={'right_inverse_exact':True,'mean_basin':mean2<=rho*rho/16,'r_le_one':r<=1,
            'contraction':K*C*C*rho<=Q(1,2),'gram_slack':gram_slack['positive_definite'],
            'gram_travel_margin':lambda0-Q(1,2)>2*sqrtM*A*r}
    return {'kind':'sparse exact projected column-block right inverse','selected_columns':selected,
            'R':matrix_json(R),'right_inverse_residual':matrix_json(residual),'C_squared':qs(C2),
            'C_upper':qs(C),'mean_upper':qs(mean_upper),'rho':qs(rho),'A_squared':qs(A2),'B_squared':qs(B2),
            'A_upper':qs(A),'B_upper':qs(B),'radius_bound':qs(r),'K_upper':qs(K),
            'lambda0':qs(lambda0),'gram_slack_LDL':gram_slack,'checks':checks,
            'finite_certificate_pass':all(checks.values()),'seconds':time.monotonic()-start,
            'mathematical_source':'charges_design9 cycle11 a-posteriori certificate; independently gate formulas'}


def checkpoint(case,path):
    path.write_text(json.dumps(case,indent=2)+'\n')


def timeout_handler(signum,frame):
    raise TimeoutError('180-second per-case execution bound')


def run_case(T,q,solver='lp'):
    tag='' if solver=='lp' else 'L2_'
    path=OUT/f'acquisition_{tag}T{T}_q{q}.json'
    delta=Q(1,2**24*(T+1)**8)
    started=time.monotonic()
    case={'T':T,'q':q,'N':2*q*q,'delta':qs(delta),'minimum_corrections':3,'maximum_corrections':5,
          'inside_universal_cardinality_regime':q>=600*(T+1),'status':'RUNNING','steps':[],
          'versions':{name:version(name) for name in ('scipy','numpy','sympy','mpmath','python-flint')},
          'python':sys.version,'solver':solver,
          'solver_policy':('at most two exact active reconstructions plus one feasible repair per step' if solver=='lp'
                           else 'one exact JJ^t solve per step; same initializer, data, Cayley and resets'),
          'stopping_rule':'after >=3 steps stop if exact mean <= beta/8 and exact Gram > I/2; otherwise max5'}
    checkpoint(case,path)
    try:
        init_start=time.monotonic()
        basis=harmonic_basis(T)
        points,initializer=centroid_initializer(q,delta)
        case['basis']=basis_json(basis);case['initializer']=initializer
        case['initial_points']=matrix_json(points);case['initializer_seconds']=time.monotonic()-init_start
        values,_=evaluate(points,basis)
        case['initial_mean_squared']=qs(mean_squared(values,basis))
        checkpoint(case,path)
        print(json.dumps({'T':T,'N':len(points),'stage':'initialized','mean':math.sqrt(qfloat(mean_squared(values,basis))),
                          'seconds':case['initializer_seconds']}),flush=True)
        for step in range(5):
            step_start=time.monotonic()
            input_points=points
            cayley,record=(correction(input_points,basis) if solver=='lp' else euclidean_correction(input_points,basis))
            points,reset_record=reset_points(cayley,delta)
            record['input_points']=matrix_json(input_points);record['reset']=reset_record
            record['output_points']=matrix_json(points);record['step']=step
            values,_=evaluate(points,basis)
            after=mean_squared(values,basis)
            record['output_mean_squared']=qs(after);record['seconds_total']=time.monotonic()-step_start
            case['steps'].append(record);checkpoint(case,path)
            print(json.dumps({'T':T,'step':step,'mean':math.sqrt(qfloat(after)),
                              'd':qfloat(Q(record['accepted_d'])),'optimality':record['exact_optimality_certified'],
                              'seconds':record['seconds_total']}),flush=True)
            beta=Q(1,2048*(T+1)**5)
            if step>=2 and after<=(beta/8)**2:
                cert,_=final_certificate(points,basis,T)
                if cert['gram_pass']:break
        final,data=final_certificate(points,basis,T)
        case['final_points']=matrix_json(points);case['final_certificate']=final
        checkpoint(case,path)
        case['a_posteriori']=right_inverse_certificate(points,basis,T,*data)
        case['status']='FINITE_ACQUISITION_CERTIFIED' if all(final[k] for k in ('mean_pass','gram_pass','unit_exact','distinct_exact')) else 'FINITE_ACQUISITION_FAILED'
    except Exception as exc:
        case['status']='FAILED';case['failure']={'type':type(exc).__name__,'message':str(exc)}
    case['seconds_total']=time.monotonic()-started
    checkpoint(case,path)
    print(json.dumps({'T':T,'status':case['status'],'corrections':len(case['steps']),
                      'seconds':case['seconds_total'],'failure':case.get('failure')}),flush=True)
    return case


def main():
    parser=argparse.ArgumentParser();parser.add_argument('--case',choices=['T2','T3','both'],default='both')
    parser.add_argument('--solver',choices=['lp','l2'],default='lp')
    args=parser.parse_args()
    OUT.mkdir(parents=True,exist_ok=True)
    signal.signal(signal.SIGALRM,timeout_handler)
    summaries=[]
    for T,q in ((2,3),(3,4)):
        if args.case!='both' and args.case!=f'T{T}':continue
        signal.alarm(180)
        try:case=run_case(T,q,args.solver)
        finally:signal.alarm(0)
        summaries.append({k:case.get(k) for k in ('T','q','N','status','seconds_total','failure')})
    (OUT/f'execution_summary_{args.solver}.json').write_text(json.dumps(summaries,indent=2)+'\n')


if __name__=='__main__':
    main()
