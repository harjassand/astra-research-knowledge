class Budget:
    def __init__(self,caps):
        self.start=time.monotonic(); self.caps=caps
        self.peak_bits=0; self.matmuls=0; self.inverses=0; self.scalar_products=0
    def check(self):
        if time.monotonic()-self.start>self.caps['max_seconds']: raise Limit('wall_time_cap')
    def observe(self,values):
        self.check()
        for x in values:
            if isinstance(x,F): sizes=(abs(x.numerator).bit_length(),x.denominator.bit_length())
            else: sizes=(abs(x).bit_length(),)
            self.peak_bits=max(self.peak_bits,*sizes)
        if self.peak_bits>self.caps['max_integer_bits']: raise Limit('integer_bit_cap')
    def predict(self,number):
        self.check()
        if number>self.caps['max_integer_bits']: raise Limit('predicted_integer_bit_cap')
    def report(self):
        peak=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        if sys.platform!='darwin': peak*=1024
        return {'seconds':time.monotonic()-self.start,'peak_integer_bits':self.peak_bits,
                'rational_matrix_products':self.matmuls,'rational_inverses':self.inverses,
                'exact_scalar_products':self.scalar_products,
                'process_peak_bytes_including_libraries':peak,'caps':self.caps}

class Mat:

        if budget: budget.observe([self.d]+[x for row in self.a for x in row])

    @classmethod

    def fr(self):return [[F(x,self.d) for x in row] for row in self.a]
    def strings(self):return [[str(x) for x in row] for row in self.fr()]

    def bits(self):return max([self.d.bit_length()]+[abs(x).bit_length() for row in self.a for x in row])
    def times(self,b):

    def scale(self,c):

    def inverse(self):

def inertia(matrix):
    """Rational symmetric congruence, including 2x2 zero-diagonal pivots."""
    if matrix.n!=matrix.m or matrix.a!=matrix.trans().a:raise Invalid('nonsymmetric_matrix')
    a=matrix.fr(); positive=negative=zero=0; budget=matrix.budget
    while a:
        n=len(a); pos=next((i for i in range(n) if a[i][i]),None)
        if pos is not None:
            order=[pos]+[i for i in range(n) if i!=pos];a=[[a[i][j] for j in order] for i in order]
            pivot=a[0][0];positive+=int(pivot>0);negative+=int(pivot<0)
            a=[[a[i][j]-a[i][0]*a[0][j]/pivot for j in range(1,n)] for i in range(1,n)]
        else:
            pair=next(((i,j) for i in range(n) for j in range(i+1,n) if a[i][j]),None)
            if pair is None:zero+=n;break
            i,j=pair;order=[i,j]+[k for k in range(n) if k not in pair];a=[[a[p][q] for q in order] for p in order]
            cross=a[0][1];positive+=1;negative+=1
            a=[[a[p][q]-(a[p][0]*a[1][q]+a[p][1]*a[0][q])/cross for q in range(2,n)] for p in range(2,n)]
        if budget:budget.observe([v for row in a for v in row])
    return [positive,negative,zero]

def ldl(matrix):
    n=matrix.n;a=matrix.fr(); lower=[[F(int(i==j)) for j in range(n)] for i in range(n)];diag=[]
    for i in range(n):
        d=a[i][i]-sum(lower[i][k]**2*diag[k] for k in range(i))
        if d<=0:raise Invalid('nonpositive_LDL_pivot')
        diag.append(d)
        for j in range(i+1,n):lower[j][i]=(a[j][i]-sum(lower[j][k]*diag[k]*lower[i][k] for k in range(i)))/d
        if matrix.budget:matrix.budget.observe(diag+[v for row in lower for v in row])
    return Mat.fractions(lower,matrix.budget),diag

def dyadic_normalizer(d):
    d=F(d)
    if d<=0:raise Invalid('nonpositive_normalization')
    exponent=d.numerator.bit_length()-d.denominator.bit_length()
    power=F(2**exponent) if exponent>=0 else F(1,2**(-exponent))
    if d<power:exponent-=1
    k=-(exponent//2)
    two=F(2**k) if k>=0 else F(1,2**(-k))
    t=d*two**2
    assert 1<=t<4
    c=F(1) if t<=F(3,2) else (F(3,4) if t<2 else F(1,2))
    s=two*c;test=d*s*s
    assert F(1,2)<=test<=F(3,2)
    return s,{'d':str(d),'k':k,'t':str(t),'mantissa':str(c),'scale':str(s),'d_scale_squared':str(test)}

def normalize_basis(B):
    lower,diag=ldl(B.trans().times(B));C=B.times(lower.inverse().trans())
    scalings=[];records=[]

def primitive_columns(Q,columns):

def greedy_columns(Q,r,K):
    selected=[];scores=[];B=None
    for k in range(r):
