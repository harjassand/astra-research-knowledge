"""Carried-state Gillespie endpoint sensor, with separate simulation cost audit.

Only the (read, initial_label) pair should be supplied to an estimator. Full
jump counts belong to the caller's audit; they are unavailable through read.
This is a floating-point exact-jump simulation algorithm, not exact arithmetic
or a theorem on simulation numerical error. No free resets after initialization.
"""
from dataclasses import dataclass
import math
import numpy as np

@dataclass(frozen=True)
class CostAudit:
    endpoint_reads: int
    elapsed_time: float
    physical_jumps: int
    random_draws: int

class EndpointSensor:
    def __init__(self, generator, initial_state, rng, charge_initial_label=True):
        q=np.array(generator,dtype=float,copy=True)
        if q.ndim!=2 or q.shape[0]!=q.shape[1] or not np.all(np.isfinite(q)):
            raise ValueError('finite square generator required')
        off=q.copy();np.fill_diagonal(off,0.)
        if np.any(off<0) or not np.allclose(q.sum(axis=1),0.,rtol=0.,atol=1e-12):
            raise ValueError('generator must have nonnegative off diagonals and zero row sums')
        if not 0<=initial_state<len(q):raise ValueError('invalid initial state')
        self.__rates=off;self.__escape=off.sum(axis=1)
        self.__state=int(initial_state);self.__rng=rng
        self.__reads=int(charge_initial_label);self.__time=0.;self.__jumps=0;self.__draws=0

    def endpoint_interface(self):
        """Return only endpoint observations; initial label is charged by default."""
        return self.read,self.__state

    def read(self,lag):
        lag=float(lag)
        if not math.isfinite(lag) or lag<=0:raise ValueError('finite positive calibrated lag required')
        remaining=lag
        while remaining>0:
            escape=self.__escape[self.__state]
            if escape==0:break
            holding=float(self.__rng.exponential(1./escape));self.__draws+=1
            if holding>=remaining:break
            remaining-=holding
            uniform=float(self.__rng.random())*escape;self.__draws+=1
            cumulative=np.cumsum(self.__rates[self.__state])
            target=int(np.searchsorted(cumulative,uniform,side='right'))
            # Floating rounding at the upper endpoint is clamped to the last positive rate.
            if target==len(cumulative):target=int(np.flatnonzero(self.__rates[self.__state])[-1])
            self.__state=target;self.__jumps+=1
        self.__time+=lag;self.__reads+=1
        return self.__state

    def audit(self):
        """Caller-only full cost audit. Do not pass this capability to estimator."""
        return CostAudit(self.__reads,self.__time,self.__jumps,self.__draws)

if __name__=='__main__':
    # Distribution check uses independently initialized fixtures, not an estimator reset.
    rng=np.random.default_rng(91237)

    read,label=sensor.endpoint_interface()
    for k in range(1000):label=read(.01 if label==0 else .02)
    cost=sensor.audit()
