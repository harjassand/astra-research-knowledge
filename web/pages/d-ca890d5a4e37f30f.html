"""Exact transcription checks for the actual second-coefficient proof.

No PDE fixture, mesh solver, or comparison potential is used. These checks do
not replace the analytic global-W, cell regularity, or signed-measure proof.
"""
from __future__ import annotations

import argparse
import json
from fractions import Fraction
from pathlib import Path

import sympy as sp


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    checks: list[dict] = []

    def record(name: str, residual, detail: str = "") -> None:
        value = sp.trigsimp(sp.simplify(residual))
        if value != 0:
            raise AssertionError(f"{name}: {value}")
        checks.append({"name": name, "pass": True, "detail": detail})

    eta, S, sigma = sp.symbols("eta S sigma", real=True)
    for n, m in [(3, 2), (4, 2), (4, 3)]:
        Y = sp.symbols(f"Y0:{n-1}", real=True)
        theta = sp.symbols(f"theta0:{m}", real=True)
        c = sp.symbols(f"c0:{m}", real=True)
        Z = (S, *Y)
        u = sp.Function(f"U{n}{m}")(*Z, *theta)
        P = sum(c[i] * sp.cos(theta[i]) for i in range(m))
        p = [sp.Function(f"p{n}{m}_{i}")(*theta) for i in range(m)]
        r = sum(Y[i] * p[i] for i in range(m))
        Q = sum(y**2 for y in Y)
        R = sp.sqrt((1 + eta*S)**2 + eta**2*Q)
        invR = sp.series(1/R, eta, 0, 3).removeO().expand()
        nu0 = sp.series((1 + eta*S)/R, eta, 0, 3).removeO().expand()
        nuY = [sp.series(eta*y/R, eta, 0, 3).removeO().expand() for y in Y]
        record(f"n{n}m{m}: radius_inverse", invR - (1-eta*S+eta**2*(S**2-Q/2)))
        record(f"n{n}m{m}: radial_direction", nu0 - (1-eta**2*Q/2))
        for i in range(n-1):
            record(f"n{n}m{m}: tangential_direction_{i}", nuY[i]-(eta*Y[i]-eta**2*S*Y[i]))

        def d1(v):
            return (-2*sum(Y[i]*sp.diff(v, theta[i]) for i in range(m))
                    +2*sum(sp.diff(v, Y[i], theta[i]) for i in range(m))
                    -2*sp.diff(v, S))

        def d2(v):
            return (2*S*sum(Y[i]*sp.diff(v, theta[i]) for i in range(m))
                    -2*sum(Y[i]*sp.diff(v, Y[i]) for i in range(n-1))
                    +sum(sp.diff(v, z, 2) for z in Z)-(n-1)*v)

        # Independent direct-E geometric operator, normalized by A^2.
        op = (-2*sum(nuY[i]*sp.diff(u, theta[i]) for i in range(m))
              +2*eta*sum(sp.diff(u, Y[i], theta[i]) for i in range(m))
              -2*eta*(nu0*sp.diff(u, S)+sum(nuY[i]*sp.diff(u, Y[i]) for i in range(n-1)))
              +eta**2*sum(sp.diff(u, z, 2) for z in Z)
              -(n-1)*eta**2*invR*u)
        truncated = sp.series(op, eta, 0, 3).removeO().expand()
        record(f"n{n}m{m}: full_D1_D2_operator", truncated-eta*d1(u)-eta**2*d2(u))
        record(f"n{n}m{m}: zeroth_Helmholtz", sum(sp.diff(P,t,2) for t in theta)+P)
        record(f"n{n}m{m}: first_corrector_source", -d1(P)-2*sum(Y[i]*sp.diff(P,theta[i]) for i in range(m)))
        expected_s_source = (2*sum(Y[i]*Y[k]*sp.diff(p[k],theta[i]) for i in range(m) for k in range(m))
                             -2*sum(sp.diff(p[i],theta[i]) for i in range(m))
                             -2*S*sum(Y[i]*sp.diff(P,theta[i]) for i in range(m))
                             +(n-1)*P)
        record(f"n{n}m{m}: second_corrector_source", -d1(r)-d2(P)-expected_s_source)
        record(f"n{n}m{m}: second_radial_term", sp.diff(expected_s_source,S)+2*sum(Y[i]*sp.diff(P,theta[i]) for i in range(m)))

        # Radial averaging premise: every individual tilted plane wave is harmonic.
        u0, ui, th = sp.symbols("u0 ui th", real=True)
        harmonic = sp.exp(sigma*u0)*sp.cos(th-sigma*ui)
        record(f"n{n}m{m}: radial_harmonic_multiplier", sp.diff(harmonic,u0,2)+sp.diff(harmonic,ui,2))

        # Independent first-order carrier/macro-shift expansion of a numerator.
        v = sp.symbols(f"v0:{n}", real=True)
        Nu1 = sp.Function(f"N1_{n}{m}")(*Z)
        Nu2 = sp.Function(f"N2_{n}{m}")(*Z)
        coeff = eta*Nu1+eta**2*(Nu2 + sigma*sum(Y[i]*v[i+1] for i in range(n-1))*Nu1
                                -sigma*sum(v[i]*sp.diff(Nu1,Z[i]) for i in range(n)))
        # N1's S derivative vanishes in the actual frozen ledger.
        record(f"n{n}m{m}: carrier_and_shift",
               coeff.coeff(eta,2)-Nu2-sigma*sum(Y[i]*v[i+1] for i in range(n-1))*Nu1
               +sigma*sum(v[i]*sp.diff(Nu1,Z[i]) for i in range(n)))

    # First trace sign checked against the explicit independently gated stripe.
    delta, h = sp.symbols("delta h", positive=True, real=True)
    tau2 = sp.sin(delta)**2
    p_old = h*sp.sin(h)-delta*sp.tan(delta)*sp.cos(h)
    p_new = -p_old
    record("stripe: p_new_equation", sp.diff(p_new,h,2)+p_new+2*sp.cos(h))
    record("stripe: left_zero_trace", p_new.subs(h,-delta))
    record("stripe: right_zero_trace", p_new.subs(h,delta))
    C = (-delta*sp.cos(2*delta)/2+sp.sin(2*delta)/4
         -delta*sp.tan(delta)*(delta+sp.sin(2*delta)/2))/sp.pi
    A_tau = 1-2*delta/sp.pi+(delta-sp.sin(2*delta)/2)/(sp.pi*tau2)
    M_r = (-sp.pi+2*delta+sp.cot(delta)-delta/sp.sin(delta)**2)/sp.pi
    C_delta = (1+2*delta*sp.tan(delta))/2
    kprime = C_delta*(sp.cos(delta)-delta/sp.sin(delta))-delta*sp.sin(delta)
    M_k = 2*kprime/(sp.pi*sp.sin(delta))
    record("stripe: full_radial_current_mean", M_r+A_tau)
    record("stripe: full_trace_current_mean", M_k-2*C/tau2)

    # Exact rational inequalities over the whole prescribed parameter band.
    tau_max = Fraction(1,100)
    gradient_lower = Fraction(49,100)-(Fraction(3,10)+Fraction(9,8)*tau_max)**2
    assert gradient_lower > Fraction(9,25)
    checks.append({"name":"uniform_band_gradient", "pass":True,
                   "exact_lower":str(gradient_lower), "target":str(Fraction(9,25))})
    # pi>3 is enough to certify b_raw<-19/10 for every tau<=1/100.
    raw_upper_using_pi3 = -2+Fraction(80,1)*tau_max/Fraction(9,1)
    assert raw_upper_using_pi3 < Fraction(-19,10)
    checks.append({"name":"uniform_raw_negative_margin", "pass":True,
                   "upper_using_pi_gt_3":str(raw_upper_using_pi3), "target":"-19/10"})
    kinetic = Fraction(1,64)
    total_margin = Fraction(-3,2)+Fraction(1,32)
    assert total_margin < 0
    checks.append({"name":"actual_trial_form_margin", "pass":True,
                   "cube_dirichlet_quotient":str(kinetic),"smooth_trial_total_upper":str(total_margin)})

    args.out.parent.mkdir(parents=True, exist_ok=True)
    result = {
        "status":"PASS",
        "checks_passed":len(checks),
        "checks":checks,
        "scope":"Exact transcription and rational margin checks only; no PDE numerical fixture or universal-proof substitution.",
        "analytic_target":"notes/pde_spectrum9/cycle13/ACTUAL_PLATEAU_SECOND_COEFFICIENT.md",
    }
    args.out.write_text(json.dumps(result, indent=2)+"\n")
    print(json.dumps({"status":result["status"],"checks_passed":len(checks),"out":str(args.out)}))


if __name__ == "__main__":
    main()
