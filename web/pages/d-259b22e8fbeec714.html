"""Finite diagnostics for the task-norm SDP reduction.

This is a numerical check, not a correctness or novelty certificate.
The theorem is derived independently in report.md.
"""
from itertools import product
from pathlib import Path
import json
import math
import time

import cvxpy as cp
import numpy as np


def inf_to_two_sq(matrix):
    signs = np.array(list(product((-1., 1.), repeat=matrix.shape[1]))).T
    return float(np.max(np.sum((matrix @ signs) ** 2, axis=0)))


def compress(matrix, rank):
    n, m = matrix.shape
    retained = cp.Variable((n, n), symmetric=True)
    dual_diag = cp.Variable(m)
    constraints = [retained >> 0, np.eye(n) - retained >> 0,
                   cp.trace(retained) <= rank,
                   cp.diag(dual_diag) - matrix.T @ (np.eye(n) - retained) @ matrix >> 0]
    problem = cp.Problem(cp.Minimize(cp.sum(dual_diag)), constraints)
    problem.solve(solver="CLARABEL", tol_gap_abs=1e-9, tol_feas=1e-9,
                  tol_gap_rel=1e-9, max_iter=300)
    if problem.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(problem.status)
    vals, vectors = np.linalg.eigh(retained.value)
    basis = vectors[:, -(2 * rank):]
    projection = basis @ basis.T
    return projection, retained.value, vals, float(problem.value)


def main():
    start = time.monotonic()
    rng = np.random.default_rng(20261007)
    records = []
    max_affinity_violation = 0.
    for trial in range(12):
        n, m, rank = 8, 7, 2
        components = rng.dirichlet(np.ones(m), size=rank)
        weights = rng.dirichlet(np.ones(rank), size=n)
        p = weights @ components
        noise = rng.dirichlet(np.full(m, .3), size=n)
        contamination = (trial + 1) / 100.
        q = (1 - contamination) * p + contamination * noise
        matrix = q / math.sqrt(n)
        p_matrix = p / math.sqrt(n)
        p_basis, _, _ = np.linalg.svd(p_matrix, full_matrices=False)
        source_projection = p_basis[:, :rank] @ p_basis[:, :rank].T
        candidate_error = inf_to_two_sq((np.eye(n) - source_projection) @ matrix)
        projection, relaxed, eigs, objective = compress(matrix, rank)
        result_error = inf_to_two_sq((np.eye(n) - projection) @ matrix)
        kl = float(np.sum(p * np.log(p / q)) / n)
        affinity = np.sum(np.sqrt(p * q), axis=1)
        row_tv = np.sum(np.abs(p - q), axis=1) / 2
        hellinger = float(2 - 2 * np.mean(affinity))
        max_affinity_violation = max(max_affinity_violation,
                                     float(np.mean(row_tv ** 2) - hellinger))
        assert np.mean(row_tv ** 2) <= hellinger + 1e-12
        assert hellinger <= kl + 1e-12
        assert result_error <= math.pi * candidate_error + 2e-7
        assert result_error <= 4 * math.pi * kl + 2e-7
        assert np.min(eigs) >= -2e-7 and np.max(eigs) <= 1 + 2e-7
        assert np.trace(relaxed) <= rank + 2e-7
        assert np.linalg.norm(projection @ projection - projection) < 1e-10
        # Norm-bounded linear readouts for every task on this finite row list.
        vals, vecs = np.linalg.eigh(projection)
        basis = vecs[:, -2 * rank:]
        states = np.column_stack((np.ones(n), basis)) / math.sqrt(2)
        assert np.max(np.linalg.norm(states, axis=1)) <= 1 + 1e-12
        max_true_task_error = 0.
        for bits in product((0., 1.), repeat=m):
            task = np.array(bits)
            signed = 2 * task - 1
            theta = np.concatenate(([1 / math.sqrt(2)],
                                     (basis.T @ (q @ signed)) / math.sqrt(2)))
            predicted = states @ theta
            max_true_task_error = max(max_true_task_error,
                                      float(np.mean((predicted - p @ task) ** 2)))
            assert np.linalg.norm(theta) <= math.sqrt((n + 1) / 2) + 1e-10
        assert max_true_task_error <= (2 * math.pi + 2) * kl + 2e-7
        records.append({"trial": trial, "contamination": contamination,
                        "kl": kl, "hellinger_squared": hellinger,
                        "source_rank_candidate_task_norm_squared": candidate_error,
                        "sdp_upper_value": objective,
                        "rounded_task_norm_squared": result_error,
                        "max_true_task_mse": max_true_task_error,
                        "retained_rank": 2 * rank})

    # Unequal prefix masses and zero-probability source prefixes: exact affinity
    # decomposition is tested separately from rank compression.
    for _ in range(100):
        n, m = 5, 6
        p_mass = rng.dirichlet(np.ones(n))
        p_mass[0] = 0
        p_mass /= np.sum(p_mass)
        q_mass = rng.dirichlet(np.ones(n))
        p_cond = rng.dirichlet(np.ones(m), size=n)
        q_cond = rng.dirichlet(np.ones(m), size=n)
        affinity = np.sum(np.sqrt(p_cond * q_cond), axis=1)
        tv = np.sum(np.abs(p_cond - q_cond), axis=1) / 2
        global_h = float(2 - 2 * np.sum(np.sqrt(p_mass * q_mass) * affinity))
        assert np.dot(q_mass, tv ** 2) <= global_h + 1e-12
        assert np.dot(p_mass, tv ** 2) <= global_h + 1e-12

    payload = {"status": "ALL_DIAGNOSTICS_PASSED", "scope": "numerical finite fixtures",
               "sdp_fixtures": len(records), "unequal_mass_affinity_fixtures": 100,
               "seconds": time.monotonic() - start,
               "max_affinity_violation": max_affinity_violation,
               "records": records}
    Path(__file__).with_name("diagnostics.json").write_text(json.dumps(payload, indent=2))
    print(json.dumps({key: value for key, value in payload.items() if key != "records"}))


if __name__ == "__main__":
    main()
