def det(a):
    a = [row[:] for row in a]
    out = F(1)
    for i in range(len(a)):
        q = next((j for j in range(i, len(a)) if a[j][i]), None)

        if q != i: a[i], a[q], out = a[q], a[i], -out

            r = a[j][i] / p
            for k in range(i, len(a)): a[j][k] -= r*a[i][k]
