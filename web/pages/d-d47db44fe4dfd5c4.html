"""Execute only the frozen source-iteration comparison or its charged sampler."""
from pathlib import Path
import argparse
import gzip
import hashlib
import json
import resource
import sys
import time
from hirai_exact import run_source_iteration
from prime_sampler import bad_prime_bound, sample_probable_prime

BASE = Path(__file__).resolve().parents[3]
OUT = BASE / "results/metric_geometry10/cycle17"


def max_rss_bytes():
    value = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    return value if sys.platform == "darwin" else value * 1024


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=("sample", "run"))
    parser.add_argument("--case")
    parser.add_argument("--field", choices=("Q", "sample", "7"))
    args = parser.parse_args()
    start = time.perf_counter()
    protocol_path = OUT / "FROZEN_PROTOCOL.json"
    protocol = json.loads(protocol_path.read_text())
    if args.mode == "sample":
        distinct = {case["input_id"]: case for case in protocol["instances"]}
        bound = sum(bad_prime_bound(case["n"], len(case["costs"]), case["height_bound"]) for case in distinct.values())
        data = sample_probable_prime(bound, protocol["failure_probability"])
        data["input_id_union"] = sorted(distinct)
        data["protocol_sha256"] = hashlib.sha256(protocol_path.read_bytes()).hexdigest()
        data["process_measured_seconds"] = time.perf_counter() - start
        data["process_cpu_seconds"] = time.process_time()
        data["peak_rss_bytes"] = max_rss_bytes()
        (OUT / "PRIME_SAMPLING.json").write_text(json.dumps(data, indent=2) + "\n")
        print(json.dumps({k: data[k] for k in ("outcome", "candidates_used", "candidate", "core_seconds", "peak_rss_bytes")}))
        return
    if args.case is None or args.field is None:
        raise SystemExit("run requires --case and --field")
    instance = next(case for case in protocol["instances"] if case["name"] == args.case)
    if args.field == "Q":
        modulus, label = None, "Q"
    elif args.field == "7":
        if instance["name"] != "triangle_capped":
            raise SystemExit("F7 negative control frozen only for triangle_capped")
        modulus, label = 7, "F7_negative_control"
    else:
        sample = json.loads((OUT / "PRIME_SAMPLING.json").read_text())
        if sample["outcome"] != "PROBABLE_PRIME":
            raise SystemExit("sampler returned UNKNOWN; do not resample silently")
        modulus, label = int(sample["candidate"]), "sample_Fp"
    source_call_start = time.perf_counter()
    summary, trace = run_source_iteration(instance, modulus)
    summary["source_call_seconds"] = time.perf_counter() - source_call_start
    stem = instance["name"] + "_" + label
    trace_path = OUT / (stem + "_trace.json.gz")
    write_start = time.perf_counter()
    with trace_path.open("wb") as raw:
        with gzip.GzipFile(fileobj=raw, mode="wb", mtime=0) as output:
            output.write(json.dumps(trace, separators=(",", ":")).encode())
    summary["trace_write_seconds"] = time.perf_counter() - write_start
    summary["trace_path"] = str(trace_path.relative_to(BASE))
    summary["trace_sha256"] = hashlib.sha256(trace_path.read_bytes()).hexdigest()
    summary["trace_bytes"] = trace_path.stat().st_size
    summary["protocol_sha256"] = hashlib.sha256(protocol_path.read_bytes()).hexdigest()
    summary["process_measured_seconds"] = time.perf_counter() - start
    summary["process_measured_scope"] = "main entry through trace writing/hashing and summary construction; excludes imports/interpreter startup and final summary write"
    summary["process_cpu_seconds"] = time.process_time()
    summary["peak_rss_bytes"] = max_rss_bytes()
    summary["source_code_sha256"] = {name: hashlib.sha256((Path(__file__).parent / name).read_bytes()).hexdigest() for name in ("hirai_exact.py", "prime_sampler.py", "run_matched_hirai.py")}
    (OUT / (stem + "_summary.json")).write_text(json.dumps(summary, indent=2) + "\n")
    print(json.dumps({k: summary[k] for k in ("instance", "field", "sweeps_completed", "best_sweep", "best_value", "certified_fixture_gap", "accuracy_met", "core_seconds", "max_field_operand_bits", "peak_rss_bytes")}))


if __name__ == "__main__":
    main()
