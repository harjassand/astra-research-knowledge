No release code, external solver, or peer implementation is imported.  A NO_GRAPH_WITNESS

from dataclasses import dataclass, asdict
from itertools import combinations


class BudgetExceeded(Exception):
    pass


class InvalidInput(Exception):
    pass


@dataclass
class Limits:
    carrier: int = 12
    lattice: int = 64
    search_nodes: int = 500_000
    operations: int = 100_000


@dataclass
class Costs:
    search_nodes: int = 0
    csp_calls: int = 0
    domain_comparisons: int = 0
    admissibility_comparisons: int = 0
    join_tests: int = 0
    forest_edges: int = 0


class DSU:
    def __init__(self, k):
        self.p = list(range(k))

    def find(self, a):
        while a != self.p[a]:
            self.p[a] = self.p[self.p[a]]
            a = self.p[a]
        return a

    def add(self, a, b):
        a, b = self.find(a), self.find(b)
        if a != b:
            self.p[b] = a

    def relation(self):
        return [[self.find(i) == self.find(j) for j in range(len(self.p))]
                for i in range(len(self.p))]


def integer(x):
    return type(x) is int


class Instance:
    def __init__(self, data, limits=None):
        self.limits = limits or Limits()
        le, d = data.get("le"), data.get("color")
        if type(le) is not list or type(d) is not list or not le or not d:
            raise InvalidInput("nonempty explicit order and color tables required")
        n, k = len(le), len(d)
        # Before any quadratic/cubic work or new arrays.
        if n > self.limits.lattice or k > self.limits.carrier:
            raise BudgetExceeded("explicit lattice/carrier dimension cap")
        if any(type(row) is not list or len(row) != n for row in le):
            raise InvalidInput("order table shape")
        if any(type(x) not in (bool, int) or x not in (0, 1)
               for row in le for x in row):
            raise InvalidInput("order entries must be boolean or literal integer 0/1")
        if any(type(row) is not list or len(row) != k for row in d):
            raise InvalidInput("color table shape")
        if any(not integer(x) or not 0 <= x < n for row in d for x in row):
            raise InvalidInput("colors must be literal bounded integer labels")
        self.n, self.k, self.le, self.d = n, k, le, d
        for a in range(n):
            if not le[a][a]:
                raise InvalidInput("non-reflexive order")
            for b in range(n):
                if a != b and le[a][b] and le[b][a]:
                    raise InvalidInput("non-antisymmetric order")
                for c in range(n):
                    if le[a][b] and le[b][c] and not le[a][c]:
                        raise InvalidInput("non-transitive order")
        bottoms = [a for a in range(n) if all(le[a])]
        tops = [a for a in range(n) if all(le[b][a] for b in range(n))]
        if len(bottoms) != 1 or len(tops) != 1:
            raise InvalidInput("not a bounded lattice")
        self.bottom, self.top = bottoms[0], tops[0]
        self.join = [[None] * n for _ in range(n)]
        for a in range(n):
            for b in range(n):
                ups = [c for c in range(n) if le[a][c] and le[b][c]]
                js = [c for c in ups if all(le[c][z] for z in ups)]
                downs = [c for c in range(n) if le[c][a] and le[c][b]]
                ms = [c for c in downs if all(le[z][c] for z in downs)]
                if len(js) != 1 or len(ms) != 1:
                    raise InvalidInput("missing unique join/meet")
                self.join[a][b] = js[0]
        for i in range(k):
            for j in range(k):
                if d[i][j] != d[j][i] or (d[i][j] == self.bottom) != (i == j):
                    raise InvalidInput("symmetry/separated diagonal")
                for h in range(k):
                    if not le[d[i][j]][self.join[d[i][h]][d[h][j]]]:
                        raise InvalidInput("triangle inequality")
        self.theta = [[[bool(le[d[i][j]][a]) for j in range(k)] for i in range(k)]
                      for a in range(n)]
        for a in range(n):
            for b in range(n):
                if not le[a][b] and not any(self.theta[a][i][j] and
                                            not self.theta[b][i][j]
                                            for i in range(k) for j in range(k)):
                    raise InvalidInput("order reflection")
        self.edges = list(combinations(range(k), 2))

    def join_defect(self, costs=None):
        for a in range(self.n):
            for b in range(a, self.n):
                if costs:
                    costs.join_tests += 1
                uf = DSU(self.k)
                for i, j in self.edges:
                    if self.theta[a][i][j] or self.theta[b][i][j]:
                        uf.add(i, j)
                if uf.relation() != self.theta[self.join[a][b]]:
                    return {"a": a, "b": b, "join": self.join[a][b],
                            "actual_relation": uf.relation()}
        return None

    def admissible(self, mapping, costs=None):
        if type(mapping) is not list or len(mapping) != self.k or any(
                not integer(x) or not 0 <= x < self.k for x in mapping):
            return False
        for i, j in self.edges:
            if costs:
                costs.admissibility_comparisons += 1
            if not self.le[self.d[mapping[i]][mapping[j]]][self.d[i][j]]:
                return False
        return True

    def placement(self, p, q, u, v, costs):
        """Binary CSP, exhaustive if UNSAT; interruption is UNKNOWN, never UNSAT."""
        costs.csp_calls += 1
        domains = [list(range(self.k)) for _ in range(self.k)]
        domains[p], domains[q] = [u], [v]

        def recurse(assigned, domains):
            costs.search_nodes += 1
            if costs.search_nodes > self.limits.search_nodes:
                raise BudgetExceeded("CSP search-node cap")
            if len(assigned) == self.k:
                return [assigned[i] for i in range(self.k)]
            var = min((i for i in range(self.k) if i not in assigned),
                      key=lambda i: len(domains[i]))
            for value in domains[var]:
                new = dict(assigned)
                new[var] = value
                reduced = [row[:] for row in domains]
                possible = True
                for j in range(self.k):
                    if j in new:
                        continue
                    keep = []
                    for w in reduced[j]:
                        costs.domain_comparisons += 1
                        if self.le[self.d[value][w]][self.d[var][j]]:
                            keep.append(w)
                    reduced[j] = keep
                    if not keep:
                        possible = False
                        break
                if possible:
                    result = recurse(new, reduced)
                    if result is not None:
                        return result
            return None

        return recurse({}, domains)


def verify(data, certificate, limits=None):
    """Polynomial verifier: no enumeration of self-maps or all seed subsets."""
    costs = Costs()
    try:
        inst = Instance(data, limits)
        maps, forests = certificate.get("maps"), certificate.get("forests")
        if type(maps) is not list or type(forests) is not dict:
            raise InvalidInput("certificate shape")
        cap = len(inst.edges) * max(0, inst.k - 1)
        if len(maps) > cap or len(maps) > inst.limits.operations:
            raise InvalidInput("operation count exceeds witness cap")
        if set(forests) != {f"{i},{j}" for i, j in inst.edges}:
            raise InvalidInput("forest keys")
        if any(not inst.admissible(m, costs) for m in maps):
            raise InvalidInput("non-admissible map")
        defect = inst.join_defect(costs)
        if defect:
            raise InvalidInput("binary join defect")
        for p, q in inst.edges:
            forest = forests[f"{p},{q}"]
            if type(forest) is not list or len(forest) > inst.k - 1:
                raise InvalidInput("forest edge cap")
            uf = DSU(inst.k)
            for e in forest:
                if type(e) is not list or len(e) != 3 or any(not integer(x) for x in e):
                    raise InvalidInput("forest entry")
                u, v, index = e
                if not (0 <= u < inst.k and 0 <= v < inst.k and 0 <= index < len(maps)):
                    raise InvalidInput("forest endpoint/index")
                if sorted([maps[index][p], maps[index][q]]) != sorted([u, v]):
                    raise InvalidInput("map does not witness marked edge")
                uf.add(u, v)
                costs.forest_edges += 1
            if uf.relation() != inst.theta[inst.d[p][q]]:
                raise InvalidInput("singleton forest misses color relation")
        return {"status": "CERTIFIED_FULL_CONGRUENCE", "costs": asdict(costs),
                "carrier_size": inst.k, "lattice_size": inst.n,
                "unary_operation_count": len(maps)}
    except BudgetExceeded as e:
        return {"status": "BUDGET_UNKNOWN", "reason": str(e), "costs": asdict(costs)}
    except (InvalidInput, AttributeError, TypeError) as e:
        return {"status": "REJECTED_CERTIFICATE", "reason": str(e), "costs": asdict(costs)}


def acquire(data, limits=None):
    costs = Costs()
    try:
        inst = Instance(data, limits)
        defect = inst.join_defect(costs)
        if defect:
            return {"status": "NO_GRAPH_WITNESS", "reason": "binary join defect",
                    "defect": defect, "costs": asdict(costs)}
        maps, indices, forests = [], {}, {}
        for p, q in inst.edges:
            c = inst.d[p][q]
            uf, forest = DSU(inst.k), []
            for u, v in inst.edges:
                if not inst.theta[c][u][v] or uf.find(u) == uf.find(v):
                    continue
                mapping = inst.placement(p, q, u, v, costs)
                if mapping is None:
                    mapping = inst.placement(p, q, v, u, costs)
                if mapping is None:
                    continue
                key = tuple(mapping)
                if key not in indices:
                    if len(maps) >= inst.limits.operations:
                        raise BudgetExceeded("output operation cap")
                    indices[key] = len(maps)
                    maps.append(mapping)
                forest.append([u, v, indices[key]])
                uf.add(u, v)
                costs.forest_edges += 1
            forests[f"{p},{q}"] = forest
            if uf.relation() != inst.theta[c]:
                return {"status": "NO_GRAPH_WITNESS", "reason": "singleton closure defect",
                        "seed": [p, q], "color": c, "closure": uf.relation(),
                        "costs": asdict(costs)}
        certificate = {"maps": maps, "forests": forests}
        check = verify(data, certificate, limits)
        if check["status"] != "CERTIFIED_FULL_CONGRUENCE":
            raise AssertionError(check)
        return {"status": "CERTIFIED_FULL_CONGRUENCE", "certificate": certificate,
                "costs": asdict(costs), "verification": check}
    except BudgetExceeded as e:
        return {"status": "BUDGET_UNKNOWN", "reason": str(e), "costs": asdict(costs)}
    except InvalidInput as e:
        return {"status": "INVALID_INPUT", "reason": str(e), "costs": asdict(costs)}
