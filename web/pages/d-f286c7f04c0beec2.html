"""Exact weighted-partial-flag oracle over Q or a caller-certified prime field.

Independent implementation of F_new(h)=F(h)+(S intersect F(h-alpha)).
It is an oracle and trace auditor, not the Hirai optimization iteration.
No external optimizer, unverified prime sampler, or rational flag lift is used.
Matrices for subspaces consist of ROW vectors spanning the subspace.

class Field:
    def __init__(self, p=None):
        if p is not None and (not isinstance(p,int) or p<2):
            raise ValueError("p must be a prime integer; primality is a caller obligation")
        self.p=p; self.ops=Counter(); self.max_bits=0
    def track(self,a):
        if self.p is None:
            self.max_bits=max(self.max_bits,abs(a.numerator).bit_length(),a.denominator.bit_length())
        else: self.max_bits=max(self.max_bits,a.bit_length())
        return a
    def elt(self,a):
        self.ops['convert']+=1
        value=Fraction(a)
        if self.p is None: return self.track(value)
        if value.denominator % self.p==0: raise ValueError("rational denominator is zero in supplied field")
        self.ops['conversion_mod']+=1
        if value.denominator==1: return self.track(value.numerator%self.p)
        self.ops['conversion_inverse']+=1; self.ops['conversion_mul']+=1
        return self.track(value.numerator*pow(value.denominator,-1,self.p)%self.p)
    def add(self,a,b):
        self.ops['add']+=1; return self.track(a+b if self.p is None else (a+b)%self.p)
    def sub(self,a,b):
        self.ops['sub']+=1; return self.track(a-b if self.p is None else (a-b)%self.p)
    def mul(self,a,b):
        self.ops['mul']+=1; return self.track(a*b if self.p is None else a*b%self.p)
    def div(self,a,b):
        self.ops['div']+=1
        if not b: raise ZeroDivisionError("zero pivot")
        return self.track(a/b if self.p is None else a*pow(b,-1,self.p)%self.p)
    def report(self):
        return {'field':'Q' if self.p is None else 'Fp','p':self.p,
                'declared_primitive_calls':dict(self.ops),'max_scalar_bits':self.max_bits,
                'p_bits':self.p.bit_length() if self.p else None,
                'count_scope':'declared field/conversion calls; excludes Fraction weight operations, internals, serialization and I/O',
                'prime_obligation':'caller supplies prime; this oracle does not certify primality' if self.p else None}


def rref(rows,n,field):
    a=[[field.elt(x) for x in row] for row in rows]
    if any(len(row)!=n for row in a): raise ValueError("row length mismatch")
    pivots=[]; k=0
    for j in range(n):
        pivot=next((i for i in range(k,len(a)) if a[i][j]),None)
        if pivot is None: continue
        a[k],a[pivot]=a[pivot],a[k]; factor=a[k][j]
        a[k]=[field.div(x,factor) for x in a[k]]
        for i in range(len(a)):
            if i!=k and a[i][j]:
                c=a[i][j]; a[i]=[field.sub(x,field.mul(c,y)) for x,y in zip(a[i],a[k])]
        pivots.append(j); k+=1
        if k==len(a): break
    return tuple(tuple(row) for row in a[:k]),tuple(pivots)


def span(rows,n,field): return rref(rows,n,field)[0]


def nullspace(rows,n,field):
    reduced,piv=rref(rows,n,field); free=[j for j in range(n) if j not in piv]
    out=[]
    for j in free:
        v=[field.elt(0) for _ in range(n)]; v[j]=field.elt(1)
        for i,k in enumerate(piv): v[k]=field.sub(field.elt(0),reduced[i][j])
        out.append(tuple(v))
    return tuple(out)


def intersection(u,v,n,field):
    if not u or not v: return ()
    # Kernel coefficients satisfy a U - b V = 0 in the ambient space.
    rows=[[*(row[j] for row in u),*(field.sub(field.elt(0),row[j]) for row in v)] for j in range(n)]
    ker=nullspace(rows,len(u)+len(v),field); out=[]
    for coeff in ker:
        out.append(tuple(sum_field((field.mul(coeff[i],u[i][j]) for i in range(len(u))),field) for j in range(n)))
    return span(out,n,field)


def sum_field(values,field):
    total=field.elt(0)
    for x in values: total=field.add(total,x)
    return total


def ambient(n,field): return tuple(tuple(field.elt(int(i==j)) for j in range(n)) for i in range(n))


@dataclass(frozen=True)
class PartialFlag:
    n: int
    levels: tuple
    spaces: tuple
    def at(self,h,field):
        h=Fraction(h)
        if h<=0: return ambient(self.n,field)
        out=()
        for w,s in zip(self.levels,self.spaces):
            if w<h: break
            out=s
        return out
    def verify(self,field,cap=None):
        if len(self.levels)!=len(self.spaces): raise ValueError("level/space count mismatch")
        previous=()
        for i,(w,s) in enumerate(zip(self.levels,self.spaces)):
            if w<=0 or (cap is not None and w>cap): raise ValueError("levels outside positive cap")
            if i and self.levels[i-1]<=w: raise ValueError("levels not strictly descending")
            if s!=span(s,self.n,field): raise ValueError("space is not canonical RREF")
            if len(s)<=len(previous) or span(previous+s,self.n,field)!=s: raise ValueError("spaces not strictly nested")
            previous=s
        return True
    def dump(self):
        return {'n':self.n,'levels':[str(v) for v in self.levels],
                'spaces':[[[str(x) for x in row] for row in s] for s in self.spaces]}


def make_flag(n,levels,spaces,field):
    result=PartialFlag(n,tuple(Fraction(w) for w in levels),tuple(span(s,n,field) for s in spaces))
    result.verify(field); return result


def from_basis(v,p,field):
    n=len(v)
    if any(len(row)!=n for row in v) or len(p)!=n: raise ValueError("basis/weight shape mismatch")
    cols=tuple(tuple(field.elt(v[i][j]) for i in range(n)) for j in range(n))
    if len(span(cols,n,field))!=n: raise ValueError("basis is singular")
    weights=tuple(Fraction(w) for w in p)
    if any(w<0 for w in weights): raise ValueError("weights must be nonnegative")
    order=sorted(range(n),key=lambda i:(-weights[i],i))
    levels=sorted(set(w for w in weights if w>0),reverse=True)
    spaces=[span(tuple(cols[i] for i in order if weights[i]>=w),n,field) for w in levels]
    return PartialFlag(n,tuple(levels),tuple(spaces))


def support(flag,s,field):
    s=span(s,flag.n,field); out=Fraction(0)
    for i,(w,a) in enumerate(zip(flag.levels,flag.spaces)):
        nextw=flag.levels[i+1] if i+1<len(flag.levels) else Fraction(0)
        out+=(w-nextw)*len(intersection(s,a,flag.n,field))
    return out


def positive_step(flag,s,alpha,cap,field):
    alpha=Fraction(alpha); cap=Fraction(cap); flag.verify(field,cap)
    if alpha<0 or cap<0: raise ValueError("alpha and cap must be nonnegative")
    s=span(s,flag.n,field)
    if not alpha or not s: return flag
    # 0-level coordinates of S can acquire positive weight.
    candidates={min(cap,w) for w in flag.levels}
    candidates.update(min(cap,w+alpha) for w in (*flag.levels,Fraction(0)))
    outlevels=[]; outspaces=[]; last=()
    for h in sorted((h for h in candidates if h>0),reverse=True):
        a=flag.at(h,field); b=intersection(s,flag.at(h-alpha,field),flag.n,field)
        w=span(a+b,flag.n,field)
        if w and w!=last:
            outlevels.append(h); outspaces.append(w); last=w
    result=PartialFlag(flag.n,tuple(outlevels),tuple(outspaces)); result.verify(field,cap); return result


def negative_step(flag,t,field):
    t=Fraction(t)
    if t<0: raise ValueError("t must be nonnegative")
    return PartialFlag(flag.n,tuple(w-t for w in flag.levels if w>t),tuple(s for w,s in zip(flag.levels,flag.spaces) if w>t))


def objective(flag,subspaces,c,u,field):
    if not (len(subspaces)==len(c)==len(u)): raise ValueError("objective length mismatch")
    total=sum((flag.levels[i]-(flag.levels[i+1] if i+1<len(flag.levels) else 0))*len(a) for i,a in enumerate(flag.spaces))
    return total+sum(Fraction(uk)*max(Fraction(0),Fraction(ck)-support(flag,s,field)) for s,ck,uk in zip(subspaces,c,u))


def squared_distance(first,second,field):
    """Exact building metric using only intersection dimensions of partial flags."""
    if first.n!=second.n: raise ValueError("ambient dimension mismatch")
    def norm2(flag):
        lastdim=0; answer=Fraction(0)
        for w,s in zip(flag.levels,flag.spaces):
            answer+=w*w*(len(s)-lastdim); lastdim=len(s)
        return answer
    dot=Fraction(0)
    for i,(w,a) in enumerate(zip(first.levels,first.spaces)):
        dw=w-(first.levels[i+1] if i+1<len(first.levels) else 0)
        for j,(z,b) in enumerate(zip(second.levels,second.spaces)):
            dz=z-(second.levels[j+1] if j+1<len(second.levels) else 0)
            dot+=dw*dz*len(intersection(a,b,first.n,field))
    answer=norm2(first)+norm2(second)-2*dot
    if answer<0: raise ArithmeticError("negative metric square")
    return answer
