import numpy as np

def boundary_nodes(n):
    return np.array([i for i in range(n+1)]+[j*(n+1)+n for j in range(1,n+1)]+[n*(n+1)+i for i in range(n-1,-1,-1)]+[j*(n+1) for j in range(n-1,0,-1)])

def forward(model,theta):
    A=model['A0']+np.einsum('j,jab->ab',theta,model['A'])
    U=np.linalg.solve(A,model['B'])
    return model['B'].T@U

def derivative(model,theta):
    A=model['A0']+np.einsum('j,jab->ab',theta,model['A']);U=np.linalg.solve(A,model['B'])
    return -np.einsum('ai,jab,bk->jik',U,model['A'],U)

def detector_frame(J,count):
    # Exact reference E-optimal output frame for d=2 and count even.
    u,s,vt=np.linalg.svd(J,full_matrices=False)
    if s[1]<1e-14:raise ValueError('Reference directional Jacobian deficient')
    w1=count*s[1]**2/(s[0]**2+s[1]**2);w2=count-w1
    plus=np.sqrt(w1/count)*u[:,0]+np.sqrt(w2/count)*u[:,1]
    minus=np.sqrt(w1/count)*u[:,0]-np.sqrt(w2/count)*u[:,1]
    Q=np.array([plus]*(count//2)+[minus]*(count//2))
    return Q

def design_matrix(model,design,theta=(0.,0.)):
    deriv=derivative(model,np.array(theta))
    return np.vstack([np.stack([q@d@g for d in deriv],axis=1) for g,q in design])

def secant_certificate(model,design,rho):
    J=design_matrix(model,design);alpha=np.linalg.svd(J,compute_uv=False)[-1]
    factor=rho*(2-rho)/(1-rho)**2
    # Per-current factor improves over one uniform operator bound while keeping proof exact.
    remainder=factor*np.sqrt(sum(np.linalg.norm(q@model['B0'].T,2)**2*np.linalg.norm(model['B0']@g)**2 for g,q in design))
    return {'reference_sigma':float(alpha),'uniform_remainder_bound':float(remainder),'global_secant_lower':float(alpha-remainder)}
