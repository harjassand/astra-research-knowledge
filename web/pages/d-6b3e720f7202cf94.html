from pathlib import Path

import hashlib

BASE=Path(__file__).resolve().parent
R=s.Rational
I=s.eye(3)
J=[s.Matrix(3,3,lambda j,k:-s.I*s.LeviCivita(i,j,k)) for i in range(3)]
checks=[]

def simp(A):return A.applyfunc(s.simplify)
def record(name,**details):checks.append({"name":name,"status":"PASS",**details})
def ket(i,j,k):
    v=s.zeros(27,1);v[9*i+3*j+k]=1;return v

w=s.symbols('w0:3',real=True)
k=s.symbols('k0:3',real=True)
K=s.diag(*k)
W=sum((w[i]*(s.kronecker_product(J[i].T,J[i],I)+
             s.kronecker_product(J[i].T,I,J[i])) for i in range(3)),s.zeros(27))
D5=3*s.kronecker_product(K,I,I)+s.kronecker_product(I,K,I)+s.kronecker_product(I,I,K)
Mminus=D5-5*W
Mplus=D5+5*W
blocks=[]
for i in range(3):
    j=(i+1)%3;ell=(i+2)%3
    bs=[ket(i,i,i),-ket(i,j,j),-ket(i,ell,ell),
        (ket(j,i,j)+ket(j,j,i))/s.sqrt(2),
        (ket(ell,i,ell)+ket(ell,ell,i))/s.sqrt(2)]
    ba=[(ket(j,i,j)-ket(j,j,i))/s.sqrt(2),
        (ket(ell,i,ell)-ket(ell,ell,i))/s.sqrt(2)]
    blocks.append(s.Matrix.hstack(*bs))
    blocks.append(s.Matrix.hstack(*ba))
    actual=simp(blocks[-2].T*Mminus*blocks[-2])
    expected=s.diag(5*k[i],3*k[i]+2*k[j],3*k[i]+2*k[ell],
                    k[i]+4*k[j],k[i]+4*k[ell])
    for r,c,v in [(0,3,-5*s.sqrt(2)*w[ell]),(1,3,-5*s.sqrt(2)*w[ell]),
                  (0,4,-5*s.sqrt(2)*w[j]),(2,4,-5*s.sqrt(2)*w[j]),
                  (3,4,-5*w[i])]:
        expected[r,c]=expected[c,r]=v
    assert actual==expected
odd_s=[];odd_a=[]
for i in range(3):

    odd_a.append((ket(i,j,ell)-ket(i,ell,j))/s.sqrt(2))
