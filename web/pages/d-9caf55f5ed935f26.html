def decode(v):
    if isinstance(v,dict) and set(v)=={'poly'}:
        return P.from_dict({tuple(mon):QQ(co) for mon,co in v['poly']})
    if isinstance(v,list):return [decode(x) for x in v]
    if isinstance(v,dict):
        return {(int(k) if k.isdecimal() else ast.literal_eval(k) if k.startswith('(') else k):decode(x)
                for k,x in v.items()}
    return v


def upoly(v):return U.from_dict({(int(i),):QQ(co) for i,co in v})


def compositions(n,k):
    if k==1:
        yield (n,);return
    for i in range(n+1):
        for tail in compositions(n-i,k-1):yield (i,)+tail


def subtract(row,weight,other):
    for col,value in other.items():
        new=row.get(col,P.zero)-weight*value
        if new:row[col]=new
        else:row.pop(col,None)

def final_intersection():

    def squares(expr):

    ev=red(-constant*inv(lead));da=red((7-2*x)/5)

    def add(v,w):
