"""Exact rational certificate for a three-sensor finite report converter.

Baseline: two independent readings per action, minimum count classifier,
uniform ties. Verifier: proposed state's action, random-walk likelihood
thresholds +-2 log 9. eta=1/10 makes both kinds of proposed output have
success threshold ceil(log_9(10 H_k/H_j))=2. All probabilities below are
Fractions, including gambler-ruin, thinning and residual inverse kernels.
This finite certificate does not prove the universal controlled theorem.
"""
from fractions import Fraction as F
from itertools import product
from math import comb
from pathlib import Path
import json


def baseline_matrix(n=2):
    rows=[]
    for i in range(3):
        row=[F(0)]*3
        for ks in product(range(n+1),repeat=3):
            prob=F(1)
            for a,k in enumerate(ks):
                p=F(1,10) if i==a else F(9,10)
                prob*=comb(n,k)*p**k*(1-p)**(n-k)
            winners=[a for a,k in enumerate(ks) if k==min(ks)]
            for a in winners:
                row[a]+=prob/F(len(winners))
        rows.append(row)
    return rows


def inverse(a):
    n=len(a)
    aug=[row[:]+[F(i==j) for j in range(n)] for i,row in enumerate(a)]
    for k in range(n):
        pivot=next(i for i in range(k,n) if aug[i][k])
        aug[k],aug[pivot]=aug[pivot],aug[k]
        divisor=aug[k][k]
        aug[k]=[x/divisor for x in aug[k]]
        for i in range(n):
            if i!=k:
                ratio=aug[i][k]
                aug[i]=[x-ratio*y for x,y in zip(aug[i],aug[k])]
    return [row[n:] for row in aug]


def matmul(a,b):
    return [[sum(x*y for x,y in zip(row,col)) for col in zip(*b)] for row in a]


eta=F(1,10)
H=[[F(4,5) if i==y else F(1,10) for y in range(3)] for i in range(3)]
A=baseline_matrix()
assert all(1-A[i][i]<=eta for i in range(3))
assert A[0][0]==F(594,625)
thinning=(1-eta)/(1+2*eta)
# Probability of hitting +2 before -2 from zero, derived by the
# exact harmonic gambler-ruin solution h(s)=(1-r^(s+2))/(1-r^4).
def pass_probability(p_up, state=0):
    ratio=(1-p_up)/p_up
    return (1-ratio**(state+2))/(1-ratio**4)

for p_up in [F(9,10),F(1,10)]:
    assert pass_probability(p_up,-2)==0
    assert pass_probability(p_up,2)==1
    for s in [-1,0,1]:
        assert pass_probability(p_up,s)==p_up*pass_probability(p_up,s+1)+(1-p_up)*pass_probability(p_up,s-1)

correct_pass=pass_probability(F(9,10))
wrong_pass=pass_probability(F(1,10))
assert correct_pass==F(81,82) and wrong_pass==F(1,82)
assert correct_pass>=1-2*eta
Q=[[thinning*sum(A[i][k]*H[k][y]*(correct_pass if i==k else wrong_pass)
                          for k in range(3))
    for y in range(3)] for i in range(3)]
assert all(Q[i][y]<=(1-eta)*H[i][y] for i in range(3) for y in range(3))
delta=[1-sum(row) for row in Q]
assert all(eta<=di<=6*eta for di in delta)
R=[[(H[i][y]-Q[i][y])/delta[i] for y in range(3)] for i in range(3)]
K=matmul(inverse(A),R)
assert all(x>=0 for row in K for x in row)
assert all(sum(row)==1 for row in K)
repaired=matmul(A,K)
assert repaired==R
final=[[Q[i][y]+delta[i]*repaired[i][y] for y in range(3)] for i in range(3)]
assert final==H
expected_verifier=2*(2*correct_pass-1)/(2*F(9,10)-1)
assert expected_verifier==2*(2*wrong_pass-1)/(2*F(1,10)-1)==F(100,41)
expected_cost=[6+expected_verifier+di*6 for di in delta]


def encode(obj):
    if isinstance(obj,F):return {"fraction":str(obj),"decimal":float(obj)}
    if isinstance(obj,list):return [encode(x) for x in obj]
    if isinstance(obj,dict):return {k:encode(v) for k,v in obj.items()}
    return obj


out=encode({"baseline_confusion":A,"thinning":thinning,
            "verification_pass_correct":correct_pass,"verification_pass_wrong":wrong_pass,
            "accepted_mass":Q,"failure_probabilities":delta,
            "residual_experiment":R,"repair_markov_kernel":K,
            "exact_final_output":final,"expected_total_cost":expected_cost,
            "all_checks":"PASS with exact Fraction arithmetic",
            "scope":"Single finite target report; universal theorem remains analytically proved separately"})
dest=Path(__file__).resolve().parents[2]/"results/check_selection/exact_finite_converter.json"
dest.write_text(json.dumps(out,indent=2)+"\n")
print(json.dumps({"checks":out["all_checks"],"expected_cost":out["expected_total_cost"],
                  "smallest_repair_probability":float(min(x for row in K for x in row))},indent=2))
