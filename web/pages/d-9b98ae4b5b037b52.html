#!/usr/bin/env python3
"""Independent exact coefficient checks for the characteristic transport gate.

The analytic coarea, moving-domain and flow estimates are proved in the gate
note. These checks only guard transcription of its local exact identities.
"""
import json
from pathlib import Path
import sympy as s

checks = {}
tau_max = s.Rational(1,100)
sig_max = s.Rational(1,800)
lo_grad2 = s.Rational(7,10)**2-(s.Rational(3,10)+tau_max+sig_max)**2
checks['dominant_shifted_band_gradient'] = lo_grad2>s.Rational(3,5)**2
checks['translated_characteristic_coarea_speed'] = s.Rational(3,5)**2-s.Rational(4,5)*sig_max>s.Rational(1,3)
checks['cell_gap_exceeds_16'] = (3*s.pi/(10*tau_max))**2>16
checks['signed_source_barrier_exceeds_2'] = 3*(2*s.Rational(3,5)**2-2*tau_max**2)>2
checks['two_currents_and_regular_integral_below_50tau'] = 48*s.exp(sig_max)<50
p,g2,tau = s.symbols('P gradP_squared tau',real=True)
# Delta(P^2)=2|grad P|^2-2P^2 since Delta P=-P.
H = 3*(tau**2-p**2)
minus_L_H = 6*g2-6*p**2-H
checks['full_signed_barrier_identity'] = s.expand(minus_L_H-3*(2*g2-tau**2-p**2))==0

r,a,K,n,mu,C,S,cp,cpp = s.symbols('r a K n mu C S cp cpp',real=True)
radial_norm = 1-K**2/r**2
lap_r = (n-1)/r+K**2/r**3
# Product rule, before grouping: E c(R) cos(at).
product = (a**2*radial_norm-a*lap_r)*C
product += (cpp*radial_norm+cp*lap_r)*C-a**2*C
product += -2*a*cp*radial_norm*C +2*a**2*mu*S-2*a*cp*mu*S
claimed = -(K**2/r**2+lap_r/a)*C
claimed += (cpp*radial_norm/a**2+cp*lap_r/a**2-2*cp*radial_norm/a)*C
claimed += 2*mu*S-2*cp*mu*S/a
checks['exact_radial_shape_product_rule'] = s.simplify(product/a**2-claimed)==0
v,c = s.symbols('v c',real=True,positive=True)
theta = 2*s.atan(s.exp(-c*v))
checks['full_extrema_characteristic_equation'] = s.simplify(s.diff(theta,v)+c*s.sin(theta))==0
checks['characteristic_cosine_is_tanh'] = s.simplify(s.cos(theta)-s.tanh(c*v))==0
checks['characteristic_radial_fold'] = s.simplify(s.diff(s.log(s.sin(theta)),v)+c*s.cos(theta))==0

rows = {}
for m in (2,3):
    e,D,Q,h,d = s.symbols('eta delta_coefficient origin_coefficient h_sigma d',real=True)
    coeff = s.symbols(f'c0:{m}',positive=True)
    jet = s.symbols(f'b0:{m}',real=True)
    M = s.symbols(f'M0:{m}',real=True)
    T = s.symbols(f'T0:{m}',real=True)
    # These averages follow from the exact radial harmonic mean. The cross
    # moments are zero by coordinate reflections. At the stated orders they
    # directly give the following independently grouped coefficients.
    ft = [-M[i]*sum(coeff)-coeff[i]*T[i]+h*coeff[i]*M[i] for i in range(m)]
    lap = 2*sum(coeff[i]*M[i]*T[i] for i in range(m))
    lap += -2*h*sum(coeff[i]*M[i]**2 for i in range(m))
    lap += (2*h-d)*D*sum(coeff)-Q*sum(coeff)-2*D*sum(jet)
    normalize = {coeff[-1]:1-sum(coeff[:-1]),jet[-1]:-sum(jet[:-1])}
    expected_ft = [-coeff[i]*T[i]-(1-h*coeff[i])*M[i] for i in range(m)]
    checks[f'{m}wave_first_crest_rows'] = all(s.simplify((ft[i]-expected_ft[i]).subs(normalize))==0 for i in range(m))
    crest = {T[i]:-(1/coeff[i]-h)*M[i] for i in range(m)}
    critical_lap = s.simplify(lap.subs(crest).subs(normalize))
    checks[f'{m}wave_shape_sum_retained_then_canceled'] = s.simplify((-2*D*sum(jet)).subs(normalize))==0
    checks[f'{m}wave_kernel_independent_crest_quadratic'] = s.simplify(critical_lap-(-2*sum(x**2 for x in M)+(2*h-d)*D-Q))==0
    t0 = s.symbols(f't0_0:{m}',real=True)
    neighboring = -2*a/r**2*sum((x+s.pi/(2*a))**2-(x-s.pi/(2*a))**2 for x in t0)
    checks[f'{m}wave_neighboring_crest_difference'] = s.simplify(neighboring+4*s.pi*sum(t0)/r**2)==0
    rows[f'{m}wave']={'gradient_first_weight':[str(s.simplify(x.subs(normalize))) for x in ft],
                     'laplacian_second_weight_before_normalization':str(lap),
                     'laplacian_second_weight_at_crest':str(critical_lap),
                     'neighboring_crest_difference':str(s.simplify(neighboring))}

R,A = s.symbols('R A',positive=True)
eta = 1/s.sqrt(A*R)
L = s.sqrt(R/A)
checks['required_return_margin'] = s.simplify(A*eta**3-L/R**2)==0
checks['actual_C3_error_is_one_eta_smaller'] = s.simplify(A*eta**4/(L/R**2)-eta)==0
checks['soft_neck_remainder_is_one_eta_smaller'] = s.simplify((eta**2/R)/(L/R**2)-eta)==0
checks['nominal_neighbor_error_is_one_eta_smaller'] = s.simplify((A*eta**3/L)*(1/A)/(L/R**2)-eta)==0
result = {'scope':'Independent finite exact coefficient/transcription checks; analytic coarea, actual global repair and stable-return proof are in MULTIWAVE_CONDUCTIVITY_GATE.md. No PDE solver or global-cover classification.',
          'checks':{k:bool(value) for k,value in checks.items()},'rows':rows}
result['all_checks_pass']=all(result['checks'].values())
out=Path('results/pde_spectrum9/cycle12/multiwave_transport_reconstruction.json')
out.write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'all_checks_pass':result['all_checks_pass'],'checks':result['checks']},indent=2))
raise SystemExit(0 if result['all_checks_pass'] else 1)
