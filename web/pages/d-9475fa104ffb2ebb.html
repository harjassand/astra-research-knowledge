def compositions(n,k):
    if k==1:
        yield(n,);return
    for i in range(n+1):
        for rest in compositions(n-i,k-1):yield(i,)+rest

def z(v):assert s.expand(v)==0

def polar():
    x=s.symbols('x0:3');v=s.symbols('v0:3');q=s.symbols('q00 q01 q02 q11 q12 q22');F=s.symbols('F')
    Q=s.Matrix([[q[0],q[1],q[2]],[q[1],q[3],q[4]],[q[2],q[4],q[5]]])
    def quartic(vec):
        vv=s.Matrix([t*t for t in vec]);return s.expand((vv.T*Q*vv)[0])
    fx=quartic(x);fv=quartic(v);t=s.symbols('t')
    # Direct polarization of the original quartic, fixing ordinary coefficient conventions.
    tvxxx=sum(v[i]*s.diff(fx,x[i]) for i in range(3))/4
    tvvxx=s.Poly(quartic([v[i]+t*x[i] for i in range(3)]),t).coeff_monomial(t*t)/6
    g=sum(x[i]*s.diff(fv,v[i]) for i in range(3))/4
    mass=[vi*vi for vi in v];sv=list(Q*s.Matrix(mass))
    z(g-sum(v[i]*sv[i]*x[i] for i in range(3)))
    pp=s.Poly(s.expand(F*F*tvxxx-3*F*g*tvvxx+2*g**3),*x)
    for i in range(3):
        mon=[0]*3;mon[i]=3
        ai=(F-2*mass[i]*sv[i])*(Q[i,i]*F-sv[i]*sv[i])
        z(pp.coeff_monomial(tuple(mon))-v[i]*ai)
    for i,j in itertools.permutations(range(3),2):
        mon=[0]*3;mon[i]=2;mon[j]=1
        hij=Q[i,j]*F*F-F*sv[j]*(sv[i]+2*Q[i,i]*mass[i])-4*F*Q[i,j]*mass[i]*sv[i]+6*mass[i]*sv[i]**2*sv[j]
        z(pp.coeff_monomial(tuple(mon))-v[j]*hij)
    bt=3*sv[0]*sv[1]*sv[2]-F*(Q[1,2]*sv[0]+Q[0,2]*sv[1]+Q[0,1]*sv[2])
    z(pp.coeff_monomial((1,1,1))-4*v[0]*v[1]*v[2]*bt)
    # All formal-F coefficient terms are homogeneous degree9 with weightF=4.
    terms=0
    for co in pp.coeffs():
        for mon,value in s.Poly(co,F,*v,*q).terms():
            assert 4*mon[0]+sum(mon[1:4])==9
            terms+=1
    return {'status':'PASS original quartic polarization',
            'generator_degree':9,'ordinary_x_triple_factor':4,'checked_coefficient_terms':terms}

def homogeneity_budget():
    # Each m,s substitutes to a degree2 numerator /sqrtF. The cleared A/H
    # have denominatorF²; B has denominatorF^(3/2). Parity removes roots.
    counts=[];maximum_den=0;max_polar_multiplier=0
    for i in range(10):
        source_count=0
        for degree in ((0,2,4) if i<9 else (1,3,5)):
            power=degree//2+2 if i<9 else (degree+3)//2
            assert power<=4
            for mon in compositions(degree,6):
                source_count+=1;maximum_den=max(maximum_den,power)
                # Multiplication by active triple absorbsvi orvj; B already has triple.
                ambient_multiplier=2*degree+4*(4-power)+(2 if i<9 else 0)
                assert ambient_multiplier==10
                max_polar_multiplier=max(max_polar_multiplier,ambient_multiplier)
        counts.append(source_count)
    assert sum(counts)==1646
    return {'status':'PASS every original parity-admitted raw source degree budget',
            'counts_per_generator':counts,'total_sources':sum(counts),
            'maximum_integer_F_power':maximum_den,'F4_K3_degree':19,
            'maximum_polar_multiplier_degree':max_polar_multiplier,
            'auxiliary_variable_inversion':False}

def pair_identities():
    a,b,c,m,n,u,w=s.symbols('a b c m n u w')
    Delta=a*c-b*b;F=a*m*m+2*b*m*n+c*n*n;p=a*m*m-c*n*n
    si=a*m+b*n;sj=b*m+c*n;sk=u*m+w*n;ck=u*m-w*n
    Ai=(F-2*m*si)*(a*F-si*si)
    Aj=(F-2*n*sj)*(c*F-sj*sj)
    Hij=b*F*F-F*sj*(si+2*a*m)-4*F*b*m*si+6*m*si*si*sj
    z(Ai+Delta*n*n*p);z(Aj-Delta*m*m*p);z(Hij-3*Delta*m*n*p)
    z(Delta*F*p-(a*Aj+2*b*Hij/3-c*Ai))
    z(F*u-si*sk-si*ck+u*p)
    z(F*w-sj*sk+sj*ck-w*p)
    Ei=F*(si*ck-u*p);Ej=F*(-sj*ck+w*p)
    z(Delta*F*m*ck-(c*Ei+b*Ej+(c*u-b*w)*F*p))
    z(Delta*F*n*ck-(-b*Ei-a*Ej+(a*w-b*u)*F*p))
    delta=a*w*w-c*u*u;N=a*w*w+2*b*u*w+c*u*u
    z(delta*F-N*p+2*ck*(a*(b*w+c*u)*m+c*(a*w+b*u)*n))
    return {'status':'PASS arbitrary independent-parameter binary identities',
            'Delta_division_premise':'positive definite principal2x2',
            'coprime_delta_division':'only on delta!=0; delta0 gives dividing gcd branch',
            'pair_lift_powers':2,'full_F6_J_from_F4_K3':True}
