class BudgetUnknown(Exception):
    def __init__(self,details):self.details=details


def preflight(k,max_vertices=2000,max_weight_bits=1000000):
    if type(k) is not int or k<0:raise ValueError('nonnegative integer rounds required')
    if type(max_vertices) is not int or max_vertices<1 or type(max_weight_bits) is not int or max_weight_bits<1:
        raise ValueError('positive integer budgets required')
    n=9+12*k
    # No identity arrays, weights, vertex lists or huge output exponents yet.
    if n>max_vertices or n*n>max_weight_bits:
        raise BudgetUnknown({'status':'BUDGET_UNKNOWN','phase':'pre_construction',
                             'rounds':k,'vertex_count':n,'absolute_weight_binary_bits':n*n})
    return n

def declared_profile(nodes):
    return [3 if node['op']=='frame' else node['expected_dimension'] for node in nodes]


def recognize(data,max_vertices=2000,max_weight_bits=1000000):

        or any(type(x) is not int for a in data.get('arrows',[]) for row in a.get('matrix',[]) for x in row)):
        return None
    if any(data.get(key)!=expected[key] for key in ('dimensions','weight','arrows','certificate')):return None
    return expected


def evaluate(nodes,field):
    bases=[]
    for i,node in enumerate(nodes):
        if node['op']=='frame':u=ambient(3,field)
        elif node['op']=='constant':
            m=node['matrix'];u=span([tuple(m[r][j] for r in range(3)) for j in range(3)],3,field)
        else:
            p,q=node['parents']
            if p>=i or q>=i:raise ValueError('chronological parents required')
            u=span(bases[p]+bases[q],3,field) if node['op']=='join' else intersection(bases[p],bases[q],3,field)
        bases.append(u)
    return bases


def check_closure(data,bases,field):
    for arrow in data['arrows']:
        t,h=arrow['tail'],arrow['head'];matrix=arrow['matrix']
        im=span([tuple(sum_field((field.mul(field.elt(matrix[r][j]),v[j]) for j in range(3)),field)
                       for r in range(3)) for v in bases[t]],3,field)
        if span(bases[h]+im,3,field)!=bases[h]:return False
    return True
