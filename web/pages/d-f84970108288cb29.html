gamma = F(4,5)*eps

# Conservative absolute Hessian bounds on [.99,1.01]^2.
hessian_bounds = [20*q+144*q**6, 20*q+192*q**6, 144*q**6,
                  30*q+192*q**6, 30*q+256*q**6, 192*q**6]
assert max(hessian_bounds) < 310
