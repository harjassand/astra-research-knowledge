def arithmetic(text, names):
    def rec(node):
        if isinstance(node, ast.Constant) and type(node.value) is int:
            return sy.Integer(node.value)
        if isinstance(node, ast.Name) and node.id in names:
            return names[node.id]
        if isinstance(node, ast.UnaryOp):
            value = rec(node.operand)
            if isinstance(node.op, ast.USub): return -value
            if isinstance(node.op, ast.UAdd): return value
        if isinstance(node, ast.BinOp):
            lhs, rhs = rec(node.left), rec(node.right)
            if isinstance(node.op, ast.Add): return lhs + rhs
            if isinstance(node.op, ast.Sub): return lhs - rhs
            if isinstance(node.op, ast.Mult): return lhs * rhs
            if isinstance(node.op, ast.Div): return lhs / rhs
            if isinstance(node.op, ast.Pow) and rhs.is_Integer:
                return lhs ** rhs
        raise ValueError(ast.dump(node))
    return rec(ast.parse(text, mode='eval').body)


def zero(expression):
    assert sy.cancel(expression) == 0, str(expression)


def generators(q, m, s):
    result = {f'A{i}': (1-2*m[i]*s[i])*(q[i,i]-s[i]**2)
              for i in range(3)}
    for i, j in itertools.permutations(range(3), 2):
        result[f'H{i}{j}'] = ((1-4*m[i]*s[i])*(q[i,j]-s[i]*s[j])
                             -2*m[i]*s[j]*(q[i,i]-s[i]**2))
    result['B'] = (3*sy.prod(s)-sum(q[j,k]*s[i]
                                 for i,j,k in [(0,1,2),(1,0,2),(2,0,1)]))
    return result


def chart_gate():
    a,b,c,d,A,B = sy.symbols('a b c d A B')
    path = BASE/'results/metric_prior/cycle13/generic_d8_divisor_factors.json'
    data = json.loads(path.read_text())
    factors = [arithmetic(t, {'a':a,'b':b,'c':c}) for t in data['factors']]
    assert len(factors)==28 and len(set(factors))==28
    # Audit every original divisor decomposition too: its factors must come
    # from the precise finite list used in the gated generic raw identity.
    for record in data['decompositions']:
        own = arithmetic(record['constant'], {})
        for text, exponent in record['factors']:
            factor=arithmetic(text, {'a':a,'b':b,'c':c})
            assert factor in factors
            own *= factor**exponent
        zero(own-arithmetic(record['divisor'], {'a':a,'b':b,'c':c}))
    assert len(data['decompositions'])==42

    # Exact sign-transparent rewrites: signs use 0<a,b<1,c<0,|c|<1.
    sign_forms = {
        0: b*(7*a*a-27)+a*c*(14*a*a+6),
        1: a*(b*b+3*c*c)+b*c*(2*a*a-6),
        2: b*c*(2*a*a+5)-7*a,
        3: a*(2*a*a-7)+5*b*c,
        4: b*b*(2*a*a-9)-a*a*c*c+a*b*c*(2*a*a+6),
        5: c*c*(2*a*a-9)-a*a*b*b+a*b*c*(2*a*a+6),
        6: c*(a*a*(2*a*a-2*b*b+13)+5*b*b)-a*b*(11*c*c+7),
        18: a*a*b*b+(9-2*b*b)*c*c-2*a*b*c*(b*b+3),
        19: b*(a-3)+2*a*c,
        22: b*(a+3)-2*a*c,
        23: a*c-b,
    }
    for i, expression in sign_forms.items(): zero(factors[i]-expression)
    simple=[9,10,11,12,13,14,15,16,24,25,26,27]
    exceptions=[7,8,17,20,21]
    assert sorted(simple+list(sign_forms)+exceptions)==list(range(28))

    # Two initial exclusions are removed by ordered-edge inequalities.
    R=factors[17]; giant=factors[8]
    L=63-33*B-2*B*B+(A-B)*(12*B+24)
    K=B*(63-33*B-16*B*B)+(A-B)*(63-18*B-16*B*B)
    zero(giant.subs(c,-d)-(7*A*B*(A+B)+a*b*d*L+d*d*K).subs({A:a*a,B:b*b}))
    zero(R.subs(c,-d)-b*(9+a*a-b*b+d*d)+6*a*d)

    cases={
        'w=0': (a*b/3, [-13*a*b/3,b*(a*a/3-3),b*(a*a/3-1),
                       a*a-9+b*b*(1-a*a/9)]),
        'u=0': (a*b, [-3*a*b,b*(a*a-3),b*(a*a-1),
                     3+a*a+b*b*(1-a*a)]),
        'z=0': (2*b/(5*a), [b*(4-25*a*a)/(5*a),-13*b/5,-3*b/5,
                            9-14*a*a+b*b*(1-4/(25*a*a))]),
    }
    chart2={a:a,b:d,c:-b}
    c2={}
    for label,(relation, expected) in cases.items():
        # Recompute all 28 transformed polynomials, not just selected entries.
        values=[sy.cancel(f.subs(chart2,simultaneous=True).subs(d,relation)) for f in factors]
        observed=[values[7],values[20],values[21],values[17]/relation]
        for p,q in zip(observed,expected): zero(p-q)
        c2[label]=[str(v) for v in observed]
    chart3={a:b,b:a,c:-2*b/(5*a)}
    c3=[sy.cancel(f.subs(chart3,simultaneous=True)) for f in factors]
    expected3={
        7:2*(a*a-b*b)/a,
        20:b*(5*a*a-6)/(5*a),
        21:b*(5*a*a-2)/(5*a),
        17:a*(9+b*b-a*a-56*b*b/(25*a*a)),
        8:7*b*b*(5*a**6+90*a*a*(1-a*a)+b*b*(33*a**4+24*a*a+36))/(25*a*a),
    }
    for i,v in expected3.items(): zero(c3[i]-v)
    # Exact certificate for the inequality R(C3)/a > 144/25:
    # R/a -144/25 = (1-A)+B+(56/25)*(1-B/A).
    zero(expected3[17]/a-sy.Rational(144,25)-
         (1-a*a+b*b+sy.Rational(56,25)*(1-b*b/(a*a))))
    exceptional={a:sy.Rational(2,5),b:sy.Rational(2,5),c:-sy.Rational(2,5)}
    q=sy.Matrix([[1,a,b],[a,1,c],[b,c,1]]).subs(exceptional)
    eps=sy.diag(-1,1,1)
    assert eps*q*eps == sy.Matrix([[1,-sy.Rational(2,5),-sy.Rational(2,5)],
                                 [-sy.Rational(2,5),1,-sy.Rational(2,5)],
                                 [-sy.Rational(2,5),-sy.Rational(2,5),1]])
    assert q.det()==sy.Rational(49,125)
    return {'complete_factor_count':28,'exact_divisor_decompositions':42,
            'factor_file_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
            'universally_safe':sorted(simple+list(sign_forms)),
            'ordered_chart_possible_zeros':[7,20,21],
            'C2_exception_curve_values':c2,
            'C3_z_curve_values':{str(i):str(v) for i,v in expected3.items()},
            'residual':'a=b=d=2/5; auxiliary sign congruent to r=-2/5',
            'residual_PD_determinant':str(q.det()),
            'strict_domain_coverage':'proved in companion note, not inferred from sampling'}


def raw_transport_gate():
    m=sy.symbols('m0:3'); s=sy.symbols('s0:3')
    q00,q11,q22,q01,q02,q12=sy.symbols('q00 q11 q22 q01 q02 q12')
    q=sy.Matrix([[q00,q01,q02],[q01,q11,q12],[q02,q12,q22]])
    own=generators(q,m,s)
    checked=0
    for perm in itertools.permutations(range(3)):
        for signs in itertools.product([-1,1],repeat=3):
            qt=sy.Matrix(3,3,lambda i,j: signs[i]*signs[j]*q[perm[i],perm[j]])
            mt=[signs[i]*m[perm[i]] for i in range(3)]
            st=[signs[i]*s[perm[i]] for i in range(3)]
            image=generators(qt,mt,st)
            for i in range(3): zero(image[f'A{i}']-own[f'A{perm[i]}'])
            for i,j in itertools.permutations(range(3),2):
                zero(image[f'H{i}{j}']-signs[i]*signs[j]*own[f'H{perm[i]}{perm[j]}'])
            zero(image['B']-sy.prod(signs)*own['B'])
            checked+=1
    u=sy.symbols('u0:3',nonzero=True)
    q=sy.Matrix([[u[0]**2,q01,q02],[q01,u[1]**2,q12],[q02,q12,u[2]**2]])
    own=generators(q,m,s)
    qt=sy.Matrix(3,3,lambda i,j:q[i,j]/(u[i]*u[j]))
    image=generators(qt,[u[i]*m[i] for i in range(3)], [s[i]/u[i] for i in range(3)])
    for i in range(3):zero(u[i]**2*image[f'A{i}']-own[f'A{i}'])
    for i,j in itertools.permutations(range(3),2):
        zero(u[i]*u[j]*image[f'H{i}{j}']-own[f'H{i}{j}'])
    zero(sy.prod(u)*image['B']-own['B'])
    return {'signed_vertex_permutation_charts':checked,
            'positive_diagonal_normalization':'all ten raw generators checked symbolically',
            'warning':'auxiliary sign changes; real hidden-axis signs do not change Q'}


def polar_gate():
    # Independently derive polarization from the quartic, by coefficient
    # extraction in t, and every cubic-x coefficient of the polar expression.
    v=sy.symbols('v0:3'); x=sy.symbols('x0:3'); t=sy.symbols('t')
    q00,q11,q22,q01,q02,q12=sy.symbols('q00 q11 q22 q01 q02 q12')
    q=sy.Matrix([[q00,q01,q02],[q01,q11,q12],[q02,q12,q22]])
    def f(y):return sum(q[i,j]*y[i]**2*y[j]**2 for i in range(3) for j in range(3))
    expansion=sy.Poly(f([v[i]+t*x[i] for i in range(3)]),t)
    F=expansion.nth(0)
    g=expansion.nth(1)/4
    T2=expansion.nth(2)/6
    T3=expansion.nth(3)/4
    S=[sum(q[i,j]*v[j]**2 for j in range(3)) for i in range(3)]
    P=sy.Poly(sy.expand(F*F*T3-3*F*g*T2+2*g**3),*x)
    A=[(F-2*v[i]**2*S[i])*(q[i,i]*F-S[i]**2) for i in range(3)]
    H={(i,j):(F-4*v[i]**2*S[i])*(q[i,j]*F-S[i]*S[j])
        -2*v[i]**2*S[j]*(q[i,i]*F-S[i]**2)
        for i,j in itertools.permutations(range(3),2)}
    B=3*sy.prod(S)-F*sum(q[j,k]*S[i] for i,j,k in [(0,1,2),(1,0,2),(2,0,1)])
    for i in range(3):
        ex=[0]*3;ex[i]=3
        zero(P.coeff_monomial(tuple(ex))-v[i]*A[i])
    for i,j in itertools.permutations(range(3),2):
        ex=[0]*3;ex[i]=2;ex[j]=1
        zero(P.coeff_monomial(tuple(ex))-v[j]*H[i,j])
    zero(P.coeff_monomial((1,1,1))-4*sy.prod(v)*B)
    return {'raw_polar_cubic_coefficients':10,
            'degree_in_v':max(sum(ex) for ex,_ in sy.Poly(P.as_expr(),*v).terms()),
            'method':'quartic t-polarization and exact coefficient extraction',
            'scope':'symbolic ternary identities; ambient extension follows unchanged because S_i may include outside indices'}


def exchangeable_residual_gate():
    path=BASE/'results/metric_prior/cycle12/exchangeable_degree8_symbolic_reduced.json'
    record=json.loads(path.read_text())
    r=sy.symbols('r'); value=-sy.Rational(2,5)
    m=sy.symbols('m0:3');s=sy.symbols('s0:3');variables=tuple(m+s)
    names={str(x):x for x in variables};names['r']=r
    q=sy.Matrix(3,3,lambda i,j:1 if i==j else value)
    raw=generators(q,m,s)
    rawterms={key:sy.Poly(p,*variables,domain=sy.QQ).terms() for key,p in raw.items()}
    for key,text in record['generator_expressions'].items():
        zero(arithmetic(text,names).subs(r,value)-raw[key])
    denominator=arithmetic(record['denominator'],names).subs(r,value)
    assert denominator.is_Rational and denominator!=0
    full={};count=0;maxdegree=0
    for source,text in zip(record['sources'],record['coefficient_numerators']):
        weight=arithmetic(text,names).subs(r,value)
        assert weight.is_Rational
        if not weight:continue
        count+=1;typ=source['type'];mon=source['monomial_exponents']
        assert len(mon)==6 and all(type(e) is int and e>=0 for e in mon)
        assert sum(mon)%2==(1 if typ=='B' else 0)
        assert sum(mon)<=(5 if typ=='B' else 4)
        for perm in itertools.permutations(range(3)):
            shift=[0]*6
            for i in range(3):shift[perm[i]]=mon[i];shift[perm[i]+3]=mon[i+3]
            key=f'A{perm[0]}' if typ=='A' else f'H{perm[0]}{perm[1]}' if typ=='H' else 'B'
            for ex,co in rawterms[key]:
                combined=tuple(ex[i]+shift[i] for i in range(6))
                maxdegree=max(maxdegree,sum(combined))
                full[combined]=full.get(combined,0)+weight*co
    unit=(0,)*6
    full[unit]=full.get(unit,0)-denominator
    assert all(co==0 for co in full.values()) and maxdegree<=8
    return {'file':str(path.relative_to(BASE)),
            'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
            'parameter':str(value),'denominator':str(denominator),
            'nonzero_source_weights':count,'full_uncompressed_auxiliary_keys':len(full),
            'maximum_source_product_degree':maxdegree,
            'identity':'sum of actual raw polynomial coefficients equals nonzero denominator'}
