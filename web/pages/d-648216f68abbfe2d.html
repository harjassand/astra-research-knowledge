def need(ok, message):
    if not ok:
        raise ValueError(message)


def enc(value):
    return independent.encode(value)


def positive_sum(x, degree):
    total = term = Q(1)
    for k in range(1, degree + 1):
        term *= x / k
        total += term
    return total
