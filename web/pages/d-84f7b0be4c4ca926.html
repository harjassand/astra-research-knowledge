def main():
    started = time.perf_counter()
    root = Path(__file__).resolve().parents[3]
    source = root/"notes/charges_design9/cycle13/TRIANGLE_GRID_POLYNOMIAL_WITNESSES.md"
    checks = {}
    delta, eps = Q(1, 100), Q(1, 10000)
    checks["elimination_lower_above_half"] = Q(1, 3)*(2-6*delta)/(1+2*delta)**2 > Q(1, 2)
    checks["elimination_error_below_three"] = 2/(1-2*delta)+2*(delta+2*delta)/(1-2*delta)**2 < 3
    checks["balanced_off_diagonal_constant"] = 36/(1-2*delta) < 40
    checks["balanced_diagonal_constant"] = 288/(1-2*delta) < 400
    checks["balanced_F_Hessian"] = 80*eps+400*eps*eps < Q(1, 32)
    checks["balanced_R_Hessian_including_coordinate_term"] = Q(12, 1000)+6*delta/Q(20000) < Q(1, 32)
    checks["combined_Hessian_below_one_sixteenth"] = 80*eps+400*eps*eps+Q(12, 1000)+6*delta/Q(20000) < Q(1, 16)
    checks["unbalanced_harmonic_no_transfer_m8_budget"] = Q(4, 100)*729/Q(2**20) < Q(1, 20000)
    checks["unbalanced_elimination_upper"] = Q(50, 47)*(2+6*delta)/(1-2*delta)**2 < 3
    # Elementary e<11/4 proof: sum through5 plus the entire6-and-up tail.
    e_upper = sum((Q(1, sp.factorial(j)) for j in range(6)), Q(0))+Q(7, 4320)
    checks["elementary_e_upper"] = e_upper < Q(11, 4)
    checks["main_target_integer"] = 11**32 < 2**113
    checks["count_target_integer"] = 11**8 < 2**28
    checks["main_tail_ratio_integer"] = 2**38*168**64 <= 255**64
    checks["count_tail_ratio_integer"] = 2**44*32**32 <= 85**32
    k = 4
    checks["main_tail_induction_base"] = 7864320000*(128*k+1)*(64*k+1)**7 <= k*2**(31*k)
    checks["count_tail_induction_base"] = 11796480000*(64*k+1)*(32*k+1)**7 <= k*2**(32*k)
    checks["placement_and_padding_base"] = Q(1204, 2**26) <= Q(k, 80000)
    checks["old_Taylor_integer_base"] = 3600000 <= 2**60
    checks["new_Taylor_integer_base"] = 3600000 <= 3**31
    # Complete polynomial differentiation with independent formal factor jets.
    x = sp.symbols("x0:3")
    c = sp.symbols("c0:3")
    a, ap, app = [sp.symbols(prefix+"0:3") for prefix in ("a", "ap", "app")]
    factors = [a[i]+ap[i]*(x[i]-c[i])+app[i]*(x[i]-c[i])**2/2 for i in range(3)]
    F = sum(factors[i]*x[(i+1)%3]*x[(i+2)%3] for i in range(3))
    substitute = dict(zip(x, c))
    symbolic_checks = []
    for i in range(3):
        j, l = (i+1)%3, (i+2)%3
        expected = ap[i]*c[j]*c[l]+a[j]*c[l]+a[l]*c[j]
        symbolic_checks.append(sp.expand(sp.diff(F, x[i]).subs(substitute)-expected) == 0)
        for j in range(i, 3):
            if i == j:
                expected = app[i]*c[(i+1)%3]*c[(i+2)%3]
            else:
                l = 3-i-j
                expected = a[l]+(ap[i]+ap[j])*c[l]
            symbolic_checks.append(sp.expand(sp.diff(F, x[i], x[j]).subs(substitute)-expected) == 0)
    checks["all_symbolic_original_derivatives"] = all(symbolic_checks)
    # Three severe amplitude scales, with genuine original-gradient checks;
    # b,z are solved from the other two exact gradient equations.
    jets = []
    for n in (1, 10**10, 10**30):
        A, B, C = Q(n*n), Q(n), Q(n)
        for signs in product((-1, 0, 1), repeat=3):
            pA, pB, pC = [2*sign*v for sign, v in zip(signs, (A, B, C))]
            t, g2, g3 = eps/16, eps/4, -eps/4
            u, v = 1+pB*t/A, 1+pC*t/A
            z, b = (g2-C*t)/(A*u), (g3-B*t)/(A*v)
            gradient = [C*b+B*z+pA*b*z, C*t+(A+pB*t)*z, B*t+(A+pC*t)*b]
            assert max(map(abs, gradient)) <= eps
            assert max(map(abs, (t, b, z))) <= 10*eps
            factors_here = [A, B, C]
            point = [t, b, z]
            first = [pA, pB, pC]
            balanced = []
            for i in range(3):
                row = []
                for j in range(3):
                    if i == j:
                        l, h = (i+1)%3, (i+2)%3
                        value = 4*factors_here[i]**2*point[l]*point[h]/(factors_here[l]*factors_here[h])
                    else:
                        l = 3-i-j
                        value = 1+(first[i]+first[j])*point[l]/factors_here[l]
                    row.append(value)
                balanced.append(row)
            row_bound = max(sum(abs(balanced[i][j]-(0 if i == j else 1)) for j in range(3)) for i in range(3))
            assert row_bound <= 80*eps+400*eps*eps
            jets.append({"largest_amplitude": str(A), "ratio_largest_to_next": str(A/B), "relative_derivative_signs": list(signs),
                         "original_gradient": list(map(str, gradient)), "coordinates": list(map(str, point)),

    # balanced estimate cannot be applied to every point of a chart.

              "checks": checks, "symbolic_derivative_identities": len(symbolic_checks), "rational_original_gradient_jets": jets,
              "exact_counts": counts, "retained_boundaries": {"whole_chart_balanced_diagonal": str(off_gradient_diagonal),
                    "missing_triangle_gradient": str(no_triangle_g1), "missing_triangle_coordinate": str(t)},
              "first_counts": {"D64k_mode": 720000*257**2, "D32k_mode": 720000*129**2},
