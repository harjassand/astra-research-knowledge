def exact_parameter_output_gram(V,g,E):
    u=mv(V.inv,V.load(g));derivs=[]
    for A in V.regions:
        du=[-x for x in mv(V.inv,mv(A,u))];full=[F(0)]+du
        bd=[full[n] for n in V.nodes];mean=sum(bd)/16
        derivs.append([x-mean for x in bd])
    return [[dot(derivs[i],mv(V.M,derivs[j]))/E for j in range(2)] for i in range(2)]

def eigen_upper_product(S,W):
    A=[[sum(S[i][k]*W[k][j] for k in range(2)) for j in range(2)] for i in range(2)]
    tr=A[0][0]+A[1][1];det=A[0][0]*A[1][1]-A[0][1]*A[1][0]
    # PSD factors ensure both eigenvalues real nonnegative. Accept rational
    # upper iff it lies to the right of the larger characteristic root.
    guess=(float(tr)+math.sqrt(max(0,float(tr*tr-4*det))))/2
    upper=F(math.ceil(guess*10**12)+2,10**12)
    while upper<tr/2 or upper**2-tr*upper+det<0:upper+=F(1,10**12)
    return upper
