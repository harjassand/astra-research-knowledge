def signfix(g):
    g=np.asarray(g);g=g/np.linalg.norm(g)
    return g if g[np.argmax(abs(g))]>=0 else -g

def unique(gs):
    ans=[]
    for g in gs:
        g=signfix(g)
        if not any(np.linalg.norm(g-h)<1e-10 for h in ans):ans.append(g)
    return ans

def mass_fraction(n=4):
    N=4*n;M=[[F(0) for _ in range(N)] for _ in range(N)]
    for i in range(N):
        j=(i+1)%N
        M[i][i]+=F(2,6*n);M[j][j]+=F(2,6*n)
        M[i][j]+=F(1,6*n);M[j][i]+=F(1,6*n)
    return M

def bilinear(M,x,y):return sum(x[i]*M[i][j]*y[j] for i in range(len(x)) for j in range(len(y)))

def quantize_trace(x):
    raw=[F(int(np.rint(v*2**32)),2**32) for v in x]
    mean=sum(raw)/len(raw)
    return [v-mean for v in raw]

def fracstr(x):return str(x.numerator)+'/'+str(x.denominator)

def finalize(model,design,label):
    V=model['boundary_values'];M=mass_fraction()
    gs=[quantize_trace(V@g) for g,q in design]
    qs=[[quantize_trace(V@row) for row in q] for g,q in design]
    energy=sum(bilinear(M,g,g) for g in gs)
    norms=[[bilinear(M,q,q) for q in rows] for rows in qs]
    assert all(sum(g)==0 for g in gs)
    assert all(sum(q)==0 for rows in qs for q in rows)
    assert energy>0 and all(n>0 for rows in norms for n in rows)
    return {'label':label,'raw_currents':[[fracstr(v) for v in g] for g in gs],
            'raw_tests':[[[fracstr(v) for v in q] for q in rows] for rows in qs],
            'total_raw_energy':fracstr(energy),'raw_test_norms':[[fracstr(n) for n in row] for row in norms],
            'normalization':'g_raw/sqrt(total_raw_energy); each q_raw/sqrt(its raw_test_norm); same definition for certificate and data',
            'mode_design_diagnostic':{'currents':[g.tolist() for g,q in design],'tests':[q.tolist() for g,q in design]},
            'floating_secant_certificate_before_rounding':secant_certificate(model,design,.05)}
