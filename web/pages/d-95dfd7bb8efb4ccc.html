def report_from_record(record):
    parity=False;residue=0
    for coin,a,y in record:
        parity^=bool(y==2 and (coin^a)==1);residue=(residue+y)%3
    return bool(parity^(residue==1)^(record[0][2]==record[-1][2]))

def execute_target(observation_oracle,private_coin,n):

def propose_record(k,n,rng):
    def observation_oracle(a):
        # Internal channel-sampler integer is NEVER retained in the record.
        u=rng.randrange(10)
        for y,prob in enumerate(MENU[a][k]):
            w=int(10*prob)
            if u<w:return y
            u-=w
    return execute_target(observation_oracle,lambda:rng.randrange(2),n)

def thresholds(h,eta):

def relative_T(record,k,p):

def convert(p,observe,rng,ledger):

def enumerate_joint(n):
    out=[]
    def visit(record,h,previous):
        t=len(record)
        if t==n:out.append((tuple(record),h,report_from_record(record)));return
        for coin in [0,1]:

    visit([],[F(1)]*3,0)
    assert all(sum(h[i] for record,h,report in out)==1 for i in range(3))
    return out
