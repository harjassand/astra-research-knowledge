def partitions(n, minimum=1):
    if n == 0:
        yield ()
    for q in range(minimum, n+1):
        for tail in partitions(n-q, q):
            yield (q,)+tail

def gibbs(H,beta):
    E,U=eigh(H); w=np.exp(-beta*(E-E[0])); w/=w.sum()
    rho=(U*w)@U.conj().T
    logz=-beta*E[0]+math.log(np.exp(-beta*(E-E[0])).sum())
    return rho,logz
