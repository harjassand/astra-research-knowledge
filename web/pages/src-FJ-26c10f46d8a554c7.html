using Map=std::unordered_map<Key,std::complex<double>,Hash>;

Map multiply(const Map& a,const Map& b) {
    Map out;
    out.reserve(a.size()*b.size());
    for(const auto& [ka,va]:a) for(const auto& [kb,vb]:b) {
        Key k{};
        for(int i=0;i<D;++i) k[i]=ka[i]+kb[i];
        out[k]+=va*vb;
    }
    return out;
}
