def recurrence(sequence, prime):
    # Standard discrepancy recurrence: polynomial coefficients have ascending
    # lags until the final reversal to ascending powers of the operator.
    current=[1]; previous=[1]; length=0; last_discrepancy=1; distance=1
    for k,value in enumerate(sequence):
        discrepancy=(value+sum(current[j]*sequence[k-j] for j in range(1,length+1)))%prime
        if discrepancy==0:
            distance+=1
            continue
        old=current[:]
        multiplier=discrepancy*pow(last_discrepancy,-1,prime)%prime
        while len(current)<len(previous)+distance: current.append(0)
        for j,v in enumerate(previous): current[j+distance]=(current[j+distance]-multiplier*v)%prime
        if 2*length<=k:
            length=k+1-length; previous=old; last_discrepancy=discrepancy; distance=1
        else: distance+=1
    return list(reversed(current[:length+1]))

def normalize(v,den):
    common=abs(den)
    for x in v: common=gcd(common,abs(x))
    if den<0: common=-common
    return [x//common for x in v],den//common
