def choose(weights,rng):
    assert all(w>=0 for w in weights) and sum(weights)>0
    den=lcm(*(x.denominator for x in weights))
    ns=[x.numerator*(den//x.denominator) for x in weights];total=sum(ns)
    while True:
        r=rng.getrandbits(total.bit_length())
        if r<total:break
    for j,x in enumerate(ns):
        if r<x:return j
        r-=x
    raise AssertionError('integer draw')

def tables(t,w):
    m=2*w;d=2**m;gmin=((1-t)/2)**m;a=d*gmin
    q=((1+t)/(1-t))**m;s=1-1/q
    ge=[((1+t)/2)**(m-k.bit_count())*((1-t)/2)**k.bit_count() for k in range(d)]
    lam=[1-gmin/g for g in ge];delta=[g-gmin for g in ge]
    c0=1/a-1;coeff=[t**k.bit_count()/a for k in range(1,d)]
    S=sum(coeff);z=S-c0
    assert sum(ge)==1 and sum(delta)==1-a and max(lam)==s and min(lam)==0
    B=max(0,q.numerator.bit_length()-q.denominator.bit_length()+1)
    h=q*(1+B);closed=1+(h.numerator+h.denominator-1)//h.denominator
    if q>1000:L=closed
    elif z<=1:L=1
    else:
        lo,hi=0,closed-1
        while lo<hi:
            mid=(lo+hi)//2
            if z*s**mid<=1:hi=mid
            else:lo=mid+1
        L=lo+1
    if q<=1000:assert z*s**(L-1)<=1
    return dict(t=t,w=w,m=m,d=d,gmin=gmin,a=a,q=q,s=s,ge=ge,lam=lam,
                delta=delta,c0=c0,coeff=coeff,S=S,z=z,L=L,closed=closed)

def axes_for_mask(k,w):
    # Sites ordered left0,right0,left1,right1,... . E=product CZ(left_i,right_i).
    # E X_left E=XZ, E X_right E=ZX, E X_left X_right E=YY, all signs+.
    out=[]
    for j in range(w):
        pair=(k>>(2*j))&3
        out.extend([['I','I'],['X','Z'],['Z','X'],['Y','Y']][pair])
    return out

def pauli_basis_draw(k,w,rng):
    axes=axes_for_mask(k,w);signs=[1 if rng.getrandbits(1) else -1 for _ in axes]
    active=[j for j,x in enumerate(axes) if x!='I']
    if active:signs[active[-1]]=prod(signs[j] for j in active[:-1])
    return [('Z' if x=='I' else x,s) for x,s in zip(axes,signs)]

def central_weights(T,r):
    a=T['a'];u=1-r*T['z']
    weights=[a*u]+[a*r*c for c in T['coeff']]
    assert min(weights)>=0 and sum(weights)==a+r*(1-a)
    return weights

def sample(T,ell,rng):
    assert ell>=T['L'];a=T['a'];alpha=1-a
    ws=[a+alpha**ell]+[a*alpha**b for b in range(1,ell)]
    assert sum(ws)==1
    b=choose(ws,rng);res=False;labels=[]
    if b:
        for j in range(ell):
            labels.append(None if j==b else choose(T['delta'] if j<b else T['ge'],rng))
        central=pauli_basis_draw(0,T['w'],rng);r=None;k=0
    else:
        res=bool(choose([a,alpha**ell],rng))
        labels=[None]+[choose(T['delta'] if res else T['ge'],rng) for _ in range(ell-1)]
        r=prod((T['lam'][j] for j in labels[1:]),start=F(1))
        k=choose(central_weights(T,r),rng);central=pauli_basis_draw(k,T['w'],rng)
    # All other sites are X eigenstates. At central b use product Pauli states.
    # Apply the supplied CZ circuit with the w gates at this midpoint deleted.
    return dict(cut=b,residual_branch=res,labels=labels,central=central,pauli_mask=k,r=r)

def sample_strip(T,n,rng):
    """Full supplied-graph sampler, n even columns, output product labels + deleted cuts.
    The retained CZ graph is implicit: rectangular nearest-neighbour edges except
    all horizontal row edges at the returned column boundaries.
    """
    assert n%2==0 and n>=2
    w=T['w'];M=n//2;L=T['L'];inputs=[[None]*w for _ in range(n)];cuts=[]
    if M<L:
        for col in range(n):
            for row in range(w):
                bit=choose([1+T['t'],1-T['t']],rng)
                inputs[col][row]=('X',1 if bit==0 else -1)
    else:
        groups=M//L;remainder=M%L;lengths=[L+remainder]+[L]*(groups-1);start=0
        for ell in lengths:
            x=sample(T,ell,rng);b=x['cut'];cuts.append(2*(start+b)+1)
            for j,label in enumerate(x['labels']):
                axes=x['central'] if j==b else [('X',-1 if label&(1<<k) else 1) for k in range(2*w)]
                for row in range(w):
                    inputs[2*(start+j)][row]=axes[2*row]
                    inputs[2*(start+j)+1][row]=axes[2*row+1]
            start+=ell
        assert start==M
    ends=[0]+cuts+[n];max_sites=w*max(y-x for x,y in zip(ends,ends[1:]))
    assert max_sites<=8*w*L
    return dict(inputs=inputs,deleted_column_boundaries=cuts,max_block_sites=max_sites)

def exact_tuple_gate(T,ell):
    a=T['a'];alpha=1-a;total=F(0);minimum=F(1);maxbits=0
    for ts in product(range(T['d']),repeat=ell-1):
        r=prod((T['lam'][j] for j in ts),start=F(1))
        e=prod((T['ge'][j] for j in ts),start=F(1))
        lhs=(a+r*alpha)*e
        rhs=a*e+alpha*prod((T['delta'][j] for j in ts),start=F(1))
        assert lhs==rhs
        cw=central_weights(T,r);total+=lhs;minimum=min(minimum,1-r*T['z'])
        maxbits=max(maxbits,r.denominator.bit_length())
    assert total==a+alpha**ell
    return dict(tuple_count=T['d']**(ell-1),exact_weight_identity=True,
                exact_total_mass=str(total),minimum_identity_coefficient=str(minimum),
                largest_r_denominator_bits=maxbits)

def float_matrix_gate(T,ell):
    I=np.eye(2,dtype=complex);X=np.array([[0,1],[1,0]],complex)
    Y=np.array([[0,-1j],[1j,0]],complex);Z=np.diag([1,-1]);paulis=dict(I=I,X=X,Y=Y,Z=Z)
    def tensor(xs):
        v=np.ones((1,1),complex)
        for x in xs:v=np.kron(v,x)
        return v
    G=tensor([(I+float(T['t'])*X)/2]*T['m']);Gp=float(T['gmin'])*np.eye(T['d']);D=G-Gp
    if T['w']!=1:raise ValueError('dense gate currently w1')
    E=np.diag([1,1,1,-1]);r=float(T['s']**(ell-1))
    cs=central_weights(T,T['s']**(ell-1));central=float(cs[0])/T['d']*np.eye(T['d'],dtype=complex)
    for k,c in enumerate(cs[1:],start=1):
        P=tensor([paulis[x] for x in axes_for_mask(k,1)])
        central+=float(c)/T['d']*(np.eye(T['d'])+P)
    err=float(np.linalg.norm(central-(Gp+r*E@D@E)))
    if ell<=4:
        Q0=tensor([Gp]+[G]*(ell-1))+tensor([D]*ell)
        Qs=[Q0]+[tensor([D]*b+[Gp]+[G]*(ell-b-1)) for b in range(1,ell)]
        telescoping=float(np.linalg.norm(sum(Qs)-tensor([G]*ell)))
    else:telescoping=None
    assert err<1e-12
    return dict(central_pauli_matrix_error=err,dense_telescoping_error=telescoping)
