def binary_rank(columns):
    piv={}
    for x in columns:
        while x:
            k=x.bit_length()-1
            if k in piv: x ^= piv[k]
            else:
                piv[k]=x
                break
    return len(piv)

def bases_fano():

def distance(A,B):
    return min(sum(len(set(x)-set(y)) for x,y in zip(A,P))
               for P in permutations(B))

def signature(A,n):
    return tuple(sum(e in b for b in A) for e in range(n))

def adjacent(A,bases_set):

    return out
