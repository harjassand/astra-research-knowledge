#!/usr/bin/env python3
"""Producer replay of a proved-corollary candidate, on frozen inputs only.

Imports producer arithmetic; consequently this is not an independent gate.
No generator/learner rerun, data draw, count change or new proposal occurs.
"""
import importlib.util
import json
from fractions import Fraction as F
from pathlib import Path
import time

BASE=Path(__file__).resolve().parents[3]
SRC=BASE/'work/pde_gate9/cycle15/actual_row_operator.py'
spec=importlib.util.spec_from_file_location('producer',SRC)
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
root=BASE/'results/pde_gate9/cycle15'
orig=json.loads((root/'FROZEN_LEARNER_OUTPUT.json').read_text())
c=orig['common_confidence']
start=time.monotonic()
z=F(c['zeta']);ep=F(c['epsilon4']);mh=F(c['observed_operator_upper'])
k0=F(3,2)*(2*z+z*z);sub=2*z*(1-z)
assert z<1 and k0<1 and sub>=0
e=(ep+k0*mh-sub)/(1-k0)
assert 0<=e<F(c['total_operator_error'])
common=dict(total_operator_error=e,zeta=z,
            condition_squared_upper=F(c['condition_squared_upper']))
W=[[F(x) for x in row] for row in c['whitener']['W']]
Wi=[[F(x) for x in row] for row in c['whitener']['W_inverse']]
S=json.loads((root/'confirmation_EXACT_STATISTICS.json').read_text())['rankone_mean_bilinear']
S=[[F(x) for x in row] for row in S]
posts={}
for name,p in orig['proposals'].items():
    R=m.quaternion_rotation(p['quaternion_integer'])
    assert R==[[F(x) for x in row] for row in p['R']]
    posts[name]=m.posterior(S,common,dict(R=R),W,Wi)
out=dict(status='PRODUCER_CANDIDATE_PENDING_INDEPENDENT_GATE',
         source_learner_freeze_sha256=m.filehash(root/'FROZEN_LEARNER_OUTPUT.json'),
         original_error=F(c['total_operator_error']),refined_error=e,k0=k0,
         lower_floor_subtraction=sub,posteriors=posts,
         no_new_samples=True,no_new_proposals=True,
         shared_confidence_event=True,seconds=time.monotonic()-start,
         scope='same-input analytic bound refinement; no comparative performance claim')
m.save('SHARPER_SAME_INPUT_POSTERIOR.json',out)
print(json.dumps(dict(error=float(e),seconds=out['seconds'],
       caps={k:[float(v['physical_column_sine_upper']),float(v['source_row_sine_upper'])]
             for k,v in posts.items()}),indent=2))
