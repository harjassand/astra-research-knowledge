"""Exact small-instance check of the XXZ-to-hard-core-boson identity.

This is a finite algebra check, not evidence for the imported FPRAS or a
canonical-sector theorem. All selected coefficients are integers and the
matrix identities are exact in NumPy's integer-valued complex arithmetic.
"""
import numpy as np

I = np.eye(2, dtype=complex)
X = np.array([[0, 1], [1, 0]], dtype=complex)
Y = np.array([[0, -1j], [1j, 0]], dtype=complex)
Z = np.diag([1, -1]).astype(complex)
n0 = np.diag([0, 1]).astype(complex)
b = np.array([[0, 1], [0, 0]], dtype=complex)
bd = b.conj().T

nsite = 3
edges = [(0, 1), (1, 2)]
alpha = [2, 1]
gamma = [1, -1]
source_b = [0, 2, 1]
field_c = [1, -2, 3]


def lift(op, i):
    out = np.array([[1]], dtype=complex)
    for j in range(nsite):
        out = np.kron(out, op if i == j else I)
    return out

Xv = [lift(X, i) for i in range(nsite)]
Yv = [lift(Y, i) for i in range(nsite)]
Zv = [lift(Z, i) for i in range(nsite)]
nv = [lift(n0, i) for i in range(nsite)]
bv = [lift(b, i) for i in range(nsite)]
bdv = [lift(bd, i) for i in range(nsite)]

K_xxz = np.zeros((2**nsite, 2**nsite), dtype=complex)
for (i, j), a, g in zip(edges, alpha, gamma):
    K_xxz -= a * (Xv[i] @ Xv[j] + Yv[i] @ Yv[j])
    K_xxz -= g * (Zv[i] @ Zv[j])
for i in range(nsite):
    K_xxz -= source_b[i] * Xv[i] + field_c[i] * Zv[i]

# K_boson = const - sum t (b_i^* b_j + b_j^* b_i)
#            + sum V n_i n_j + sum nu_i n_i + sum Omega_i (b_i+b_i^*)
# with t=2 alpha, V=-4 gamma, nu_i=2 c_i+2 sum_{e incident} gamma_e,
# Omega_i=-b_i, and const=-sum_e gamma_e-sum_i c_i.
nu = [2 * field_c[i] + 2 * sum(g for (u, v), g in zip(edges, gamma) if i in (u, v))
      for i in range(nsite)]
constant = -sum(gamma) - sum(field_c)
K_boson = constant * np.eye(2**nsite, dtype=complex)
for (i, j), a, g in zip(edges, alpha, gamma):
    K_boson -= 2 * a * (bdv[i] @ bv[j] + bdv[j] @ bv[i])
    K_boson += (-4 * g) * (nv[i] @ nv[j])
for i in range(nsite):
    K_boson += nu[i] * nv[i] - source_b[i] * (bv[i] + bdv[i])

assert np.array_equal(K_xxz, K_boson)
assert all(abs(-4 * g) <= 2 * (2 * a) for a, g in zip(alpha, gamma))
assert all(-s <= 0 for s in source_b)
print("EXACT_MATRIX_IDENTITY: PASS")
print("n=3 path, alpha=[2,1], gamma=[1,-1], fields included")
print("hard-core parameters: t=[4,2], V=[-4,4], nu=", nu,
      "Omega=", [-x for x in source_b], "constant=", constant)
