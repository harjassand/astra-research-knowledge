def rank(rows,p=2,ncols=None):
    if not rows:return 0
    a=[list(r) for r in rows]; nr=len(a); nc=len(a[0]) if ncols is None else ncols; r=0
    for c in range(nc):
        k=next((k for k in range(r,nr) if a[k][c]%p),None)
        if k is None:continue
        a[r],a[k]=a[k],a[r]; inv=pow(a[r][c]%p,-1,p); a[r]=[(z*inv)%p for z in a[r]]
        for k in range(nr):
            if k!=r and a[k][c]%p:
                fac=a[k][c]; a[k]=[(u-fac*v)%p for u,v in zip(a[k],a[r])]
        r+=1
        if r==nr:break
    return r

def transpose(A):return list(map(list,zip(*A)))
def lincomb(A,l,p=2):return [[sum(l[k]*A[k][i][j] for k in range(len(A)))%p for j in range(len(A[0][0]))] for i in range(len(A[0]))]
def mv(A,v,p=2):return [sum(a*b for a,b in zip(row,v))%p for row in A]
def F(A,x,p=2):return [mv(transpose(a),x,p) for a in A]
def vectors(n,p=2):return list(product(range(p),repeat=n))
def zeros(n):return [0]*n

def gf_mul(a,b,poly,n):
    out=0
    while b:
        if b&1:out^=a
        b>>=1;a<<=1
        if a>>n:a^=poly
    return out

def gf_pow(a,k,poly,n):
    out=1
    while k:
        if k&1:out=gf_mul(out,a,poly,n)
        a=gf_mul(a,a,poly,n);k>>=1
    return out

def gf_solve(M,b,poly,n):
    A=[list(row)+[val] for row,val in zip(M,b)]; N=len(M)
    for c in range(N):
        k=next(k for k in range(c,N) if A[k][c]); A[c],A[k]=A[k],A[c]
        inv=gf_pow(A[c][c],(1<<n)-2,poly,n); A[c]=[gf_mul(z,inv,poly,n) for z in A[c]]
        for k in range(N):
            if k!=c and A[k][c]:
                f=A[k][c]; A[k]=[u^gf_mul(f,v,poly,n) for u,v in zip(A[k],A[c])]
    return [row[-1] for row in A]

def polynomial_remainder(a,b):
    while a and a.bit_length()>=b.bit_length():a^=b<<(a.bit_length()-b.bit_length())
    return a

def pgcd(a,b):
    while b:a,b=b,polynomial_remainder(a,b)
    return a

def irreducible(poly,n):
    v=2
    for i in range(1,n+1):
        v=gf_mul(v,v,poly,n)
        if i<=n//2 and pgcd(v^2,poly)!=1:return False
    return v==2

def qpoly_matrix(coeffs,poly,n):
    cols=[]
    for k in range(n):
        z=0
        for j,c in enumerate(coeffs):z^=gf_mul(c,gf_pow(1<<k,1<<j,poly,n),poly,n)
        cols.append([(z>>i)&1 for i in range(n)])
    return transpose(cols)

def acquire_qpoly(A,poly,n):
    assert irreducible(poly,n)
    M=[[gf_pow(1<<k,1<<j,poly,n) for j in range(n)] for k in range(n)]
    coeff=[]
    for mat in A:
        values=[sum(mat[i][k]<<i for i in range(n)) for k in range(n)]
        c=gf_solve(M,values,poly,n); assert qpoly_matrix(c,poly,n)==mat;coeff.append(c)
    independent=rank([sum(a,[]) for a in A])==len(A)
    s=max(j for c in coeff for j,v in enumerate(c) if v)
    return {'independent':independent,'q_degree':s,'certified_min_rank':n-s,'coefficients':coeff,'irreducible':True}

def exact_law(A,R,p=2,a=None,c=None,d=None,extra_h=None):
    m=len(A); nx=len(A[0]); ny=len(A[0][0]); ls=vectors(m,p); xs=vectors(nx,p); ys=vectors(ny,p)
    if a is None:a=[zeros(nx) for _ in range(m)]
    if c is None:c=[zeros(ny) for _ in range(m)]
    if d is None:d=zeros(m)
    C=Fraction(1)+Fraction(p**m-1,p**R)
    xmass={x:Fraction() for x in xs}; ranks=[]; consistency_reject=0
    for l in ls:
        Al=lincomb(A,l,p); r=rank(Al,p);ranks.append(r)
        assert not any(l) or r>=R
        cl=mv(transpose(c),l,p)
        # Uniform solutions A_lambda^T x=-c_lambda; their size, if nonempty, p^(nx-r).
        ker=[x for x in xs if mv(transpose(Al),x,p)==[(-z)%p for z in cl]]
        if not ker:
            consistency_reject+=1;continue
        assert len(ker)==p**(nx-r)
        branch=Fraction(1,C) if not any(l) else Fraction(1,p**r*C)
        for x in ker:xmass[x]+=branch/len(ker)
    mass_per_solution=Fraction(p**m,p**(nx+ny))/C
    counts={b:0 for b in ls}; accepted={b:Fraction() for b in ls}; rankhist={}; distinct_point_masses=set()
    for x in xs:
        Fx=[[(z+w)%p for z,w in zip(row,ci)] for row,ci in zip(F(A,x,p),c)]
        rx=rank(Fx,p);rankhist[rx]=rankhist.get(rx,0)+1
        assert xmass[x]==Fraction(p**(m-rx),p**nx)/C
        shift=[(sum(ai[j]*x[j] for j in range(nx))+di)%p for ai,di in zip(a,d)]
        if extra_h is not None:shift=[(u+v)%p for u,v in zip(shift,extra_h(x))]
        for y in ys:
            b=tuple((z+w)%p for z,w in zip(mv(Fx,y,p),shift))
            pointmass=xmass[x]/p**(ny-rx)
            assert pointmass==mass_per_solution
            distinct_point_masses.add(pointmass)
            counts[b]+=1;accepted[b]+=pointmass
    assert len(distinct_point_masses)==1
    return {'q':p,'m':m,'nx':nx,'ny':ny,'R':R,'envelope_C':str(C),
        'combination_rank_histogram':{str(r):ranks.count(r) for r in sorted(set(ranks))},
        'fiber_rank_histogram':rankhist,'counts_by_target':{' '.join(map(str,k)):v for k,v in counts.items()},
        'acceptance_by_target':{' '.join(map(str,k)):str(v) for k,v in accepted.items()},
        'max_expected_trials':str(max(Fraction(1,v) for v in accepted.values() if v)),
        'exact_uniformity':True,'one_trial_probability_per_solution':str(mass_per_solution),
        'inconsistent_dual_branches':consistency_reject}

def binary_trap(d):
    # F(e0)=E11; F(ej)=E11+e2 e_j^T; F(e'_j)=E11+(e1+e2)e_j^T.
    A=[]; E=[[int(i==0 and j==0) for j in range(d)] for i in range(2)]; gens=[E]
    for kind in range(2):
        for j in range(d):
            B=[row[:] for row in E]
            B[1][j]^=1
            if kind:B[0][j]^=1
            gens.append(B)
    mats=[[[B[i][j] for j in range(d)] for B in gens] for i in range(2)]
    x0=(1,)+(0,)*(2*d); base=F(mats,x0);assert rank(base)==1
    neigh=[]
    for j in range(len(x0)):
        x=list(x0);x[j]^=1;neigh.append(rank(F(mats,x)))
    ranks=[rank(lincomb(mats,l)) for l in vectors(2) if any(l)]
    assert ranks==[d,d,d];assert max(neigh)==1
    witness=[0]*len(x0);witness[1]=1;witness[2]=1
    # two first-kind directions give first row 0, second e1+e2: rank one. Use base + two directions instead.
    witness[0]=1
    assert rank(F(mats,witness))==2
    return {'d':d,'min_nonzero_combination_rank':min(ranks),'initial_x':x0,'initial_rank':1,
            'single_flip_ranks':neigh,'escaping_x':witness,'escaping_rank':2,'matrices':mats}
