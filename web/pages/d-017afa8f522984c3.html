#!/usr/bin/env python3
"""Read-only exact controls for the authorized exposed reconstruction.

SymPy only. Controls an overcomplete singular embedded frame, scalar
geometry, noncommuting reference blocks, and the separate block-column
lemma. This is diagnostic evidence, not formal/general proof verification.
"""
import sympy as s

count = 0


def check(x):
    global count
    assert x
    count += 1


def simp(x):
    return s.simplify(x)


def eq(a, b):
    check(a.shape == b.shape)
    for x in a-b:
        check(simp(x) == 0)


def norm2(a):
    return simp(s.trace(a.conjugate().T*a))


def nonnegative(x):
    check(simp(x).is_nonnegative is True)


h = s.Matrix([[2,1],[1,1]])
h2, h4 = h*h, h**4
u = s.Matrix([[1/s.sqrt(2), s.Rational(1,2), s.Rational(1,2)],
              [0,1/s.sqrt(2),-1/s.sqrt(2)]])
eq(u*u.T, s.eye(2))
v = u.T
proj = v*u
eq(proj*proj, proj)
check(proj.rank() == 2 and proj.rows == 3)
b0 = simp(v*h2*u)
h0 = simp(v*h*u)
di = [simp((v*h4*u)[i,i]) for i in range(3)]
for x in di:
    nonnegative(x)
    check(x != 0)
eq(h0*h0, b0)
eq(b0*b0, v*h4*u)
check(b0.det() == 0)
weights = [x/47 for x in di]
check(sum(weights) == 1)
gram = s.Matrix([[simp(b0[i,j]*b0[j,i]/s.sqrt(di[i]*di[j]))
                  for j in range(3)] for i in range(3)])
for i in range(3):
    check(simp(sum(b0[i,j]*b0[j,i]/di[i] for j in range(3))) == 1)


def scalar_check():
    a0 = s.diag(s.Rational(101,100),s.Rational(99,100))
    zz = norm2(a0)
    a = v*a0*u
    f0 = s.diag(*[simp((h0*a*h0)[i,i]/di[i]) for i in range(3)])
    ap = simp(h0*f0*h0)
    score = simp(sum(di[i]*f0[i,i]**2 for i in range(3))/zz)
    energy = 1-score
    dd = simp(score-norm2(ap)/zz)
    residual = simp(norm2(a-b0*f0)/zz)
    nonnegative(energy)
    nonnegative(dd)
    nonnegative(energy-dd)
    check(simp(norm2(b0*f0-f0*b0)/(2*zz)-dd) == 0)
    check(simp(norm2(a-ap)/zz-energy+dd) == 0)
    rr = ap-(b0*f0+f0*b0)/2
    nonnegative(dd/2-norm2(rr)/zz)
    nonnegative((1+1/s.sqrt(2))*energy-residual)
    # Reconstruct the physical canonical map from Mi=ui ui* and its
    # pure posterior pi=(H²ui)(H²ui)*/di. No inverse of singular B is used.
    rho = a0*a0/zz
    out = s.zeros(2)
    g = 0
    for i in range(3):
        ui = u[:,i]
        post = h2*ui*ui.T*h2/di[i]
        eq(post*post, post)
        check(simp(s.trace(post)) == 1)
        qi = simp((ui.T*rho*ui)[0])
        out += qi*post
        g += simp(abs((ui.T*h2*a0*ui)[0])**2/(di[i]*zz))
    diff = simp(rho-out)
    check(s.trace(diff) == 0)
    err2 = simp(-diff.det())
    nonnegative(1-g-err2)
    nonnegative(residual-(1-g))
    nonnegative((1+1/s.sqrt(2))*energy-err2)


def block_check():
    x = s.Matrix([[0,1],[1,0]])
    z = s.diag(1,-1)
    eps = s.Rational(1,100)
    # Order is B_system then reference; swapping tensor factors is unitary.
    a_phys = s.kronecker_product(h2,s.eye(2)) + eps*(s.kronecker_product(z,x)+s.kronecker_product(x,z))
    check(a_phys.is_positive_definite)
    zz = norm2(a_phys)
    w = s.kronecker_product(v,s.eye(2))
    a = simp(w*a_phys*w.T)
    bb = s.kronecker_product(b0,s.eye(2))
    hh = s.kronecker_product(h0,s.eye(2))
    fblocks = [simp((hh*a*hh)[2*i:2*i+2,2*i:2*i+2]/di[i]) for i in range(3)]
    f0 = s.diag(*fblocks)
    ap = simp(hh*f0*hh)
    score = simp(sum(di[i]*norm2(fblocks[i]) for i in range(3))/zz)
    energy = 1-score
    dd = simp(score-norm2(ap)/zz)
    check(fblocks[0]*fblocks[1] != fblocks[1]*fblocks[0])
    nonnegative(energy)
    nonnegative(dd)
    nonnegative(energy-dd)
    check(simp(norm2(bb*f0-f0*bb)/(2*zz)-dd) == 0)
    check(simp(norm2(a-ap)/zz-energy+dd) == 0)
    rr = ap-(bb*f0+f0*bb)/2
    nonnegative(dd/2-norm2(rr)/zz)
    residual = simp(norm2(a-bb*f0)/zz)
    nonnegative((1+1/s.sqrt(2))*energy-residual)
    out = s.zeros(6)
    for i in range(3):
        ai = a[:,2*i:2*i+2]
        bi = bb[:,2*i:2*i+2]
        vi = bi/s.sqrt(di[i])
        eq(vi.T*vi,s.eye(2))
        qi = ai.T*ai
        zi = ai*vi.T
        fp = vi*(s.sqrt(di[i])*fblocks[i])*vi.T
        eq(zi*zi.T,ai*ai.T)
        eq(zi.T*zi,vi*qi*vi.T)
        check(simp(norm2(zi-fp)-norm2(ai-bi*fblocks[i])) == 0)
        out += vi*qi*vi.T
    physical_out = simp(w.T*out*w/zz)
    physical_rho = a_phys*a_phys/zz
    # In dimension 4, e² <= ||rho-out||_2² by trace/HS Cauchy. This exact
    # sufficient certificate checks actual error without numerical roots.
    trace_upper = norm2(physical_rho-physical_out)
    nonnegative((2+s.sqrt(2))*energy-trace_upper)


def column_lemma_checks():
    a,b,k=s.symbols('a b k',real=True)
    eqn=(a*a+b*b+k*k)*(b*b+k*k)-(a+b)**2*k*k
    check(s.expand(eqn-((a*b-k*k)**2+b*b*k*k+b**4)) == 0)
    for n in range(1,9):
        eps=s.Rational(1,2**n)
        zz=s.Matrix([[1,eps],[-eps,0]])
        ff=s.diag(1,0)
        comm=zz*zz.T-zz.T*zz
        # Its eigenvalues are +/-2 eps, so (half trace norm)^2=4 eps².
        check(set(comm.eigenvals()) == {-2*eps,2*eps})
        q=norm2(zz)
        dist=norm2(zz-ff)
        nonnegative(2*q*dist-4*eps**2)
        check(simp(4*eps**2/(q*dist)) == 2/(1+2*eps**2))


scalar_check()
block_check()
column_lemma_checks()
