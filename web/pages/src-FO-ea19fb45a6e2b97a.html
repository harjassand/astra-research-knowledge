#!/usr/bin/env python3
"""Independent arithmetic for the FO symmetry-control note (stdlib only)."""
import json
import math

# GaAs solid heat content Shomate fit from Yamaguchi et al. (1989), Table 1.
# H(T)-H(298.15 K) = a*T + b*T^2 + c/T + d, in J/mol.
a, b, c, d = 47.3, 3.99e-3, 2.40e5, -1.53e4
Tm = 1514.0
Tref = 298.15
L = 106.72e3  # J/mol, measured fusion enthalpy from the same source
C = 0.02       # measured modulation amplitude in Fraser et al. (1999)
Hs = a * Tm + b * Tm**2 + c / Tm + d
melt_threshold = L / (2 * Hs + L)
max_melt_fraction = 2 * C * Hs / ((1 - C) * L)

# Universal interference bound from |sum A* B| <= ||A|| ||B||.
# For balanced one- and two-photon rates, the mathematical ceiling is 1.
max_interference_fraction = 2 * math.sqrt(1.0 * 1.0) / (1.0 + 1.0)

# Source-selected (111) geometry, in the cubic [100],[010],[001] basis.
ew = (0.0, 1 / math.sqrt(2), -1 / math.sqrt(2))
e2 = (2 / math.sqrt(6), -1 / math.sqrt(6), -1 / math.sqrt(6))
# Td scalar tensor: T_ijk=1 for all distinct indices, 0 otherwise.
contraction = sum(e2[i] * ew[j] * ew[k]
                  for i in range(3) for j in range(3) for k in range(3)
                  if len({i, j, k}) == 3)

out = {
    "heat_fit_source": "Yamaguchi et al. 1989, Table 1, GaAs(s), valid 800-1514 K",
    "Tref_K": Tref,
    "Tm_K": Tm,
    "solid_sensible_heat_J_per_mol": Hs,
    "fusion_enthalpy_J_per_mol": L,
    "full_melt_one_side_while_other_below_solid_threshold_C_strictly_above": melt_threshold,
    "measured_phase_odd_population_modulation_C": C,
    "max_hot_side_melt_fraction_if_other_side_stays_solid": max_melt_fraction,
    "universal_Cauchy_ceiling_balanced_pathways": max_interference_fraction,
    "omega_polarization_cubic_basis": ew,
    "two_omega_polarization_cubic_basis": e2,
    "Td_scalar_tensor_contraction": contraction,
    "expected_contraction_exact": "-sqrt(2/3)",
}
print(json.dumps(out, indent=2, sort_keys=True))
