"""Exact logarithmic amplification witness; finite values are supporting only."""
from pathlib import Path
import json
import sympy as s
ROOT=Path(__file__).resolve().parents[2]
b,theta=s.symbols('b theta',positive=True)
def ep(rate):
    q=s.Matrix([[-rate-b,rate,b],[1,-2,1],[1,1,-2]])
    tau=[(-q).minor_submatrix(i,i).det() for i in range(3)]
    pi=[z/sum(tau) for z in tau]
    assert all(s.cancel(z)==0 for z in s.Matrix(1,3,pi)*q)
    # Only ordinary pair0<->1 and rare pair0<->2 contribute.
    return (pi[0]*rate-pi[1])*s.log(rate)+(pi[2]-b*pi[0])*s.log(1/b),pi
base,_=ep(s.Integer(1));changed,pi=ep(1+theta)
coefficient=s.simplify(pi[2].subs(b,0)-s.Rational(1,6))
assert s.simplify(coefficient-theta/(6*(2+theta)))==0
# The ordinary log term is bounded; stationary differences converge at b=0.
# Thus the exact coefficient above equals lim (changed-base)/log(1/b).
records=[]
for bt in [s.Rational(1,10**k) for k in [3,6,12,24,48]]:
    difference=(changed-base).subs({b:bt,theta:s.Rational(1,100)})
    ratio=(difference/s.log(1/bt)).evalf(50)
    records.append({'b':str(bt),'EP_difference_decimal':str(difference.evalf(50)),
                    'difference_over_log_decimal':str(ratio)})
result={'scope':'Exact stationary/log coefficient witness. Finite EP ratios illustrate convergence, not a universal numerical proof.',
        'baseline_row_rates':'q01=1, q10=q12=q20=q21=1, q02=b',
        'perturbation':'q01=1+theta, same support and other rates',
        'exact_lim_EP_difference_over_log':str(coefficient),
        'theta_0.01_exact_limit':str(coefficient.subs(theta,s.Rational(1,100))),
        'records':records}
(ROOT/'results/r17_nonequilibrium/ep_nuisance.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
