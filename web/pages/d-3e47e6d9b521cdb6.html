"""Exact finite checks for the c03_l04 Gaussian-filter boundary gadgets."""
import json
from fractions import Fraction as Q


def mm(A, B):
    return [[sum((A[i][k] * B[k][j] for k in range(len(B))), Q(0))
             for j in range(len(B[0]))] for i in range(len(A))]


def tr(A):
    return [list(row) for row in zip(*A)]


def diag(v):
    return [[v[i] if i == j else Q(0) for j in range(len(v))]
            for i in range(len(v))]

# Two noncommuting fermionic occupation filters. Basis is |00>,|10>,|01>,|11>.
c, s = Q(3, 5), Q(4, 5)
U = [
    [Q(1), Q(0), Q(0), Q(0)],
    [Q(0), c, -s, Q(0)],
    [Q(0), s, c, Q(0)],
    [Q(0), Q(0), Q(0), Q(1)],
]
P = diag([Q(1), Q(0), Q(1), Q(0)])  # first mode empty
Qrot = mm(mm(U, P), tr(U))          # rotated-mode-empty projector
K = mm(Qrot, P)                      # sequential accepted Kraus operator
assert mm(P, P) == P
assert mm(Qrot, Qrot) == Qrot
KdagK = mm(tr(K), K)
assert KdagK == diag([Q(1), Q(0), c*c, Q(0)])
assert c*c == Q(9, 25)
assert mm(KdagK, KdagK) != KdagK
# On the maximally mixed two-mode Gaussian input, accepted output eigenweights
# are proportional to 1 and 9/25, unlike any rank-two hard projection.
trace_weight = Q(1) + c*c
normalized_eigenweights = (Q(1) / trace_weight, c*c / trace_weight)
assert normalized_eigenweights == (Q(25, 34), Q(9, 34))

# A minimal adaptive accept/reject instrument: first measure P.  If P occurs,
# continue with Q and accept only Q; if I-P occurs, accept immediately.  Its
# two accepted Kraus operators are QP and I-P, and its acceptance effect is
# their squared sum.  This is neither a (scaled) projector nor a Gaussian
# positive operator: after normalization, Wick factorization fails.
I = diag([Q(1)] * 4)
K_accept_0 = mm(Qrot, P)
K_accept_1 = [[I[i][j] - P[i][j] for j in range(4)] for i in range(4)]
accept_effect = [
    [sum((K_accept_0[k][i] * K_accept_0[k][j]
          + K_accept_1[k][i] * K_accept_1[k][j]
          for k in range(4)), Q(0))
     for j in range(4)] for i in range(4)
]
assert accept_effect == diag([Q(1), Q(1), c*c, Q(1)])
assert mm(accept_effect, accept_effect) != accept_effect
effect_trace = sum((accept_effect[i][i] for i in range(4)), Q(0))
adaptive_effect_state = [accept_effect[i][i] / effect_trace for i in range(4)]
adaptive_effect_n1 = adaptive_effect_state[1] + adaptive_effect_state[3]
adaptive_effect_n2 = adaptive_effect_state[2] + adaptive_effect_state[3]
adaptive_effect_n1n2 = adaptive_effect_state[3]
adaptive_effect_wick_gap = adaptive_effect_n1n2 - adaptive_effect_n1 * adaptive_effect_n2
assert (adaptive_effect_n1, adaptive_effect_n2, adaptive_effect_n1n2) == (
    Q(25, 42), Q(17, 42), Q(25, 84)
)
assert adaptive_effect_wick_gap == Q(25, 441)
# All off-diagonal normal/anomalous contractions vanish for this diagonal state.
# Hence a Gaussian state would have n1n2=n1*n2, which is false.

# Sequential adaptive occupation filtering on two entangled pair states.
# For the accepted data patterns {00,10,11}, compute the exact diagonal law.
patterns = [(0, 0), (1, 0), (1, 1)]
prob = Q(1, len(patterns))
n1 = sum((prob * x for x, y in patterns), Q(0))
n2 = sum((prob * y for x, y in patterns), Q(0))
n1n2 = sum((prob * x * y for x, y in patterns), Q(0))
assert (n1, n2, n1n2) == (Q(2, 3), Q(1, 3), Q(1, 3))
assert n1n2 != n1 * n2
number_probabilities = tuple(
    sum((prob for x, y in patterns if x + y == k), Q(0))
    for k in range(3)
)
parity_probabilities = (
    sum((prob for x, y in patterns if (x + y) % 2 == 0), Q(0)),
    sum((prob for x, y in patterns if (x + y) % 2 == 1), Q(0)),
)
assert number_probabilities == (Q(1, 3), Q(1, 3), Q(1, 3))
assert parity_probabilities == (Q(2, 3), Q(1, 3))

# A terminal two-particle projection of two independent BCS pairs yields
# omega=e12+e34. Its exterior square is exactly 2 e1234, so it is not decomposable.
exterior_square_coefficient = 2
assert exterior_square_coefficient != 0
