def scale_interval(c, interval):
    lo,hi=interval
    return (c*lo,c*hi) if c>=0 else (c*hi,c*lo)

def log_interval(y,m):
    assert y>0
    x=y;k=0
    while x>=2:x/=2;k+=1
    while x<1:x*=2;k-=1
    def unit(z):
        total=F(0);power=z
        for j in range(m):
            total+=2*power/F(2*j+1);power*=z*z
        tail=2*power/(F(2*m+1)*(1-z*z))
        return total,total+tail
    lx=unit((x-1)/(x+1));l2=scale_interval(F(k),unit(F(1,3)))
    return lx[0]+l2[0],lx[1]+l2[1]

def endpoint_interval(b,t,tolerance):

    assert 0<x<1
    q=[row[:] for row in Q0];q[0][2]+=b;q[0][0]-=b
    transition=[[F(i==j)+q[i][j]/rho for j in range(3)] for i in range(3)]
    assert all(v>=0 for row in transition for v in row)
    # Alternating exponential remainder, and positive uniformization tail.
    total=F(1);term=F(1)
    for j in range(10000):
        nxt=term*x/F(j+1)
        elo,ehi=(total-nxt,total) if j%2==0 else (total,total+nxt)
        if ehi-elo<=tolerance/8:break
        total+=(-1 if (j+1)%2 else 1)*nxt;term=nxt
    else:raise RuntimeError('exponential fixture cap')
    row=[F(1),F(0),F(0)];weight=F(1);partial=F(0)

def invert_frequency(beta,t,z,e):
    low,high=beta,min(4*beta,F(1));tau=e*beta*t/F(32);w=e*beta/F(4)
    evaluations=0
    while high-low>w:
        mid=(low+high)/2
        lo,hi=endpoint_interval(mid,t,2*tau);evaluations+=1
        if lo<=z<=hi:return mid,evaluations,'probability_tolerance'
        if z<lo:high=mid
        else:low=mid
    return (low+high)/2,evaluations,'parameter_width'
