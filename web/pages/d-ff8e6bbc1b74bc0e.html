"""Rebuild every original generator and replay the isolated-star certificate.

No creator/checker/reduction imports. A coefficientwise QQ[t] remainder
identity is sufficient; no field irreducibility or numerical root is assumed.
"""
import gzip
import hashlib
import itertools
import json
import time
from fractions import Fraction
from pathlib import Path

import sympy as sy


BASE=Path(__file__).resolve().parents[3]


def main():
    start=time.monotonic()
    path=BASE/'results/metric_prior/cycle14/isolated_star_rational_D8_certificate.json.gz'
    record=json.loads(gzip.decompress(path.read_bytes()))
    t=sy.symbols('t');m=sy.symbols('m0:3');s=sy.symbols('s0:3')
    variables=tuple(m+s)
    assert record['variables']==[str(v) for v in variables]
    assert record['parameters']==['a=t','b=t','c=0']
    q=sy.Matrix([[1,t,t],[t,1,0],[t,0,1]])
    own=[(1-2*m[i]*s[i])*(1-s[i]**2) for i in range(3)]
    own += [(1-4*m[i]*s[i])*(q[i,j]-s[i]*s[j])-2*m[i]*s[j]*(1-s[i]**2)
            for i,j in itertools.permutations(range(3),2)]
    own += [3*sy.prod(s)-t*s[1]-t*s[2]]
    assert len(own)==10
    terms=[]
    for expression in own:
        source=[]
        for mon,co in sy.Poly(expression,*variables,domain=sy.QQ.poly_ring(t)).terms():
            source.append((mon,[(int(ex[0]),Fraction(str(value)))
                               for ex,value in sy.Poly(co,t,domain=sy.QQ).terms()]))
        terms.append(source)
    total={};degree_max=0;bits=0
    for stored in record['terms']:
        gi=stored['generator'];shift=tuple(stored['auxiliary_monomial'])
        assert type(gi) is int and 0<=gi<10
        assert len(shift)==6 and all(type(v) is int and v>=0 for v in shift)
        assert sum(shift)%2==(1 if gi==9 else 0)
        assert sum(shift)<=(5 if gi==9 else 4)
        values=[Fraction(v) for v in stored['coefficient_ascending_t_basis']]
        assert len(values)==4 and any(values)
        bits=max(bits,max(max(abs(v.numerator).bit_length(),v.denominator.bit_length()) for v in values))
        for ex,coefficients in terms[gi]:
            mon=tuple(ex[i]+shift[i] for i in range(6))
            degree_max=max(degree_max,sum(mon))
            row=total.setdefault(mon,[Fraction(0)]*5)
            for i,value in enumerate(values):
                for j,co in coefficients:
                    assert i+j<=4
                    row[i+j]+=value*co
    assert len(record['terms'])==588 and degree_max<=8
    total.setdefault((0,)*6,[Fraction(0)]*5)[0]-=1
    h=[Fraction(-528),Fraction(0),Fraction(1127),Fraction(0),Fraction(1302)]
    assert [Fraction(x) for x in record['report']['minpoly_ascending']]==h
    nonzero_quotients=0
    for mon,row in total.items():
        coefficient=row[4]/h[4]
        remainder=[row[i]-coefficient*h[i] for i in range(5)]
        assert not any(remainder), f'raw QQ[t] coefficient remainder failed at {mon}'
        nonzero_quotients+=bool(coefficient)
    # Independent complete exception and PD-domain checks.
    A,B=sy.symbols('A B')
    p=1302*A*B-40*A+1167*B-528
    assert sy.expand(p-p.xreplace({A:B,B:A})-1207*(B-A))==0
    assert sy.expand(p.subs(B,A)-(1302*A*A+1127*A-528))==0
    g=1302*A*A+1127*A-528
    assert g.subs(A,0)==-528 and g.subs(A,sy.Rational(1,2))==361
    assert sy.Poly(g,A).count_roots(0,sy.Rational(1,2))==1
    assert sy.Poly(1302*t**4+1127*t*t-528,t).count_roots(0,1)==1
    assert q.det()==1-2*t*t
    record_out={
        'status':'PASS complete coefficientwise original-generator identity modulo h',
        'certificate_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
        'source_path':str(path.relative_to(BASE)),
        'ten_raw_generators_reconstructed':True,'raw_terms':588,
        'complete_auxiliary_coefficient_keys':len(total),
        'maximum_auxiliary_product_degree':degree_max,
        'maximum_stored_rational_coefficient_bits':bits,
        'coefficient_reduction_ring':'QQ[t]/(1302t4+1127t2-528)',
        'every_original_auxiliary_coefficient_remainder':'zero',
        'nonzero_h_quotient_coefficients':nonzero_quotients,
        'exception_exhaustion':'p(A,B)-p(B,A)=1207(B-A); A=B yields h(A)',
        'positive_root_PD':'unique 0<t2<1/2, determinant1-2t2>0',
        'scope':'Together with previously gated two-chart real cover outside this isolated orbit and lower-edge identities, all real PD zero-edge triples have D8. Rational-hidden zero-edge acquisition was already complete. Dense positive-product exceptions remain open.',
        'seconds':time.monotonic()-start}
    output=BASE/'results/metric_use10/cycle14/isolated_star_gate.json'
    output.write_text(json.dumps(record_out,indent=2)+'\n')
    print(json.dumps({'status':record_out['status'],'raw_terms':588,
                      'maximum_auxiliary_product_degree':degree_max,
                      'seconds':record_out['seconds'],'output':str(output)}))


if __name__=='__main__':main()
