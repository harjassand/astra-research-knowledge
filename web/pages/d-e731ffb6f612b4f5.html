def ceil_log2(n: int) -> int:
    if n < 1:
        raise ValueError("positive integer required")
    return (n - 1).bit_length()


def primality_certificate(n: int) -> dict:
    """A deterministic trial-division proof or smallest found divisor."""
    if n < 2:
        return {"n": n, "prime": False, "divisor": None, "divisions": 0}
    if n == 2:
        return {"n": n, "prime": True, "divisor": None, "divisions": 0}
    count = 1
    if n % 2 == 0:
        return {"n": n, "prime": False, "divisor": 2, "divisions": count}
    d = 3
    while d * d <= n:
        count += 1
        if n % d == 0:
            return {"n": n, "prime": False, "divisor": d, "divisions": count}
        d += 2
    return {"n": n, "prime": True, "divisor": None, "divisions": count,
            "last_odd_tested": d - 2}


@dataclass
class Acquisition:
    records: list[dict] = field(default_factory=list)
    order_attempts: list[dict] = field(default_factory=list)

    def inner_pool(self, lower: int, capacity: int) -> list[dict]:
        q = lower + 1
        if q <= 2:
            q = 2
        elif q % 2 == 0:
            q += 1
        rows = []
        total = 0
        while total < capacity:
            cert = primality_certificate(q)
            self.records.append({"kind": "q", **cert})
            if cert["prime"]:
                ell = q.bit_length() - 1
                rows.append({"q": q, "ell": ell, "certificate": cert})
                total += ell
            q = q + 1 if q == 2 else q + 2
        return rows

    def outer_pool(self, q: int, capacity: int) -> list[dict]:
        rows = []
        total = 0
        multiplier = 1
        while total < capacity:
            p = multiplier * q + 1
            cert = primality_certificate(p)
            self.records.append({"kind": "p", "q": q,
                                 "multiplier": multiplier, **cert})
            if cert["prime"]:
                h = 2
                while True:
                    gamma = pow(h, multiplier, p)
                    accepted = gamma != 1 and gamma != 0
                    self.order_attempts.append({"p": p, "q": q, "h": h,
                                                "gamma": gamma,
                                                "accepted": accepted})
                    if accepted:
                        assert pow(gamma, q, p) == 1
                        break
                    h += 1
                ell = p.bit_length() - 1
                rows.append({"p": p, "ell": ell, "gamma": gamma,
                             "h": h, "certificate": cert})
                total += ell
            multiplier += 1
        return rows

    def summary(self) -> dict:
        return {"candidate_count": len(self.records),
                "accepted_count": sum(r["prime"] for r in self.records),
                "rejected_count": sum(not r["prime"] for r in self.records),
                "trial_divisions": sum(r["divisions"] for r in self.records),
                "order_attempt_count": len(self.order_attempts),
                "max_candidate_bits": max((r["n"].bit_length()
                                           for r in self.records), default=0)}


class ExactRandom:
    def __init__(self, seed: int):
        self.rng = random.Random(seed)
        self.bits = 0
        self.rejections = 0

    def below(self, n: int) -> int:
        if n < 1:
            raise ValueError("empty distribution")
        if n == 1:
            return 0
        bits = (n - 1).bit_length()
        while True:
            self.bits += bits
            r = self.rng.getrandbits(bits)
            if r < n:
                return r
            self.rejections += 1

    def weighted(self, rows: list[dict]) -> dict:
        r = self.below(sum(row["ell"] for row in rows))
        for row in rows:
            if r < row["ell"]:
                return row
            r -= row["ell"]
        raise AssertionError("weight table")


def validate_pool(q_rows: list[dict], p_rows: list[dict], q: int,
                  q_lower: int, q_capacity: int, p_capacity: int) -> None:
    q_values = [r["q"] for r in q_rows]
    p_values = [r["p"] for r in p_rows]
    if len(set(q_values)) != len(q_values) or len(set(p_values)) != len(p_values):
        raise ValueError("repeated primes cannot add capacity")
    if q not in q_values:
        raise ValueError("selected q absent")
    for r in q_rows:
        n = r["q"]
        if n <= q_lower or not primality_certificate(n)["prime"]:
            raise ValueError("invalid inner prime")
        if r["ell"] != n.bit_length() - 1:
            raise ValueError("invalid inner weight")
    for r in p_rows:
        p = r["p"]
        if not primality_certificate(p)["prime"] or (p - 1) % q:
            raise ValueError("invalid outer prime")
        if r["ell"] != p.bit_length() - 1:
            raise ValueError("invalid outer weight")
        if not (1 < r["gamma"] < p) or pow(r["gamma"], q, p) != 1:
            raise ValueError("invalid order-q element")
    if sum(r["ell"] for r in q_rows) < q_capacity:
        raise ValueError("insufficient inner capacity")
    if sum(r["ell"] for r in p_rows) < p_capacity:
        raise ValueError("insufficient outer capacity")


class LinearExponentialOracle:
    """Fixed integer f_j(X),g_j(X),h_j=1, hidden from the tester.

    Modular powers and reductions are actual computations, not a unit-cost
    real-exponential or target-coefficient oracle. Counts describe this
    supplied toy provider only, not general neural-network cost.
    """
    def __init__(self, coefficient_pairs: list[tuple[int, int]],
                 exponent_pairs: list[tuple[int, int]]):
        self.fs = tuple(coefficient_pairs)
        self.gs = tuple(exponent_pairs)
        self.calls = []
        self.modular_pow_calls = 0
        self.modular_mul_calls = 0
        self.constant_reduction_bits = 0

    def __call__(self, p: int, q: int, a: int, u: list[int], v: list[int]):
        assert len(u) == len(v) == 1
        result = 0
        for (fc, fa), (gc, ga) in zip(self.fs, self.gs):
            self.constant_reduction_bits += sum(abs(t).bit_length()
                                               for t in (fc, fa, gc, ga))
            fv = (fc % p + (fa % p) * u[0]) % p
            gv = (gc % q + (ga % q) * v[0]) % q
            result = (result + fv * pow(a, gv, p)) % p
            self.modular_pow_calls += 1
            self.modular_mul_calls += 3
        query_bits = p.bit_length() * (len(u) + 1) + q.bit_length() * (len(v) + 1)
        record = {"p": p, "q": q, "a": a, "u": u, "v": v,
                  "response": result, "query_payload_upper_bits": query_bits,
                  "response_upper_bits": p.bit_length()}
        self.calls.append(record)
        return result


def one_query(oracle, params: dict, q_rows: list[dict], acquisition: Acquisition,
              rng: ExactRandom) -> tuple[dict, list[dict]]:
    selected = rng.weighted(q_rows)
    q = selected["q"]
    required = 6 * ceil_log2(params["k"] * params["w"])
    p_rows = acquisition.outer_pool(q, required)
    p_selected = rng.weighted(p_rows)
    p = p_selected["p"]
    r = rng.below(q - 1) + 1
    a = pow(p_selected["gamma"], r, p)
    u = [rng.below(p) for _ in range(params["n"])]
    v = [rng.below(q) for _ in range(params["n"])]
    start = time.perf_counter()
    answer = oracle(p, q, a, u, v)
    oracle_seconds = time.perf_counter() - start
    return {"q": q, "p": p, "r": r, "a": a, "u": u, "v": v,
            "answer": answer, "accept_zero": answer in (0, None),
            "oracle_seconds": oracle_seconds}, p_rows


def multiplication_matrix(c: tuple[int, ...], q: int) -> list[list[int]]:
    """Multiply by C modulo the monic 1+z+...+z^(q-1)."""
    n = q - 1
    columns = []
    for shift in range(n):
        a = [0] * (len(c) + shift)
        for j, coefficient in enumerate(c):
            a[j + shift] += coefficient
        for j in range(len(a) - 1, n - 1, -1):
            coefficient = a[j]
            for t in range(n + 1):
                a[j - n + t] -= coefficient
        columns.append((a + [0] * n)[:n])
    return [[columns[j][i] for j in range(n)] for i in range(n)]


def bareiss_det(a: list[list[int]]) -> int:
    a = [row[:] for row in a]
    n = len(a)
    if n == 1:
        return a[0][0]
    denominator = 1
    sign = 1
    for k in range(n - 1):
        pivot_row = next((i for i in range(k, n) if a[i][k]), None)
        if pivot_row is None:
            return 0
        if pivot_row != k:
            a[k], a[pivot_row] = a[pivot_row], a[k]
            sign *= -1
        pivot = a[k][k]
        for i in range(k + 1, n):
            for j in range(k + 1, n):
                numerator = pivot * a[i][j] - a[i][k] * a[k][j]
                assert numerator % denominator == 0
                a[i][j] = numerator // denominator
            a[i][k] = 0
        denominator = pivot
    return sign * a[-1][-1]


def modular_nullity(a: list[list[int]], p: int) -> int:
    a = [[v % p for v in row] for row in a]
    m = len(a)
    rank = 0
    for col in range(m):
        row = next((r for r in range(rank, m) if a[r][col]), None)
        if row is None:
            continue
        a[rank], a[row] = a[row], a[rank]
        inv = pow(a[rank][col], -1, p)
        a[rank] = [v * inv % p for v in a[rank]]
        for r in range(rank + 1, m):
            factor = a[r][col]
            a[r] = [(x - factor * y) % p for x, y in zip(a[r], a[rank])]
        rank += 1
    return m - rank


def primitive_roots(p: int, q: int) -> list[int]:
    for h in range(2, p):
        gamma = pow(h, (p - 1) // q, p)
        if gamma != 1:
            return [pow(gamma, r, p) for r in range(1, q)]
    raise AssertionError("order-q subgroup")


def polynomial_value(c: tuple[int, ...], a: int, p: int) -> int:
    value = 0
    for coefficient in reversed(c):
        value = (value * a + coefficient) % p
    return value


def root_diagnostics() -> dict:
    start = time.perf_counter()
    checked = 0
    nonzero_root_events = 0
    max_det_bits = 0
    randomizer = random.Random(131313)
    row_reports = []
    for q in (3, 5, 7, 11):
        acquisition = Acquisition()
        pool = acquisition.outer_pool(q, 18)
        pool = pool[:3]
        if q in (3, 5):
            vectors = product((-2, -1, 0, 1, 2), repeat=q)
        else:
            vectors = []
            for _ in range(250):
                support = randomizer.sample(range(q), randomizer.randrange(1, min(q, 5)))
                v = [0] * q
                for i in support:
                    v[i] = randomizer.choice((-3, -2, -1, 1, 2, 3))
                vectors.append(tuple(v))
        per_q = 0
        for c in vectors:
            if not any(c) or sum(v != 0 for v in c) == q:
                continue
            matrix = multiplication_matrix(c, q)
            resultant = bareiss_det(matrix)
            assert resultant != 0
            norm = sum(abs(v) for v in c)
            assert abs(resultant) <= norm ** (q - 1)
            max_det_bits = max(max_det_bits, abs(resultant).bit_length())
            norm_product = 1
            weighted_roots = 0
            for row in pool:
                p = row["p"]
                roots = sum(polynomial_value(c, a, p) == 0
                            for a in primitive_roots(p, q))
                assert modular_nullity(matrix, p) == roots
                assert resultant % (p ** roots) == 0
                norm_product *= p ** roots
                weighted_roots += row["ell"] * roots
                nonzero_root_events += roots > 0
            assert abs(resultant) % norm_product == 0
            assert norm_product <= norm ** (q - 1)
            assert weighted_roots <= (q - 1) * ceil_log2(norm)
            checked += 1
            per_q += 1
        row_reports.append({"q": q, "checked": per_q,
                            "p_pool": [r["p"] for r in pool]})

    # Decisive interface failures, not counterexamples to the admitted lemma.
    full_support = bareiss_det(multiplication_matrix((1, 1, 1, 1, 1), 5)) == 0
    composite_vector = (1, 0, 0, 1, 0, 0, 1, 0, 0)  # Phi_9, degree6, support3.
    composite_det_zero = bareiss_det(multiplication_matrix(composite_vector, 9)) == 0
    # Same prime counted three times falsely promises a capacity factor3.
    repeated_prime_actual = Fraction(1)
    repeated_prime_false_bound = Fraction(ceil_log2(11), 3 * (11 .bit_length() - 1))
    # C_p=p is degree0/support1 at each characteristic, but input changes with p.
    adaptive_pool = (11, 31, 41)
    adaptive_false_bound = Fraction(ceil_log2(max(adaptive_pool)),
                                   sum(p.bit_length() - 1 for p in adaptive_pool))
    return {"status": "PASS", "vectors_checked": checked,
            "prime_root_checks": checked * 3,
            "nonzero_root_events": nonzero_root_events,
            "max_determinant_bits": max_det_bits, "rows": row_reports,
            "seconds": time.perf_counter() - start,
            "obstructions": {
                "full_support_zero_norm": full_support,
                "composite_q_sparse_zero_norm": composite_det_zero,
                "duplicate_prime_actual_error": str(repeated_prime_actual),
                "duplicate_prime_false_bound": str(repeated_prime_false_bound),
                "p_adaptive_constant_actual_error": "1",
                "p_adaptive_constant_false_bound": str(adaptive_false_bound)}}


def exact_characteristic_fixture() -> tuple[dict, dict]:
    """P=1-exp(103): exact error from q=103, with d0/k2/w103.

    Diagnostic enumerates roots; the tester itself never performs this scan.
    """
    start = time.perf_counter()
    acquisition = Acquisition()
    q_rows = acquisition.inner_pool(2, 6 * 3 * ceil_log2(103))
    capacity = sum(r["ell"] for r in q_rows)
    accepted = Fraction(0)
    evaluations = 0
    pools = {}
    for row in q_rows:
        q = row["q"]
        p_rows = acquisition.outer_pool(q, 6 * ceil_log2(206))
        pools[str(q)] = p_rows
        p_capacity = sum(p["ell"] for p in p_rows)
        conditional = Fraction(0)
        for p_row in p_rows:
            p = p_row["p"]
            roots = primitive_roots(p, q)
            zeros = sum((1 - pow(a, 103 % q, p)) % p == 0 for a in roots)
            evaluations += len(roots)
            conditional += Fraction(p_row["ell"] * zeros, p_capacity * (q - 1))
        accepted += Fraction(row["ell"], capacity) * conditional
    predicted_exact = Fraction(next(r["ell"] for r in q_rows if r["q"] == 103), capacity)
    assert accepted == predicted_exact
    bound = Fraction(3 * ceil_log2(103), capacity) + Fraction(ceil_log2(206), 48)
    assert accepted <= bound < Fraction(1, 2)
    return {"status": "PASS", "P": "1-exp(103)", "n": 0,
            "k": 2, "d": 0, "w": 103, "q_count": len(q_rows),
            "q_min": q_rows[0]["q"], "q_max": q_rows[-1]["q"],
            "inner_capacity": capacity, "exact_acceptance": str(accepted),
            "theorem_upper_bound": str(bound), "root_evaluations": evaluations,
            "seconds": time.perf_counter() - start,
            "acquisition": acquisition.summary()}, {
                "q_pool": q_rows, "p_pools": pools,
                "candidates": acquisition.records,
                "order_attempts": acquisition.order_attempts}


def charged_prototype() -> tuple[dict, dict]:
    start = time.perf_counter()
    k, d, n, w = 4, 1, 1, 1 << 256
    params = {"k": k, "d": d, "n": n, "w": w}
    constraints = k + k * (k - 1) // 2
    q_capacity = 6 * constraints * ceil_log2(w)
    p_capacity = 6 * ceil_log2(k * w)
    q_lower = max(k, 6 * d * (k * k + 1))
    acquisition = Acquisition()
    q_start = time.perf_counter()
    q_rows = acquisition.inner_pool(q_lower, q_capacity)
    q_seconds = time.perf_counter() - q_start
    fs = [(1, w), (-1, -w), (2, 1), (3, -1)]
    gs = [(j * j, j + 1) for j in range(k)]
    oracle = LinearExponentialOracle(fs, gs)
    rng = ExactRandom(131313)
    query_start = time.perf_counter()
    query, p_rows = one_query(oracle, params, q_rows, acquisition, rng)
    query_seconds = time.perf_counter() - query_start
    gate_start = time.perf_counter()
    validate_pool(q_rows, p_rows, query["q"], q_lower, q_capacity, p_capacity)
    pool_gate_seconds = time.perf_counter() - gate_start
    actual_q_capacity = sum(row["ell"] for row in q_rows)
    actual_p_capacity = sum(row["ell"] for row in p_rows)
    # The global proof uses the promised p capacity for every lazy q;
    # observed p minima/capacity alone are not treated as global minima.
    certified_bound = (Fraction(constraints * ceil_log2(w), actual_q_capacity)
                       + Fraction(d * (k * k + 1), q_rows[0]["q"])
                       + Fraction(ceil_log2(k * w), p_capacity))
    assert certified_bound < Fraction(1, 2)
    zero_oracle = LinearExponentialOracle([(0, 0)] * k, gs)
    zero_answer = zero_oracle(query["p"], query["q"], query["a"], query["u"], query["v"])
    assert zero_answer == 0
    assert len(oracle.calls) == len(zero_oracle.calls) == 1
    # Admission/membership mutations are explicit and retain their failure.
    mutants = {}
    bad = p_rows + [dict(p_rows[0])]
    try:
        validate_pool(q_rows, bad, query["q"], q_lower, q_capacity, p_capacity)
        mutants["duplicate_outer_prime"] = "MISSED"
    except ValueError as error:
        mutants["duplicate_outer_prime"] = str(error)
    bad = [dict(r) for r in p_rows]
    bad[0]["gamma"] = 1
    try:
        validate_pool(q_rows, bad, query["q"], q_lower, q_capacity, p_capacity)
        mutants["identity_generator"] = "MISSED"
    except ValueError as error:
        mutants["identity_generator"] = str(error)
    bad = [dict(r) for r in q_rows]
    bad[0]["q"] = 105
    try:
        validate_pool(bad, p_rows, query["q"], q_lower, q_capacity, p_capacity)
        mutants["composite_inner_prime"] = "MISSED"
    except ValueError as error:
        mutants["composite_inner_prime"] = str(error)
    assert "MISSED" not in mutants.values()
    pool_storage_bits = sum(r["q"].bit_length() + r["ell"].bit_length() for r in q_rows)
    pool_storage_bits += sum(r["p"].bit_length() + r["ell"].bit_length()
                             + r["gamma"].bit_length() for r in p_rows)
    report = {"status": "PASS", "params": params,
              "required_q_capacity": q_capacity, "actual_q_capacity": actual_q_capacity,
              "required_p_capacity": p_capacity, "actual_used_p_capacity": actual_p_capacity,
              "q_pool_count": len(q_rows), "used_p_pool_count": len(p_rows),
              "q_min": q_rows[0]["q"], "q_max": q_rows[-1]["q"],
              "used_p_min": p_rows[0]["p"], "used_p_max": p_rows[-1]["p"],
              "max_q_bits": q_rows[-1]["q"].bit_length(),
              "max_used_p_bits": p_rows[-1]["p"].bit_length(),
              "pool_payload_upper_bits": pool_storage_bits,
              "guaranteed_error_upper_bound": str(certified_bound),
              "single_nonzero_query": query,
              "single_zero_query_response": zero_answer,
              "random_bits_consumed": rng.bits, "sampling_rejections": rng.rejections,
              "input_constant_bits_read_nonzero_query": oracle.constant_reduction_bits,
              "oracle_modular_mul_calls": oracle.modular_mul_calls,
              "oracle_modular_pow_calls": oracle.modular_pow_calls,
              "acquisition": acquisition.summary(), "mutant_rejections": mutants,
              "timing_seconds": {"q_acquisition": q_seconds,
                                   "lazy_p_acquisition_plus_sampling_and_query": query_seconds,
                                   "fresh_trial_division_pool_gate": pool_gate_seconds,
                                   "nonzero_query_only": query["oracle_seconds"],
                                   "prototype_total": time.perf_counter() - start},
              "control": {"June_source_required_q_greater_than_2kw": str(2 * k * w),
                          "June_source_q_minimum_bit_length": (2 * k * w).bit_length(),
                          "large_field_source_not_executed": True,
                          "no_compiler_or_end_to_end_speedup_claim": True}}
    log = {"parameters": params, "coefficient_pairs": fs, "exponent_pairs": gs,
           "q_pool": q_rows, "used_p_pool": p_rows,
           "selected_q": query["q"], "candidate_records": acquisition.records,
           "order_attempts": acquisition.order_attempts,
           "nonzero_oracle_log": oracle.calls, "zero_oracle_log": zero_oracle.calls}
    return report, log
