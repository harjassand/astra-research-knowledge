def p_kernel(x,y,r):
    z,w=(x+y)/2,(x-y)/2
    a,b=np.full_like(w,1-r),np.full_like(w,r)
    mask=np.abs(w)>1e-10
    a[mask]=np.sinh(2*(1-r)*w[mask])/np.sinh(2*w[mask])
    b[mask]=np.sinh(2*r*w[mask])/np.sinh(2*w[mask])
    return np.exp(-z)*a+np.exp(z)*b-np.exp((2*r-1)*z)
