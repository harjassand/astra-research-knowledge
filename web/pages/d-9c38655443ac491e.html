def acquire(U,M,seed=20261006,max_trials=30,p=None,x=None):
    n=M.rows;Mi=M.inv();A=[Mi*S for S in U]
    ij=[(i,j) for i in range(n) for j in range(i,n)]
    Hb=[]
    for i,j in ij:
        H=s.zeros(n);H[i,j]=H[j,i]=1;Hb.append(H)
    Xb=[Mi*H for H in Hb]
    Avec=s.Matrix.hstack(*[s.Matrix(n*n,1,list(a)) for a in A])
    # Every annihilator of U must annihilate each Jordan product.
    ann=Avec.T.nullspace();constraints=[]
    for a in A:
        J=s.Matrix.hstack(*[s.Matrix(n*n,1,list(X*a+a*X)) for X in Xb])
        for f in ann:constraints.append((f.T*J).tolist()[0])
    K=s.Matrix(constraints)
    ns=K.nullspace();X=[sum((v[i]*Xb[i] for i in range(len(Xb))),s.zeros(n)) for v in ns]
    # Center of the multiplier Jordan algebra: its primitive spaces are false-twin blocks.
    comm=[]
    for a in X:
        comm.append(s.Matrix.hstack(*[s.Matrix(n*n,1,list(b*a-a*b)) for b in X]))
    centers=s.Matrix.vstack(*comm).nullspace()
    Z=[sum((v[i]*X[i] for i in range(len(X))),s.zeros(n)) for v in centers]
    rng=random.Random(seed)
    for trial in range(max_trials):
        w=[rng.randrange(-2*n*n,2*n*n+1) for _ in Z]
        H=sum((c*a for c,a in zip(w,Z)),s.zeros(n))
        fac=s.factor_list(H.charpoly().as_expr())[1]
        if len(fac)!=len(Z) or any(s.total_degree(f)!=1 for f,mult in fac):continue
        eig=H.eigenvects()
        if len(eig)!=len(Z) or any(not lam.is_Rational or len(v)!=mult for lam,mult,v in eig):continue
        blocks=[s.Matrix.hstack(*v) for _,_,v in eig]
        axes=[];block_trials=[]
        for W in blocks:
            k=W.cols
            if k==1:axes.append(W);block_trials.append(0);continue
            if p is None or x is None:raise ValueError('nontrivial false-twin blocks require quartic coefficients')
            y=s.symbols('z:'+str(k));sub={x[i]:sum(W[i,j]*y[j] for j in range(k)) for i in range(n)}
            restricted=s.Poly(s.expand(p.subs(sub,simultaneous=True)),*y)
            f4=sum(c*__import__('math').prod(yj**ej for yj,ej in zip(y,e)) for e,c in restricted.as_dict().items() if sum(e)==4)
            metric=W.T*M*W
            for bt in range(max_trials):
                h=[rng.randrange(-2*k*k,2*k*k+1) for _ in range(k)]
                A=metric.inv()*s.hessian(f4,y).subs(dict(zip(y,h)))/12
                fs=s.factor_list(A.charpoly().as_expr())[1]
                if len(fs)!=k or any(s.total_degree(f)!=1 or mult!=1 for f,mult in fs):continue
                ev=A.eigenvects();U=s.Matrix.hstack(*[vs[0] for _,_,vs in ev])
                axes.append(W*U);block_trials.append(bt+1);break
            else:raise ValueError('UNKNOWN: unresolved pure-power block within budget')
        V=s.Matrix.hstack(*axes);assert V.det()!=0
        return V,{'multiplier_dimension':len(X),'center_dimension':len(Z),
                  'block_dimensions':[W.cols for W in blocks],'block_trials':block_trials,
                  'constraint_matrix_shape':list(K.shape),
                  'trial':trial+1,'random_weights':w,'axes_acquired':n,'supplied_hidden_graph':False}
    raise ValueError('UNKNOWN: no split center multiplier within trial budget')


def discover_jordan(p,x):
    def callback(U,Minv,rng):
        return acquire(U,Minv.inv(),seed=rng.randrange(2**32),p=p,x=x)
    return discover(p,x,edge_acquirer=callback,any_multiplicity=True)

    for n in [4,5,6,7,9]:

        LV=L*V;assert all(sum(LV[i,j]!=0 for i in range(n))==1 for j in range(n))
        r.update({'n':n,'seconds':dt,'input_coefficients':len(s.Poly(p,*x).terms()),
