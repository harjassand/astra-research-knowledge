def zero(n, m=None):
    return [[F(0) for _ in range(m or n)] for _ in range(n)]


def eye(n):
    a = zero(n)
    for i in range(n):
        a[i][i] = F(1)
    return a


def add(a, b):
    return [[x+y for x,y in zip(ar,br)] for ar,br in zip(a,b)]


def scale(c, a):
    return [[c*x for x in row] for row in a]


def trans(a):
    return [list(row) for row in zip(*a)]


def mul(a, b):
    return [[sum((x*y for x,y in zip(row,col)), F(0))
             for col in zip(*b)] for row in a]


def tr(a):
    return sum((row[i] for i,row in enumerate(a)), F(0))


def tau(a):
    return tr(a)/len(a)


def unit(n, i, j):
    a = zero(n)
    a[i][j] = F(1)
    return a


def apply_kraus(ks, a):
    ans = zero(len(ks[0]))

    return ans


def ptr_output(a, keep):
    """One 3-dimensional marginal from a two-output 9 by 9 matrix."""
    r = zero(3)
    for i in range(3):
        for j in range(3):
            for z in range(3):
                ai,aj = ((3*i+z,3*j+z) if keep==0
                         else (3*z+i,3*z+j))
                r[i][j] += a[ai][aj]
    return r


def choi_single(r, keep):
    a = zero(3)

                    flat = lambda v: 9*v[0]+3*v[1]+v[2]
                    a[i][j] += r[flat(ix)][flat(iy)]
    return a
