#!/usr/bin/env python3
"""Symbolic checks for the actual-E, finite Dirichlet stripe parametrix.

No global PDE solver or numerical proof is claimed. The analytic localization,
soft derivative and stable return estimates remain in the companion note.
"""
from __future__ import annotations

import argparse
import json
import math
from pathlib import Path

import sympy as sp


def run() -> dict:
    h, v, b = sp.symbols("h v b", real=True)
    R, t, a, K, n = sp.symbols("R t a K n", positive=True)
    g = sp.Function("g")(R, t)
    E = sp.exp(-a * R)
    mu = t / R
    rr = 1 - K**2 / R**2
    lapR = (n - 1) / R + K**2 / R**3
    dR = K**2 / R**2 + lapR / a

    def slow_lap(q):
        return (rr * sp.diff(q, R, 2) + lapR * sp.diff(q, R)
                + 2 * mu * sp.diff(q, R, t) + sp.diff(q, t, 2))

    def Bop(q):
        return -2 * mu * q + 2 / a * (sp.diff(q, t) + mu * sp.diff(q, R))

    def Cop(q):
        return (-2 / a * (rr * sp.diff(q, R) + mu * sp.diff(q, t))
                + slow_lap(q) / a**2 - dR * q)

    # Compute the transformed Laplacian by differentiating E*g before using
    # any claimed operator coefficient. Its phi, phi', phi'' coefficients
    # must be Cop(g)+g, Bop(g), and g respectively.
    phi_coefficient = slow_lap(E * g) / (a**2 * E)
    phiprime_coefficient = 2 / (a * E) * (
        sp.diff(E * g, t) + mu * sp.diff(E * g, R))
    checks = {
        "exact_operator_phi_coefficient": sp.simplify(phi_coefficient - Cop(g) - g) == 0,
        "exact_operator_phiprime_coefficient": sp.simplify(phiprime_coefficient - Bop(g)) == 0,
    }

    inverse_cache = {}
    inverse_records = []

    def L0(f):
        return sp.diff(f, h, 2) + f

    def inverse(f):
        f = sp.trigsimp(sp.expand(f))
        key = sp.srepr(f)
        if key in inverse_cache:
            return inverse_cache[key]
        # Variation of constants with zero data at h=0, then adjust both
        # homogeneous constants to impose the ACTUAL stripe walls +/-b.
        cv = sp.integrate(sp.cos(v) * f.subs(h, v), (v, 0, h))
        sv = sp.integrate(sp.sin(v) * f.subs(h, v), (v, 0, h))
        up = sp.trigsimp(sp.expand(sp.sin(h) * cv - sp.cos(h) * sv))
        ub, umb = up.subs(h, b), up.subs(h, -b)
        ac = -(ub + umb) / (2 * sp.cos(b))
        bs = -(ub - umb) / (2 * sp.sin(b))
        u = sp.trigsimp(sp.expand(up + ac * sp.cos(h) + bs * sp.sin(h)))
        record = {
            "forcing": str(f),
            "solution": str(u),
            "L0_inverse": sp.trigsimp(sp.expand(L0(u) - f)) == 0,
            "left_trace": sp.trigsimp(u.subs(h, -b)) == 0,
            "right_trace": sp.trigsimp(u.subs(h, b)) == 0,
        }
        inverse_records.append(record)
        inverse_cache[key] = u
        return u

    # Keep separable terms, rather than silently truncating the exact slow
    # coefficients. There are 2, 4, 8 terms at orders one, two, three.
    stages = [[(sp.Integer(1), -sp.sin(h))]]
    stage_records = []
    for order in (1, 2, 3):
        new_terms = []
        identity_checks = []
        for coeff, fast in stages[-1]:
            bf, cf = Bop(coeff), Cop(coeff)
            inv_derivative = inverse(sp.diff(fast, h))
            inv_fast = inverse(fast)
            new_terms.extend([(-bf, inv_derivative), (-cf, inv_fast)])
            identity_checks.extend([
                sp.trigsimp(sp.expand(L0(inv_derivative) - sp.diff(fast, h))) == 0,
                sp.trigsimp(sp.expand(L0(inv_fast) - fast)) == 0,
            ])
        stages.append(new_terms)
        stage_records.append({
            "order": order,
            "term_count": len(new_terms),
            "all_inverse_identities": all(identity_checks),
            "slow_coefficient_ops": [int(sp.count_ops(q)) for q, _ in new_terms],
            "fast_coefficient_ops": [int(sp.count_ops(f)) for _, f in new_terms],
        })

    p = b * sp.tan(b) * sp.cos(h) - h * sp.sin(h)
    r_D = (h * sp.cos(h) - b * sp.cot(b) * sp.sin(h)) / 2
    U1 = sum(q * f for q, f in stages[1])
    checks["first_corrector_exact"] = sp.trigsimp(sp.expand(U1 - mu * p - dR * r_D)) == 0
    checks["finite_telescope_exact"] = all(x["all_inverse_identities"] for x in stage_records)
    checks["all_Dirichlet_inverse_checks"] = all(
        x["L0_inverse"] and x["left_trace"] and x["right_trace"] for x in inverse_records)

    zeta = sp.symbols("zeta", real=True)
    csc_adjoint = sp.diff(1 / sp.sin(zeta), zeta, 2) + 1 / sp.sin(zeta)
    checks["csc_adjoint_exact"] = sp.trigsimp(csc_adjoint - 2 / sp.sin(zeta)**3) == 0

    hs, mm = sp.symbols("h_sigma mu_symbol", real=True)
    crest_quadratic = 2 * mm * (-(1 - hs) * mm) - 2 * hs * mm**2
    checks["smoothing_cancels_from_crest_quadratic"] = sp.expand(crest_quadratic + 2 * mm**2) == 0

    tc0, rr0 = sp.symbols("t0 r0", positive=True)
    neighboring_crests = -2 * a / rr0**2 * ((tc0 + sp.pi / (2*a))**2 - (tc0 - sp.pi / (2*a))**2)
    checks["neighboring_crest_difference_exact"] = sp.simplify(neighboring_crests + 4*sp.pi*tc0/rr0**2) == 0

    bb, ss = 0.1, 0.025
    excess_bound = 4 * (bb + ss) * math.exp(ss) * bb * math.tan(bb) / math.cos(bb + ss)**3
    checks["fixed_finite_kernel_B_bound"] = excess_bound < 0.006 and excess_bound < math.pi
    eta, rj = sp.symbols("eta rj", positive=True)
    checks["required_scale_identity"] = sp.simplify(
        (sp.sqrt(rj/a) / rj**2).subs(rj, 1/(a*eta**2)) - a*eta**3) == 0

    return {
        "scope": "Exact symbolic operator/finite Dirichlet recursion checks; analytic folded-return gate remains separate.",
        "sympy_version": sp.__version__,
        "checks": checks,
        "all_checks_pass": all(checks.values()),
        "B_excess_bound_at_max_parameters": excess_bound,
        "stages": stage_records,
        "inverse_count": len(inverse_records),
        "inverse_records": inverse_records,
    }


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    result = run()
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({k: result[k] for k in ("scope", "all_checks_pass", "checks", "stages", "inverse_count", "B_excess_bound_at_max_parameters")}, indent=2))
    raise SystemExit(0 if result["all_checks_pass"] else 1)
