def fnum(x):return float(F(x))
def physical_port(model):
    P=np.zeros((len(model['points']),len(model['boundary_nodes'])))
    P[model['boundary_nodes']]=model['boundary_mass']
    return P[1:]

def operator_rows(plan):
    E=fnum(plan['total_raw_energy']);rows=[]
    for g,block,norms in zip(plan['raw_currents'],plan['raw_tests'],plan['raw_test_norms']):
        g=np.array([fnum(x) for x in g])/np.sqrt(E)
        for q,N in zip(block,norms):
            q=np.array([fnum(x) for x in q])/np.sqrt(fnum(N));rows.append(np.outer(q,g).reshape(-1))
    return np.array(rows)

class Oracle:
    def __init__(self,model,port=None):
        self.model=model;self.P=physical_port(model) if port is None else port
        self.cache={};self.factorizations=0;self.RHS=0;self.requests=0
    def get(self,theta):
        self.requests+=1;key=tuple(np.asarray(theta,dtype=float))
        if key not in self.cache:
            A=self.model['A0']+np.einsum('j,jab->ab',theta,self.model['A'])
            self.cache[key]=self.P.T@np.linalg.solve(A,self.P)
            self.factorizations+=1;self.RHS+=self.P.shape[1]
        return self.cache[key]

def fit(oracle,R,grid,pred,obs):
    residual=np.linalg.norm(pred-obs,axis=1);k=int(np.argmin(residual));best=grid[k].copy();score=residual[k]
    spacing=.1/40;evals=0
    for level in range(5):
        lo=np.maximum(-.05,best-spacing);hi=np.minimum(.05,best+spacing)
        xs=np.linspace(lo[0],hi[0],7);ys=np.linspace(lo[1],hi[1],7)
        candidates=np.vstack([np.array(list(itertools.product(xs,ys))),best])
        values=np.array([R@oracle.get(theta).reshape(-1) for theta in candidates]);evals+=len(candidates)
        residual=np.linalg.norm(values-obs,axis=1);k=int(np.argmin(residual))
        if residual[k]<=score:best=candidates[k].copy();score=float(residual[k])
        spacing/=3
    return best,float(score),evals

def summary(records):
    out={}
    for label in ['random_single','optimized_single','optimized_multi']:
        for sigma in [.0003,.001,.003]:
            data=[z for z in records if z['method']==label and z['noise_SD']==sigma]
            errors=np.array([z['parameter_error'] for z in data]);bounds=[z['global_grid_error_bound_diagnostic'] for z in data if z['global_grid_error_bound_diagnostic'] is not None]
            out[label+'_'+str(sigma)]={'trials':len(data),'RMSE':float(np.sqrt(np.mean(errors**2))),'median_error':float(np.median(errors)),'p95_error':float(np.quantile(errors,.95)),'maximum_error':float(max(errors)),'positive_certificate_trials':len(bounds),'max_error_over_bound':max([z['parameter_error']/z['global_grid_error_bound_diagnostic'] for z in data if z['global_grid_error_bound_diagnostic'] is not None],default=None)}
    return out
