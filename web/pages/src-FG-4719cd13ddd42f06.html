#!/usr/bin/env python3
"""Exact, no-pickle checker for the new scalar/tensor boundary separation.

Dependencies: Python3.10+ and SymPy. The JSON stores only rational polynomial
coefficient arrays, not expressions to evaluate. No NumPy/solver is imported.
The analytic implication and scope of the obstruction are proved separately.
"""
from pathlib import Path
import hashlib
import itertools
import json
import sympy as s
from general_c4 import build

ROOT=Path(__file__).parent
DATA=ROOT/'TENSOR_CERTIFICATE.json'

def main():
    data=json.loads(DATA.read_text());g=build();A,C,E,L=g['parameters'];x=g['coordinates'];h=g['jets'];q=g['q']
    m=L**4-48*L**2+384
    assert s.Poly(m,L).is_irreducible
    lo,hi=map(s.Rational,data['root_interval'])
    assert s.Poly(m,L).count_roots(lo,hi)==1
    assert lo*lo>10 and hi*hi<s.Rational(51,5)
    def red(f):return s.rem(s.Poly(s.expand(f),L,domain=s.QQ),s.Poly(m,L)).as_expr()
    def dec(row):
        assert len(row)==4
        return sum(s.Rational(c)*L**i for i,c in enumerate(row))
    def matrix(rows):return s.Matrix([[dec(v) for v in row] for row in rows])
    def interval(poly,a=lo,b=hi):
        lower=upper=s.Rational(0)
        for cc in s.Poly(poly,L).all_coeffs():
            pp=[lower*a,lower*b,upper*a,upper*b]
            lower=min(pp)+cc;upper=max(pp)+cc
        return lower,upper
    V=matrix(data['scalar_frame']);T=matrix(data['T'])
    uu=s.Matrix(list(map(dec,data['u'])));vv=s.Matrix(list(map(dec,data['v'])))
    alpha,beta=map(dec,data['tensor_weights'])
    assert alpha==0 and beta==s.Rational(1,4)
    assert V.shape==(6,4) and T.shape==(4,4) and T==T.T
    Tmins=[red(T[:i,:i].det()) for i in range(1,5)]
    Tbounds=[interval(f) for f in Tmins]
    assert all(a>0 for a,b in Tbounds)
    F=sum(pp*ff for pp,ff in zip([1,4,0,L],g['basis']))
    HF=s.hessian(F,x);B1=g['B1'];B2=g['B2']
    diss=(B1.T*HF*B1+B2.T*HF*B2)[0]
    omega={ij:sum(dec(row[n])*x[i]*x[j] for n,(i,j) in enumerate(itertools.combinations_with_replacement(range(4),2)))
           for ij,row in zip(itertools.combinations(range(4),2),data['omega'])}
    def om(i,j):return omega[(i,j)] if i<j else -omega[(j,i)]
    for i,j,k in itertools.combinations(range(4),3):
        assert s.expand(s.diff(om(j,k),x[i])-s.diff(om(i,k),x[j])+s.diff(om(i,j),x[k]))==0
    corr=sum(omega[(i,j)]*(B1[i]*B2[j]-B1[j]*B2[i]) for i,j in omega)
    a,b,c,d=x;cx=a+s.I*c;cy=b+s.I*d
    u=a*a+c*c;v0=b*b+d*d
    theta=s.Matrix([-c,-d,a,b]);zeta=s.Matrix([-b,a,-d,c]);theta_a=s.Matrix([-c,d,a,-b])
    co=[(651*L**3-564*L**2-32928*L+17280)/16384,
        -(61*L**3-848*L**2-1968*L+35936)/4096,
        -(27*L**3-2976*L**2-528*L+22944)/8192,
        (61*L**3-19*L**2-430*L+400)/512,
        -(221*L**3-188*L**2-3008*L+4992)/16384]
    primitive=co[0]*(u+v0)*theta+co[1]*q*theta+co[2]*(u+v0)*zeta+co[3]*q*zeta+co[4]*(u-v0)*theta_a
    Da=primitive.jacobian(x)
    for ii,jj in omega:
        assert s.expand(Da[jj,ii]-Da[ii,jj]-omega[(ii,jj)])==0
    X=h[0]+s.I*h[3];Y=h[1]+s.I*h[4];Z=h[2]+s.I*h[5]
    bminus=s.Matrix([cx*X-s.I*cy*Z,cx*Z-s.I*cy*X,cy*Y-s.I*cx*Y])
    tensor=s.expand((vv.T*bminus)[0]);tensor=s.re(tensor).expand()**2+s.im(tensor).expand()**2
    zp=V.T*g['zplus'];zm=V.T*g['zminus']
    identity=s.expand(diss+corr-(zp.T*T*zp+zm.T*T*zm)[0]-beta*tensor)
    assert all(red(cc)==0 for cc in s.Poly(identity,*(tuple(x)+h)).coeffs())

    # Global rank-one threshold: leading3block PD and determinant square.
    k,j=s.symbols('k j',real=True);v=L*L
    M=s.Matrix([[12+8*k*k,16*k,-2*L*k,L*(3+k*k)],
                [16*k,8+12*k*k,-L*(1+3*k*k),2*L*k],
                [-2*L*k,-L*(1+3*k*k),4+8*k*k,0],
                [L*(3+k*k),2*L*k,0,8+4*k*k]])
    R=s.Matrix([1,k,0,0]);assert s.expand((R.T*HF*R)[0]-(x.T*M*x)[0])==0
    assert s.expand(M[:2,:2].det()-(96*(k*k-s.Rational(1,4))**2+90))==0
    D3=s.expand(M[:3,:3].det()).subs(k**6,j**3).subs(k**4,j**2).subs(k**2,j)
    lower=s.Rational(12,5)*((j-2)**2*(14*j+5)+89)
    assert s.expand(D3-lower-12*(s.Rational(51,5)-v)*(6*j**3+j*j+4*j+1))==0
    c1=-48*(v*4-14*v-8*4*2+24*4+16*4+16*2)
    c2=768*(v-5)
    square=(2*c2*k*k+c1*(1+k*k)**2)**2
    assert all(red(cc)==0 for cc in s.Poly(s.expand(4*c2*M.det()-square),k).coeffs())
    assert interval(c2)[0]>0

    # The two acoustic null vectors and minor-removal rigidity.
    jp=j*j+(s.Rational(3,8)*v-8)*j+1
    gb=s.groebner([m,jp],j,L,order='lex',domain=s.QQ)
    red2=lambda f:gb.reduce(s.expand(f))[1]
    AA=s.Matrix([[12+8*j,16*j,-2*j],[16,8+12*j,-1-3*j],[-2*v,-v*(1+3*j),4+8*j]])
    num=AA.adjugate()*s.Matrix([-3-j,-2,0]);ra,rb,rc=num.applyfunc(red2);D=red2(AA.det())
    assert s.expand(num[2]+24*v*(j-1)**2)==0
    vec=s.Matrix([L*ra,L*rb,rc,D])
    # Fourth acoustic equation after dividing its harmless k factor.
    last=L*(3+j)*(L*ra)+2*L*j*(L*rb)+(8+4*j)*D
    assert red2(last)==0
    vp=s.Matrix([L*ra,L*j*ra,L*j*rb,-j*rc,-D,-j*D]).applyfunc(red2)
    vm=s.Matrix([-L*j*rb,-L*rb,-L*ra,-D,-j*rc,-rc]).applyfunc(red2)

    # Complete common6 scalar family against arbitrary closed quadratic forms.
    xpairs=list(itertools.combinations_with_replacement(range(4),2))
    hpairs=list(itertools.combinations_with_replacement(range(6),2))
    fpairs=list(itertools.combinations(range(4),2))
    mon=[x[i]*x[j]*h[k]*h[l] for i,j in xpairs for k,l in hpairs]
    lookup={s.Poly(mm,*(tuple(x)+h)).monoms()[0]:i for i,mm in enumerate(mon)}
    def coeff(poly):
        ans=s.zeros(210,1)
        for exp,c in s.Poly(s.expand(poly),*(tuple(x)+h)).terms():ans[lookup[exp]]=c
        return ans
    ww=s.symbols('w0:60');xmon=[x[i]*x[j] for i,j in xpairs]
    forms={ij:sum(ww[10*n+i]*mm for i,mm in enumerate(xmon)) for n,ij in enumerate(fpairs)}
    def omw(i,j):return forms[(i,j)] if i<j else -forms[(j,i)]
    eq=[]
    for ii,jj,kk in itertools.combinations(range(4),3):
        f=s.diff(omw(jj,kk),x[ii])-s.diff(omw(ii,kk),x[jj])+s.diff(omw(ii,jj),x[kk])
        eq.extend(s.expand(f).coeff(xx) for xx in x)
    Cl,_=s.linear_eq_to_matrix(eq,ww);assert Cl.rank()==15
    NN=s.Matrix.hstack(*Cl.nullspace());assert NN.cols==45
    OO=s.Matrix.hstack(*[coeff(mm*(B1[i]*B2[j]-B1[j]*B2[i])) for i,j in fpairs for mm in xmon])
    DD=s.Matrix.hstack(*(OO*NN).T.nullspace()).T
    pairs6=list(itertools.combinations_with_replacement(range(6),2))
    scalar=s.Matrix.hstack(*[coeff((g['zplus'][i]*g['zplus'][j]+g['zminus'][i]*g['zminus'][j])*(1 if i==j else 2)) for i,j in pairs6])
    assert (DD*scalar).rank()==15
    Sg=g['matrix'].subs({A:1,C:4,E:0})
    gauges=s.Matrix([[Sg[i,j].diff(t) for t in g['free']] for i,j in pairs6]);assert gauges.rank()==6
    # All full closed scalar solutions therefore equal this six-gauge family.
    tvals=[-2,3*L**3/64-9*L/4,L*L/4-10,3*L*L/8-3,L**3/8-L,-L**3/64+L/4]
    SS=Sg.subs(dict(zip(g['free'],tvals))).applyfunc(red)
    assert (SS*vp).applyfunc(red2)==s.zeros(6,1)
    assert (SS*vm).applyfunc(red2)==s.zeros(6,1)
    assert red(SS[2:5,2:5].det())==74*v-768
    assert 74*s.Rational(51,5)-768<0

    # A nonzero real6minor proves uniqueness over all real gauge choices,
    # without assuming those choices lie in any prescribed number field.
    un=json.loads((ROOT/'SCALAR_UNIQUENESS.json').read_text())
    ll,lh=map(s.Rational,un['L_interval']);jl,jh=map(s.Rational,un['j_interval'])
    assert s.Poly(m,L).count_roots(ll,lh)==1 and lo<ll<lh<hi
    def box(poly):
        lower=upper=s.Rational(0)
        for cc in s.Poly(poly,L).all_coeffs():
            cl=cu=s.Rational(0)
            for c in s.Poly(cc,j).all_coeffs():
                pp=[cl*jl,cl*jh,cu*jl,cu*jh];cl=min(pp)+c;cu=max(pp)+c
            pp=[lower*ll,lower*lh,upper*ll,upper*lh];lower=min(pp)+cl;upper=max(pp)+cu
        return lower,upper
    assert box(jp.subs(j,jl))[0]>0 and box(jp.subs(j,jh))[1]<0
    MM=s.Matrix.hstack(*[(Sg.diff(t)*vp).col_join(Sg.diff(t)*vm) for t in g['free']]).applyfunc(red2)
    determinant=red2(MM[un['rows'],:].det(method='domain-ge'));bd=box(determinant)
    assert bd[0]>0 or bd[1]<0

    # Source basis actions giving full scalar12 averaging reduction.
    # Input quarter-turn: x'=y,y'=-x,X'=Z,Y'=-Y,Z'=X.
    rot={x[0]:x[1],x[1]:-x[0],x[2]:x[3],x[3]:-x[2],
         h[0]:h[2],h[1]:-h[1],h[2]:h[0],h[3]:h[5],h[4]:-h[4],h[5]:h[3]}
    refl={x[0]:x[0],x[1]:-x[1],x[2]:-x[2],x[3]:x[3],
          h[0]:h[0],h[1]:-h[1],h[2]:h[2],h[3]:-h[3],h[4]:h[4],h[5]:-h[5]}
    assert (g['zplus'].subs(rot,simultaneous=True)+g['zminus']).applyfunc(s.expand)==s.zeros(6,1)
    assert (g['zminus'].subs(rot,simultaneous=True)-g['zplus']).applyfunc(s.expand)==s.zeros(6,1)
    assert (g['zplus'].subs(refl,simultaneous=True)-g['zplus']).applyfunc(s.expand)==s.zeros(6,1)
    assert (g['zminus'].subs(refl,simultaneous=True)+g['zminus']).applyfunc(s.expand)==s.zeros(6,1)
    assert s.expand(F.subs(rot,simultaneous=True)-F)==s.expand(F.subs(refl,simultaneous=True)-F)==0
    bplus=s.Matrix([cx*X+s.I*cy*Z,cx*Z+s.I*cy*X,cy*Y+s.I*cx*Y])
    assert (bplus.subs(rot,simultaneous=True)+s.I*bplus).applyfunc(s.expand)==s.zeros(3,1)
    assert (bminus.subs(rot,simultaneous=True)-s.I*bminus).applyfunc(s.expand)==s.zeros(3,1)
    assert (bplus.subs(refl,simultaneous=True)-s.conjugate(bplus)).applyfunc(s.expand)==s.zeros(3,1)
    assert (bminus.subs(refl,simultaneous=True)-s.conjugate(bminus)).applyfunc(s.expand)==s.zeros(3,1)
    angle={x[0]:(x[0]-x[2])/s.sqrt(2),x[1]:(x[1]-x[3])/s.sqrt(2),
           x[2]:(x[0]+x[2])/s.sqrt(2),x[3]:(x[1]+x[3])/s.sqrt(2)}
    for ii in range(3):
        angle[h[ii]]=(h[ii]-h[ii+3])/s.sqrt(2)
        angle[h[ii+3]]=(h[ii]+h[ii+3])/s.sqrt(2)
    assert (g['zplus'].subs(angle,simultaneous=True)-g['zplus']).applyfunc(s.expand)==s.zeros(6,1)
    assert (g['zminus'].subs(angle,simultaneous=True)-g['zminus']).applyfunc(s.expand)==s.zeros(6,1)
    assert (bplus.subs(angle,simultaneous=True)-s.I*bplus).applyfunc(s.expand)==s.zeros(3,1)
    assert (bminus.subs(angle,simultaneous=True)-s.I*bminus).applyfunc(s.expand)==s.zeros(3,1)
    features=list(g['zplus'])+list(g['zminus'])
    for block in [bplus,bminus]:
        features+=list(block.applyfunc(lambda v:s.re(v).expand()))
        features+=list(block.applyfunc(lambda v:s.im(v).expand()))
    bilinear=[xx*hh for xx in x for hh in h]
    fmap=s.Matrix([[s.Poly(f,*(tuple(x)+h)).coeff_monomial(mm) for mm in bilinear] for f in features])
    assert fmap.shape==(24,24) and fmap.rank()==24
    result={'status':'PASS','tensor_certificate_sha256':hashlib.sha256(DATA.read_bytes()).hexdigest(),
            'root_polynomial':str(m),'root_interval':data['root_interval'],
            'universal_tensor_heat_identity':True,'closed_two_form':True,
            'compact_five_term_primitive':True,
            'positive_T_principal_minors':4,'tensor_weight':'1/4',
            'global_acoustic_PSD':True,'critical_null_identity':True,
            'full_closed_quadratic_form_dimension':45,'common6_scalar_image_rank':15,
            'real_gauge_unique':True,'scalar_forced_negative_principal_minor':'74L²-768<0',
            'scalar12_averaging_actions':True,
            'complete_24_feature_basis_rank':24,'scalar_tensor_group_actions':True,
            'minor_removal_rigidity':'gamma=0 because D>0,chi<0 and Dq=(D-jchi)/D>0',
            'scope':'New lambda4 sharp slice and scalar/tensor representation separation; not all C4/full9, novelty/external validation UNKNOWN.'}
    (ROOT/'TENSOR_VERIFICATION.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result,indent=2))
