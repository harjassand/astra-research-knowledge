def floor_grid(x): return F(x.numerator * GRID // x.denominator, GRID)
def ceil_grid(x): return -floor_grid(-x)

class I:
    def __init__(self, lo, hi=None, rounded=True):
        lo, hi = F(lo), F(lo if hi is None else hi)
        if lo > hi: raise ValueError("reversed interval")
        self.lo = floor_grid(lo) if rounded else lo
        self.hi = ceil_grid(hi) if rounded else hi
    @staticmethod
    def of(x): return x if isinstance(x, I) else I(x)
    def __add__(self, other):

        y=I.of(other); return I(self.lo+y.lo, self.hi+y.hi)
    __radd__=__add__
    def __neg__(self): return I(-self.hi,-self.lo)
    def __sub__(self,other): return self+-I.of(other)
    def __rsub__(self,other): return I.of(other)+-self
    def __mul__(self,other):
        COUNTS["interval_arithmetic_calls"] += 1
        y=I.of(other); vals=[a*b for a in (self.lo,self.hi) for b in (y.lo,y.hi)]
        return I(min(vals),max(vals))
    __rmul__=__mul__
    def __truediv__(self,other):
        COUNTS["interval_arithmetic_calls"] += 1
        y=I.of(other)
        if y.lo <= 0 <= y.hi: raise ValueError("division through zero")
        return self*I(1/y.hi,1/y.lo)
    def __rtruediv__(self,other): return I.of(other)/self
    def json(self):

@lru_cache(maxsize=20000)
def exp_scalar(x):

@lru_cache(maxsize=20000)
def log_base_1_2(x):
    # Exact identity log x = 2 sum_{j>=0} z^(2j+1)/(2j+1), z=(x-1)/(x+1).
    x=F(x)
    if not 1 <= x <= 2: raise ValueError("log normalization")

        if tail <= F(1,1<<(BITS+8)): break

    return I((2*acc).lo,(2*acc).hi+tail)

@lru_cache(maxsize=20000)
def log_scalar(x):
    x=F(x); COUNTS["log_scalar_queries"] += 1
    if x <= 0: raise ValueError("nonpositive log")

    while x >= 2: x/=2; k+=1
    return log_base_1_2(x)+k*log_base_1_2(F(2))

def mean_log(log_values, weights, exponent):
    chk(len(log_values)==len(weights) and sum(weights,F(0))==1)
    chk(all(w>=0 for w in weights) and 0 < exponent < 1)
    return log_i(sum((w*exp_i(exponent*v) for w,v in zip(weights,log_values)),I(0)))/exponent

def poisson_tail_majorants(lam,J):
    # e^-lam <= 1; ratios yield exact rational, conservative upper bounds.
    chk(lam>=0 and J+1>lam)
    probability=lam**(J+1)/factorial(J+1)/(1-lam/F(J+2))
    first_moment=lam*lam**J/factorial(J)/(1-lam/F(J+1))
    return probability,first_moment

def triangle_two_spin_pressure(alpha,J,coupling):
    """Exact finite-volume N=2 aggregation for symmetric p=2 coupling.

    Self-loops contribute a mean-zero additive constant. Cross-edge increment
    D is 0 with probability 1/2 and +/-1 with probability 1/4 each. Conditional
    partition is 4*cosh(coupling*D), up to that additive constant.
    """
    lam=2*alpha; law={0:F(1)}; pois=exp_scalar(-lam); total=I(0)
    cardinalities=[]
    for k in range(J+1):
        chk(sum(law.values(),F(0))==1)
        cond=sum((prob*logcosh(coupling*d)/2 for d,prob in law.items()),I(0))
        total+=pois*cond; cardinalities.append(len(law))
        nextlaw={}
        for d,p in law.items():
            for step,w in ((0,F(1,2)),(-1,F(1,4)),(1,F(1,4))):
                nextlaw[d+step]=nextlaw.get(d+step,F(0))+p*w
        law=nextlaw;pois=pois*lam/F(k+1)
    _,moment=poisson_tail_majorants(lam,J)
    # logcosh(coupling*D)/2 <= |coupling|*k/2.
    return log_scalar(F(2))+total+I(0,abs(coupling)*moment/2),cardinalities

def one_level_trial(alpha,J,coupling,mag,exponent):
    """r=1, root law delta_(fair +/-mag), h=0. Finite binomial sums.

    Physical symmetric coupling signs permute the fair message signs, so every
    fixed physical realization has the same conditional power mean. Labels
    remain independent throughout; no physical disorder is annealed inside it.
    """
    ec=exp_scalar(coupling); em=exp_scalar(-coupling)
    c=(ec+em)/2; sh=(ec-em)/2
    up=c+mag*sh; down=c-mag*sh
    lam=2*alpha;pois=exp_scalar(-lam);total=I(0)
    for k in range(J+1):
        values=[];weights=[]
        for j in range(k+1):
            vp=I(1);vm=I(1)
            for _ in range(j): vp*=up;vm*=down
            for _ in range(k-j):vp*=down;vm*=up
            values.append(log_i((vp+vm)/2));weights.append(F(comb(k,j),2**k))
        total+=pois*mean_log(values,weights,exponent)
        pois=pois*lam/F(k+1)
    _,moment=poisson_tail_majorants(lam,J)
    site=total+I(-abs(coupling)*moment,abs(coupling)*moment)
    edge=mean_log([log_i(c+mag*mag*sh),log_i(c-mag*mag*sh)],[F(1,2)]*2,exponent)
    return log_scalar(F(2))+site-alpha*edge

def independent_field_pressure(t): return log_scalar(F(2))+logcosh(t)

def derivative_certificate(t,epsilon):
    records=[]
    for k in range(1,24):
        h=F(1,2**k)
        left=(independent_field_pressure(t)-independent_field_pressure(t-h))/h

        interval=I(left.lo,right.hi)
        records.append({"k":k,"h":str(h),"slope":interval.json()})
        if interval.hi-interval.lo < epsilon:return interval,records
    raise RuntimeError("predeclared derivative cap exceeded")

def magnetization_chernoff(m,step):
    # Public C=0, rational safe compactification R=(2C+1)/(1-|m|)=2.
    R=F(1)/(1-abs(m));chk((R/step).denominator==1)
    values=[]

    best_lo=max(v.lo for _,v in values);best_hi=max(v.hi for _,v in values)
    # Nearest grid point <= step/2 away; objective Lipschitz <= 1+|m|.
    out=I(best_lo,best_hi+(1+abs(m))*step/2)
    return out,{"R":str(R),"step":str(step),"queries":len(values),"best_grid_t":str(max(values,key=lambda tv:tv[1].lo)[0])}
