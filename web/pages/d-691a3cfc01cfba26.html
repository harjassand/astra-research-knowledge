#!/usr/bin/env python3
"""Exact finite rational checks of the N77 stationary formula on a rate box.

The proof for every n and every positive rate vector is the symbolic phase
flux and Poisson detailed-balance calculation in REPORT.txt. This program
checks the formulas at all 32 corners plus the center of [1,2]^5 and n=0..30.
"""
from fractions import Fraction as F
from itertools import product

rate_vectors = [tuple(map(F, bits)) for bits in product((1, 2), repeat=5)]
rate_vectors.append((F(3, 2),) * 5)
checks = 0

for rates in rate_vectors:
    k1, k2, k3, k4, k5 = rates
    mu = k4 / k5
    for n in range(31):
        q = (2 * k1,
             (n + 1) * k2,
             2 * k3 * (n + 1) * (n + 2) * (n + 3))
        w = (1 / q[0], 1 / q[1], 1 / q[2])
        Z = sum(w)
        phase = tuple(z / Z for z in w)
        assert sum(phase) == 1
        assert all(phase[i] * q[i] == phase[(i + 1) % 3] * q[(i + 1) % 3]
                   for i in range(3))
        # Verify both pathwise invariants on every phase, with D arbitrary.
        states = ((n, 0, 2), (n + 1, 1, 1), (n + 3, 2, 0))
        assert all(b + c == 2 for _, b, c in states)
        assert all(a - b * (b + 1) // 2 == n for a, b, _ in states)
        for d_count in range(21):
            # Unnormalized Poisson weights avoid floating-point exp(-mu).
            rho_d = mu ** d_count
            for m in range(1, d_count + 1):
                rho_d /= m
            rho_next = rho_d * mu / (d_count + 1)
            birth_flux = rho_d * (2 * k4)
            death_flux = rho_next * (2 * k5 * (d_count + 1))
            assert birth_flux == death_flux
            checks += 1

print(f"rate_vectors={len(rate_vectors)}; n_values=31; exact_fraction=YES")
print(f"phase_flux_checks={len(rate_vectors)*31*3}; poisson_balance_checks={checks}; PASS")
print("symbolic scope: all n>=0 and all positive rates by explicit formulas; finite checks are corroboration only")
