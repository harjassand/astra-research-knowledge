from fractions import Fraction as F
import math,json,time
from pathlib import Path
import numpy as np
from freeze_designs import mass_fraction,bilinear,fracstr

ROOT=Path(__file__).resolve().parents[3];OUT=ROOT/'results/rank_pde/cycle2'
def dot(x,y):return sum((a*b for a,b in zip(x,y)),F(0))
def mv(A,x):return [dot(row,x) for row in A]

def inverse(A):
    n=len(A);R=[row.copy()+[F(i==j) for j in range(n)] for i,row in enumerate(A)]
    for j in range(n):
        p=next(i for i in range(j,n) if R[i][j]);R[j],R[p]=R[p],R[j]
        z=R[j][j];R[j]=[v/z for v in R[j]]
        for i in range(n):
            if i!=j and R[i][j]:
                z=R[i][j];R[i]=[v-z*w for v,w in zip(R[i],R[j])]
    return [row[n:] for row in R]

def pd(A):
    n=len(A);L=[[F(0)]*n for _ in range(n)];D=[]
    for i in range(n):
        for j in range(i):L[i][j]=(A[i][j]-sum((L[i][k]*D[k]*L[j][k] for k in range(j)),F(0)))/D[j]
        d=A[i][i]-sum((L[i][k]**2*D[k] for k in range(i)),F(0))
        if d<=0:return False
        D.append(d);L[i][i]=F(1)
    return True

def exact_assembly():
    n=4;points=[(F(i,n),F(j,n)) for j in range(n+1) for i in range(n+1)]
    mats=[[[F(0)]*25 for _ in range(25)] for _ in range(3)]
    for j in range(n):
        for i in range(n):
            ids=[j*5+i,j*5+i+1,(j+1)*5+i+1,(j+1)*5+i]
            lab=1 if i<2 and j>=2 else (2 if i>=2 and j==0 else 0)
            for tri in [(ids[0],ids[1],ids[2]),(ids[0],ids[2],ids[3])]:
                p=[points[k] for k in tri]
                det=(p[1][0]-p[0][0])*(p[2][1]-p[0][1])-(p[2][0]-p[0][0])*(p[1][1]-p[0][1])
                area=abs(det)/2
                grad=[((p[(k+1)%3][1]-p[(k+2)%3][1])/det,(p[(k+2)%3][0]-p[(k+1)%3][0])/det) for k in range(3)]
                for a,b in [(a,b) for a in range(3) for b in range(3)]:
                    z=area*dot(grad[a],grad[b]);mats[0][tri[a]][tri[b]]+=z
                    if lab:mats[lab][tri[a]][tri[b]]+=z
    K=[row[1:] for row in mats[0][1:]];regions=[[row[1:] for row in mat[1:]] for mat in mats[1:]]
    assert pd(K)
    # PSD regional and complementary stiffnesses follow directly from sums
    # of area*(gradient dot gradient), requiring no numerical eigenvalue test.
    return K,regions

class Verifier:
    def __init__(self):
        start=time.perf_counter();self.K,self.regions=exact_assembly();self.inv=inverse(self.K);self.M=mass_fraction()
        self.nodes=list(range(5))+[9,14,19,24]+[23,22,21,20]+[15,10,5]
        self.setup_seconds=time.perf_counter()-start
    def load(self,v):
        values=mv(self.M,v);full=[F(0)]*25
        for i,z in zip(self.nodes,values):full[i]=z
        assert sum(full)==0
        return full[1:]
    def certify(self,plan):
        start=time.perf_counter();gs=[[F(v) for v in row] for row in plan['raw_currents']]
        qs=[[[F(v) for v in row] for row in block] for block in plan['raw_tests']]
        assert all(sum(g)==0 for g in gs) and all(sum(q)==0 for block in qs for q in block)
        E=sum(bilinear(self.M,g,g) for g in gs);assert E==F(plan['total_raw_energy'])>0
        norms=[[bilinear(self.M,q,q) for q in block] for block in qs]
        assert norms==[[F(v) for v in block] for block in plan['raw_test_norms']]
        G=[[F(0)]*2 for _ in range(2)];remainder_sum=F(0);norm_uppers=[];checks=0
        for g,block,Ns in zip(gs,qs,norms):
            b=self.load(g);u=mv(self.inv,b);bqs=[self.load(q) for q in block];uqs=[mv(self.inv,bq) for bq in bqs]
            for uq,N in zip(uqs,Ns):
                row=[-dot(uq,mv(A,u)) for A in self.regions]
                for i in range(2):
                    for j in range(2):G[i][j]+=row[i]*row[j]/(E*N)
            H=[[dot(bq,uq) for uq in uqs] for bq in bqs]
            eigguess=np.linalg.eigvalsh(np.array(H,dtype=float)/np.sqrt(np.outer(np.array(Ns,dtype=float),np.array(Ns,dtype=float))))[-1]
            upper=F(math.ceil(float(eigguess)*10**12)+2,10**12)
            while not pd([[upper*Ns[i]*F(i==j)-H[i][j] for j in range(len(block))] for i in range(len(block))]):upper+=F(1,10**12);checks+=1
            norm_uppers.append(upper);remainder_sum+=upper*dot(b,u)/E
        guess=math.sqrt(max(0,np.linalg.eigvalsh(np.array(G,dtype=float))[0]));alpha=F(max(0,math.floor(guess*10**12)-2),10**12)
        while alpha>0:
            a,b=G[0][0]-alpha**2,G[1][1]-alpha**2
            if a>=0 and b>=0 and a*b>=G[0][1]**2:break
            alpha-=F(1,10**12);checks+=1
        factor=F(1,20)*(2-F(1,20))/(1-F(1,20))**2;remainder_sq=factor**2*remainder_sum
        beta=F(math.ceil(math.sqrt(float(remainder_sq))*10**12)+2,10**12)
        while beta**2<remainder_sq:beta+=F(1,10**12);checks+=1
        kappa=alpha-beta
        # Upper parameter Lipschitz constant, useful for the global grid slack.
        lipschitz=F(math.ceil(math.sqrt(float(G[0][0]+G[1][1]))*10**12)+2,10**12)+beta
        assert (lipschitz-beta)**2>=G[0][0]+G[1][1]
        vals=[v for row in G for v in row]+[E,remainder_sq]+[v for block in norms for v in block]
        return {'label':plan['label'],'status':'POSITIVE' if kappa>0 else 'UNKNOWN','reference_sigma_lower':fracstr(alpha),'remainder_upper':fracstr(beta),'global_secant_lower':fracstr(kappa),'parameter_Lipschitz_upper':fracstr(lipschitz),'normalized_Gram':[[fracstr(v) for v in row] for row in G],'output_norm_upper':[fracstr(v) for v in norm_uppers],'raw_total_energy':fracstr(E),'max_recorded_rational_bitlength':max(max(abs(v.numerator).bit_length(),v.denominator.bit_length()) for v in vals),'order_bound_retry_count':checks,'seconds':time.perf_counter()-start,'scope':'All secants in theta box[-1/20,1/20]^2 for exact normalized raw physical traces; finite Galerkin map, not interval-certified floating observations'}

def main():
    plans=json.loads((OUT/'frozen_designs.json').read_text());V=Verifier()
    ans={k:V.certify(plans[k]) for k in ['optimized_single','optimized_multi']}
    ans['random']=[V.certify(p) for p in plans['random']]
    ans['costs']={'setup_seconds':V.setup_seconds,'certificates':130,'certificate_seconds':sum(v['seconds'] for v in [ans['optimized_single'],ans['optimized_multi']]+ans['random']),'exact_reference_inverse_dimension':24}
    (OUT/'exact_certificates.json').write_text(json.dumps(ans,indent=2))
    print(json.dumps({k:ans[k] for k in ['optimized_single','optimized_multi','costs']},indent=2))
    print({'positive_random':sum(p['status']=='POSITIVE' for p in ans['random']),'minimum_random_kappa':min(float(F(p['global_secant_lower'])) for p in ans['random'])})
if __name__=='__main__':main()
