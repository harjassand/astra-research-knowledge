def check(name,expr):
    r=s.factor(s.expand(expr))
    assert r==0,(name,r)
    out[name]={'pass':True,'residual':'0'}
def matrix_check(name,M):
    r=[s.factor(s.expand(z)) for z in M]
    assert all(z==0 for z in r),(name,r)
    out[name]={'pass':True,'residuals':['0']*len(r)}

hxx,hxy,hyy,g1x,g1y,g2x,g2y=s.symbols('hxx hxy hyy G1x G1y G2x G2y')
H=s.Matrix([[hxx,hxy,g1x,g2x],[hxy,hyy,g1y,g2y],
            [g1x,g1y,0,0],[g2x,g2y,0,0]])
check('zero_section_block_determinant',H.det()-(g1x*g2y-g1y*g2x)**2)

x,y,rho,eta=s.symbols('x y rho eta')
a,b,c,d,ax,bx,cx,dx=s.symbols('a b c d ax bx cx dx')
delta=a*d-b*c
W=ax*b-a*bx
KK=cx*b-a*dx
JG=s.Matrix([[2*y*ax+4*c+4*x*cx,2*a],
             [2*y*bx+4*d+4*x*dx,2*b]])
check('one_base_variable_Jacobian',JG.det()-4*(y*W-2*delta+2*x*KK))
delta_derivative=ax*d+a*dx-bx*c-b*cx
check('unimodular_derivative_relation',KK-(ax*d-bx*c)+delta_derivative)

# Canonical constant-first-row form and global inverse, no parameter choice.
c0,d0,rr=s.symbols('c0 d0 r_at_x')
delta0=a*d0-b*c0
z=y+2*x*rr
G1=2*a*z+4*x*c0
G2=2*b*z+4*x*d0
xr=(-b*G1+a*G2)/(4*delta0)
zr=(d0*G1-c0*G2)/(2*delta0)
yr=zr-2*xr*rr
check('one_variable_inverse_x',xr-x)
check('one_variable_inverse_y',yr-y)
u,v,rr_target=s.symbols('u v r_at_recovered_x')
xt=(-b*u+a*v)/(4*delta0)
zt=(d0*u-c0*v)/(2*delta0)
yt=zt-2*xt*rr_target
check('one_variable_right_inverse_G1',2*a*(yt+2*xt*rr_target)+4*xt*c0-u)
check('one_variable_right_inverse_G2',2*b*(yt+2*xt*rr_target)+4*xt*d0-v)

# Contracted-zero jets can hold without vanishing individual quadratic jets.
R=s.Matrix([rho,eta])
linear=y*rho+2*x*eta
null=s.Matrix([2*x,-y])
Theta=R+null*linear**2
ThetaInv=R-null*linear**2
matrix_check('nontrivial_contracted_zero_inverse',ThetaInv.subs({rho:Theta[0],eta:Theta[1]},simultaneous=True)-R)
check('nontrivial_contracted_zero_det1',Theta.jacobian((rho,eta)).det()-1)
check('nontrivial_contracted_zero_potential',2*y*Theta[0]+4*x*Theta[1]-2*linear)
contract=2*y*s.hessian(Theta[0],(rho,eta))+4*x*s.hessian(Theta[1],(rho,eta))
matrix_check('nontrivial_contracted_zero_jet',contract)
assert s.hessian(Theta[0],(rho,eta))!=s.zeros(2)
assert s.hessian(Theta[1],(rho,eta))!=s.zeros(2)
