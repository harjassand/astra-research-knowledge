#!/usr/bin/env python3
"""Exact interval certificate for the final branching programme's fixed-iid converse.

No physical observations. Fractions compute its state-specific two-action
occupancies; rational atanh series enclose all logarithms. This compares FULL
logical execution records, not only the projected Boolean terminal report.
"""
from fractions import Fraction as F
from pathlib import Path
import json

G=((F(8,10),F(1,10),F(1,10)),(F(2,10),F(7,10),F(1,10)),(F(1,10),F(2,10),F(7,10)))
MENU=(G,tuple(G[(i+1)%3] for i in range(3)))

def log_interval(x, terms=48):
    if x == 1: return F(0),F(0)
    if x < 1:
        lo,hi=log_interval(1/x,terms)
        return -hi,-lo
    z=(x-1)/(x+1)
    lower=2*sum((z**(2*k+1)/F(2*k+1) for k in range(terms)),F(0))
    tail=2*z**(2*terms+1)/(F(2*terms+1)*(1-z*z))
    return lower,lower+tail

def action_counts(i,n):
    previous=(F(1),F(0),F(0));counts=[F(0),F(0)]
    for t in range(n):
        if t%2==0: freq=(previous[0]+previous[1],previous[2])
        else: freq=((previous[0]+previous[1])/2,previous[2]+(previous[0]+previous[1])/2)
        counts=[counts[a]+freq[a] for a in range(2)]
        previous=tuple(sum((freq[a]*MENU[a][i][y] for a in range(2)),F(0)) for y in range(3))
    assert sum(counts)==n
    return counts

def kl_interval(i,j,a):
    terms=[(MENU[a][i][y],log_interval(MENU[a][i][y]/MENU[a][j][y])) for y in range(3)]
    return tuple(sum((p*interval[k] for p,interval in terms),F(0)) for k in range(2))

def certify(n=1024):
    occupancy=[action_counts(i,n) for i in range(3)]
    rates={}
    for i in range(3):
        for j in range(3):
            if i==j: continue
            interval=tuple(sum((occupancy[i][a]*kl_interval(i,j,a)[k] for a in range(2)),F(0))/n for k in range(2))
            rates[(i,j)]=interval
    directions=((1,0),(0,2),(2,1))
    for i,j in directions: assert rates[(i,j)][0]>rates[(j,i)][1]
    l9,u9=log_interval(F(9));Dupper=F(4,5)*u9
    coefficient_lower=sum((rates[pair][0] for pair in directions),F(0))/(2*Dupper)
    assert coefficient_lower>1
    return {'horizon':n,'preserved_output':'FULL actions/observations/original-controller-coins record and Boolean projection',
        'excluded_output_claim':'No lower bound for Boolean report alone',
        'physical_samples_used':0,'log_terms':48,
        'action_occupancy_fractions':[[str(x/n) for x in row] for row in occupancy],
        'action_occupancy_decimal':[[float(x/n) for x in row] for row in occupancy],
        'directed_KL_per_step_intervals':{f'{i}>{j}':[str(x) for x in v] for (i,j),v in rates.items()},
        'pair_directions':directions,'fixed_iid_worst_state_cost_coefficient_lower':str(coefficient_lower),
        'coefficient_lower_decimal':float(coefficient_lower),'coefficient_exceeds_one_exactly':True,
        'lower_expected_source_count':float(coefficient_lower*n),
        'source_model':'Bernoulli .1 in own action, .9 in others; known laws, unit readings',
        'comparator':'Any fixed iid action mixture; arbitrary retention/stopping/report garbling; full-record law exact',
        'proof':'Stopped source KL D(q_i+q_j)E_iN; target controlled-record KL=sum_b E_iN_b KL(G_i^b||G_j^b); summing three strongest unordered constraints gives worststate >= n sum_pair g_pair/(2D). Rational lower targetlogs and upper sourcelog certify strict excess.'}

if __name__=='__main__':
