def binom(n,p):
    if p == 0: return np.r_[1., np.zeros(n)]
    if p == 1: return np.r_[np.zeros(n), 1.]
    return np.array([math.exp(math.lgamma(n+1)-math.lgamma(k+1)-math.lgamma(n-k+1)+k*math.log(p)+(n-k)*math.log1p(-p)) for k in range(n+1)])

def record_fixture(n,p,r):
    w=binom(n,p)
    wd=w*(np.arange(n+1)-n*p)/(p*(1-p))
    kernels=np.array([binom(r,k/n) for k in range(n+1)])
    Q=w@kernels; dQ=wd@kernels
    ids=Q>1e-300
    fi=float(np.sum(dQ[ids]**2/Q[ids]))
    s=np.arange(r+1)/r
    variance=float(np.sum(Q*(s-p)**2))
    target=p*(1-p)*(1/n+(1-1/n)/r)
    iid=binom(r,p)
    tv=float(np.abs(Q-iid).sum()/2)
    assert abs(Q.sum()-1)<2e-12
    assert abs(variance-target)<2e-12
    assert fi <= n/(p*(1-p))*(1+1e-12)
    return dict(n=n,p=p,r=r,variance=variance,variance_formula=target,record_fisher=fi,input_fisher=n/(p*(1-p)),tv_from_iid=tv,pinsker_bound=min(1.,math.sqrt(r/(2*n))))

def block_fixture(n,k,m,r):
    """Full diagonal operator calculation, not merely substitution into formula."""
    configs=np.array([tuple(int(j>>i)&1 for i in range(n)) for j in range(2**n) if j.bit_count()==k],float)
    size=len(configs); q=k/n
    # Original shell probabilities; measured first bit. Twirl all block entries.
    sectors={}
    for a,row in enumerate(configs):
        key=(int(row[:m].sum()),tuple(row[m:]))
        sectors.setdefault(key,[]).append(a)
    state=np.ones(size)/size; defects=[]; exterior_defects=[]
    for bit in (0,1):
        branch=state*(configs[:,0]==bit)
        recovered=np.zeros(size)
        for ids in sectors.values(): recovered[ids]=branch[ids].sum()/len(ids)
        ideal=(q if bit else 1-q)*state
        defects.append(float(np.abs(recovered-ideal).sum()))
        # Any block CPTP recovery keeps these exterior subnormalized marginals.
        outside={}
        for a,row in enumerate(configs):
            key=tuple(row[m:]);outside[key]=outside.get(key,0)+(branch[a]-ideal[a])
        exterior_defects.append(sum(abs(x) for x in outside.values()))
    weights=np.zeros(m+1)
    for s in range(m+1):
        if 0<=k-s<=n-m:
            weights[s]=math.comb(m,s)*math.comb(n-m,k-s)/math.comb(n,k)
    epsilon=2*float(np.dot(weights,np.abs(np.arange(m+1)/m-q)))
    variance=q*(1-q)*(n-m)/(m*(n-1))
    kernels=np.array([binom(r,s/m) for s in range(m+1)])
    Q=weights@kernels
    mse=float(Q@((np.arange(r+1)/r-q)**2))
    predicted=variance+(q*(1-q)-variance)/r
    assert abs(sum(defects)-epsilon)<2e-13
    assert abs(sum(exterior_defects)-epsilon)<2e-13
    assert abs(mse-predicted)<2e-13
    return dict(n=n,k=k,m=m,r=r,operator_defect=sum(defects),universal_exterior_lower_bound=sum(exterior_defects),hypergeometric_epsilon=epsilon,precision_floor=variance,record_mse=mse,record_mse_formula=predicted)

def shell(n,k):
    a=np.array([int(j).bit_count()==k for j in range(2**n)],float)
    return np.diag(a/a.sum())

def charge_decoder_fixture():
    n=6; ks=[1,4]; states={k:shell(n,k) for k in ks}
    weights=np.array([int(j).bit_count() for j in range(2**n)])
    def recover(x):
        d=np.real(np.diag(x)); out=np.zeros_like(x)
        for q in range(n+1):
            target=next((k for k in ks if abs(q-k)<=1),ks[0])
            out+=d[weights==q].sum()*states[target]
        return out
    rng=np.random.default_rng(20261006); errors=[]
    for _ in range(30):
        K=rng.normal(size=(2,2))+1j*rng.normal(size=(2,2))
        K/=np.linalg.svd(K,compute_uv=False)[0]
        K=np.kron(K,np.eye(2**(n-1)))
        for k in ks:
            b=K@states[k]@K.conj().T
            errors.append(float(np.linalg.svd(recover(b)-np.trace(b)*states[k],compute_uv=False).sum()))
    # Binary measurement on the first spin.
    K=np.kron(np.diag([0.,1.]),np.eye(2**(n-1)))
    pi=.3; sigma=pi*states[1]+(1-pi)*states[4]
    defects=[]
    for P in (K,np.eye(2**n)-K):
        b=P@sigma@P
        defects.append(float(np.abs(np.diag(recover(b)-np.trace(b)*sigma)).sum()))
    predicted=4*pi*(1-pi)*abs(1/n-4/n)
    assert max(errors)<2e-13
    assert abs(sum(defects)-predicted)<2e-13
    return dict(n=n,code_weights=ks,random_contractions=30,max_arbitrary_kraus_error=max(errors),mixture_pi=pi,mixture_instrument_defect=sum(defects),predicted_mixture_defect=predicted)

def ssep_fixture(n=7,k=3):
    cfg=np.array([tuple(int(j>>i)&1 for i in range(n)) for j in range(2**n) if j.bit_count()==k],float)
    index={tuple(row):i for i,row in enumerate(cfg)}; m=len(cfg)
    L=np.zeros((m,m)); A=np.zeros((n,n))
    for i in range(n):
        j=(i+1)%n
        A[i,i]+=1; A[j,j]+=1; A[i,j]-=1; A[j,i]-=1
        for a,row in enumerate(cfg):
            other=row.copy();other[i],other[j]=row[j],row[i]
            b=index[tuple(other)]
            if a!=b: L[a,a]+=1; L[a,b]-=1
    ev,U=np.linalg.eigh(L); av,V=np.linalg.eigh(A)
    eta=cfg[:,0]; q=k/n; chi=q*(1-q)*n/(n-1)
    data=[]
    for t in [0.,.01,.1,1.,10.]:
        E=(U*np.exp(-t*ev))@U.T
        heat=(V*np.exp(-t*av))@V.T
        actual=E@(eta/m)
        predicted=(cfg@heat[0])/m
        branch_error=float(np.abs(actual-predicted).sum())
        defect=float(2*np.abs(actual-q/m).sum())
        bound=2*math.sqrt(max(0.,chi*(np.dot(heat[0],heat[0])-1/n)))
        cov_actual=float((eta-q)@E@(eta-q)/m)
        cov_pred=chi*(heat[0,0]-1/n)
        assert branch_error<2e-13
        assert abs(cov_actual-cov_pred)<2e-13
        assert defect<=bound+2e-13
        data.append(dict(t=t,exact_defect=defect,heat_bound=bound,branch_identity_error=branch_error,covariance=cov_actual,covariance_formula=cov_pred))
    modes=av[av>1e-10]
    greens=float(np.sum(1/modes)/n)
    assert abs(greens-(n*n-1)/(12*n))<2e-13
    # Integrate covariance exactly via eigenvalue formula.
    centered=eta-q; coeff=(U.T@centered)**2/m
    variance=[]
    for T in [.1,1.,10.,100.]:
        positive=ev>1e-10
        rates=ev[positive]; coeffs=coeff[positive]
        actual=float(2*np.sum(coeffs*(T/rates+np.expm1(-rates*T)/rates**2))/T**2)
        predicted=float(2*chi*np.sum(T/modes+np.expm1(-modes*T)/modes**2)/(n*T*T))
        bound=2*chi*greens/T
        assert abs(actual-predicted)<2e-13
        assert actual<=bound+2e-13
        variance.append(dict(T=T,variance=actual,one_particle_formula=predicted,upper_bound=bound))
    return dict(n=n,k=k,configuration_count=m,branch_and_covariance_checks=data,variance_checks=variance,green=greens,long_time_variance_coefficient=2*chi*greens,cycle_coefficient=q*(1-q)*(n+1)/6)

def green_torus(L,d):
    vals=[]
    for k in itertools.product(range(L),repeat=d):
        value=4*sum(math.sin(math.pi*a/L)**2 for a in k)
        if value>1e-14:vals.append(value)
    return sum(1/x for x in vals)/(L**d)

def minimax_fixture(n):
    a=math.sqrt(n)/2; risk=1/(4*(math.sqrt(n)+1)**2)
    errors=[]
    for p in np.linspace(.001,.999,101):
        estimate=(np.arange(n+1)+a)/(n+2*a)
        errors.append(abs(float(binom(n,float(p))@((estimate-p)**2))-risk))
    assert max(errors)<2e-13
    return dict(n=n,beta_prior_a=a,exact_constant_minimax_risk=risk,max_direct_risk_error=max(errors))
