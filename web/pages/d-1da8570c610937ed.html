#!/usr/bin/env python3
"""Independent exact interval certificate for one negative axial Hankel form.

The high-precision step chooses a candidate rational vector only. The sign
claim is certified afterward using Fraction arithmetic and dyadic outward
intervals for every exp(-9d^2/400).
"""
from fractions import Fraction
from math import comb, factorial
from pathlib import Path
import json
import mpmath as mp

N = 72
DELTA_NUM, DELTA_DEN = 81, 50
D_BITS = 256
D = 1 << D_BITS
RANGE_SQUARES = 10
TAYLOR_ODD = 35
TAYLOR_EVEN = 34
RAT_BITS10 = 50


def exponent_x(k: int) -> Fraction:
    # delta*(k-N/2)^2/N = (81/50)*(k-36)^2/72 = 9*(k-36)^2/400.
    return Fraction(9 * (k - 36) ** 2, 400)


def ceil_fraction(x: Fraction) -> int:
    return -((-x.numerator) // x.denominator)


def floor_fraction(x: Fraction) -> int:
    return x.numerator // x.denominator


def exp_interval(distance: int) -> tuple[int, int]:
    """Return integer l,h with l/D <= exp(-9 distance^2/400) <= h/D."""
    x = Fraction(9 * distance * distance, 400)
    y = x / (1 << RANGE_SQUARES)
    assert 0 <= y <= Fraction(1, 16)
    term = Fraction(1)
    lo_series = term
    hi_series = term
    for r in range(1, TAYLOR_ODD + 1):
        term *= -y / r
        lo_series += term
        if r <= TAYLOR_EVEN:
            hi_series += term
    assert lo_series <= hi_series and lo_series > 0
    lo = floor_fraction(lo_series * D)
    hi = ceil_fraction(hi_series * D)
    for _ in range(RANGE_SQUARES):
        lo = (lo * lo) // D
        hi = ceil_fraction(Fraction(hi * hi, D))
    return lo, hi


def mp_entry(i: int, j: int, scale_exp: list[int]) -> mp.mpf:
    k = i + j + 1
    x = mp.mpf(9) * (k - 36) ** 2 / 400
    return mp.power(2, scale_exp[i] + scale_exp[j]) * mp.exp(-x) / comb(N, k)


def main() -> None:
    # Candidate selection only; all final vector entries are rounded rationals.
    mp.mp.dps = 110
    n = 36  # H1 indices 0,...,floor((N-1)/2).
    scale_exp = []
    for i in range(n):
        k = 2 * i + 1
        x = mp.mpf(9) * (k - 36) ** 2 / 400
        log2_binom = mp.log(comb(N, k), 2)
        scale_exp.append(int(mp.nint((log2_binom + x / mp.log(2)) / 2)))

    H = mp.matrix([[mp_entry(i, j, scale_exp) for j in range(n)] for i in range(n)])
    A = H[: n - 1, : n - 1]
    b = H[: n - 1, n - 1]
    x = mp.lu_solve(A, b)
    schur = H[n - 1, n - 1] - sum(b[i] * x[i] for i in range(n - 1))
    assert schur < 0

    # z=(-A^{-1}b,1) is the Schur-complement witness. Rationalize at 50 places.
    scale10 = 10**RAT_BITS10
    z = [Fraction(int(mp.nint(-x[i] * scale10)), scale10) for i in range(n - 1)]
    z.append(Fraction(1))
    v = [z[i] * (1 << scale_exp[i]) for i in range(n)]

    # Aggregate exact rational coefficients of each q_k/N! term.
    coeff = {k: Fraction(0) for k in range(1, N)}
    for i in range(n):
        for j in range(n):
            k = i + j + 1
            coeff[k] += v[i] * v[j] / comb(N, k)

    # Outward interval sum at the common dyadic scale D.
    total_lo = 0
    total_hi = 0
    exp_cache = {d: exp_interval(d) for d in range(37)}
    for k in range(1, N):
        c = coeff[k]
        if c == 0:
            continue
        lo, hi = exp_cache[abs(k - 36)]
        if c > 0:
            total_lo += floor_fraction(c * lo)
            total_hi += ceil_fraction(c * hi)
        else:
            total_lo += floor_fraction(c * hi)
            total_hi += ceil_fraction(c * lo)

    assert total_hi < 0, (total_lo, total_hi)
    result = {
        "status": "CERTIFIED_NEGATIVE",
        "matrix": "H1=(q_(i+j+1))_{0<=i,j<=35}",
        "N": N,
        "delta": "81/50",
        "q_k": "k!(72-k)! exp[-(81/50)(k-36)^2/72]",
        "normalization": "H1/72!; positive normalization, sign unchanged",
        "congruence": "A_ij=2^(e_i+e_j) q_(i+j+1)/72!",
        "scale_exponents_e_i": scale_exp,
        "witness": "v_i=2^e_i z_i, z_i rounded to denominator 10^50; z_35=1",
        "z_rational": [[str(a.numerator), str(a.denominator)] for a in z],
        "quadratic_form": "v^T(H1/72!)v = sum_(k=1)^71 c_k exp[-9(k-36)^2/400]",
        "coefficient_c_k": {str(k): [str(c.numerator), str(c.denominator)] for k, c in coeff.items() if c},
        "dyadic_bits": D_BITS,
        "range_squares": RANGE_SQUARES,
        "alternating_series_bounds": {"lower_degree_odd": TAYLOR_ODD, "upper_degree_even": TAYLOR_EVEN},
        "quadratic_interval_scaled_numerators": [str(total_lo), str(total_hi)],
        "quadratic_interval_denominator": str(D),
        "approx_candidate_schur_only": mp.nstr(schur, 40),
        "note": "Floating point was used only to choose z. The displayed rational witness and interval are independently checkable by integer/rational arithmetic."
    }
    out = Path(__file__).with_suffix(".json")
    out.write_text(json.dumps(result, indent=2) + "\n")
    print("wrote", out)
    print("scaled interval:", total_lo, total_hi, "/", D)
    print("decimal endpoints:", mp.nstr(mp.mpf(total_lo) / D, 25), mp.nstr(mp.mpf(total_hi) / D, 25))
    print("candidate Schur (not used for certification):", mp.nstr(schur, 30))


if __name__ == "__main__":
    main()
