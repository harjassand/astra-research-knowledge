def cross(a, b):
    return [a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]]


def determinant(rows):
    rows = [list(row) for row in rows]
    n, sign, old = len(rows), 1, 1
    for k in range(n - 1):
        pivot = next((i for i in range(k, n) if rows[i][k] != 0), None)
        if pivot is None:
            return 0
        if pivot != k:
            rows[k], rows[pivot] = rows[pivot], rows[k]
            sign = -sign
        value = rows[k][k]
        for i in range(k+1, n):
            for j in range(k+1, n):
                numerator = rows[i][j]*value-rows[i][k]*rows[k][j]
                assert numerator % old == 0
                rows[i][j] = numerator // old
            rows[i][k] = 0
        old = value
    return sign * rows[-1][-1]


def triangle_certificate(instance):
    assert instance["n"] == 3 and instance["costs"] == [2, 2, 2]
    matrices = instance["subspaces"]
    normals = [cross([r[0] for r in s], [r[1] for r in s]) for s in matrices]
    det = determinant(normals)
    assert det != 0 and all(any(x for x in row) for row in normals)
    caps = [Fraction(u) for u in instance["caps"]]
    assert all(0 <= u <= 1 for u in caps)
    ordered = sorted(range(3), key=lambda i: caps[i])
    i, j, k = ordered
    a, b, c = [caps[index] for index in ordered]
    total = min(a+b+c, 1+a, Fraction(3, 2))
    if total == a+b+c:
        primal = caps
        dual = {"kind": "all cap penalties", "y": [2, 2, 2]}
    elif total == Fraction(3, 2):
        primal = [Fraction(1, 2)] * 3
        dual = {"kind": "ambient uniform", "p": [1, 1, 1]}
    else:
        primal = [Fraction(0)]*3
        primal[i] = a
        primal[j] = max(a, 1-c)
        primal[k] = 1-primal[j]
        line = cross(normals[j], normals[k])
        assert any(line)
        assert sum(x*y for x,y in zip(normals[i], line)) != 0
        assert sum(x*y for x,y in zip(normals[j], line)) == 0
        assert sum(x*y for x,y in zip(normals[k], line)) == 0
        y = [0]*3
        y[i] = 2
        dual = {"kind": "pair intersection line", "line": line, "lambda": 2, "y": y}
    assert all(0 <= x <= u for x, u in zip(primal, caps))
    assert all(primal[u]+primal[v] <= 1 for u in range(3) for v in range(u+1,3))
    assert sum(primal) == total
    return {"name": instance["name"], "normal_determinant": det, "acquired_normals": normals,
            "primal": [str(x) for x in primal], "exact_value": str(2*total), "dual": dual,
            "scope": "Three independent plane normals, uniform cost2, arbitrary rational caps; universal class proof supplies all-subspace feasibility."}


def decomposition_certificate(instance):
    n, spaces = instance["n"], instance["subspaces"]
    assert n % 2 == 0 and len(spaces) == n and instance["costs"] == [2]*n
    half = n//2
    first = [[entry for s in spaces[:half] for entry in s[i]] for i in range(n)]
    second = [[entry for s in spaces[half:] for entry in s[i]] for i in range(n)]
    determinants = [determinant(first), determinant(second)]
    assert all(d != 0 for d in determinants)
    caps = [Fraction(u) for u in instance["caps"]]
    assert all(0 <= u <= 1 for u in caps)
    lower_t, upper_t = 1-min(caps[half:]), min(caps[:half])
    assert lower_t <= upper_t
    primal = [lower_t]*half + [1-lower_t]*half
    assert all(0 <= x <= u for x,u in zip(primal,caps))
    assert 2*sum(primal) == n
    return {"name": instance["name"], "decomposition_determinants": determinants,
            "mixing_interval": [str(lower_t), str(upper_t)], "primal": [str(x) for x in primal],
            "exact_value": str(n), "dual": {"kind": "ambient uniform", "p": [1]*n},
            "scope": "Two acquired direct decompositions into2-planes and nonempty cap-compatible mixing interval; universal direct-sum argument supplies all-subspace feasibility."}
