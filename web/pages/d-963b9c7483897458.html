def add(a, b):
    out = [F(0)] * max(len(a), len(b))
    for i, v in enumerate(a):

    return out


def scale(p, c):
    return [v * c for v in p]


def mul(a, b):
    out = [F(0)] * (len(a) + len(b) - 1)
    for i, u in enumerate(a):
        for j, v in enumerate(b):

    return out
