def transpose(a): return [list(x) for x in zip(*a)]
def mm(a,b): return [[sum(x*y for x,y in zip(row,col)) for col in zip(*b)] for row in a]
def eye(n): return [[int(i==j) for j in range(n)] for i in range(n)]
def power(a,n):
 out=eye(len(a))
 for _ in range(n): out=mm(out,a)
 return out
def moment(a,n,v=None):
 if v is None: return power(a,n)[0][0]
 p=power(a,n);return sum(v[i]*p[i][j]*v[j] for i in range(len(v)) for j in range(len(v)))
def matrix_log_contraction(e,terms=200):
 n=len(e);q=[[int(i==j)-e[i][j] for j in range(n)] for i in range(n)]
 out=[[0.]*n for _ in range(n)];p=eye(n)
 for h in range(1,terms+1):
  p=mm(p,q)
  for i in range(n):
   for j in range(n): out[i][j]-=p[i][j]/h
 return out

def reaction_e(lam,k,delta):
 w=math.sqrt(k*k+delta*delta);f=math.exp(-(lam+k));s=math.sinh(w)/w
 return [[f*(math.cosh(w)+k*s),-delta*f*s],[-delta*f*s,f*(math.cosh(w)-k*s)]]

class Oracle:
 def __init__(self,e,sigma,seed):
  self._e=e;self._sigma=sigma;self._rng=random.Random(seed)
  self.cost={'zero_observations':0,'pulse_observations':0,'actions':0,'wait_time':0.,'protocols':set()}
 def zero(self):
  self.cost['zero_observations']+=1;self.cost['actions']+=2
  return self._rng.gauss(0,self._sigma)
 def assay(self,h):
  self.cost['pulse_observations']+=1;self.cost['actions']+=4;self.cost['wait_time']+=h;self.cost['protocols'].add(h)
  return self._rng.gauss(float(moment(self._e,h)),self._sigma)
 def ledger(self):
  return dict(self.cost,protocols=sorted(self.cost['protocols']))

def calibrate(oracle,eta):
 # D=(Z1-Z2)/sqrt2 cancels any constant zero-protocol mean.
 c=math.log(2)/2-.25;m=math.ceil(math.log(1/eta)/c)
 s=sum(((oracle.zero()-oracle.zero())/math.sqrt(2))**2 for _ in range(m))/m
 return {'pairs':m,'S':s,'sigma_squared_upper':2*s}

def acquire(oracle,H,tau,eta,mode):
 # No access to the private kinetic matrix or sigma in either algorithm.
 cal=calibrate(oracle,eta/2);u=cal['sigma_squared_upper']
 if mode=='moment_certificate':
  choose=H*(H-1)/2;w=tau/(2*(H+6*choose))
  R=max(1,math.ceil(2*u*math.log(8/eta)/(w*w)))
  a=max(0.,min(1.,sum(oracle.assay(1) for _ in range(R))/R))
  c=max(0.,min(1.,sum(oracle.assay(2) for _ in range(R))/R))
  aL,aU=max(0.,a-w),min(1.,a+w);cL,cU=max(0.,c-w),min(1.,c+w)
  rU=max(0.,cU-aL*aL);rL=max(0.,cL-aU*aU)
  errors=[h*w+h*(h-1)*rU/2 for h in range(1,H+1)]
  pred=[a**h for h in range(1,H+1)]
  status='CERTIFIED' if max(errors)<=tau else 'UNKNOWN'
  if cU<aL*aL:status='UNKNOWN_MODEL_OR_NOISE'
  report={'a':a,'variance_interval':[rL,rU],'mean_width':w,'predictions':pred,'error_bounds':errors,'status':status}
 else:
  # Competent direct impulse-response retaining control covers every h<=H.
  w=tau;R=max(1,math.ceil(2*u*math.log(4*H/eta)/(w*w)))
  pred=[max(0.,min(1.,sum(oracle.assay(h) for _ in range(R))/R)) for h in range(1,H+1)]
  report={'predictions':pred,'error_bounds':[tau]*H,'status':'CERTIFIED'}
 return dict(report,calibration=cal,repetitions_per_protocol=R,cost=oracle.ledger())

def exact_two_protocol_obstruction():
 a,b=F(4,5),F(1,50);models=[[[a,b],[b,F(1,5)]],[[a,b],[b,F(3,5)]]]

def cyclic_family():
