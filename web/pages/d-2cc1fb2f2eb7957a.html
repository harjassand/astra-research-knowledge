begin=time.monotonic()
x,y,rho,eta=s.symbols('x y rho eta')
V=(x,y,rho,eta)
out={}
def check(name,expr):
    r=s.factor(s.expand(expr))
    assert r==0,(name,r)
    out[name]={'pass':True,'residual':'0'}
def matrix_check(name,M):
    r=[s.factor(s.expand(z)) for z in M]
    assert all(z==0 for z in r),(name,r)
    out[name]={'pass':True,'residuals':['0']*len(r)}
def degree(f):
    return -1000 if f==0 else s.Poly(s.expand(f),*V).total_degree()
def hom(f,n):
    P=s.Poly(s.expand(f),*V)
    return s.Add(*(co*s.prod(z**k for z,k in zip(V,mon))
                   for mon,co in P.terms() if sum(mon)==n))

P=s.Matrix([[1,4],[s.Rational(1,2),3]])
M=s.Matrix([[x,x**2-1],[1,x]])
D=(P*M*P.inv()).applyfunc(s.expand)
check('chart_det1',D.det()-1)
matrix_check('chart_polynomial_inverse',D*D.adjugate()-s.eye(2))
c=s.Matrix([[2*y,4*x]])
cplus=s.Matrix([[-3,4]])
cminus=-cplus
matrix_check('firstjet_plus',cplus*D.subs(x,1)-cplus)
matrix_check('firstjet_minus',cminus*D.subs(x,-1)-cplus)

     {x:-1,y:s.Rational(3,2),rho:0,eta:0}]
