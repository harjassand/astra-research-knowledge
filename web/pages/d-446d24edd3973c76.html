PR,aa,bb,cc=ring('a,b,c',QQ)

N=2*r**4+13*r**2-7*r

def zero(expr):
    assert sy.cancel(expr)==0,str(expr)


def polynomial_record(v):
    if isinstance(v,dict) and list(v)==['poly']:
        return PR.from_dict({tuple(mon):QQ(co) for mon,co in v['poly']})
    if isinstance(v,list):return [polynomial_record(x) for x in v]
    if isinstance(v,dict):
        return {(int(k) if k.isdecimal() else ast.literal_eval(k) if k.startswith('(') else k):
                polynomial_record(x) for k,x in v.items()}
    return v


def upoly(terms):
    return RR.from_dict({(int(power),):QQ(co) for power,co in terms})


def subtract(row,coef,other):
    for col,val in other.items():
        nv=row.get(col,PR.zero)-coef*val
        if nv:row[col]=nv
        else:row.pop(col,None)


def compositions(total,n):
    if n==1:
        yield (total,)
        return
    for i in range(total+1):
        for suffix in compositions(total-i,n-1):yield (i,)+suffix


def real_domain_and_factors():
    a,b,c,r0,t=sy.symbols('a b c r t')
    fpath=BASE/'results/metric_prior/cycle13/generic_d8_divisor_factors.json'
    original=[sy.sympify(v,locals={'a':a,'b':b,'c':c}) for v in json.loads(fpath.read_text())['factors']]
    assert len(original)==28
    p7=original[6]
    zero(b*p7-c*p7.subs({b:c,c:b},simultaneous=True)
         -(b*b-c*c)*(b*c*(5-2*a*a)-7*a))
    zero((p7/c).subs(c,7*a/(b*(5-2*a*a)))
         +4*a*a*(a*a+1)*(a*a+3)/(5-2*a*a))
    nr=2*r0**4+13*r0*r0-7*r0; dr=2*r0*r0+11*r0-5
    gr=4*r0**3+2*r0*r0+15*r0-5
    Tr=nr/dr
    zero(p7.subs({a:r0,b:t,c:t},simultaneous=True)-t*(nr-dr*t*t))
    zero(Tr-r0*r0-r0*(1-r0)*(11*r0-7)/dr)
    zero(Tr-(1+r0)/2-(r0-1)*gr/(2*dr))
    assert sy.gcd(nr,dr)==1
    assert gr.subs(r0,sy.Rational(1,4))<0<gr.subs(r0,sy.Rational(1,3))
    assert dr.subs(r0,sy.Rational(2,5))<0<dr.subs(r0,sy.Rational(3,7))
    assert sy.Poly(gr,r0).count_roots(0,sy.oo)==1
    assert sy.Poly(dr,r0).count_roots(0,sy.oo)==1

    C0=28*r0**4+42*r0**3+236*r0*r0+241*r0-135
    C1=4*r0**3+12*r0*r0+59*r0-23
    C7=64*r0**7+120*r0**6+1084*r0**5+2090*r0**4+4678*r0**3+5617*r0*r0-7196*r0+1575
    Sr=2*r0**6+3*r0**5+46*r0**4+51*r0**3+132*r0*r0-106*r0+16
    alternate=[sy.expand(v.subs({a:t,b:t,c:r0},simultaneous=True)) for v in original]
    # Exact complete alternate-chart factor list is freshly substituted.
    normalized={}
    for i,value in enumerate(alternate):
        tp=sy.Poly(value,t)
        parity={mon[0]%2 for mon,_ in tp.terms()}
        # Some simple affine factors mix t parities; their unit-interval signs
        # are elementary and are retained as literal expressions in the report.
        if len(parity)==1:
            chi=next(iter(parity))
            val=sy.cancel(sum(co*Tr**((mon[0]-chi)//2) for mon,co in tp.terms()))
            normalized[i]=(chi,val)
    zero(normalized[0][1]-(r0-1)*C0/dr)
    zero(normalized[1][1]-r0*(r0-1)*C1/dr)
    zero(normalized[5][1]-4*r0*r0*(r0-1)*Sr/dr**2)
    zero(normalized[18][1]+4*r0*r0*(r0-1)*Sr/dr**2)
    zero(normalized[6][1]+Tr*(r0-1)*(11*r0-7))
    zero(normalized[8][1]+r0*r0*(r0-1)**2*(2*r0**3+13*r0-7)*C7/dr**3)
    assert normalized[0][0]==normalized[1][0]==1
    for p in (C0,C1):
        assert all(co>0 for co in sy.Poly(sy.diff(p,r0),r0).all_coeffs())
        assert p.subs(r0,sy.Rational(1,3))<0<p.subs(r0,sy.Rational(7,11))
    for p in (C7,Sr):
        assert all(co>0 for co in sy.Poly(sy.diff(p,r0,2),r0).all_coeffs())
        assert sy.diff(p,r0).subs(r0,sy.Rational(7,11))>0
        assert p.subs(r0,sy.Rational(7,11))>0
    assert sy.diff(C7,r0).subs(r0,sy.Rational(1,3))<0<C7.subs(r0,sy.Rational(1,3))
    assert sy.diff(Sr,r0).subs(r0,sy.Rational(1,4))<0
    assert Sr.subs(r0,sy.Rational(1,5))>0>Sr.subs(r0,sy.Rational(1,4))
    assert Sr.subs(r0,sy.Rational(1,3))<0
    assert sy.Poly(Sr,r0).count_roots(0,sy.Rational(1,3))==1
    assert sy.Poly(Sr,r0).count_roots(sy.Rational(7,11),1)==0
    factorization=sy.factor_list(Sr,r0)[1]
    assert len(factorization)==1 and factorization[0][0].as_poly(r0).degree()==6
    return {'status':'PASS independently derived full real p7 domain and exceptional locus',
       'domain':'(0,eta) union [7/11,1), G(eta)=0,1/4<eta<1/3',
       'smallest_first_nonintersection':'b!=c implies at least one p7 chart nonzero',
       'alternate_factors':[{ 'index':i,'actual_substitution':str(val),
              'removed_t_parity':normalized[i][0] if i in normalized else None,
              'exact_rational_restriction':str(normalized[i][1]) if i in normalized else None}
              for i,val in enumerate(alternate)],
       'exception':'unique alpha rootS in(1/5,1/4), plus previously gated exchangeable r7/11',
       'S_QQ_irreducible':True,
       'factor_source_sha256':hashlib.sha256(fpath.read_bytes()).hexdigest()}


def replay():
    basepath=BASE/'results/metric_prior/cycle13/constant_schur_reduction.json.gz'
    curvepath=BASE/'results/metric_prior/cycle15/p7_curve_scalar_reduction.json.gz'
    dagpath=BASE/'results/metric_prior/cycle15/p7_targeted_unit_dag.json.gz'
    base=polynomial_record(json.loads(gzip.decompress(basepath.read_bytes())))
    curve=json.loads(gzip.decompress(curvepath.read_bytes()))
    dag=json.loads(gzip.decompress(dagpath.read_bytes()))
    raw,A,B,C,m0,m1,m2,s0,s1,s2=ring('a,b,c,m0,m1,m2,s0,s1,s2',QQ)
    mm=[m0,m1,m2];ss=[s0,s1,s2]
    q=[[raw.one,A,B],[A,raw.one,C],[B,C,raw.one]]
    generators=[(1-2*mi*si)*(1-si*si) for mi,si in zip(mm,ss)]
    generators.extend((1-4*mm[i]*ss[i])*(q[i][j]-ss[i]*ss[j])
                      -2*mm[i]*ss[j]*(1-ss[i]*ss[i])
                      for i,j in itertools.permutations(range(3),2))
    generators.append(3*s0*s1*s2-C*s0-B*s1-A*s2)
    coefficients=[]
    for g in generators:
        table={}
        for powers,co in g.items():
            table.setdefault(powers[3:],{})[powers[:3]]=co
        coefficients.append({powers:PR.from_dict(co) for powers,co in table.items()})
    assert coefficients==base['generators']
    expected={(g,mon) for g in range(10)
              for deg in ((0,2,4) if g<9 else (1,3,5))
              for mon in compositions(deg,6)}
    provided={(g,tuple(mon)) for g,mon in base['sources']}
    assert len(base['sources'])==len(expected)==1646 and provided==expected
    allmon={mon for deg in (0,2,4,6,8) for mon in compositions(deg,6)}
    assert set(map(tuple,base['basis']))==allmon and len(base['basis'])==1897
    basis={tuple(mon):i for i,mon in enumerate(base['basis'])}
    unitcol=basis[(0,)*6]
    ownrows=[]
    maximum_degree=0
    for gi,shift in base['sources']:
        row={}
        for mon,co in coefficients[gi].items():
            product=tuple(a+b for a,b in zip(mon,shift))
            maximum_degree=max(maximum_degree,sum(product))
            row[basis[product]]=co
        ownrows.append(row)
    assert ownrows==base['rows'] and maximum_degree==8
    pivots={}
    for col,(source,divisor,history) in base['dag'].items():
        row=dict(ownrows[source])
        for old,co in history:
            assert old in pivots
            subtract(row,co,pivots[old])
        assert divisor and divisor.is_ground
        dv=divisor[(0,0,0)]
        assert row[col]==divisor
        row={k:v/dv for k,v in row.items()}
        assert row==base['pivots'][col] and row[col]==PR.one
        pivots[col]=row
    residual={}
    for source,stored,history in base['pending']:
        row=dict(ownrows[source])
        for col,co in history:subtract(row,co,pivots[col])
        assert row==stored
        residual[source]=row
    checked_rows={};nodes=[];nonconstant=0;degrees=[]
    for nid,node in enumerate(dag['nodes']):
        den=upoly(node['divisor'])
        assert den and den.gcd(S).is_ground
        nonconstant+=not den.is_ground
        if node['kind']=='initial':
            rid=node['rowid'];gi,shift=base['sources'][rid]
            # Actual generator sign character under the simultaneous action
            # (b,c)->(-b,-c),(m2,s2)->(-m2,-s2), independently checked by terms.
            generator_chars={(powers[1]+powers[2]+powers[5]+powers[8])%2
                             for powers in generators[gi]}
            assert len(generator_chars)==1
            chi=(sum(generator_chars)+shift[2]+shift[5])%2
            assert chi==curve['row_source_characters'][str(rid)]
            power=curve['row_D_powers'][str(rid)]
            row={}
            for col,poly in residual[rid].items():
                aux=base['basis'][col];delta=(aux[2]+aux[5])%2
                out=RR.zero
                for (i,j,k),co in poly.items():
                    exponent=j+k+chi-delta
                    assert exponent>=0 and exponent%2==0
                    e=exponent//2
                    assert e<=power
                    out+=co*r**i*N**e*D**(power-e)
                if out:row[col]=out
            stored={int(col):upoly(v) for col,v in curve['rows'][str(rid)].items()}
            assert row==stored
            checked_rows[rid]=row
        else:
            assert node['kind']=='eliminate'
            left=node['left'];right=node['right']
            assert 0<=left<nid and 0<=right<nid
            p=upoly(node['p']);co=upoly(node['coefficient'])
            row={col:p*val for col,val in nodes[left].items()}
            for col,val in nodes[right].items():
                nv=row.get(col,RR.zero)-co*val
                if nv:row[col]=nv
                else:row.pop(col,None)
        result={}
        for col,val in row.items():
            quotient,remainder=val.div(den)
            assert not remainder
            if quotient:result[col]=quotient
            degrees.append(quotient.degree())
        nodes.append(result)
    final=nodes[dag['finalnode']]
    stored={int(col):upoly(v) for col,v in dag['finalrow'].items()}
    assert final==stored=={unitcol:RR.one}
    assert set(dag['used_nodes'])==set(range(len(nodes)))
    assert D.gcd(S).is_ground and N.gcd(S).is_ground
    return {'status':'PASS independently rebuilt original raw D8 unit identity',
        'original_generators':10,'exhaustive_legal_sources':1646,'exhaustive_even_monomials':1897,
        'constant_history_pivots':len(pivots),'residual_histories':len(residual),
        'selected_original_rows':len(checked_rows),'exact_polynomial_DAG_nodes':len(nodes),
        'all_divisors_coprime_to_S':True,'nonconstant_divisor_count':int(nonconstant),
        'final_raw_unit':'1','maximum_auxiliary_product_degree':maximum_degree,
        'maximum_replayed_parameter_degree':max(degrees),
        'modular_rank_used_for_proof':False,'creator_or_checker_imported':False,
        'sources':{str(p.relative_to(BASE)):hashlib.sha256(p.read_bytes()).hexdigest()
                   for p in (basepath,curvepath,dagpath)}}
