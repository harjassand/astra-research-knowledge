from sympy.polys.domains import QQ
from sympy.polys.rings import ring

BASE=Path(__file__).resolve().parents[3]
a,b,c,r,u,A,E=s.symbols('a b c r u A E')
source=BASE/'sources/metric_use10/cycle16/generic_d8_divisor_factors.json'
if not source.is_file():source=BASE/'results/metric_prior/cycle13/generic_d8_divisor_factors.json'
factors=[s.sympify(v,locals={'a':a,'b':b,'c':c}) for v in json.loads(source.read_text())['factors']]

def square_convert(v):
    ans=0
    for (i,j),co in s.Poly(v,r,u).terms():
        assert i%2==j%2==0,(i,j)
        ans+=co*A**(i//2)*E**(j//2)
    return s.expand(ans)

def main():
    start=time.monotonic();d=7/(2*r*r+5)
    plargest=s.cancel(factors[6].subs({a:u,b:r*d/u,c:r},simultaneous=True)/r)
    vmiddle=s.cancel(factors[8].subs({a:r*d/u,b:u,c:r},simultaneous=True)/(r*r))
    raw=[]
    for name,val in [('P',plargest),('V',vmiddle)]:
        num,den=s.fraction(val)
        content,p=s.Poly(num,r,u).primitive()
        raw.append(square_convert(p.as_expr()))
        print(name,raw[-1],flush=True)
        print(name+' denominator',s.factor(den),'content',content,flush=True)
    p,v=raw
    resultant=s.resultant(p,v,E)
    const,fs=s.factor_list(resultant,A)
    print('resultant factors',[(str(f),m) for f,m in fs],flush=True)
    out={'status':'PASS independent original-factor final intersection and tie branch', 'P':str(p),'V':str(v),
         'resultant_constant':str(const),'resultant_factors':[(str(f),m) for f,m in fs],
         'subresultant_degrees':[int(s.degree(q,E)) for q in s.subresultants(p,v,E)],
         'source_sha256':hashlib.sha256(source.read_bytes()).hexdigest(),
         'scope':'Final chart-intersection and separate tie only; full rawD8 additionally needs M-curve and safe/domain gates.',
         'seconds':time.monotonic()-start}
    high=[q for q,m in fs if s.degree(q,A)>1]
    assert len(high)==1
    Rexpr=high[0];assert s.Poly(Rexpr,A).is_irreducible
    for q,m in fs:
        if q==Rexpr:continue
        assert s.degree(q,A)==1
        root=s.solve(q,A)[0];assert root<=0 or root>=1
    subs=s.subresultants(p,v,E);linear=s.Poly(subs[-2],E)
    assert linear.degree()==1
    K,x=ring('A',QQ);mod=K.from_expr(Rexpr)
    red=lambda q:q.rem(mod)
    def inv(q):
        v,w,g=q.gcdex(mod);assert g.is_ground and g
        ans=red(v/g[(0,)]);assert red(q*ans)==K.one
        return ans
    lead=red(K.from_expr(linear.coeff_monomial(E)));constant=red(K.from_expr(linear.coeff_monomial(1)))
    ev=red(-constant*inv(lead));dv=red(7*inv(2*x+5));einv=inv(ev)
    def evalparam(expr):
        ans=K.zero
        for (i,j),co in s.Poly(expr,A,E).terms():ans+=QQ.from_sympy(co)*red(x**i*ev**j)
        return red(ans)
    assert evalparam(p)==evalparam(v)==K.zero
    sq=[red(x),ev]
    def add(v,w):
        ans=dict(v)
        for ch,co in w.items():
            val=red(ans.get(ch,K.zero)+co)
            if val:ans[ch]=val
            else:ans.pop(ch,None)
        return ans
    def mul(v,w):
        ans={}
        for ch,co in v.items():
            for hc,bc in w.items():
                val=red(co*bc)
                for j in range(2):
                    if ch[j] and hc[j]:val=red(val*sq[j])
                dst=tuple((ch[j]+hc[j])%2 for j in range(2))
                ans=add(ans,{dst:val})
        return ans
    def power(v,n):
        ans={(0,0):K.one}
        for _ in range(n):ans=mul(ans,v)
        return ans
    # Alternate chart(c,a,b)=(u,r,rd/u).
    charts=[{(0,1):K.one},{(1,0):K.one},{(1,1):red(dv*einv)}]
    norms=[]
    for index,factor in enumerate(factors):
        value={}
        for mon,co in s.Poly(factor,a,b,c).terms():
            term={(0,0):K(QQ.from_sympy(co))}
            for j,n in enumerate(mon):term=mul(term,power(charts[j],n))
            value=add(value,term)
        norm={(0,0):K.one}
        for sr,su in itertools.product((-1,1),repeat=2):
            norm=mul(norm,{ch:co*sr**ch[0]*su**ch[1] for ch,co in value.items()})
        assert set(norm)=={(0,0)}
        nv=norm[(0,0)]
        assert nv and nv.gcd(mod).is_ground,(index,str(nv))
        norms.append({'index':index,'norm':[[mon[0],str(co)] for mon,co in sorted(nv.items())],'gcd_R':'1'})
    out.update({'R_irreducible':True,'R':str(Rexpr),
                'E_coefficients':[[mon[0],str(co)] for mon,co in sorted(ev.items())],

        norm=s.Poly(val.as_expr()*val.as_expr().subs(u,-u),u).rem(modulus)
        assert norm.degree()==0 and norm.as_expr()!=0,(index,norm)

    dv=7/(2*rt*rt+5)

                'M_nonzero':str(M),'chart':'(t,7/11,t)','all28_norms':tied}
    out['seconds']=time.monotonic()-start
