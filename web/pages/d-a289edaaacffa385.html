class Poly:
    def __init__(self,n,terms=None):
        self.n=n; self.terms={m:int(c) for m,c in (terms or {}).items() if c}
    @classmethod
    def const(cls,n,c): return cls(n,{(0,)*n:int(c)})
    @classmethod
    def var(cls,n,i):
        m=[0]*n; m[i]=1; return cls(n,{tuple(m):1})
    def __add__(self,other):
        if not isinstance(other,Poly): other=Poly.const(self.n,other)
        d=self.terms.copy()
        for m,c in other.terms.items(): d[m]=d.get(m,0)+c
        return Poly(self.n,d)
    __radd__=__add__
    def __neg__(self): return Poly(self.n,{m:-c for m,c in self.terms.items()})
    def __sub__(self,other): return self+(-other)
    def __mul__(self,other):
        if not isinstance(other,Poly): other=Poly.const(self.n,other)
        d={}
        for a,c in self.terms.items():
            for b,e in other.terms.items():
                m=tuple(x+y for x,y in zip(a,b)); d[m]=d.get(m,0)+c*e
        return Poly(self.n,d)
    __rmul__=__mul__
    def derivative(self,i):
        d={}
        for m,c in self.terms.items():
            if m[i]:
                t=list(m); t[i]-=1; d[tuple(t)]=c*m[i]
        return Poly(self.n,d)
    def evaluate(self,x,ring=None):
        if ring is None:
            return sum((Q(c)*product(Q(t)**e for t,e in zip(x,m))
                        for m,c in self.terms.items()),Q(0))
        answer=ring.elt(0)

            answer=ring.add(answer,term)
        return answer
    def dump(self): return [{'monomial':list(m),'coefficient':c} for m,c in sorted(self.terms.items())]

class UnitRing:
    """Z/modulus with unit-only division; modulus is a power of supplied p."""
    def __init__(self,modulus,p,cost): self.modulus=modulus; self.p=p; self.cost=cost
    def elt(self,x): self.cost['convert']+=1; return int(x)%self.modulus
    def add(self,a,b):self.cost['add']+=1; return (a+b)%self.modulus
    def sub(self,a,b):self.cost['sub']+=1; return (a-b)%self.modulus
    def mul(self,a,b):self.cost['mul']+=1; return a*b%self.modulus
    def div(self,a,b):
        if b%self.p==0:raise ZeroDivisionError('nonunit modular pivot')
        self.cost['unit_div']+=1;return a*pow(b,-1,self.modulus)%self.modulus


def solve_square(matrix,rhs,ring):
    n=len(matrix)
    if any(len(row)!=n for row in matrix) or len(rhs)!=n:raise ValueError('square dimensions')
    a=[[ring.elt(x) for x in row]+[ring.elt(y)] for row,y in zip(matrix,rhs)]
    for j in range(n):
        k=next((i for i in range(j,n) if a[i][j]%ring.p),None)
        if k is None:raise ValueError('selected Jacobian not invertible modulo p')
        a[j],a[k]=a[k],a[j];pivot=a[j][j]
        a[j]=[ring.div(x,pivot) for x in a[j]]
        for i in range(n):

                c=a[i][j];a[i]=[ring.sub(x,ring.mul(c,y)) for x,y in zip(a[i],a[j])]
    return [row[-1] for row in a]


def chart(data):

        for r in range(dim):

    n=len(coords); lookup={triple:i for i,triple in enumerate(variables)}; y=[]

def independent_rows(matrix,p,n):
    f=Field(p);selected=[];rank=0
    for i,row in enumerate(matrix):
        rr,_=rref([matrix[j] for j in selected]+[row],n,f)
        if len(rr)>rank:selected.append(i);rank=len(rr)
        if rank==n:break
    return selected,f.report()


def reconstruct(a,modulus,bound):
    a%=modulus
    if modulus<=2*bound*bound: return None
    r0,r1=modulus,a;t0,t1=0,1
    while abs(r1)>bound:
        q=r0//r1;r0,r1=r1,r0-q*r1;t0,t1=t1,t0-q*t1
    if t1==0 or abs(t1)>bound or gcd(r1,t1)!=1:return None
    if t1<0:r1,t1=-r1,-t1
    if gcd(t1,modulus)!=1 or (r1-a*t1)%modulus:return None
    return Q(r1,t1)

def original_closure(data,bases):
    f=Field()
    for arrow in data['arrows']:
        t,h=arrow['tail'],arrow['head'];m=arrow['matrix'];dim=data['dimensions'][h]
        image=[[sum((Q(m[r][l])*col[l] for l in range(len(col))),Q(0)) for r in range(dim)]
               for col in bases[t]]
        if len(rref(bases[h]+image,dim,f)[0])!=len(bases[h]):return False
    return True
