\[ \|\mathcal K\|_\diamond\le \frac{\kappa}{d^2}. \]

\[ \boxed{ \Psi=\mathcal D+\frac{\mathcal K}{\kappa} } \tag{1} \]
is CPTP. Consequently,
\[ \boxed{ \kappa(\mathcal D-\mathrm{id})+\mathcal K = \kappa(\Psi-\mathrm{id}). } \tag{2} \]
The right side is a valid Poisson-jump generator: at rate \(\kappa\), apply the channel \(\Psi\).

\[ J(\mathcal D)=\frac{I_{d^2}}{d^2}, \qquad \|J(\mathcal K)\|_\infty \le \|J(\mathcal K)\|_1 \le \|\mathcal K\|_\diamond. \]

\[ J(\Psi) = \frac{I_{d^2}}{d^2} + \frac{J(\mathcal K)}{\kappa} \ge0. \]

\[ \|\mathcal K\|_\diamond\le \frac{\kappa}{2d^2}, \]

\[ J(\Psi)\ge \frac{I_{d^2}}{2d^2}. \tag{3} \]

\[ \Pr(\text{fault before }W) = \frac{\eta}{v+\eta} = O(\eta). \tag{12} \]

\[ \begin{aligned} P_r &=\mathbb E\!\left[(1-e^{-\eta W})^r\right]\\ &= \frac{\Gamma(1+v/\eta)\Gamma(r+1)} {\Gamma(r+1+v/\eta)}. \end{aligned} \tag{13} \]

\[ \boxed{ P_r=\frac1{\binom{a+r}{r}}. } \tag{14} \]

\[ P_r\sim a!\,r^{-a}. \]
