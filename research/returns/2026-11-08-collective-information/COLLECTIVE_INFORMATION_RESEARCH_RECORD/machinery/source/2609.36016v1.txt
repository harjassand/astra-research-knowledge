 // Restore the saved color scheme preference, or
 // enact the browser preference if "automatic", 
 // without expecting DOM load to have completed.
 //
 // Also restore any saved readingmode and ToC display preferences.
 function initializeReadingPreferences() {
 let saved_theme = localStorage.getItem("ar5iv_theme") || "automatic";
 if (saved_theme === "automatic") {
 if (window.matchMedia("(prefers-color-scheme: dark)").matches) {
 saved_theme = "dark";
 }
 }
 if (saved_theme == "dark") {
 document.documentElement.setAttribute("data-theme", "dark");
 } else {
 document.documentElement.setAttribute("data-theme", "light");
 }
 const tocDisplay = localStorage.getItem('arxiv_html_paper_toc_display');
 if (tocDisplay) {
 document.documentElement.setAttribute("data-toc-display", tocDisplay);
 }
 const readingMode = localStorage.getItem('arxiv_html_paper_reading_mode');
 if (readingMode) {
 document.documentElement.setAttribute("data-reading-mode", readingMode);
 }
 }
 // Run as soon as JS starts, to minimize repainting
 initializeReadingPreferences();
 /* Reading mode hides the chrome and the empty .arxiv-html-header marker: the pinned JS reads
 the marker's display to know which way to toggle. The marker draws no box. */
 .arxiv-html-header { display: contents; }
 html[data-reading-mode="enabled"] :is(.arxiv-html-site-chrome, .arxiv-html-header, .ds-site-footer) { display: none; }
 /* The paper controls' icons are blocks, so they sit centred (the pinned JS shows the theme
 icon by making its span a block). */
 .arxiv-html-site-chrome .ds-site-header-tools svg { display: block; width: 18px; height: 18px; flex-shrink: 0; fill: currentColor; }
 /* ar5iv dims every image in dark mode and puts it on a white wash, for paper figures;
 arXiv's own render as they are (the logo on the bar, the funder logos on their white
 chips). Important: ar5iv's dark rules are unlayered, as specific as
 :root:not([data-theme]) img:not(.ltx_no_dark_filter). */
 :is(.ds-site-header, .ds-site-footer) img { filter: none !important; }
 .ds-site-header img { background-color: transparent !important; }
 /* Wide (the theme's 1280px): the theme lays the body out as a grid whose full-width top rows
 were the band's and its sticky reader bar's. The band takes the first and scrolls away; the
 header, the paper's bar (no site navigation, the paper's controls), takes the second and
 stays in view, the one sticky bar, with the anchor clearance that requires (DESIGN-POLICIES).
 The controls show their names. */
 @media (min-width: 1280px) {
 html { scroll-padding-top: 80px; }
 .arxiv-html-site-chrome { display: contents; }
 .arxiv-html-site-chrome > #ds-announcement { grid-area: announce; }
 .arxiv-html-site-chrome > .ds-site-header { grid-area: header; position: sticky; top: 0; z-index: 3; }
 .arxiv-html-site-chrome .tool-icon { display: none; }
 }
 /* Narrow: the header scrolls away, and the controls show icons (names kept for screen
 readers; Why HTML? goes). The theme's contents panel is fixed below where its old sticky
 bar was, and hidden only once the reader has closed it, so until then it is an invisible box
 over the header and the paper: it shows only while open, under the chrome, in its place in
 the page. The paper's footer (build logs, reporting instructions) keeps the text's side
 inset; the theme gives it none, so it met the screen edges. */
 @media (max-width: 1279px) {
 .arxiv-html-site-chrome .tool-wide { display: none; }
 .arxiv-html-site-chrome .tool-label { position: absolute; width: 1px; height: 1px; overflow: hidden; clip: rect(0 0 0 0); white-space: nowrap; }
 html:not([data-toc-display="block"]) .ltx_page_navbar { display: none; }
 .ltx_page_navbar { position: absolute; top: auto; }
 .arxiv-html-footer { padding-inline: 1rem; }
 }
